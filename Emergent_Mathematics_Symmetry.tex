\part{Symmetry}

\section{Spectral Theory}

\emph{Question.} Given a system's linear law of motion and a measurement, which frequencies will appear, and with what strengths?


\begin{definition}[Linear spaces and maps]
Let $k$ be $\R$ or $\C$.  A \emph{linear space} over $k$ is an abelian group
$V$ with scalar multiplication satisfying
\[
a(v+w)=av+aw,\quad (a+b)v=av+bv,\quad
(ab)v=a(bv),\quad 1v=v.
\]
A map $T:V\to W$ is \emph{linear} when
$T(av+bw)=aT(v)+bT(w)$.  Its kernel and image are
\[
\ker T=\{v:T(v)=0\},\qquad \im T=\{T(v):v\in V\}.
\]
For $T:V\to V$, a scalar $\lambda$ is an \emph{eigenvalue} when
$\ker(T-\lambda I)\ne0$; a nonzero vector in this kernel is an
\emph{eigenvector}.
\end{definition}

\begin{definition}[Dual spaces and tensor products]
The \emph{dual space} of $V$ is
$V^\vee=\Hom_k(V,k)$.  Its canonical evaluation pairing is
\[
\operatorname{ev}_V:V^\vee\times V\longrightarrow k,
\qquad (\varphi,v)\longmapsto\varphi(v).
\]
The algebraic transpose of $T:V\to W$ is
$T^\vee:W^\vee\to V^\vee$, defined by $T^\vee(\varphi)=\varphi\circ T$.

The \emph{tensor product} $V\otimes W$ is characterized by a bilinear map
$(v,w)\mapsto v\otimes w$ with the following universal property: every
bilinear map $b:V\times W\to X$ factors uniquely as
\[
V\times W\longrightarrow V\otimes W
\xrightarrow{\,\widetilde b\,}X.
\]
The evaluation pairing induces a linear map
$\operatorname{ev}_V:V^\vee\otimes V\to k$.
\end{definition}

\begin{definition}[Finite-dimensional linear spaces]
A linear space $V$ is \emph{finite-dimensional} when there is a
coevaluation map
\[
\operatorname{coev}_V:k\longrightarrow V\otimes V^\vee
\]
satisfying the snake identities
\[
(\id_V\otimes\operatorname{ev}_V)
\circ(\operatorname{coev}_V\otimes\id_V)=\id_V,
\]
\[
(\operatorname{ev}_V\otimes\id_{V^\vee})
\circ(\id_{V^\vee}\otimes\operatorname{coev}_V)=\id_{V^\vee}.
\]
\end{definition}

\begin{definition}[Trace and dimension]
For a finite-dimensional linear space $V$, let
$s_{V,V^\vee}:V\otimes V^\vee\to V^\vee\otimes V$ exchange the factors.
For $T\in\End(V)$, define
\[
\operatorname{tr}(T)
=\operatorname{ev}_V\circ s_{V,V^\vee}
 \circ(T\otimes\id_{V^\vee})\circ\operatorname{coev}_V,
\qquad
\dim V=\operatorname{tr}(\id_V).
\]
\end{definition}


\begin{definition}[Exterior powers and determinant]
For $m\geq0$, the $m$th \emph{exterior power} of a linear space $V$ is
\[
\Lambda^mV
=V^{\otimes m}/
\langle v_1\otimes\cdots\otimes v_m:
v_i=v_j\text{ for some }i\ne j\rangle,
\]
and the image of $v_1\otimes\cdots\otimes v_m$ is written
$v_1\wedge\cdots\wedge v_m$.  An endomorphism $T$ induces
\[
\Lambda^mT(v_1\wedge\cdots\wedge v_m)
=Tv_1\wedge\cdots\wedge Tv_m.
\]
If $V$ has finite dimension $n$, its \emph{determinant} is the scalar
$\det(T)$ defined by the action on the highest nonzero exterior power:
\[
\Lambda^nT=\det(T)\id_{\Lambda^nV}.
\]
\end{definition}

\begin{exercise}[Exterior powers and determinant]
For a finite-dimensional linear space $V$ of dimension $n$, prove
\[
\dim\Lambda^mV=\binom{n}{m},
\qquad
\Lambda^mV=0\quad(m>n).
\]
Use functoriality of the highest exterior power to prove
\[
\det(ST)=\det(S)\det(T),
\qquad
T\text{ is invertible}\Longleftrightarrow\det(T)\ne0.
\]
\end{exercise}

\begin{definition}[Symmetric powers]
For a linear space $V$, the $m$th \emph{symmetric power} is
\[
\operatorname{Sym}^mV
=V^{\otimes m}/\langle v_1\otimes\cdots\otimes v_m
-v_{\sigma(1)}\otimes\cdots\otimes v_{\sigma(m)}:\sigma\in S_m\rangle.
\]
An endomorphism $T$ induces $\operatorname{Sym}^mT$ on
$\operatorname{Sym}^mV$.
\end{definition}

\begin{exercise}[Symmetric-power trace identity]
Let $V$ be a finite-dimensional complex linear space and $T\in\End(V)$.
Prove that
\[
\sum_{m=0}^{\infty}
\operatorname{tr}(\operatorname{Sym}^mT)z^m
=\frac1{\det(I-zT)}
\]
for $|z|$ smaller than the reciprocal of the spectral radius of $T$.
For $T=I$, deduce
\[
\sum_{m=0}^{\infty}\dim(\operatorname{Sym}^mV)z^m
=(1-z)^{-\dim V}.
\]
\end{exercise}


\begin{definition}[Hilbert spaces and orthogonality]
An \emph{inner product} on a complex linear space is conjugate-linear in its
first argument, linear in its second, positive definite, and satisfies
$\langle x,y\rangle=\overline{\langle y,x\rangle}$.  Its norm is
$\|x\|=\sqrt{\langle x,x\rangle}$.  A \emph{Hilbert space} is an inner-product
space complete for this norm.

Vectors are \emph{orthogonal} when their inner product is zero.  For a
subspace $M\subseteq H$, define
\[
M^\perp=\{x:\langle m,x\rangle=0\text{ for every }m\in M\}.
\]
\end{definition}

\begin{definition}[Bounded operators and adjoints]
A linear map $T:H\to K$ between Hilbert spaces is \emph{bounded} when
\[
\|T\|=\sup_{\|x\|\leq1}\|Tx\|<\infty.
\]
An \emph{adjoint} of $T$ is an operator $T^*:K\to H$ satisfying
\[
\langle x,Ty\rangle=\langle T^*x,y\rangle.
\]
An operator is self-adjoint when $T=T^*$, unitary when $T^*T=TT^*=I$, and
normal when $T^*T=TT^*$.  An \emph{orthogonal projection} is an operator
$P$ satisfying $P=P^*=P^2$.
\end{definition}

\begin{definition}[Trace and Hilbert--Schmidt classes]
For $x,y\in H$, let
\[
\theta_{x,y}(\xi)=\langle y,\xi\rangle x.
\]
The algebra $\mathcal K(H)$ of \emph{compact operators} is the norm closure
of the finite-rank operators.
The \emph{trace class} $S_1(H)$ consists of operators admitting a
representation
\[
A=\sum_{j=1}^{\infty}\theta_{x_j,y_j},
\qquad
\sum_{j=1}^{\infty}\|x_j\|\,\|y_j\|<\infty,
\]
where the series converges in operator norm, with norm the infimum of the
displayed sums.  Define
\[
\Tr(A)=\sum_{j=1}^{\infty}\langle y_j,x_j\rangle.
\]
The \emph{Hilbert--Schmidt class} is
\[
S_2(H)=\{A\in B(H):A^*A\in S_1(H)\},
\qquad
\|A\|_2^2=\Tr(A^*A).
\]
\end{definition}

\begin{exercise}[Hilbert-space foundations]
Prove the Cauchy--Schwarz inequality, the projection theorem, and the Riesz
representation theorem.  Deduce the existence of adjoints and prove
\[
\|T\|^2=\|T^*T\|,\qquad
(\im T)^\perp=\ker T^*.
\]
For every closed subspace $M\subseteq H$, prove
$H=M\oplus M^\perp$ and construct the orthogonal projection $P_M:H\to M$.
\end{exercise}

\begin{exercise}[Trace ideals]
Prove that $S_1(H)$, $\Tr$, and $S_2(H)$ are well-defined. Prove that
$\langle A,B\rangle_{S_2}=\Tr(A^*B)$ makes $S_2(H)$ a Hilbert space and
that the pairing $(A,T)\mapsto\Tr(AT)$ identifies $S_1(H)^*$ isometrically
with $B(H)$.
\end{exercise}

\begin{definition}[Fredholm determinant]
For $A\in S_1(H)$ with nonzero eigenvalues
$\lambda_1(A),\lambda_2(A),\ldots$, repeated according to algebraic
multiplicity, define
\[
\det(I+zA)=\prod_{j=1}^{\infty}(1+z\lambda_j(A)).
\]
\end{definition}

\begin{exercise}[Fredholm determinant and trace identity]
Prove that the defining product converges absolutely and locally uniformly
in $z$ and agrees with the exterior-power determinant when $A$ has finite
rank.  Prove
\[
\det(I-zA)
=\exp\!\left(-\sum_{m=1}^{\infty}
\frac{z^m}{m}\Tr(A^m)\right)
\]
for $|z|<\|A\|^{-1}$.  Use the product to show that the determinant is
entire and that its zeros are the reciprocals of the negatives of the
nonzero eigenvalues of $A$.  Construct the completed exterior powers of
$H$ and deduce
\[
\det(I+zA)=\sum_{m=0}^{\infty}
z^m\Tr(\Lambda^mA).
\]
\end{exercise}


\begin{definition}[Measure and \texorpdfstring{$L^p$}{Lp} spaces]
A \emph{$\sigma$-algebra} $\mathcal F$ on a set $X$ contains
$\varnothing$ and is closed under complements and countable unions.  A
\emph{measure} is a countably additive map
$\mu:\mathcal F\to[0,\infty]$ with $\mu(\varnothing)=0$, and
$(X,\mathcal F,\mu)$ is a
\emph{measure space}.  A map is \emph{measurable} when inverse images of
measurable sets are measurable.  For $1\leq p<\infty$, define
\[
L^p(X,\mu)=
\left\{f:\int_X|f|^p\,d\mu<\infty\right\}\big/\text{equality a.e.}
\]
with its usual norm, and let $L^\infty(X,\mu)$ consist of essentially
bounded measurable functions.
\end{definition}

\begin{definition}[Probability spaces and convergence]
A \emph{probability space} is a measure space
$(\Omega,\mathcal F,\mathbb P)$ with $\mathbb P(\Omega)=1$.  A
\emph{random variable} is a measurable map $X:\Omega\to Y$, its
\emph{law} is $\mathbb P_X(B)=\mathbb P(X^{-1}(B))$, and, when it is
scalar-valued and integrable, its \emph{expectation} is its integral.
A random vector $X=(X_1,\ldots,X_N)$ has \emph{joint density} $p$ with
respect to a measure $\mu$ when
\[
\mathbb P(X\in B)=\int_Bp\,d\mu.
\]
The Gaussian law $N(m,v)$ on $\R$ has density
\[
\frac1{\sqrt{2\pi v}}\exp\!\left[-\frac{(x-m)^2}{2v}\right].
\]
A sequence of random variables in a metric space
$(Y,d)$ \emph{converges in probability} to $X$ when
\[
\mathbb P(d(X_N,X)>\varepsilon)\longrightarrow0
\qquad(\varepsilon>0).
\]
It \emph{converges in distribution}, written $X_N\Longrightarrow X$,
when
\[
\int f\,d\mathbb P_{X_N}\longrightarrow\int f\,d\mathbb P_X
\]
for every bounded continuous $f$.
\end{definition}

\begin{definition}[Gamma and beta integrals]
For $s,t>0$, define
\[
\Gamma(s)=\int_0^\infty x^{s-1}e^{-x}\,dx,
\qquad
B(s,t)=\int_0^1x^{s-1}(1-x)^{t-1}\,dx.
\]
\end{definition}

\begin{exercise}[\texorpdfstring{$L^p$}{Lp} integration]
Construct the Lebesgue integral from nonnegative simple functions.  Prove
the monotone convergence theorem, Fatou's lemma, the dominated convergence
theorem, and Tonelli's and Fubini's theorems, stating the hypotheses that
distinguish the last two.  Prove H\"older's and Minkowski's inequalities
and deduce that $L^p(X,\mu)$ is complete for $1\leq p\leq\infty$.  For
$1<p<\infty$ with $p^{-1}+q^{-1}=1$, identify the continuous dual of
$L^p$ with $L^q$ under the usual semifiniteness hypotheses.
\end{exercise}

\begin{exercise}[Beta--gamma identities]
Use Tonelli's theorem and the change of variables
$x=ru$, $y=r(1-u)$ to prove
\[
B(s,t)=\frac{\Gamma(s)\Gamma(t)}{\Gamma(s+t)}.
\]
Use integration by parts to prove
$\Gamma(s+1)=s\Gamma(s)$, and use the pushforward of Euclidean measure
under the norm map to obtain
\[
\Gamma\!\left(\frac12\right)=\sqrt\pi,
\qquad
\int_{\R^n}e^{-|x|^2/2}\,dx=(2\pi)^{n/2}.
\]
State the uses of Tonelli, Fubini, monotone convergence, and dominated
convergence in the derivation.
\end{exercise}



\begin{definition}[$C^*$-algebras, spectrum, and positivity]
A \emph{$C^*$-algebra} is a complete normed complex algebra $A$ with a
conjugate-linear involution satisfying
\[
(ab)^*=b^*a^*,\qquad \|ab\|\leq\|a\|\|b\|,\qquad
\|a^*a\|=\|a\|^2.
\]
For a unital $C^*$-algebra, define
\[
\spec_A(a)=\{\lambda\in\C:a-\lambda1\text{ is not invertible}\}.
\]
An element is \emph{positive}, written $a\geq0$, when $a=b^*b$ for some
$b\in A$.  It is \emph{self-adjoint} when $a=a^*$ and \emph{normal} when
$a^*a=aa^*$.
\end{definition}

\begin{definition}[Gelfand spectrum]
For a commutative $C^*$-algebra $A$, let $\widehat A$ be the space of
nonzero homomorphisms $\chi:A\to\C$ with its weak-$*$ topology.  The
\emph{Gelfand transform} is
\[
\Gamma:A\longrightarrow C_0(\widehat A),
\qquad \Gamma(a)(\chi)=\chi(a).
\]
\end{definition}

\begin{definition}[Positive functionals and states]
A linear functional $\varphi:A\to\C$ on a unital $C^*$-algebra is
\emph{positive} when $\varphi(a^*a)\geq0$ for every $a\in A$.  A
\emph{state} is a positive functional satisfying $\varphi(1)=1$.
\end{definition}

\begin{exercise}[Gelfand theory]
Prove that $\spec_A(a)$ is nonempty and compact and that
\[
r(a)=\lim_{n\to\infty}\|a^n\|^{1/n}.
\]
For normal $a$, prove $r(a)=\|a\|$.  Prove that positive elements have
unique positive square roots and that $a\geq0$ if and only if $a=a^*$ and
$\spec_A(a)\subseteq[0,\infty)$.  Prove that the Gelfand transform is an
isometric $*$-isomorphism.  For every normal element $a$ of a unital
$C^*$-algebra, prove that there is a unique unital isometric
$*$-homomorphism
\[
C(\spec(a))\longrightarrow C^*(1,a),\qquad f\longmapsto f(a),
\]
sending the coordinate function to $a$, and prove the continuous spectral
mapping theorem.
\end{exercise}


\begin{definition}[Projection-valued measures]
A \emph{projection-valued measure} on a measurable space $X$ is a map $E$
from measurable subsets of $X$ to orthogonal projections on $H$ satisfying
\[
E(\varnothing)=0,\qquad E(X)=I,
\qquad
E\!\left(\bigsqcup_{j=1}^\infty B_j\right)
=\sum_{j=1}^\infty E(B_j)
\]
for pairwise disjoint measurable sets, where the sum converges in the strong
operator topology.  The spectral integral $\int_X f\,dE$ for bounded
measurable $f$ is defined by uniform approximation from simple functions.
For a topological space $X$, the measure is \emph{regular} when every
scalar measure $B\mapsto\langle x,E(B)y\rangle$ is a regular Borel measure.
\end{definition}

\begin{exercise}[\texorpdfstring{$C_0(X)$}{C0(X)} representations]
Let $X$ be locally compact Hausdorff.  A representation
$\pi:C_0(X)\to B(H)$ is \emph{nondegenerate} when
$\overline{\pi(C_0(X))H}=H$.  Prove that every nondegenerate
$*$-representation has a unique regular
projection-valued measure $E_\pi$ satisfying
\[
\pi(f)=\int_X f(x)\,dE_\pi(x),
\qquad f\in C_0(X).
\]
\end{exercise}

\begin{exercise}[Bounded normal spectral theorem]
For a normal operator $T\in B(H)$, apply the preceding representation
theorem to the continuous functional calculus
$C(\spec(T))\to B(H)$ and prove that there is a unique projection-valued
measure $E_T$ on $\spec(T)$ satisfying
\[
T=\int_{\spec(T)}z\,dE_T(z),
\qquad
f(T)=\int_{\spec(T)}f(z)\,dE_T(z).
\]
Deduce that a normal operator on a finite-dimensional Hilbert space has the
intrinsic spectral decomposition
\[
H=\bigoplus_{\lambda\in\spec(T)}E_T(\{\lambda\})H,
\qquad
T|_{E_T(\{\lambda\})H}=\lambda I.
\]
\end{exercise}

\begin{exercise}[Compact spectral theorem]
Prove that every trace-class operator is compact.  For a compact normal
operator $K$, prove that every nonzero spectral value is an eigenvalue of
finite multiplicity, that the nonzero eigenvalues can accumulate only at
zero, and that
\[
K=\sum_{\lambda\in\spec(K)\setminus\{0\}}
\lambda E_K(\{\lambda\})
\]
in operator norm.  For $K\geq0$ in $S_1(H)$, prove
\[
\Tr K
=\sum_{\lambda\in\spec(K)\setminus\{0\}}
\lambda\dim E_K(\{\lambda\})H<\infty.
\]
\end{exercise}

\begin{definition}[Unbounded self-adjoint operators]
An unbounded operator is a linear map $A:\mathcal D(A)\to H$ on a dense
domain.  Its adjoint has domain
\[
\mathcal D(A^*)=
\{y\in H:x\mapsto\langle y,Ax\rangle
\text{ is bounded on }\mathcal D(A)\}.
\]
For $y\in\mathcal D(A^*)$, the vector $A^*y$ is determined by
$\langle A^*y,x\rangle=\langle y,Ax\rangle$ for
$x\in\mathcal D(A)$.  The operator is \emph{self-adjoint} when $A=A^*$,
including equality of domains, and \emph{closed} when its graph is closed
in $H\oplus H$.
\end{definition}

\begin{exercise}[Unbounded spectral theorem]
For a self-adjoint operator $A$, prove that $A\pm iI$ are bijective from
$\mathcal D(A)$ to $H$ with bounded inverses and that the Cayley transform
\[
C_A=(A-iI)(A+iI)^{-1}
\]
is unitary with $\ker(I-C_A)=0$.  Apply the bounded normal spectral theorem
to $C_A$ and the inverse Cayley map to prove that there is a unique
projection-valued measure $E_A$ on $\R$ satisfying
\[
A=\int_\R\lambda\,dE_A(\lambda),\qquad
\mathcal D(A)=
\left\{x:\int_\R\lambda^2\,d\langle x,E_A(\lambda)x\rangle<\infty\right\}.
\]
Use $E_A$ to prove that $e^{-itA}$ is a strongly continuous unitary group
and that every strongly continuous one-parameter unitary group has a unique
self-adjoint generator.
\end{exercise}

\begin{exercise}[Spring and string frequencies]
Let $V$ be a finite-dimensional real linear space with positive-definite
mass form $m$ and nonnegative stiffness form $k$.  Use $m$ to define the
self-adjoint operator $A$ by $k(x,y)=m(x,Ay)$.  Prove that solutions of
$\ddot x+Ax=0$ decompose intrinsically through the spectral projections of
$A$ and that the observed angular frequencies are $\sqrt{\lambda}$ for
$\lambda\in\spec(A)\setminus\{0\}$.

For a stretched string of length $L$, wave speed $c$, and fixed endpoints,
use the self-adjoint Dirichlet realization of $-c^2d^2/dx^2$ to prove
\[
\omega_n=\frac{n\pi c}{L},
\qquad
u_n(x)=\sin\!\left(\frac{n\pi x}{L}\right).
\]
Deduce the harmonic resonance ratios and the nodal locations measured by
driving the string at each $\omega_n$.
\end{exercise}

\begin{exercise}[Spectral lines and time evolution]
\label{ex:spectral-lines}
Let $A$ be self-adjoint, let $B\in B(H)$, and set
$B(t)=e^{itA/\hbar}Be^{-itA/\hbar}$.  For eigenvalues $\lambda,\mu$ of
$A$, write $P_\lambda=E_A(\{\lambda\})$ and prove
\[
P_\mu B(t)P_\lambda
=e^{it(\mu-\lambda)/\hbar}P_\mu BP_\lambda.
\]
Prove in the weak operator topology that
\[
B(t)=\int_{\R}\!\int_{\R}
e^{it(\mu-\lambda)/\hbar}\,dE_A(\mu)B\,dE_A(\lambda).
\]
Deduce that the pure-point response measured through $B$ can contain a
spectral line at $(\mu-\lambda)/\hbar$ only when
$P_\mu BP_\lambda\ne0$, and identify vanishing spectral blocks as
selection rules predicting absent lines.
\end{exercise}

\begin{definition}[Airy function]
The Airy function is the oscillatory integral
\[
\operatorname{Ai}(x)=\frac1\pi\int_0^\infty
\cos\!\left(\frac{t^3}{3}+xt\right)\,dt.
\]
Let $a_n>0$ be determined by
$\operatorname{Ai}(-a_n)=0$, ordered increasingly.
\end{definition}

\begin{exercise}[Airy spectrum of the quantum bouncer]
On $L^2((0,\infty),dz)$ consider the self-adjoint Dirichlet realization
\[
\widehat H=-\frac{\hbar^2}{2m}\frac{d^2}{dz^2}+mgz,
\qquad \psi(0)=0.
\]
Prove that $\operatorname{Ai}''(x)=x\operatorname{Ai}(x)$ and that
$\operatorname{Ai}$ decays as $x\to+\infty$.  With
\[
\ell_g=\left(\frac{\hbar^2}{2m^2g}\right)^{1/3},
\]
derive
\[
E_n=mg\ell_g a_n,
\qquad
\psi_n(z)=C_n\operatorname{Ai}\!\left(\frac z{\ell_g}-a_n\right).
\]
Use the spectral projections of $\widehat H$ to prove that a periodic
perturbation $B\cos(\Omega t)$ can produce a transition from the $n$th to
the $k$th level only through the block
$E_{\widehat H}(\{E_k\})BE_{\widehat H}(\{E_n\})$, at the resonant
frequency
\[
\Omega_{kn}=\frac{mg\ell_g}{\hbar}|a_k-a_n|.
\]
Deduce the transmission thresholds $z_n=\ell_g a_n$ for a horizontal
absorber above a gravitationally bound beam and state how measured
threshold heights and resonance frequencies determine $g$.
\end{exercise}

\section{Abelian Harmonic Analysis}

\emph{Question.} Given the shape and arrangement of a wave source or scatterer, where will its emitted or scattered intensity be concentrated or vanish?


\begin{definition}[Groups, homomorphisms, and quotients]
A \emph{group} is a set $G$ with an associative multiplication, an identity,
and inverses. A \emph{homomorphism} $\phi:G\to H$ preserves multiplication.
A subgroup $N\leq G$ is \emph{normal} if $gNg^{-1}=N$ for every $g\in G$;
the \emph{quotient group} $G/N$ consists of cosets with induced
multiplication.
\end{definition}

\begin{definition}[Actions, orbits, and stabilizers]
A \emph{left action} of $G$ on $X$ is a homomorphism
$G\to\operatorname{Bij}(X)$. The \emph{orbit} and \emph{stabilizer} of
$x\in X$ are $Gx=\{gx:g\in G\}$ and $G_x=\{g:gx=x\}$.
\end{definition}

\begin{definition}[Representations and intertwiners]
A \emph{representation} of $G$ on $V$ is a homomorphism
$\rho:G\to\operatorname{GL}(V)$. An \emph{intertwiner}
$T:(V,\rho)\to(W,\sigma)$ satisfies $T\rho(g)=\sigma(g)T$. A subspace is
\emph{invariant} if it is preserved by every $\rho(g)$; a nonzero
representation is \emph{irreducible} if it has no proper nonzero invariant
subspace. Direct sums and tensor products carry the actions
$\rho\oplus\sigma$ and $g\mapsto\rho(g)\otimes\sigma(g)$.
\end{definition}

\begin{definition}[Topological spaces]
A \emph{topology} on $X$ is a family of subsets containing $\varnothing,X$
and closed under arbitrary unions and finite intersections. A map is
\emph{continuous} when inverse images of open sets are open. A space is
\emph{Hausdorff} when distinct points have disjoint neighborhoods,
\emph{second countable} when its topology is generated by a countable family
of open sets, \emph{compact} when every open cover has a finite subcover,
and \emph{locally compact} when every point has a neighborhood with compact
closure. The \emph{Borel sets} form the $\sigma$-algebra generated by the
open sets. A Borel measure is \emph{regular} when it is inner regular on
open sets and outer regular on Borel sets.
\end{definition}

\begin{definition}[Regular representation]
Let $G$ be a locally compact group with left Haar measure $\mu$.  For
$f,g\in C_c(G)$, define
\[
(f*g)(x)=\int_G f(y)g(y^{-1}x)\,d\mu(y).
\]
The \emph{left regular representation} on $L^2(G)$ is
\[
(\lambda_s\xi)(t)=\xi(s^{-1}t).
\]
\end{definition}

\begin{exercise}[Haar regular representation]
Identify Haar measure on $\R^n$, $\mathbb Z^n$, and $\mathbb T^n$. Construct
normalized Haar measure on a compact Lie group from a left-invariant metric.
Prove associativity of convolution, the $L^1$ convolution inequality, and
$\|f*g\|_2\leq\|f\|_1\|g\|_2$. Prove that $g\mapsto\lambda_g$ is a strongly
continuous unitary representation. For a finite-dimensional representation
of a compact group, average an inner product over $G$ to obtain an invariant
one.
\end{exercise}

\begin{definition}[Pontryagin dual]
For a locally compact abelian group $G$, its \emph{Pontryagin dual}
$\widehat G$ is the group of continuous characters $\chi:G\to U(1)$, with
pointwise multiplication and the compact-open topology. For $f\in L^1(G)$,
define
\[
\widehat f(\chi)=\int_G f(x)\overline{\chi(x)}\,d\mu(x).
\]
The \emph{double-dual map} sends
$x\in G$ to the character $\chi\mapsto\chi(x)$ of $\widehat G$.
\end{definition}

\begin{exercise}[Abelian Fourier duality]
Identify
\[
\widehat{\R^n}\cong\R^n,
\qquad
\widehat{\mathbb Z^n}\cong\mathbb T^n,
\qquad
\widehat{\mathbb T^n}\cong\mathbb Z^n,
\]
and verify the double-dual map in each case. Show that the group Fourier
transform recovers Fourier transforms on $\R^n$ and Fourier series on
$\mathbb T^n$. For a closed subgroup $H\leq G$, define its annihilator
$H^\perp\subseteq\widehat G$ and prove
$\widehat{G/H}\cong H^\perp$.
\end{exercise}



\begin{definition}[Fourier series]
With normalized Haar measure on $\T=\R/2\pi\Z$, define
\[
\widehat f(k)=\int_\T f(x)e^{-ikx}\,dx,\qquad k\in\Z.
\]
The \emph{Fourier series} of $f$ is
$\sum_{k\in\Z}\widehat f(k)e^{ikx}$.
\end{definition}

\begin{definition}[Schwartz space and convolution]
The \emph{Schwartz space} $\mathcal S(\R^n)$ consists of the smooth
functions $f$ satisfying
\[
\sup_{x\in\R^n}|x^\alpha\partial^\beta f(x)|<\infty
\]
for all multi-indices $\alpha,\beta$.  For integrable functions, define
\[
(f*g)(x)=\int_{\R^n}f(y)g(x-y)\,dy.
\]
\end{definition}

\begin{definition}[Fourier transform]
Let $V$ be an $n$-dimensional real Euclidean space with Lebesgue measure
$dq$, and equip $V^*$ with the dual measure $dp$.  For
$f\in\mathcal S(V)$, the semiclassical Fourier transform is
\[
(\mathcal F_\hbar f)(p)
=(2\pi\hbar)^{-n/2}\int_Ve^{-ip(q)/\hbar}f(q)\,dq.
\]
Write $\mathcal F=\mathcal F_1$.
\end{definition}

\begin{exercise}[Fourier inversion and Plancherel]
Let $G$ be a second-countable locally compact abelian group.  Prove that
every irreducible unitary representation is a character.  Prove that Haar
measure on $\widehat G$ can be normalized so that the group Fourier
transform extends to a unitary map
\[
L^2(G)\longrightarrow L^2(\widehat G),
\]
and prove Fourier inversion when both $f$ and $\widehat f$ are integrable.
Prove that $(e^{ikx})_{k\in\Z}$ is a complete orthonormal family in
$L^2(\T)$ and derive
\[
f=\sum_{k\in\Z}\widehat f(k)e^{ikx},
\qquad
\|f\|_2^2=\sum_{k\in\Z}|\widehat f(k)|^2.
\]
Prove Fourier inversion on $\mathcal S(V)$ and prove that
$\mathcal F_\hbar$ extends uniquely to a unitary operator on $L^2(V)$.
For $v\in V$, prove
\[
\mathcal F_\hbar(-i\hbar d_v)\mathcal F_\hbar^{-1}\widehat f(p)
=p(v)\widehat f(p).
\]
Substitute the plane wave
$\psi(q,t)=e^{i(k(q)-\omega t)}$, with $k\in V^*$, into the free
Schr\"odinger equation and prove
\[
\hbar\omega=\frac{\hbar^2\|k\|^2}{2m}.
\]
Set $p=\hbar k$ and $E=\hbar\omega$.  Derive
\[
E(p)=\frac{\|p\|^2}{2m},
\qquad
v_{\rm group}=\frac{p^\sharp}{m},
\]
where $p^\sharp\in V$ is the Euclidean-dual vector to $p$.
Deduce the time-of-flight displacement of a wave packet concentrated near
$p_0$ and the de Broglie relation between its measured wavelength and
momentum.
\end{exercise}


\begin{definition}[Bessel functions]
For $\alpha>-1$, the Bessel function of the first kind is
\[
J_\alpha(z)
=\sum_{m=0}^{\infty}
\frac{(-1)^m}{m!\,\Gamma(m+\alpha+1)}
\left(\frac z2\right)^{2m+\alpha},
\]
with a branch of $z^\alpha$ fixed when $\alpha\notin\Z$.
\end{definition}

\begin{exercise}[Radial Bessel transform]
Prove
\[
z^2J_\alpha''(z)+zJ_\alpha'(z)
+(z^2-\alpha^2)J_\alpha(z)=0
\]
and the recurrence identities
\[
J_{\alpha-1}(z)+J_{\alpha+1}(z)=\frac{2\alpha}{z}J_\alpha(z),
\qquad
J_{\alpha-1}(z)-J_{\alpha+1}(z)=2J_\alpha'(z).
\]
Average the Euclidean Fourier kernel over spheres and express the Fourier
transform of a radial Schwartz function on $\R^n$ as a Hankel transform
with kernel $J_{n/2-1}$.  Derive the corresponding inversion identity.
\end{exercise}

\begin{exercise}[Bessel diffraction]
Let $j_{\nu,k}$ be the $k$th positive zero of $J_\nu$.  Separate the wave
equation on a circular membrane of radius $R$ with Dirichlet boundary and
prove that its normal-mode frequencies are
\[
\omega_{m,k}=\frac cRj_{|m|,k},
\]
with radial factor $J_{|m|}(j_{|m|,k}r/R)$.  Deduce the nodal circles and
angular nodal lines and state how measured resonance ratios determine the
ratios of Bessel zeros without knowledge of $c/R$.

For a uniformly illuminated circular aperture of radius $a$, average the
Fraunhofer kernel over the aperture and derive
\[
\frac{I(\theta)}{I(0)}
=\left[
\frac{2J_1(ka\sin\theta)}{ka\sin\theta}
\right]^2.
\]
Prove that the dark-ring angles satisfy
$ka\sin\theta_k=j_{1,k}$ and obtain the small-angle prediction
$\theta_1\sim j_{1,1}\lambda/(2\pi a)$.  Identify which membrane
resonances and diffraction minima measure zeros of the same Bessel
functions.
\end{exercise}

\begin{definition}[Lattices and reciprocal lattices]
A \emph{lattice} in $\R^n$ is a subgroup
$\Gamma=A\mathbb Z^n$ for some invertible linear map $A:\R^n\to\R^n$. Its
covolume is $\vol(\Gamma)=|\det A|$, and a fundamental cell is
$F=A[0,1)^n$. Its \emph{reciprocal lattice} is the annihilator
\[
\Gamma^*=\{\xi\in\R^n:\xi\cdot\gamma\in2\pi\mathbb Z
\text{ for every }\gamma\in\Gamma\}.
\]
\end{definition}

\begin{definition}[Periodic summation and structure factors]
For $f\in\mathcal S(\R^n)$, its \emph{$\Gamma$-periodization} is
\[
P_\Gamma f(x)=\sum_{\gamma\in\Gamma}f(x+\gamma).
\]
For a finite motif $B\subset\R^n$ with scattering weights $c_b$, its
\emph{structure factor} is
\[
S_B(q)=\sum_{b\in B}c_b e^{-iq\cdot b}.
\]
\end{definition}

\begin{exercise}[Poisson summation]
\label{ex:poisson-summation}
Prove that $P_\Gamma f$ converges with all derivatives and compute its
Fourier coefficients. With the Fourier-transform normalization above, prove
\[
P_\Gamma f(x)=\frac{(2\pi)^{n/2}}{\vol(\Gamma)}
 \sum_{\xi\in\Gamma^*}\mathcal Ff(\xi)e^{i\xi\cdot x}.
\]
Set $x=0$ to obtain the Poisson summation formula. Derive the Jacobi theta
transformation by applying it to a Gaussian.
\end{exercise}

\begin{exercise}[Crystal diffraction and Bragg peaks]
Let $\Gamma_N$ be a finite rectangular portion of $\Gamma$. In the
single-scattering approximation, prove that the amplitude at momentum
transfer $q$ factors as
\[
\mathcal A_N(q)
=S_B(q)\sum_{\gamma\in\Gamma_N}e^{-iq\cdot\gamma}.
\]
Use geometric sums and Poisson summation to show that the normalized
intensity concentrates at $q\in\Gamma^*$ as the crystal grows. For elastic
scattering with incident and outgoing wavevectors of magnitude
$2\pi/\lambda$, derive the Laue condition and, for planes spaced by $d$, the
Bragg relation
\[
2d\sin\theta=m\lambda.
\]
Show how zeros of $S_B$ produce missing reflections. Compare the peak
condition with
\href{https://doi.org/10.1098/rspa.1913.0040}
{W.~H. Bragg and W.~L. Bragg, \emph{Proceedings of the Royal Society A}
\textbf{88} (1913)}.
\end{exercise}


\begin{definition}[Integral lattices and theta series]
A \emph{Euclidean lattice} $L\subset\R^d$ is a discrete subgroup for which
$\R^d/L$ is compact.  Its dual is
\[
L^*=\{y\in\R^d:\langle x,y\rangle\in\Z
\text{ for every }x\in L\}.
\]
It is \emph{integral} if $L\subseteq L^*$, \emph{unimodular} if
$L=L^*$, and \emph{even} if $\|x\|^2\in2\Z$ for every $x\in L$.
For $\tau$ in the upper half-plane and $q=e^{2\pi i\tau}$, its theta series
is
\[
\Theta_L(\tau)=\sum_{x\in L}q^{\|x\|^2/2}.
\]
Its coefficient of $q^n$ counts lattice vectors of squared length $2n$.
\end{definition}

\begin{exercise}[Poisson duality for lattice theta series]
Apply Exercise~\ref{ex:poisson-summation} to a Gaussian and prove
\[
\Theta_L(-1/\tau)
=\frac{(-i\tau)^{d/2}}{\vol(L)}\Theta_{L^*}(\tau),
\]
using the branch positive on the positive imaginary axis.  Show that
evenness gives $\Theta_L(\tau+1)=\Theta_L(\tau)$.  Interpret the first
identity as momentum--winding duality for a compact free boson and as the
reciprocal-lattice relation behind diffraction.  Explain why a general
lattice theta series transforms with its dual rather than being a scalar
modular form by itself.
\end{exercise}


\begin{definition}[The modular group and modular forms]
The modular group $SL_2(\Z)$ acts on the upper half-plane by
\[
\gamma\tau=\frac{a\tau+b}{c\tau+d},
\qquad
\gamma=\begin{pmatrix}a&b\\c&d\end{pmatrix},
\]
and is generated by $S:\tau\mapsto-1/\tau$ and
$T:\tau\mapsto\tau+1$.  A \emph{modular form of integral weight $k$} for
$SL_2(\Z)$ is a holomorphic function $f$ on the upper half-plane satisfying
\[
f(\gamma\tau)=(c\tau+d)^k f(\tau)
\]
and holomorphic at the cusp, meaning that its Fourier expansion
$f(\tau)=\sum_{n\geq0}a_nq^n$ has no negative powers.  It is a
\emph{cusp form} if $a_0=0$.
\end{definition}

\begin{definition}[Theta and eta functions]
Set
\[
\begin{aligned}
\vartheta_2(\tau)&=\sum_{n\in\Z}q^{(n+1/2)^2/2},\\
\vartheta_3(\tau)&=\sum_{n\in\Z}q^{n^2/2},\\
\vartheta_4(\tau)&=\sum_{n\in\Z}(-1)^nq^{n^2/2},
\end{aligned}
\qquad
\eta(\tau)=q^{1/24}(q;q)_\infty.
\]
\end{definition}

\begin{exercise}[Theta transformations]
\label{ex:theta-eta-transformations}
Use Poisson summation on shifted and phase-twisted Gaussians to derive
\[
\vartheta_2(-1/\tau)=(-i\tau)^{1/2}\vartheta_4(\tau),
\quad
\vartheta_3(-1/\tau)=(-i\tau)^{1/2}\vartheta_3(\tau),
\quad
\vartheta_4(-1/\tau)=(-i\tau)^{1/2}\vartheta_2(\tau).
\]
Prove the $T$-transformation laws directly from the series and derive
\[
\eta(\tau+1)=e^{\pi i/12}\eta(\tau),
\qquad
\eta(-1/\tau)=(-i\tau)^{1/2}\eta(\tau),
\]
using the holomorphic square root on the upper half-plane.  Establish the
Jacobi triple product, first as an identity of formal Laurent series,
\[
\sum_{n\in\Z}z^nq^{n^2/2}
=\prod_{m\geq1}(1-q^m)
 (1+zq^{m-1/2})(1+z^{-1}q^{m-1/2}),
\]
and use specializations to recover product forms for the theta functions.
Explain how the two sides encode charge sectors and independent oscillator
modes, respectively.
\end{exercise}


\begin{definition}[Eisenstein series and the discriminant]
Let $\sigma_r(n)=\sum_{d\mid n}d^r$.  The normalized Eisenstein series of
weights four and six are
\[
E_4(\tau)=1+240\sum_{n\geq1}\sigma_3(n)q^n,
\qquad
E_6(\tau)=1-504\sum_{n\geq1}\sigma_5(n)q^n.
\]
The modular discriminant is the weight-twelve cusp form
\[
\Delta(\tau)=\eta(\tau)^{24}
=q\prod_{n\geq1}(1-q^n)^{24}
=\frac{E_4(\tau)^3-E_6(\tau)^2}{1728}.
\]
\end{definition}

\begin{exercise}[Eisenstein series and the $E_8$ shell]
Construct Eisenstein series by summing $(m\tau+n)^{-k}$ over nonzero lattice
points for even $k>2$, prove their modular transformation law, and obtain the
displayed Fourier expansions for $E_4$ and $E_6$.  For a positive-definite
even unimodular lattice, first use
consistency of the $S$ and $T$ laws to prove that its dimension $d$ is
divisible by eight, and then show that its theta series is a modular form of
weight $d/2$.
Construct the $E_8$ lattice as
\[
 \{x\in\Z^8:\textstyle\sum_i x_i\equiv0\pmod2\}
 \cup
 \{x\in(\Z+\tfrac12)^8:\textstyle\sum_i x_i\equiv0\pmod2\}.
\]
Enumerate its vectors of squared length two and obtain $240$, matching the
coefficient of $q$ in $E_4$.
\end{exercise}

\begin{exercise}[Theta diffraction]
Let $T_L=\R^d/L$ carry its flat metric, and normalize $L^*$ by
$\langle x,\xi\rangle\in\Z$.  Prove
\[
\Tr(e^{t\Delta_{T_L}})
=\sum_{\xi\in L^*}e^{-4\pi^2t\|\xi\|^2}
=\Theta_{L^*}(4\pi it).
\]
Apply Poisson duality to derive the short-time image sum and its leading
term $\vol(T_L)(4\pi t)^{-d/2}$.  For a periodic array with motif $B$,
prove that the elastic intensity at reciprocal vector $2\pi\xi$ is
proportional to
\[
|S_B(2\pi\xi)|^2.
\]
Prove that the heat-trace exponents and diffraction-peak radii determine
the same reciprocal squared lengths $\|\xi\|^2$.  Deduce the predicted
peak locations, systematic absences, and thermal decay rates from the
theta coefficients of $L^*$.
\end{exercise}

\section{Compact Harmonic Analysis}

\emph{Question.} When a physical system is unchanged by rotations, which angular states can occur and which transitions have nonzero amplitude?

\begin{exercise}[Maschke's theorem]
Let $G$ be finite and let the characteristic of $k$ not divide $|G|$.
Average an arbitrary projection over $G$ to construct an invariant
complement to every invariant subspace. Deduce that every finite-dimensional
representation is a direct sum of irreducible representations.
\end{exercise}

\begin{exercise}[Schur's lemma]
Prove that a nonzero intertwiner between irreducible complex
finite-dimensional representations is an isomorphism and that every
endomorphism of an irreducible representation is scalar. Deduce uniqueness
of irreducible multiplicities in a decomposition.
\end{exercise}

\begin{definition}[Characters]
The \emph{character} of a finite-dimensional representation $V$ is the class
function $\chi_V(g)=\Tr(\rho_V(g))$. For class functions on a finite group,
define
\[
\langle f,h\rangle_G=\frac1{|G|}\sum_{g\in G}\overline{f(g)}h(g).
\]
\end{definition}

\begin{exercise}[Character orthogonality and decomposition]
Use averaging of linear maps and Schur's lemma to prove orthonormality of the
irreducible characters. Prove that they form a complete orthonormal family
in the class functions and that
$\langle\chi_U,\chi_V\rangle_G=\dim\Hom_G(U,V)$. Deduce the multiplicity
formula and the character table constraints.
\end{exercise}

\begin{definition}[Topological and smooth manifolds]
A \emph{topological manifold} of dimension $n$ is a Hausdorff,
second-countable space locally homeomorphic to $\R^n$. A \emph{smooth
manifold} is a topological manifold with a maximal atlas whose transition
maps are smooth. A map of smooth manifolds is \emph{smooth} when its chart
representatives are smooth; it is a \emph{diffeomorphism} when it has a
smooth inverse.
\end{definition}

\begin{definition}[Tangent and cotangent spaces]
The \emph{tangent space} $T_pM$ is the vector space of derivations
$v:C^\infty(M)\to\R$ at $p$, satisfying
$v(fg)=f(p)v(g)+g(p)v(f)$. The \emph{cotangent space} is
$T_p^*M=(T_pM)^*$. The differential of $F:M\to N$ is
$dF_p(v)(f)=v(f\circ F)$. The disjoint union
$TM=\coprod_{p\in M}T_pM$ has a canonical smooth vector-bundle structure:
if $x=(x^1,\ldots,x^n)$ is a coordinate chart on $U$, its bundle chart is
\[
TM|_U\longrightarrow x(U)\times\R^n,
\qquad
v\in T_pM\longmapsto
\bigl(x(p),v(x^1),\ldots,v(x^n)\bigr).
\]
The cotangent bundle $T^*M$ has the dual bundle charts.
\end{definition}

\begin{definition}[Locally compact groups and Haar measure]
A \emph{topological group} is a group with a Hausdorff topology for which
multiplication and inversion are continuous.  It is \emph{locally compact}
when every point has a compact neighborhood.  A \emph{Radon measure} is a
Borel measure finite on compact sets, outer regular on Borel sets, and inner
regular on open sets.  A \emph{left Haar measure} on a locally compact group
is a nonzero left-invariant Radon measure, so
$\mu(gE)=\mu(E)$.  Its \emph{modular function} is determined by
\[
\mu(Eg)=\Delta_G(g)^{-1}\mu(E).
\]
Write $C_b(G)$, $C_0(G)$, and $C_c(G)$ for the continuous functions that
are bounded, vanish at infinity, and have compact support, respectively.
\end{definition}

\begin{definition}[Matrix coefficients]
A \emph{unitary representation} of a topological group $G$ on a Hilbert
space $H_\pi$ is a homomorphism $\pi:G\to U(H_\pi)$ such that
$g\mapsto\pi(g)x$ is continuous for every $x\in H_\pi$.  Its
\emph{matrix coefficients} are
\[
g\longmapsto\langle y,\pi(g)x\rangle,
\qquad x,y\in H_\pi.
\]
It is \emph{irreducible} when it has no proper nonzero closed invariant
subspace.  The \emph{unitary dual} $\widehat G$ is the set of
unitary-equivalence classes of irreducible unitary representations.
\end{definition}

\begin{definition}[Lie groups]
A \emph{Lie group} is a topological group with a smooth manifold structure
for which multiplication and inversion are smooth. Its \emph{Lie algebra} is
$\mathfrak g=T_eG$ with bracket induced by left-invariant vector fields. A
\emph{one-parameter subgroup} is a smooth homomorphism $\R\to G$. The
\emph{exponential map} $\exp:\mathfrak g\to G$ sends $X$ to the value at
$1$ of the one-parameter subgroup with initial velocity $X$.
\end{definition}

\begin{exercise}[Exponential map]
Prove existence and uniqueness of the one-parameter subgroup generated by
$X\in\mathfrak g$. For a closed matrix group, prove that it is
$t\mapsto e^{tX}$, identify the commutator bracket, and compute the first
nontrivial terms of the Baker--Campbell--Hausdorff formula.
\end{exercise}

\begin{definition}[Infinitesimal representations]
For a finite-dimensional smooth representation
$\rho:G\to\operatorname{GL}(V)$, its
\emph{infinitesimal representation} is
\[
d\rho:\mathfrak g\to\End(V),\qquad
d\rho(X)=\left.\frac{d}{dt}\right|_{0}\rho(\exp tX).
\]
\end{definition}


\begin{exercise}[Peter--Weyl theorem]
Let $G$ be compact with normalized Haar measure.  Prove that matrix
coefficients of irreducible unitary representations are dense in $C(G)$ and
$L^2(G)$ and that
\[
L^2(G)\cong
\mathop{\widehat{\bigoplus}}_{\pi\in\widehat G}
H_\pi\otimes H_\pi^*.
\]
Show that the left regular representation acts as
$\pi\otimes I_{H_\pi^*}$ on the $\pi$-summand.  Derive Schur
orthogonality for irreducible matrix coefficients and the multiplicity
$d_\pi=\dim H_\pi$.
For $f\in L^2(G)\subseteq L^1(G)$, define
$\widehat f(\pi)=\int_Gf(g)\pi(g)^*\,dg$ and prove
\[
f=\sum_{\pi\in\widehat G}
(\dim H_\pi)\Tr\!\bigl(\widehat f(\pi)\pi(\,\cdot\,)\bigr),
\qquad
\|f\|_2^2=\sum_{\pi\in\widehat G}
(\dim H_\pi)\|\widehat f(\pi)\|_{\rm HS}^2
\],
where the first sum converges in $L^2(G)$.  For $f\in C(G)$, construct
finite matrix-coefficient approximate identities and prove uniform
convergence of the corresponding summability means to $f$.
\end{exercise}

\begin{exercise}[Character inversion]
Let $G$ be compact with normalized Haar measure, and let $f$ be a finite
linear combination of irreducible characters.  For an irreducible unitary
representation $\pi$, set
\[
\widehat f(\pi)=\int_G f(s)\pi(s^{-1})\,ds.
\]
If $f$ is central, use Schur's lemma to write
$\widehat f(\pi)=c_\pi I_{d_\pi}$.  Derive
\[
f(e)=\sum_{\pi\in\widehat G}d_\pi\Tr(\widehat f(\pi))
=\sum_{\pi\in\widehat G}d_\pi^2c_\pi
\]
and prove that convolution by $f$ acts by the scalar $c_\pi$ on every
copy of $\pi$ in $L^2(G)$.  Extend the identity to the natural Sobolev
classes for which the character series converges absolutely.
\end{exercise}

\begin{definition}[Rotations and spin]
Define
\[
\begin{aligned}
SO(3)&=\{R\in\End(\R^3):R^*R=\id,\ \det R=1\},\\
SU(2)&=\{U\in\End(\C^2):U^*U=\id,\ \det U=1\}.
\end{aligned}
\]
An \emph{angular-momentum representation} is a unitary representation of
$SU(2)$; with Planck constant $\hbar$, its infinitesimal generators
$J_1,J_2,J_3$ are normalized by
$[J_i,J_j]=i\hbar\varepsilon_{ijk}J_k$.
\end{definition}

\begin{exercise}[$SU(2)$ double-covers $SO(3)$]
Let $SU(2)$ act by conjugation on the real space of traceless Hermitian
$2\times2$ operators. Prove that the resulting homomorphism
$SU(2)\to SO(3)$ is surjective with kernel $\{\pm\id\}$. Deduce the effect of
a $2\pi$ rotation in integer- and half-integer-spin representations and the
$4\pi$ restoration observed by neutron interferometry.
\end{exercise}

\begin{exercise}[Irreducible representations of $SU(2)$]
Using raising and lowering operators, prove that every finite-dimensional
irreducible smooth complex representation is indexed by
$j\in\{0,\frac12,1,\ldots\}$, has dimension $2j+1$, and has
$J_3$-eigenvalues $m\hbar$ for $m=-j,-j+1,\ldots,j$. Compute the spectrum of
$J^2=J_1^2+J_2^2+J_3^2$. For $j=\frac12$, relate the two $J_3$ outcomes to
the two Stern--Gerlach beams.
\end{exercise}

\begin{exercise}[Clebsch--Gordan decomposition]
Prove
\[
V_{j_1}\otimes V_{j_2}\cong
\bigoplus_{j=|j_1-j_2|}^{j_1+j_2}V_j.
\]
Construct coupled angular-momentum eigenstates from highest-weight vectors
and compute the singlet and triplet decomposition of
$V_{1/2}\otimes V_{1/2}$.
\end{exercise}

\begin{definition}[Ordinary hypergeometric series]
For $m\in\Z_{\geq0}$, define
\[
(a)_0=1,
\qquad
(a)_m=a(a+1)\cdots(a+m-1)\quad(m\geq1).
\]
Where the quotient is defined,
$(a)_m=\Gamma(a+m)/\Gamma(a)$; the product gives its continuation at
nonpositive integral $a$.  If no denominator parameter $b_j$ is a
nonpositive integer, the \emph{generalized hypergeometric series} is
\[
{}_rF_s\!\left(
\begin{matrix}a_1,\ldots,a_r\\b_1,\ldots,b_s\end{matrix};z
\right)
=\sum_{m=0}^\infty
\frac{(a_1)_m\cdots(a_r)_m}
{(b_1)_m\cdots(b_s)_m}\frac{z^m}{m!}.
\]
It is interpreted as a finite sum when a numerator parameter is a
nonpositive integer.  Otherwise it converges for every $z$ when $r\leq s$,
for $|z|<1$ when $r=s+1$, and only at $z=0$ when $r>s+1$.
The Gauss hypergeometric operator is
\[
\mathcal L_{a,b,c}
=z(1-z)\frac{d^2}{dz^2}
+\bigl(c-(a+b+1)z\bigr)\frac{d}{dz}-ab.
\]
\end{definition}

\begin{exercise}[Gauss hypergeometric theorem]
Prove
\[
{}_1F_0\!\left(\begin{matrix}a\\-\end{matrix};z\right)
=(1-z)^{-a}
\]
and prove that ${}_2F_1(a,b;c;z)$ is the solution of
$\mathcal L_{a,b,c}f=0$ analytic at $z=0$ with $f(0)=1$.  Derive Euler's
integral and transformation
\[
{}_2F_1(a,b;c;z)
=\frac{\Gamma(c)}{\Gamma(b)\Gamma(c-b)}
\int_0^1u^{b-1}(1-u)^{c-b-1}(1-zu)^{-a}\,du,
\]
\[
{}_2F_1(a,b;c;z)
=(1-z)^{c-a-b}{}_2F_1(c-a,c-b;c;z),
\]
and Gauss's summation
\[
{}_2F_1(a,b;c;1)
=\frac{\Gamma(c)\Gamma(c-a-b)}
{\Gamma(c-a)\Gamma(c-b)}.
\]
State the parameter conditions for each identity and extend them by
analytic continuation.
\end{exercise}

\begin{definition}[Jacobi polynomials]
For $\alpha,\beta>-1$, the \emph{Jacobi polynomial} is
\[
P_n^{(\alpha,\beta)}(x)
=\frac{(\alpha+1)_n}{n!}
{}_2F_1\!\left(
\begin{matrix}-n,n+\alpha+\beta+1\\\alpha+1\end{matrix};
\frac{1-x}{2}
\right).
\]
\end{definition}

\begin{exercise}[Jacobi spectral equation]
Prove
\[
\left((1-x^2)\frac{d^2}{dx^2}
+\bigl(\beta-\alpha-(\alpha+\beta+2)x\bigr)\frac{d}{dx}\right)
P_n^{(\alpha,\beta)}
=-n(n+\alpha+\beta+1)P_n^{(\alpha,\beta)}.
\]
Derive Jacobi orthogonality on $[-1,1]$ with weight
$(1-x)^\alpha(1+x)^\beta$ and derive the three-term recurrence.
\end{exercise}

\begin{exercise}[Spherical harmonics and atomic orbitals]
Identify $S^2$ with $SO(3)/SO(2)$ and $L^2(S^2)$ with the right
$SO(2)$-invariant subspace of $L^2(SO(3))$. Use Peter--Weyl to obtain
\[
L^2(S^2)\cong\mathop{\widehat{\bigoplus}}_{\ell=0}^\infty V_\ell
\]
and prove that the functions
$Y_{\ell m}\propto D^\ell_{m0}$, $-\ell\leq m\leq\ell$, form a complete
orthogonal family in $V_\ell$. For the unit round sphere, set
\[
\Delta_{S^2}
=\frac1{\sin\theta}\partial_\theta
  \bigl(\sin\theta\,\partial_\theta\bigr)
 +\frac1{\sin^2\theta}\partial_\phi^2.
\]
Derive
$\Delta_{S^2}Y_{\ell m}=-\ell(\ell+1)Y_{\ell m}$ and completeness of the
spherical harmonics.  Derive the Jacobi realization
\[
Y_{\ell m}(\theta,\phi)
=C_{\ell m}e^{im\phi}(\sin\theta)^{|m|}
P_{\ell-|m|}^{(|m|,|m|)}(\cos\theta)
\]
with $C_{\ell m}$ fixed by unit norm. Separate the Coulomb Hamiltonian in spherical
coordinates, identify its angular factor, and recover
$\psi_{n\ell m}(r,\theta,\phi)=R_{n\ell}(r)Y_{\ell m}(\theta,\phi)$.

\end{exercise}

\begin{definition}[Clebsch--Gordan and Wigner coefficients]

\end{definition}

\section{Semisimple Harmonic Analysis}

\emph{Question.} After a localized pulse is released in a medium whose area grows exponentially with distance, what signal is measured at radius $r$ and time $t$?

\begin{definition}[Cartan and Iwasawa decompositions]
Let $G$ be a connected real semisimple Lie group with finite center and let
$\mathfrak g$ be its Lie algebra.  A \emph{Cartan involution} gives a vector
space decomposition
\[
\mathfrak g=\mathfrak k\oplus\mathfrak p
\]
into its $+1$ and $-1$ eigenspaces; the subgroup $K$ with Lie algebra
$\mathfrak k$ is maximal compact, and $G/K$ is a noncompact Riemannian
symmetric space.  Choose a maximal abelian subspace
$\mathfrak a\subseteq\mathfrak p$, put $A=\exp\mathfrak a$, choose positive
restricted roots, and let $N$ be the subgroup generated by their root
spaces.  The \emph{Iwasawa decomposition} is the diffeomorphism
\[
K\times A\times N\longrightarrow G,
\qquad(k,a,n)\longmapsto kan.
\]
If $\mathfrak a^+$ is a positive Weyl chamber, the \emph{Cartan
decomposition} is $G=K\overline{A^+}K$.  Write
$M=Z_K(A)$ and $W=N_K(A)/M$, and let $\rho$ be half the sum of the positive
restricted roots, counted with multiplicity.
\end{definition}

\begin{definition}[Spherical transform]
For $\lambda\in\mathfrak a^*$, the normalized \emph{spherical principal
series} is
\[
\pi_\lambda
=\operatorname{Ind}_{MAN}^G(1_M\otimes e^{i\lambda}\otimes1_N),
\]
where normalized induction incorporates the modular factor determined by
$\rho$.  It contains a unit $K$-fixed vector $v_\lambda$, and its
\emph{zonal spherical function} is
\[
\varphi_\lambda(g)
=\langle v_\lambda,\pi_\lambda(g)v_\lambda\rangle.
\]
It is $K$-bi-invariant, satisfies $\varphi_\lambda(e)=1$, and is a joint
eigenfunction of the $G$-invariant differential operators on $G/K$.

For $f\in C_c^\infty(K\backslash G/K)$, its \emph{spherical transform} is
\[
\widetilde f(\lambda)
=\int_G f(g)\varphi_{-\lambda}(g)\,dg.
\]
Harish-Chandra's $c$-function is the coefficient governing the leading
asymptotics of $\varphi_\lambda(\exp H)$ as
$H\to\infty$ in a positive chamber.
\end{definition}

\begin{exercise}[Harish-Chandra transform on $\mathbb H^2$]
For the sign convention in which the Laplace--Beltrami operator is

\qquad \varphi_\lambda(0)=1,

 f(r)\varphi_\lambda(r)\sinh r\,dr.
\]

$\lambda^2$ give orthogonal generalized eigenfunctions after truncation to
$[0,R]$ with a common self-adjoint boundary condition.  Compare the

its oscillation parameter is continuous.  Contrast this with the discrete
Dirichlet spectrum in the finite hyperbolic-drum exercise below.
\end{exercise}

\begin{definition}[Discrete-series characters]
A \emph{distribution} on a Lie group $G$ is a continuous linear functional on
$C_c^\infty(G)$ with its usual test-function topology. Let $\pi$ be an
irreducible admissible unitary representation of a connected
real reductive group.  Although the infinite-dimensional operator $\pi(g)$
usually has no trace, for $f\in C_c^\infty(G)$ the integrated operator
\[
\pi(f)=\int_G f(g)\pi(g)\,dg
\]
is trace class.  The \emph{Harish-Chandra character} is the
conjugation-invariant distribution
\[
\Theta_\pi(f)=\Tr(\pi(f)).
\]
An irreducible unitary representation belongs to the \emph{discrete series}
if it has a nonzero matrix coefficient square-integrable modulo the center.
\end{definition}

\begin{exercise}[Spherical Plancherel formula]

=C_G\int_{\mathfrak a^*}|\widetilde f(\lambda)|^2

{\Gamma(\tfrac12+i\lambda)},
\qquad

\]
with the constants fixed by the same normalization.

\end{exercise}

\section{Kirillov Orbit Method}

\emph{Question.} How does the phase produced by exchanging position and momentum shifts constrain a particle's quantum states and oscillator transitions?


\begin{definition}[Coadjoint action and stabilizers]
For a Lie group $G$ with Lie algebra $\mathfrak g$, define
\[
\Ad_g=d(h\mapsto ghg^{-1})_e,
\qquad
(\Ad_g^*\ell)(X)=\ell(\Ad_{g^{-1}}X).
\]
For $\ell\in\mathfrak g^*$, define
\[
\mathcal O_\ell=\Ad^*(G)\ell,
\qquad
G_\ell=\{g\in G:\Ad_g^*\ell=\ell\},
\]
and
\[
\mathfrak g_\ell
=\{X\in\mathfrak g:\ell([X,Y])=0
\text{ for every }Y\in\mathfrak g\}.
\]
For $X\in\mathfrak g$, set
\[
(\ad_X^*\ell)(Y)=-\ell([X,Y]).
\]
\end{definition}

\begin{definition}[Kirillov form]
On $T_\ell\mathcal O_\ell$, define
\[
\omega_\ell(\ad_X^*\ell,\ad_Y^*\ell)=\ell([X,Y]).
\]
For $\eta=\Ad_g^*\ell$, transport $\omega_\ell$ by the coadjoint
action to define $\omega_\eta$.
\end{definition}

\begin{exercise}[Geometry of coadjoint orbits]
Prove
\[
T_\ell\mathcal O_\ell
=\{\ad_X^*\ell:X\in\mathfrak g\}
\cong\mathfrak g/\mathfrak g_\ell.
\]
Prove that the Kirillov form is well-defined, alternating, nondegenerate,
and $G$-invariant.  Prove the cyclic identity
\[
\ell([X,[Y,Z]])+\ell([Y,[Z,X]])+\ell([Z,[X,Y]])=0
\]
and identify it with the closedness identity for the invariant two-form
$\omega$ on $\mathcal O_\ell$.
\end{exercise}


\begin{definition}[Orbital representations]
Let $G$ be connected, simply connected, and nilpotent.  A
\emph{polarization} at $\ell\in\mathfrak g^*$ is a Lie subalgebra
$\mathfrak p\subseteq\mathfrak g$ satisfying
\[
\ell([\mathfrak p,\mathfrak p])=0,
\qquad
\dim\mathfrak p
=\frac{\dim\mathfrak g+\dim\mathfrak g_\ell}{2}.
\]
For $P=\exp(\mathfrak p)$, define
\[
\chi_\ell(\exp X)=e^{i\ell(X)}.
\]
The \emph{orbital representation}
$\pi_\ell=\operatorname{Ind}_P^G\chi_\ell$ acts on measurable
functions $F:G\to\C$ satisfying
\[
F(gp)=\chi_\ell(p)^{-1}F(g),
\qquad
\int_{G/P}|F(g)|^2\,d(gP)<\infty,
\]
by
\[
(\pi_\ell(g_0)F)(g)=F(g_0^{-1}g),
\]
where $d(gP)$ is a $G$-invariant quotient measure.
\end{definition}

\begin{exercise}[Kirillov correspondence]
Prove that every $\ell\in\mathfrak g^*$ admits a polarization, that
$P=\exp(\mathfrak p)$ is closed, and that $\chi_\ell$ is a unitary
character.  Prove that $\pi_\ell$ is irreducible and independent of the
polarization up to unitary equivalence.  Prove
\[
\pi_\ell\simeq\pi_{\ell'}
\Longleftrightarrow
\mathcal O_\ell=\mathcal O_{\ell'}
\]
and that
\[
\mathfrak g^*/G\longrightarrow\widehat G,
\qquad
\mathcal O_\ell\longmapsto[\pi_\ell],
\]
is a bijection.
\end{exercise}

\begin{exercise}[Kirillov character formula]
Let $f$ be a function on $G$ for which $f\circ\exp\in\mathcal S(\mathfrak g)$,
choose $dX$ so that $\exp_*dX$ is Haar measure, and define
\[
\mathcal F_Gf(\eta)
=\int_{\mathfrak g}f(\exp X)e^{i\eta(X)}\,dX.
\]
If $\dim\mathcal O_\ell=2d$, define its orbital measure by
\[
d\mu_{\mathcal O_\ell}
=(2\pi)^{-d}\frac{\omega^d}{d!}.
\]
For
\[
\pi_\ell(f)=\int_G f(g)\pi_\ell(g)\,dg,
\]
prove that $\pi_\ell(f)$ is trace class and that
\[
\Tr\pi_\ell(f)
=\int_{\mathcal O_\ell}\mathcal F_Gf(\eta)\,
d\mu_{\mathcal O_\ell}(\eta).
\]
Compare the correspondence and character formula with
\href{https://doi.org/10.1070/RM1962v017n04ABEH004118}
{Kirillov, \emph{Russian Mathematical Surveys} \textbf{17} (1962)}.
\end{exercise}


\begin{definition}[Projective representations]
A \emph{normalized multiplier} on a locally compact group $G$ is a
continuous map
$\sigma:G\times G\to\T$ satisfying
\[
\sigma(x,e)=\sigma(e,x)=1,
\qquad
\sigma(x,y)\sigma(xy,z)=\sigma(y,z)\sigma(x,yz).
\]
A \emph{$\sigma$-projective unitary representation} is a strongly continuous map
$U:G\to U(H)$ satisfying
\[
U_xU_y=\sigma(x,y)U_{xy},
\qquad U_e=I.
\]
Multipliers $\sigma$ and $\sigma'$ are \emph{cohomologous} when there is a
continuous map $b:G\to\T$, $b(e)=1$, such that
\[
\sigma'(x,y)=b(x)b(y)\overline{b(xy)}\,\sigma(x,y).
\]
\end{definition}


\begin{definition}[Symplectic phase space]
A \emph{symplectic form} on a finite-dimensional real linear space $E$ is a
nondegenerate alternating bilinear form $\omega:E\times E\to\R$.  On
$E=V\oplus V^*$, define
\[
\omega\bigl((q,p),(q',p')\bigr)=p(q')-p'(q).
\]
For $\lambda\in\R$, define
\[
\sigma_{\lambda\omega}(z,z')=e^{i\lambda\omega(z,z')/2}.
\]
The Weyl multiplier is $\sigma_\omega$.  For $a\in C^\infty(E)$, define
$X_a:E\to E$ by
\[
\omega(X_a(z),u)=-da_z(u)
\qquad(z,u\in E),
\]
and define the canonical Poisson bracket by
\[
\{a,b\}(z)=-\omega(X_a(z),X_b(z)).
\]
\end{definition}

\begin{exercise}[The Weyl multiplier]
Prove that $\sigma_{\lambda\omega}$ is a normalized multiplier for every
$\lambda\in\R$.
\end{exercise}


\begin{definition}[Commutator bicharacter]
For a multiplier $\sigma$ on an abelian group $A$, define
\[
c_\sigma(x,y)=\sigma(x,y)\overline{\sigma(y,x)}.
\]
\end{definition}

\begin{exercise}[Commutator bicharacter]
Prove that $c_\sigma$ is an alternating bicharacter and depends only on the
cohomology class of $\sigma$.  For the Weyl multiplier on $V\oplus V^*$,
prove
\[
c_\sigma\bigl((q,p),(q',p')\bigr)
=e^{i(p(q')-p'(q))}.
\]
\end{exercise}

\begin{exercise}[Linear Darboux theorem]
Prove that every finite-dimensional symplectic space is symplectically
isomorphic to
\[
(\R^{2n},\omega_0),
\qquad
\omega_0\bigl((q,p),(q',p')\bigr)=p(q')-p'(q).
\]
Prove
\[
c_{\sigma_{\lambda\omega}}(z,z')=e^{i\lambda\omega(z,z')}.
\]
\end{exercise}


\begin{definition}[Heisenberg group]
For a symplectic space $(E,\omega)$, its \emph{Heisenberg group} is
\[
\mathbb H(E,\omega)=E\times\R
\]
with multiplication
\[
(z,s)(z',s')
=(z+z',s+s'+\tfrac12\omega(z,z')).
\]
Write
$\mathbb H_n=\mathbb H(\R^n\oplus(\R^n)^*,\omega)$ for the canonical
symplectic form on $\R^n\oplus(\R^n)^*$.  Its Lie algebra is
$\mathfrak h(E,\omega)=E\oplus\R Z$ with
\[
[(z,sZ),(z',s'Z)]=\omega(z,z')Z.
\]
For $E=V\oplus V^*$ and $\lambda\ne0$, the \emph{Schr\"odinger
representation} on $L^2(V)$ is
\[
\bigl(\pi_\lambda(q,p,s)\xi\bigr)(x)
=e^{i\lambda(s+p(x-q/2))}\xi(x-q).
\]
For $\hbar>0$, $\ell\in V^*$, and $v\in V$, define on $\mathcal S(V)$
\[
Q_\ell\psi(x)=\ell(x)\psi(x),
\qquad
P_v\psi(x)=-i\hbar\,d\psi_x(v).
\]
\end{definition}

\begin{exercise}[Central extensions]
Prove that $\mathbb H(E,\omega)$ is a locally compact group with center
$\{0\}\times\R$ and that
\[
\pi_\lambda(0,s)=e^{i\lambda s}I.
\]
Prove that projective representations of $E$ with multiplier
$e^{i\lambda\omega/2}$ correspond to representations of
$\mathbb H(E,\omega)$ with central character $e^{i\lambda s}$.
\end{exercise}

\begin{exercise}[Canonical commutation and Weyl relations]
Prove
\[
[Q_\ell,P_v]=i\hbar\ell(v)I.
\]
For
\[
(T_q\psi)(x)=\psi(x-q),
\qquad
(M_p\psi)(x)=e^{ip(x)}\psi(x),
\]
prove
\[
M_pT_q=e^{ip(q)}T_qM_p.
\]

\]
and identify $W$ with the restriction of $\pi_1$ to
$V\oplus V^*\subset\mathbb H(V\oplus V^*,\omega)$.
\end{exercise}


\begin{exercise}[Schr\"odinger model]
Identify

For $(\xi,\lambda)\in E^*\oplus\R Z^*$, prove

\begin{exercise}[Stone--von Neumann theorem]
Deduce from the Kirillov correspondence that $\pi_\lambda$ is

\end{exercise}


\begin{definition}[Coefficient transforms]
Let $(X,\mu)$ be a measure space and let $U:X\to U(H)$ be a weakly
measurable map.  The \emph{coefficient transform} of the unitary system
$U$ is
\[
\mathcal V_U:S_1(H)\longrightarrow L^\infty(X),
\qquad
\mathcal V_U(A)(x)=\Tr(AU_x).
\]
\end{definition}

\begin{definition}[Fourier--Wigner transform]
Let $V$ be an $n$-dimensional real linear space with Lebesgue measure $dq$,
and equip $V^*$ with the dual measure $dp$ for which
$(2\pi)^{-n/2}\int_Ve^{-ip(q)}f(q)\,dq$ is unitary on $L^2$.
For $q\in V$ and $p\in V^*$, define
\[
(T_q\xi)(x)=\xi(x-q),
\qquad
(M_p\xi)(x)=e^{ip(x)}\xi(x).
\]
For $z=(q,p)\in V\oplus V^*$, define
\[
W(z)=e^{-ip(q)/2}M_pT_q.
\]
The \emph{Fourier--Wigner transform} is
\[
\mathcal W:S_1(L^2(V))\longrightarrow C_b(V\oplus V^*),
\qquad
\mathcal W(A)(z)=(2\pi)^{-n/2}\Tr(AW(z)).
\]
\end{definition}

\begin{exercise}[Fourier--Wigner Plancherel]
Prove
\[
W(z)^*=W(-z).
\]
For $A,B\in S_1(L^2(V))\cap S_2(L^2(V))$, prove the Fourier--Wigner
orthogonality identity
\[
\int_{V\oplus V^*}
\overline{\mathcal W(A)(z)}\mathcal W(B)(z)\,dz
=\Tr(A^*B).
\]
Deduce that $\mathcal W$ extends uniquely to a unitary map
\[
S_2(L^2(V))\longrightarrow L^2(V\oplus V^*)
\]
and, whenever $\mathcal W(A)\in L^1(V\oplus V^*)$, prove
\[
A=(2\pi)^{-n/2}\int_{V\oplus V^*}
\mathcal W(A)(z)W(z)^*\,dz
\]
in the weak operator topology.
\end{exercise}


\begin{exercise}[Heisenberg Plancherel and inversion]
\label{ex:heisenberg-harmonic-analysis}
For $V=\R^n$ and $f$ satisfying
$f\circ\exp\in\mathcal S(\mathfrak h_n)$, prove
\[
\Tr\pi_\lambda(f)
=(2\pi)^n|\lambda|^{-n}
\int_\R f(0,0,s)e^{i\lambda s}\,ds.
\]
Derive this identity both from the Schr\"odinger kernel and from the
Kirillov character formula on $\mathcal O_\lambda$.  For
$f\in L^1(\mathbb H_n)\cap L^2(\mathbb H_n)$, use Haar measure
$dq\,dp\,ds$ and the scalar Fourier transform
$\widehat h(\lambda)=\int_\R h(s)e^{-i\lambda s}\,ds$, and define
\[
\widehat f(\lambda)
=\int_{\mathbb H_n}f(g)\pi_\lambda(g)^*\,dg.
\]
Derive the Plancherel measure
$(2\pi)^{-(n+1)}|\lambda|^n\,d\lambda$ and prove
\[
\|f\|_2^2
=(2\pi)^{-(n+1)}\int_{\R\setminus\{0\}}
\|\widehat f(\lambda)\|_{\rm HS}^2|\lambda|^n\,d\lambda.
\]
Prove the inversion formula and the distributional orthogonality of Weyl
operators.  Prove that representations with trivial central character have
zero Plancherel measure.
\end{exercise}


\begin{definition}[Hermite polynomials and functions]
\label{def:hermite-functions}
Define the physicists' Hermite polynomials by
\[
e^{2xt-t^2}=\sum_{n=0}^{\infty}H_n(x)\frac{t^n}{n!}.
\]
The normalized \emph{Hermite functions} are
\[
h_n(x)=\frac{e^{-x^2/2}H_n(x)}
{(2^nn!\sqrt\pi)^{1/2}}.
\]
\end{definition}

\begin{exercise}[Mehler kernel]
Prove that $(h_n)_{n\geq0}$ is a complete orthonormal family in
$L^2(\R)$ and, for $|r|<1$, derive Mehler's identity
\[
\sum_{n=0}^{\infty}r^nh_n(x)h_n(y)
=\frac1{\sqrt{\pi(1-r^2)}}
\exp\!\left(
-\frac{(1+r^2)(x^2+y^2)-4rxy}{2(1-r^2)}
\right).
\]
\end{exercise}

\begin{exercise}[Oscillator spectroscopy]
For
\[
\widehat H=-\frac{\hbar^2}{2m}\frac{d^2}{dx^2}
+\frac{m\omega^2x^2}{2},
\]
use the Hermite functions to prove that
\[
\psi_n(x)=\left(\frac{m\omega}{\hbar}\right)^{1/4}
h_n\!\left(\sqrt{\frac{m\omega}{\hbar}}x\right),
\qquad
E_n=\hbar\omega\left(n+\frac12\right)
\]
give the complete spectral decomposition of $\widehat H$.  Construct the
creation and annihilation operators and compute
\[
\langle\psi_m,Q\psi_n\rangle
=\sqrt{\frac{\hbar}{2m\omega}}
\bigl(\sqrt{n+1}\,\delta_{m,n+1}
+\sqrt n\,\delta_{m,n-1}\bigr).
\]
Deduce the spectral-line prediction
$|E_{n\pm1}-E_n|/\hbar=\omega$ and the relative line strengths for a
linearly coupled oscillator.  For $\alpha\in\C$, prove convergence and
normalization of
\[
\psi_\alpha=e^{-|\alpha|^2/2}
\sum_{n\geq0}\frac{\alpha^n}{\sqrt{n!}}\psi_n,
\]
show that its number statistics are Poisson with mean $|\alpha|^2$, and
derive its sinusoidal position expectation.  Analytically continue
Mehler's identity with the usual $t-i0$ prescription to obtain the
oscillator propagator
\[
K_t(x,y)=
\sqrt{\frac{m\omega}{2\pi i\hbar\sin(\omega t)}}
\exp\!\left[
\frac{im\omega}{2\hbar\sin(\omega t)}
\bigl((x^2+y^2)\cos(\omega t)-2xy\bigr)
\right].
\]
Use these formulas to predict the spectral spacing, coherent oscillation
frequency, occupation-number distribution, and revival period measured in
an oscillator experiment.

\part{Symmetry}

\section{Spectral Theory}



\begin{exercise}[Vector spaces, duality, trace, and dimension]
Let $k$ be a field.  A \emph{vector space} over $k$ is an abelian group
$V$ with scalar multiplication satisfying
\[
a(v+w)=av+aw,\quad (a+b)v=av+bv,\quad
(ab)v=a(bv),\quad 1v=v.
\]
A map $T:V\to W$ is \emph{linear} when
$T(av+bw)=aT(v)+bT(w)$.  Its kernel and image are
\[
\ker T=\{v:T(v)=0\},
\qquad
\im T=\{T(v):v\in V\}.
\]

The \emph{dual space} of $V$ is
$V^\vee=\Hom_k(V,k)$.  Its canonical evaluation pairing is
\[
\operatorname{ev}_V:V^\vee\times V\longrightarrow k,
\qquad
(\varphi,v)\longmapsto\varphi(v).
\]
The \emph{tensor product} $V\otimes W$ is characterized by a bilinear map
$(v,w)\mapsto v\otimes w$ with the following universal property: every
bilinear map
$b:V\times W\to X$ factors uniquely as
\[
V\times W\longrightarrow V\otimes W
\xrightarrow{\,\widetilde b\,}X.
\]
The evaluation pairing induces a linear map
$\operatorname{ev}_V:V^\vee\otimes V\to k$.

A vector space $V$ is \emph{finite-dimensional} when there is a
coevaluation map
\[
\operatorname{coev}_V:k\longrightarrow V\otimes V^\vee
\]
satisfying the snake identities
\[
(\id_V\otimes\operatorname{ev}_V)
\circ(\operatorname{coev}_V\otimes\id_V)=\id_V,
\]
\[
(\operatorname{ev}_V\otimes\id_{V^\vee})
\circ(\id_{V^\vee}\otimes\operatorname{coev}_V)=\id_{V^\vee}.
\]

Let $s_{V,V^\vee}:V\otimes V^\vee\to V^\vee\otimes V$ exchange the
factors.  For $T\in\End(V)$, define
\[
\operatorname{tr}(T)
=\operatorname{ev}_V\circ s_{V,V^\vee}
 \circ(T\otimes\id_{V^\vee})\circ\operatorname{coev}_V.
\]
The \emph{dimension} of $V$ is the scalar
\[
\dim_k(V)=\operatorname{tr}(\id_V)\in k.
\]

\begin{enumerate}[label=(\alph*)]
\item Let $c,d:k\to V\otimes V^\vee$ both satisfy the snake identities with
$\operatorname{ev}_V$.  Apply
\[
\id_V\otimes\operatorname{ev}_V\otimes\id_{V^\vee}
\]
to $d\otimes c:k\to V\otimes V^\vee\otimes V\otimes V^\vee$ and use the
snake identity for $d$ to identify the result with $c$.  Use the snake
identity for $c$ to identify the same result with $d$, and deduce $c=d$.
\item Prove that $T\mapsto\operatorname{tr}(T)$ is linear and that
\[
\operatorname{tr}(ST)=\operatorname{tr}(TS)
\qquad(S,T\in\End(V)).
\]
\end{enumerate}
\end{exercise}

\begin{exercise}[Exterior algebra, rank, and determinant]
For $m\geq0$, the $m$th \emph{exterior power} of a vector space $V$ is
\[
\Lambda^mV
=V^{\otimes m}/
\langle v_1\otimes\cdots\otimes v_m:
v_i=v_j\text{ for some }i\ne j\rangle,
\]
and the image of $v_1\otimes\cdots\otimes v_m$ is written
$v_1\wedge\cdots\wedge v_m$.  An endomorphism $T$ induces
\[
\Lambda^mT(v_1\wedge\cdots\wedge v_m)
=Tv_1\wedge\cdots\wedge Tv_m.
\]
The products
\[
\Lambda^rV\otimes\Lambda^sV\longrightarrow\Lambda^{r+s}V,
\qquad
(u_1\wedge\cdots\wedge u_r)\otimes
(v_1\wedge\cdots\wedge v_s)
\longmapsto
u_1\wedge\cdots\wedge u_r\wedge v_1\wedge\cdots\wedge v_s
\]
make $\Lambda^\bullet V=\bigoplus_{m\geq0}\Lambda^mV$ the
\emph{exterior algebra} of $V$.

For a finite-dimensional vector space $V$, an integer
$n\in\mathbb Z_{\geq0}$ satisfying
\[
\Lambda^nV\ne0,
\qquad
\Lambda^{n+1}V=0
\]
is its \emph{exterior rank}, written $\rank(V)=n$, when it exists and is
unique.  When $\rank(V)=n$, the \emph{determinant} of $T\in\End(V)$ is the
scalar $\det(T)$ satisfying
\[
\Lambda^nT=\det(T)\id_{\Lambda^nV}.
\]

Let $V$ be finite-dimensional.
\begin{enumerate}[label=(\alph*)]
\item Prove that the exterior algebra is graded-commutative:
\[
\xi\wedge\eta=(-1)^{rs}\eta\wedge\xi,
\qquad
\xi\in\Lambda^rV,\quad\eta\in\Lambda^sV.
\]
\item Prove that the exterior rank exists and is unique, that every
$\Lambda^mV$ is finite-dimensional, and that every endomorphism of
$\Lambda^nV$ is a unique scalar multiple of the identity.
\item Prove
\[
\dim_k(V)=n\,1_k,
\qquad
\dim_k(\Lambda^mV)=\binom{n}{m}1_k,
\qquad
\Lambda^mV=0\quad(m>n).
\]
Deduce that $\dim_k(V)$ determines $n$ when $\operatorname{char}k=0$, whereas
it determines only the residue class of $n$ modulo $p$ when
$\operatorname{char}k=p>0$.
\item Prove that the determinant is well-defined.  Use functoriality of the
highest exterior power to prove
\[
\det(ST)=\det(S)\det(T),
\qquad
T\text{ is invertible}\Longleftrightarrow\det(T)\ne0.
\]
\item For $T\in\End(V)$, extend scalars to $k[z]$ and prove
\[
\det(I+zT)=\sum_{m=0}^n
\operatorname{tr}(\Lambda^mT)z^m.
\]
Recover $(1+z)^n=\sum_m\binom nmz^m$ by taking $T=I$.
\end{enumerate}
\end{exercise}

\begin{exercise}[Rising factorials and the binomial series]
For $a\in\C$ and $m\in\Z_{\geq0}$, define
\[
(a)_0=1,
\qquad
(a)_m=a(a+1)\cdots(a+m-1)\quad(m\geq1).
\]
For $|z|<1$, define the \emph{binomial series}
\[
B_a(z)=\sum_{m=0}^{\infty}\frac{(a)_m}{m!}z^m.
\]

Prove that the binomial series converges for $|z|<1$ and satisfies
\[
(1-z)B_a'(z)=aB_a(z),
\qquad B_a(0)=1.
\]
For the branch normalized to equal $1$ at $z=0$, deduce
\[
B_a(z)=(1-z)^{-a}.
\]
For $a=-n$, $n\in\Z_{\geq0}$, prove that the series terminates and, after
replacing $z$ by $-z$, recover
\[
(1+z)^n=\sum_{m=0}^n\binom{n}{m}z^m.
\]
\end{exercise}

\begin{exercise}[Hilbert spaces and bounded operators]
An \emph{inner product} on a complex vector space is conjugate-linear in its
first argument, linear in its second, positive definite, and satisfies
$\langle x,y\rangle=\overline{\langle y,x\rangle}$.  Its norm is
$\|x\|=\sqrt{\langle x,x\rangle}$.  A \emph{Hilbert space} is an inner-product
space complete for this norm.

Vectors are \emph{orthogonal} when their inner product is zero.  For a
subspace $M\subseteq H$, define
\[
M^\perp=\{x:\langle m,x\rangle=0\text{ for every }m\in M\}.
\]

A linear map $T:H\to K$ between Hilbert spaces is \emph{bounded} when
\[
\|T\|=\sup_{\|x\|\leq1}\|Tx\|<\infty.
\]
Write $B(H,K)$ for the bounded operators and $B(H)=B(H,H)$.
An \emph{adjoint} of $T$ is an operator $T^*:K\to H$ satisfying
\[
\langle x,Ty\rangle=\langle T^*x,y\rangle.
\]
An \emph{orthogonal projection} is an operator $P$ satisfying
$P=P^*=P^2$.

\begin{enumerate}[label=(\alph*)]
\item Prove the Cauchy--Schwarz inequality, the projection theorem, and the Riesz
representation theorem.  Deduce the existence of adjoints and prove
\[
\|T\|^2=\|T^*T\|,\qquad
(\im T)^\perp=\ker T^*.
\]
\item For every closed subspace $M\subseteq H$, prove
$H=M\oplus M^\perp$ and construct the orthogonal projection $P_M:H\to M$.
\end{enumerate}
\end{exercise}

\begin{exercise}[\texorpdfstring{$L^p$}{Lp} geometry]
For a measure space $(X,\mathcal F,\mu)$ and $1\leq p<\infty$, define
\[
L^p(X,\mu)=
\left\{f:X\to\C\text{ measurable}:
\int_X|f|^p\,d\mu<\infty\right\}\big/\text{equality a.e.},
\qquad
\|f\|_p=\left(\int_X|f|^p\,d\mu\right)^{1/p}.
\]
Let $L^\infty(X,\mu)$ consist of essentially bounded measurable functions
modulo equality almost everywhere, with
\[
\|f\|_\infty=\operatorname*{ess\,sup}_{x\in X}|f(x)|.
\]

Prove H\"older's and Minkowski's inequalities and deduce that
$L^p(X,\mu)$ is complete for $1\leq p\leq\infty$.  For $1<p<\infty$ with
$p^{-1}+q^{-1}=1$, assume that $(X,\mathcal F,\mu)$ is $\sigma$-finite and
prove that
\[
L^q(X,\mu)\longrightarrow L^p(X,\mu)^*,
\qquad
g\longmapsto\left(f\mapsto\int_X\overline{g}f\,d\mu\right)
\]
is an isometric isomorphism.
\end{exercise}

\begin{exercise}[Gamma and beta integrals]
For $s,t>0$, define
\[
\Gamma(s)=\int_0^\infty x^{s-1}e^{-x}\,dx,
\qquad
B(s,t)=\int_0^1x^{s-1}(1-x)^{t-1}\,dx.
\]

Use Tonelli's theorem and the change of variables
$x=ru$, $y=r(1-u)$ to prove
\[
B(s,t)=\frac{\Gamma(s)\Gamma(t)}{\Gamma(s+t)}.
\]
Use integration by parts to prove
$\Gamma(s+1)=s\Gamma(s)$, and use the pushforward of Euclidean measure
under the norm map to obtain
\[
\Gamma\!\left(\frac12\right)=\sqrt\pi,
\qquad
\int_{\R^n}e^{-|x|^2/2}\,dx=(2\pi)^{n/2}.
\]
State the uses of Tonelli, Fubini, monotone convergence, and dominated
convergence in the derivation.
\end{exercise}



\begin{exercise}[$C^*$-algebras and spectra]
A \emph{$C^*$-algebra} is a complete normed complex algebra $A$ with a
conjugate-linear involution satisfying
\[
(ab)^*=b^*a^*,\qquad \|ab\|\leq\|a\|\|b\|,\qquad
\|a^*a\|=\|a\|^2.
\]
For a unital $C^*$-algebra, define
\[
\spec_A(a)=\{\lambda\in\C:a-\lambda1\text{ is not invertible}\}.
\]
An element is \emph{normal} when $a^*a=aa^*$.

Prove that $\spec_A(a)$ is nonempty and compact and that
\[
r(a)=\lim_{n\to\infty}\|a^n\|^{1/n}.
\]
For normal $a$, prove $r(a)=\|a\|$.
\end{exercise}

\begin{exercise}[Commutative Gelfand theory]
For a commutative $C^*$-algebra $A$, a \emph{character} is a nonzero
homomorphism $\chi:A\to\C$.  Let $\widehat A$ be the space of characters
with its weak-$*$ topology.  The \emph{Gelfand transform} is
\[
\Gamma:A\longrightarrow C_0(\widehat A),
\qquad \Gamma(a)(\chi)=\chi(a).
\]

Prove that every character of a commutative $C^*$-algebra is continuous
and that the Gelfand transform is an isometric $*$-isomorphism
$A\cong C_0(\widehat A)$.
\end{exercise}

\begin{exercise}[Continuous functional calculus]
An element $a$ of a $C^*$-algebra is \emph{self-adjoint} when $a=a^*$ and
\emph{positive}, written $a\geq0$, when $a=b^*b$ for some $b$.

For every normal element $a$ of a unital $C^*$-algebra, apply commutative
Gelfand theory to $C^*(1,a)$ and prove that there is a unique unital isometric
$*$-homomorphism
\[
C(\spec(a))\longrightarrow C^*(1,a),\qquad f\longmapsto f(a),
\]
sending the coordinate function to $a$, and prove the continuous spectral
mapping theorem.  Deduce that positive elements have unique positive square
roots and that $a\geq0$ if and only if $a=a^*$ and
$\spec_A(a)\subseteq[0,\infty)$.
\end{exercise}


\begin{exercise}[Positive functionals and scalar measures]
For a locally compact Hausdorff space $X$, a linear functional
$\varphi:C_c(X)\to\C$ is \emph{positive} when $\varphi(f)\geq0$ for every
real-valued $f\geq0$.

Let $X$ be locally compact Hausdorff.  Prove that every positive linear
functional $\varphi:C_c(X)\to\C$ that is bounded on functions supported in
each compact set has a unique regular Borel measure $\mu_\varphi$ satisfying
\[
\varphi(f)=\int_Xf\,d\mu_\varphi.
\]
\end{exercise}

\begin{exercise}[Complex Riesz--Markov representation]
Prove that the real and imaginary parts of a bounded functional on
$C_0(X)$ are differences of positive functionals.  Apply the positive
Riesz--Markov theorem to prove that every $\varphi\in C_0(X)^*$ has a
unique finite complex regular measure $\mu_\varphi$, and prove
$\|\varphi\|=|\mu_\varphi|(X)$.
\end{exercise}

\begin{exercise}[Spectral distributions of states]
For a unital $C^*$-algebra, a positive linear functional is one satisfying
$\varphi(a^*a)\geq0$ for every $a$, and a \emph{state} is a positive
functional with $\varphi(1)=1$.

Let $A$ be a unital $C^*$-algebra, let $a=a^*\in A$, and let $\varphi$ be a
state.  Apply the continuous functional calculus and Riesz--Markov theorem
to prove that there is a unique Borel probability measure
$\mu_{\varphi,a}$ on $\spec(a)$ such that
\[
\varphi(f(a))=\int_{\spec(a)}f(\lambda)\,d\mu_{\varphi,a}(\lambda),
\qquad f\in C(\spec(a)).
\]
Deduce
\[
\varphi(a)=\int\lambda\,d\mu_{\varphi,a}(\lambda),
\qquad
\varphi(a^2)-\varphi(a)^2
=\int(\lambda-\varphi(a))^2\,d\mu_{\varphi,a}(\lambda).
\]
\end{exercise}

\begin{exercise}[\texorpdfstring{$C_0(X)$}{C0(X)} representations]
A \emph{projection-valued measure} on a measurable space $X$ is a map $E$
from measurable subsets of $X$ to orthogonal projections on $H$ satisfying
\[
E(\varnothing)=0,\qquad E(X)=I,
\qquad
E\!\left(\bigsqcup_{j=1}^\infty B_j\right)
=\sum_{j=1}^\infty E(B_j)
\]
for pairwise disjoint measurable sets, where the sum converges in the strong
operator topology.  The spectral integral $\int_X f\,dE$ for bounded
measurable $f$ is defined by uniform approximation from simple functions.
It is regular when each scalar measure
$B\mapsto\langle x,E(B)y\rangle$ is a regular Borel measure.

Let $X$ be locally compact Hausdorff.  A representation
$\pi:C_0(X)\to B(H)$ is \emph{nondegenerate} when
$\overline{\pi(C_0(X))H}=H$.  Prove that every nondegenerate
$*$-representation determines scalar measures by applying the preceding
exercise to $f\mapsto\langle x,\pi(f)y\rangle$.  Use polarization and
multiplicativity to assemble them into a unique regular projection-valued
measure $E_\pi$ satisfying
\[
\pi(f)=\int_X f(x)\,dE_\pi(x),
\qquad f\in C_0(X).
\]
\end{exercise}

\begin{exercise}[Bounded normal spectral theorem]
For a normal operator $T\in B(H)$, apply the preceding representation
theorem to the continuous functional calculus
$C(\spec(T))\to B(H)$ and prove that there is a unique projection-valued
measure $E_T$ on $\spec(T)$ satisfying
\[
T=\int_{\spec(T)}z\,dE_T(z),
\qquad
f(T)=\int_{\spec(T)}f(z)\,dE_T(z).
\]
Deduce that a normal operator on a finite-dimensional Hilbert space has the
intrinsic spectral decomposition
\[
H=\bigoplus_{\lambda\in\spec(T)}E_T(\{\lambda\})H,
\qquad
T|_{E_T(\{\lambda\})H}=\lambda I.
\]
\end{exercise}

\begin{exercise}[Compact spectral theorem]
An operator $K\in B(H)$ is compact when it maps the closed unit ball to a
set with compact closure.

For a compact normal operator $K$, prove that every nonzero spectral value is an eigenvalue of
finite multiplicity, that the nonzero eigenvalues can accumulate only at
zero, and that
\[
K=\sum_{\lambda\in\spec(K)\setminus\{0\}}
\lambda E_K(\{\lambda\})
\]
in operator norm, and prove that the spectral subspaces in the displayed
sum are mutually orthogonal.
\end{exercise}

\begin{exercise}[Unbounded self-adjoint operators and the Cayley transform]
An unbounded operator is a linear map $A:\mathcal D(A)\to H$ on a dense
domain.  Its adjoint has domain
\[
\mathcal D(A^*)=
\{y\in H:x\mapsto\langle y,Ax\rangle
\text{ is bounded on }\mathcal D(A)\}.
\]
For $y\in\mathcal D(A^*)$, the vector $A^*y$ is determined by
$\langle A^*y,x\rangle=\langle y,Ax\rangle$ for
$x\in\mathcal D(A)$.  The operator is \emph{self-adjoint} when $A=A^*$,
including equality of domains.

For a self-adjoint operator $A$, prove that $A\pm iI$ are bijective from
$\mathcal D(A)$ to $H$ with bounded inverses and that the Cayley transform
\[
C_A=(A-iI)(A+iI)^{-1}
\]
is unitary with $\ker(I-C_A)=0$.  Prove conversely that every unitary $C$
with $\ker(I-C)=0$ determines a self-adjoint operator through the inverse
Cayley transform.
\end{exercise}

\begin{exercise}[Unbounded spectral theorem]
Apply the bounded normal spectral theorem to $C_A$ and transport its
projection-valued measure by the inverse Cayley map.  Prove that there is a unique
projection-valued measure $E_A$ on $\R$ satisfying
\[
A=\int_\R\lambda\,dE_A(\lambda),\qquad
\mathcal D(A)=
\left\{x:\int_\R\lambda^2\,d\langle x,E_A(\lambda)x\rangle<\infty\right\}.
\]
\end{exercise}

\begin{exercise}[Stone's theorem]
Use the measurable functional calculus of $E_A$ to prove that $e^{-itA}$
is a strongly continuous unitary group with generator $-iA$.  Conversely,
let $(U_t)_{t\in\R}$ be a strongly continuous unitary group and define
\[
\mathcal D(A)=\left\{x:\lim_{t\to0}
i\frac{U_tx-x}{t}\text{ exists}\right\},
\qquad
Ax=\lim_{t\to0}i\frac{U_tx-x}{t}.
\]
Prove that $\mathcal D(A)$ is dense and that $A$ is symmetric.  Use the
strong integrals
\[
i\int_0^\infty e^{-t}U_t\,dt,
\qquad
-i\int_{-\infty}^0e^tU_t\,dt
\]
to prove that $A-iI$ and $A+iI$ are surjective.  Deduce that $A$ is
self-adjoint, prove $U_t=e^{-itA}$, and prove uniqueness of $A$.
\end{exercise}

\begin{exercise}[Spring and string frequencies]
Let $V$ be a finite-dimensional real vector space with positive-definite
mass form $m$ and nonnegative stiffness form $k$.  Use $m$ to define the
self-adjoint operator $A$ by $k(x,y)=m(x,Ay)$.  Prove that solutions of
$\ddot x+Ax=0$ decompose intrinsically through the spectral projections of
$A$ and that the observed angular frequencies are $\sqrt{\lambda}$ for
$\lambda\in\spec(A)\setminus\{0\}$.

For a stretched string of length $L$, wave speed $c$, and fixed endpoints,
let $D$ be the self-adjoint operator on $L^2((0,L);\C)$ given by
\[
D\phi=-c^2\phi'',
\]
on the domain of functions $\phi$ for which $\phi$ and $\phi'$ are
absolutely continuous on $[0,L]$, $\phi''\in L^2((0,L))$, and
$\phi(0)=\phi(L)=0$.  For the wave equation
\[
\partial_t^2u+Du=0,
\]
prove that a nonzero time-harmonic solution
$u(t,x)=\operatorname{Re}(e^{i\omega t}\phi(x))$, with $\omega>0$,
exists exactly when, for some $n\geq1$,
\[
\omega=\omega_n=\frac{n\pi c}{L},
\qquad
\phi\in(\C\phi_n)\setminus\{0\},
\qquad
\phi_n(x)=\sqrt{\frac2L}\sin\!\left(\frac{n\pi x}{L}\right),
\]
Deduce the harmonic resonance ratios and the nodal locations measured by
driving the string at each $\omega_n$.
\end{exercise}

\begin{exercise}[Spectral lines and time evolution]
\label{ex:spectral-lines}
Let $A$ be self-adjoint, let $B\in B(H)$, and set
$B(t)=e^{itA/\hbar}Be^{-itA/\hbar}$.  For eigenvalues $\lambda,\mu$ of
$A$, write $P_\lambda=E_A(\{\lambda\})$ and prove
\[
P_\mu B(t)P_\lambda
=e^{it(\mu-\lambda)/\hbar}P_\mu BP_\lambda.
\]
Use the functional calculus to prove the factorized weak identity
\[
B(t)=
\left(\int_{\R}e^{it\mu/\hbar}\,dE_A(\mu)\right)
B
\left(\int_{\R}e^{-it\lambda/\hbar}\,dE_A(\lambda)\right).
\]
When $A$ has pure-point spectrum, deduce in the weak operator topology that
\[
B(t)=\sum_{\lambda,\mu}
e^{it(\mu-\lambda)/\hbar}P_\mu BP_\lambda.
\]
For an initial state $\psi$, define the line strength
\[
I_{\mu\lambda}(\psi)=\|P_\mu BP_\lambda\psi\|^2.
\]
Prove that a line at $(\mu-\lambda)/\hbar$ occurs only when this quantity is
nonzero, and identify vanishing spectral blocks as selection rules.
\end{exercise}

\begin{exercise}[Hermite polynomials and functions]
Define the physicists' Hermite polynomials by
\[
e^{2xt-t^2}=\sum_{n=0}^{\infty}H_n(x)\frac{t^n}{n!}.
\]
The normalized \emph{Hermite functions} are
\[
h_n(x)=\frac{e^{-x^2/2}H_n(x)}
{(2^nn!\sqrt\pi)^{1/2}}.
\]

Use the generating function and the Gaussian integral to prove that
$(h_n)_{n\geq0}$ is an orthonormal family in $L^2(\R)$.  Derive the
Hermite differential equation and recurrence relations.
\end{exercise}

\begin{exercise}[Mehler kernel]
For $|r|<1$, use the Hermite generating function and Gaussian integration
to derive Mehler's identity
\[
\sum_{n=0}^{\infty}r^nh_n(x)h_n(y)
=\frac1{\sqrt{\pi(1-r^2)}}
\exp\!\left(
-\frac{(1+r^2)(x^2+y^2)-4rxy}{2(1-r^2)}
\right).
\]
\end{exercise}

\begin{exercise}[Hermite completeness]
Let $K_r$ be the integral operator with the Mehler kernel for $0<r<1$.
Use its explicit Gaussian form to prove
\[
K_rf\longrightarrow f\qquad\text{in }L^2(\R)\quad(r\uparrow1).
\]
Prove that the image of $K_r$ lies in the closed linear span of the Hermite
functions and deduce that $(h_n)_{n\geq0}$ is complete in $L^2(\R)$.
\end{exercise}

\begin{exercise}[Oscillator spectroscopy]
The \emph{Schwartz space} $\mathcal S(\R)$ consists of the smooth functions
$\psi$ satisfying
\[
\sup_{x\in\R}(1+|x|)^m|\psi^{(r)}(x)|<\infty
\]
for all $m,r\geq0$.  On $\mathcal S(\R)$ let
\[
(Q\psi)(x)=x\psi(x),
\qquad
(P\psi)(x)=-i\hbar\psi'(x).
\]
For
\[
\widehat H=-\frac{\hbar^2}{2M}\frac{d^2}{dx^2}
+\frac{M\omega^2x^2}{2},
\]
use the Hermite functions to prove that
\[
\psi_n(x)=\left(\frac{M\omega}{\hbar}\right)^{1/4}
h_n\!\left(\sqrt{\frac{M\omega}{\hbar}}x\right),
\qquad
E_n=\hbar\omega\left(n+\frac12\right)
\]
give the complete spectral decomposition of $\widehat H$.  Construct the
creation and annihilation operators and compute
\[
\langle\psi_k,Q\psi_n\rangle
=\sqrt{\frac{\hbar}{2M\omega}}
\bigl(\sqrt{n+1}\,\delta_{k,n+1}
+\sqrt n\,\delta_{k,n-1}\bigr).
\]
Deduce the spectral-line prediction
$|E_{n\pm1}-E_n|/\hbar=\omega$ and the relative line strengths for a
linearly coupled oscillator.  For $\alpha\in\C$, prove convergence and
normalization of
\[
\psi_\alpha=e^{-|\alpha|^2/2}
\sum_{n\geq0}\frac{\alpha^n}{\sqrt{n!}}\psi_n,
\]
show that its number statistics are Poisson with mean $|\alpha|^2$, and
derive its sinusoidal position expectation.  Analytically continue
Mehler's identity with the usual $t-i0$ prescription to obtain the
oscillator propagator for $t\notin(\pi/\omega)\Z$,
\[
K_t(x,y)=
\sqrt{\frac{M\omega}{2\pi i\hbar\sin(\omega t)}}
\exp\!\left[
\frac{iM\omega}{2\hbar\sin(\omega t)}
\bigl((x^2+y^2)\cos(\omega t)-2xy\bigr)
\right].
\]
Determine the Maslov phase obtained by continuation across each caustic.
Prove
\[
e^{-2\pi i\widehat H/(\hbar\omega)}=-I,
\qquad
e^{-4\pi i\widehat H/(\hbar\omega)}=I,
\]
and distinguish the projective and observable revival period
$2\pi/\omega$ from the exact operator period $4\pi/\omega$.
Use these formulas to predict the spectral spacing, coherent oscillation
frequency, occupation-number distribution, and revival period measured in
an oscillator experiment.
\end{exercise}

\begin{exercise}[Airy functions and quantum-bouncer spectroscopy]
The Airy function $\operatorname{Ai}$ is the unique real solution of
\[
\operatorname{Ai}''(x)=x\operatorname{Ai}(x)
\]
normalized by
\[
\operatorname{Ai}(x)\sim
\frac{x^{-1/4}}{2\sqrt\pi}e^{-2x^{3/2}/3}
\qquad(x\to+\infty).
\]
Let $a_n>0$ be determined by
$\operatorname{Ai}(-a_n)=0$, ordered increasingly.

On $L^2((0,\infty),dz)$ consider the self-adjoint Dirichlet realization
\[
\widehat H=-\frac{\hbar^2}{2M}\frac{d^2}{dz^2}+Mgz,
\qquad \psi(0)=0.
\]
With
\[
\ell_g=\left(\frac{\hbar^2}{2M^2g}\right)^{1/3},
\]
derive
\[
E_n=Mg\ell_g a_n,
\qquad
\psi_n(z)=C_n\operatorname{Ai}\!\left(\frac z{\ell_g}-a_n\right).
\]
Use the spectral projections of $\widehat H$ to prove that a periodic
perturbation $B\cos(\Omega t)$ can produce a transition from the $n$th to
the $k$th level only through the block
$E_{\widehat H}(\{E_k\})BE_{\widehat H}(\{E_n\})$, at the resonant
frequency
\[
\Omega_{kn}=\frac{Mg\ell_g}{\hbar}|a_k-a_n|.
\]
Deduce the classical turning heights $z_n=\ell_g a_n$.  Model an absorber
with lower face at height $h$, width $w$, and loss strength $\Gamma>0$ by

\gamma_n(h)=\frac{2\Gamma}{\hbar}
\int_h^{h+w}|\psi_n(z)|^2\,dz
\]
and the transmission factor $e^{-T\gamma_n(h)}$ after flight time $T$.
Use the Airy asymptotic to prove that the onset occurs across a height

\end{exercise}

\section{Quantization}

\begin{exercise}[Differential forms and Cartan calculus]
A \emph{smooth manifold} of dimension $n$ is a Hausdorff, second-countable
space with a maximal atlas of homeomorphisms to open subsets of $\R^n$
whose transition maps are smooth.  A map of smooth manifolds is
\emph{smooth} when its chart representatives are smooth; it is a
\emph{diffeomorphism} when it has a smooth inverse.

The \emph{tangent space} $T_pM$ is the vector space of derivations
$v:C^\infty(M)\to\R$ at $p$, satisfying
\[
v(fg)=f(p)v(g)+g(p)v(f).
\]
The differential of a smooth map $F:M\to N$ is
\[
dF_p:T_pM\longrightarrow T_{F(p)}N,
\qquad
dF_p(v)(f)=v(f\circ F).
\]
Define
\[
TM=\coprod_{p\in M}T_pM.
\]
A \emph{smooth vector bundle} is a smooth map $\pi:E\to M$ locally
isomorphic to $U\times W\to U$ for a finite-dimensional real vector space
$W$, by fiberwise linear maps.  A section is a smooth map $s:M\to E$
satisfying $\pi s=\id_M$; write $\Gamma(E)$ for the space of sections.
The bundle $TM$ has the bundle structure induced by the differentials of
chart maps, and a \emph{vector field} is a section of $TM$.

The cotangent space at $p$ is
$T_p^*M=(T_pM)^*$, and
$T^*M=\coprod_{p\in M}T_p^*M$ has the dual vector-bundle structure.

A \emph{differential $k$-form} on $M$ is a smooth section of
$\Lambda^kT^*M$; write $\Omega^k(M)$ for the space of differential
$k$-forms.  The exterior
products on the cotangent spaces define
\[
\wedge:\Omega^r(M)\times\Omega^s(M)\longrightarrow\Omega^{r+s}(M).
\]
For a smooth map $F:M\to N$, define
\[
(F^*\alpha)_p(v_1,\ldots,v_k)
=\alpha_{F(p)}(dF_pv_1,\ldots,dF_pv_k).
\]
For a vector field $X$, define contraction by
\[
(\iota_X\alpha)_p(v_2,\ldots,v_k)
=\alpha_p(X_p,v_2,\ldots,v_k).
\]
The \emph{exterior derivative} is the family
$d:\Omega^k(M)\to\Omega^{k+1}(M)$ satisfying $df(X)=Xf$, the graded
Leibniz rule, $d^2=0$, and $F^*d=dF^*$.
A \emph{local flow} of $X$ is a smooth local family $\Phi_t$ satisfying
$\Phi_0=\id$, $\Phi_{t+s}=\Phi_t\Phi_s$ whenever both sides are defined,
and $\left.\frac d{dt}\right|_{t=0}\Phi_t(p)=X_p$.  Define
\[
\mathcal L_X\alpha
=\left.\frac d{dt}\right|_{t=0}\Phi_t^*\alpha.
\]

Verify that the differentials of chart maps give $TM$ and $T^*M$ the
stated smooth vector-bundle structures and that
$d(G\circ F)=dG\circ dF$.  Construct $d$ locally and prove its uniqueness.
For contraction
$\iota_X$, prove Cartan's identity
\[
\mathcal L_X=d\iota_X+\iota_Xd.
\]
For $A\in\Omega^1(M)$, put $F=dA$.  Prove that replacing $A$ by
$A+d\chi$, with $\chi\in C^\infty(M)$, leaves $F$ unchanged.  Use
Cartan's identity to derive
\[
\mathcal L_XA=\iota_XF+d(\iota_XA).
\]
\end{exercise}

\begin{exercise}[Local Poincar\'e lemma]
Let $U$ be a star-shaped open neighborhood of $0$ in a finite-dimensional
real vector space.  For $\alpha\in\Omega^k(U)$ with $k>0$, define
\[
(K\alpha)_x(v_1,\ldots,v_{k-1})
=\int_0^1t^{k-1}\alpha_{tx}(x,v_1,\ldots,v_{k-1})\,dt.
\]
Use Cartan's identity for the radial flow to prove
\[
dK+Kd=\id
\]
on positive-degree forms.  Deduce that every closed form of positive degree
on $U$ is exact, with a primitive vanishing at the origin.
\end{exercise}

\begin{exercise}[Integration of forms and Stokes' theorem]
An \emph{orientation} of an $n$-manifold is a continuous choice of component
of the nonzero elements in $\Lambda^nT_p^*M$. The integral $\int_M\omega$ of a
compactly supported top form is defined using orientation-preserving charts
and a partition of unity.

Prove that the integral is independent of charts and partition of unity.
For an oriented manifold with boundary, prove
\[
\int_Md\omega=\int_{\partial M}\omega.
\]
Deduce the fundamental theorem of calculus, Green's theorem, the divergence
theorem, and the classical curl theorem. For time-dependent magnetic
two-form $B$ and electric one-form $E$, prove the equivalence of
\[
\int_{\partial S}E=-\frac d{dt}\int_SB
\qquad\text{and}\qquad
dE=-\partial_tB,
\]
recovering the observed flux--emf law.
\end{exercise}

\begin{exercise}[de Rham cohomology]
A differential form $\omega$ is \emph{closed} if $d\omega=0$ and
\emph{exact} if $\omega=d\eta$ for some form $\eta$. Since $d^2=0$, the
\emph{de Rham cohomology} of $M$ is
\[
H_{\mathrm{dR}}^k(M)
=\frac{\ker(d:\Omega^k(M)\to\Omega^{k+1}(M))}
{\im(d:\Omega^{k-1}(M)\to\Omega^k(M))}.
\]
Pullback induces maps on de Rham cohomology, and the wedge product makes
$H_{\mathrm{dR}}^*(M)$ a graded-commutative algebra.

Use the local Poincar\'e lemma and Stokes' theorem
to prove that an exact $k$-form has zero integral over
every closed oriented $k$-manifold mapped smoothly into $M$. Prove
$H_{\mathrm{dR}}^1(S^1)\cong\R$, with the isomorphism given by integration.

On $\R^2\setminus\{0\}$, show that
\[
\alpha=\frac{-y\,dx+x\,dy}{x^2+y^2}
\]
is closed and has integral $2\pi$ around the unit circle, hence is not exact.
Compute $H_{\mathrm{dR}}^1(\R^2\setminus\{0\})=\R\cdot[\alpha]$. Deduce that a
closed electromagnetic field strength has local potentials, while it has a
global potential $F=dA$ exactly when $[F]=0$ in de Rham cohomology.
\end{exercise}

\begin{exercise}[Cotangent phase space]
For the projection $\pi:T^*Q\to Q$, define the tautological one-form by
\[
\theta_\alpha(v)=\alpha(d\pi(v)),
\qquad \alpha\in T^*Q,\quad v\in T_\alpha(T^*Q).
\]
The canonical two-form on $T^*Q$ is $\omega=d\theta$.

For a finite-dimensional real vector space $V$, construct the canonical
identifications
\[
TV\cong V\times V,
\qquad
T^*V\cong V\times V^*.
\]
Prove that a phase $S\in C^\infty(Q)$ determines the momentum
$p=dS_q\in T_q^*Q$.  For a diffeomorphism
$F:Q\to Q$, construct its cotangent lift and prove that it preserves
$\theta$ and $d\theta$.  Under $T^*V\cong V\times V^*$, prove
\[
\omega\bigl((q,p),(q',p')\bigr)=p(q')-p'(q).
\]
\end{exercise}

\begin{exercise}[Symplectic manifolds and Hamiltonian fields]
A \emph{symplectic form} on a smooth manifold $M$ is a closed
nondegenerate two-form $\omega$.  A \emph{symplectic vector space} is a
finite-dimensional real vector space equipped with a nondegenerate
alternating bilinear form; it carries the corresponding constant
symplectic form.  For
$f\in C^\infty(M)$, define
\[
\iota_{X_f}\omega=-df,
\qquad
\{f,g\}=-\omega(X_f,X_g).
\]
The local flow of $X_f$ is the \emph{Hamiltonian flow} of $f$.
A diffeomorphism $F:(M,\omega_M)\to(N,\omega_N)$ is a
\emph{canonical transformation} when $F^*\omega_N=\omega_M$.

Prove that the canonical two-form $d\theta$ on $T^*Q$ is nondegenerate and
hence symplectic.
Prove that the Poisson bracket is antisymmetric,
satisfies the Leibniz rule and Jacobi identity, and is preserved by
canonical transformations.  For the Hamiltonian flow $\Phi_t$ of $H$,
prove
\[
\frac{d}{dt}f(\Phi_t(x))=\{f,H\}(\Phi_t(x)).
\]
\end{exercise}

\begin{exercise}[Liouville's theorem]
\label{ex:liouville-theorem}
On a $2n$-dimensional symplectic manifold, prove that Hamiltonian flow
preserves $\omega$, the volume form $\omega^n/n!$, and every density
depending only on the Hamiltonian.  On a symplectic vector space, prove
that the infinitesimal generator of a linear Hamiltonian flow has trace
zero and that the flow has determinant one.
\end{exercise}

\begin{exercise}[Moment maps and Noether charges]
A \emph{Lie group} is a topological group with a smooth manifold structure
for which multiplication and inversion are smooth.  Its Lie algebra
$\mathfrak g$ is $T_eG$ with the bracket induced by left-invariant vector
fields.  For a smooth left action of $G$ on $M$ and $\xi\in\mathfrak g$,
define the fundamental vector field
\[
\xi_M(x)=d(g\mapsto gx)_e(\xi).
\]
Define
\[
\Ad_g=d(h\mapsto ghg^{-1})_e,
\qquad
(\Ad_g^*\ell)(\xi)=\ell(\Ad_{g^{-1}}\xi).
\]

An action of $G$ on $(M,\omega)$ is \emph{Hamiltonian} when it preserves
$\omega$ and has an equivariant map $\mu:M\to\mathfrak g^*$ satisfying
\[
\iota_{\xi_M}\omega=-d\mu^\xi,
\qquad
\mu^\xi(x)=\langle\mu(x),\xi\rangle,
\qquad
\mu(gx)=\Ad_g^*\mu(x).
\]
The map $\mu$ is the \emph{moment map}.

Prove
\[
X_{\mu^\xi}=\xi_M,
\qquad
\{\mu^\xi,\mu^\eta\}=\mu^{[\xi,\eta]}.
\]
If a Hamiltonian $H$ is $G$-invariant, prove Noether's conservation law
\[
\frac d{dt}\mu^\xi(\Phi_t(x))=0
\]
for its Hamiltonian flow $\Phi_t$.

Let $G$ act on a manifold $Q$ and lift the action to $T^*Q$.  Prove that
the lifted action preserves the tautological form and is Hamiltonian with
\[
\langle\mu(\alpha_q),\xi\rangle
=\alpha_q(\xi_Q(q)).
\]
For the distinguished Euclidean configuration space $Q=\R^3$, identify
the moment maps for translations and rotations with linear momentum $p$
and angular momentum $q\times p$.  Deduce conservation of $p$ for a
translation-invariant Hamiltonian and conservation of $q\times p$ for
\[
H(q,p)=\frac{\lVert p\rVert^2}{2m}+V(\lVert q\rVert).
\]
\end{exercise}

\begin{exercise}[Symplectic reduction]
Let a compact Lie group $G$ act on $(M,\omega)$ with equivariant moment
map $\mu:M\to\mathfrak g^*$.  Suppose $0$ is a regular value, meaning
$d\mu_x$ is surjective for every $x\in\mu^{-1}(0)$, and suppose $G$ acts
freely on $\mu^{-1}(0)$.  Use as theorem input that
\[
M/\!\!/G=\mu^{-1}(0)/G
\]
is a smooth manifold and the quotient map
$\pi:\mu^{-1}(0)\to M/\!\!/G$ is a submersion.

For the inclusion $i:\mu^{-1}(0)\hookrightarrow M$, prove that
$i^*\omega$ is invariant and vanishes on the tangent directions to the
$G$-orbits.  Prove that there is a unique symplectic form $\omega_{\rm red}$
such that
\[
\pi^*\omega_{\rm red}=i^*\omega,
\qquad
\dim(M/\!\!/G)=\dim M-2\dim G.
\]
If $H$ is $G$-invariant, prove that its restriction descends to
$H_{\rm red}$ and that the Hamiltonian flow of $H$ on $\mu^{-1}(0)$
projects to the Hamiltonian flow of $H_{\rm red}$.  Write the reduced
equations governing motion at the imposed value of the conserved Noether
charges.
\end{exercise}

\begin{exercise}[Coadjoint orbits and the Kirillov form]
For $\ell\in\mathfrak g^*$, define
\[
\mathcal O_\ell=\Ad^*(G)\ell,
\qquad
G_\ell=\{g\in G:\Ad_g^*\ell=\ell\},
\]
and
\[
\mathfrak g_\ell
=\{X\in\mathfrak g:\ell([X,Y])=0
\text{ for every }Y\in\mathfrak g\}.
\]
For $X\in\mathfrak g$, set
\[
(\ad_X^*\ell)(Y)=-\ell([X,Y]).
\]

On $T_\ell\mathcal O_\ell$, define
\[
\omega_\ell(\ad_X^*\ell,\ad_Y^*\ell)=\ell([X,Y]).
\]
For $\eta=\Ad_g^*\ell$, transport $\omega_\ell$ by the coadjoint
action to define $\omega_\eta$.

Prove
\[
T_\ell\mathcal O_\ell
=\{\ad_X^*\ell:X\in\mathfrak g\}
\cong\mathfrak g/\mathfrak g_\ell.
\]
Prove that the Kirillov form is well-defined, alternating, nondegenerate,
and $G$-invariant.  Prove the cyclic identity
\[
\ell([X,[Y,Z]])+\ell([Y,[Z,X]])+\ell([Z,[X,Y]])=0
\]
and prove directly that it is the cyclic identity satisfied by the
transported field $\omega$ on $\mathcal O_\ell$.
Prove that
\[
\mu_{\mathcal O}:\mathcal O_\ell\longrightarrow\mathfrak g^*,
\qquad
\mu_{\mathcal O}(\eta)=-\eta,
\]
is an equivariant moment map with the sign conventions above.

For the distinguished group
\[
SO(3)=\{R\in\operatorname{End}(\R^3):R^*R=\id,\ \det R=1\},
\]
use the Euclidean inner product and
orientation to identify $\mathfrak{so}(3)$ and its dual with $\R^3$, with
bracket $[X,Y]=X\times Y$.  Prove that $\mathcal O_\ell$ is the sphere of
radius $L=\lVert\ell\rVert$, that
\[
\omega_\ell(X\times\ell,Y\times\ell)
=\ell\mathbin{\cdot}(X\times Y),
\]
and that the Kirillov form is $L^{-1}$ times the oriented Euclidean area
form on
this sphere.  Deduce

\int_{\mathcal O_\ell}\omega=4\pi L.
\]
\end{exercise}

\begin{exercise}[Contact manifolds and symplectization]
Let $Y$ be a smooth manifold of dimension $2n+1$.  A one-form $\alpha$ is
a \emph{contact form} when
\[
\alpha\wedge(d\alpha)^n
\]
is nowhere zero.  Its kernel $\xi=\ker\alpha$ is the associated cooriented
\emph{contact structure}; contact forms defining the same cooriented
contact structure differ by multiplication by a positive smooth function.  The
\emph{Reeb vector field} $R_\alpha$ is defined by
\[
\alpha(R_\alpha)=1,
\qquad
\iota_{R_\alpha}d\alpha=0.
\]
An immersed $n$-manifold $\iota:L\to Y$ is \emph{Legendrian} when
$\iota^*\alpha=0$.

Let $(M,\omega=d\lambda)$ be an exact symplectic manifold.  On
$M\times\R$, with coordinate function $u$ on the second factor, prove that
\[
\alpha=du-\lambda
\]
is a contact form and that $R_\alpha$ is the infinitesimal generator of
translation in the $\R$ factor.  For a manifold $X$, identify
$J^1X=T^*X\times\R$ with this contactification and prove that
\[
j^1f:X\longrightarrow J^1X,
\qquad
j^1f(x)=(df_x,f(x)),
\]
is Legendrian for every $f\in C^\infty(X)$.  Prove that every section of
$J^1X\to X$ with Legendrian image is of this form.

For a contact manifold $(Y,\alpha)$ and the positive coordinate $r$ on
$\R_{>0}$, prove that
\[
\Omega=d(r\alpha)
\]
is symplectic on $\R_{>0}\times Y$.  Determine the vector field $Z$
satisfying $\iota_Z\Omega=r\alpha$ and prove that multiplication of
$\alpha$ by a positive function gives a symplectomorphic symplectization.
\end{exercise}


\begin{exercise}[Connections, curvature, and holonomy]
A \emph{connection} on a vector bundle $E\to M$ is an $\R$-bilinear map
$\nabla:\Gamma(TM)\times\Gamma(E)\to\Gamma(E)$, linear over smooth
functions in the first variable and satisfying
$\nabla_X(fs)=X(f)s+f\nabla_Xs$. Along a curve $\gamma$ it induces a
covariant derivative $D/dt$; its solutions $Ds/dt=0$ define
\emph{parallel transport} between endpoint fibers.

The \emph{curvature} of $\nabla$ is the $\operatorname{End}(E)$-valued
two-form
\[
R^\nabla(X,Y)s
=\nabla_X\nabla_Ys-\nabla_Y\nabla_Xs-\nabla_{[X,Y]}s.
\]
The \emph{holonomy group} at $p$ consists of the automorphisms of $E_p$
obtained by parallel transport around loops based at $p$.

Prove existence, uniqueness, composition, and inverse properties of parallel
transport. Prove that curvature is tensorial and gives the leading failure
of transport around an infinitesimal parallelogram to return a vector to
itself. Deduce that a flat connection has path-independent transport on every
simply connected coordinate domain.
\end{exercise}

\begin{exercise}[Principal connections and gauge covariance]
A \emph{principal $G$-bundle} is a smooth map
$P\to M$ with a free right $G$-action, transitive on each fiber, and local
$G$-equivariant trivializations.  For a homomorphism
$\rho:G\to\operatorname{GL}(W)$, its associated vector bundle is
$P\times_GW=(P\times W)/((pg,w)\sim(p,\rho(g)w))$.
A \emph{principal connection} is an
equivariant $\mathfrak g$-valued one-form $A$ on $P$ reproducing fundamental
vertical fields. It induces connections on every associated bundle, and its
\emph{curvature} is
$F_A=dA+\frac12[A\wedge A]$. A \emph{gauge transformation} is a
$G$-equivariant bundle automorphism covering $\id_M$.

Prove $d_AF_A=0$.  For a gauge transformation $\Phi$, prove
$F_{\Phi^*A}=\Phi^*F_A$ and that holonomy changes by conjugacy.  Prove that
a flat connection has trivial holonomy on contractible loops and is
gauge-equivalent to a product connection on a connected base exactly when
its holonomy homomorphism is trivial.  For $G=U(1)$, compute the holonomy
around a
generator of the fundamental group and exhibit the obstruction to removing
a flat connection globally.
\end{exercise}



\begin{exercise}[Integral curvature and circle cocycles]
A \emph{good cover} of a manifold is an open cover
$\{U_i\}_{i\in I}$ for which every nonempty finite intersection is
contractible.  A \emph{smooth complex vector bundle} is a smooth vector
bundle locally modeled on a finite-dimensional complex vector space by
fiberwise complex-linear product maps; it is a \emph{complex line bundle}
when its complex rank is one.  A \emph{Hermitian vector bundle} carries a
smoothly varying positive-definite Hermitian form, and a connection on it
is \emph{unitary} when it preserves that form.  Let
$F\in\Omega^2(M;i\R)$ be closed and choose, on a good cover,
\[
dA_i=F|_{U_i}.
\]
On $U_i\cap U_j$, choose $f_{ij}:U_i\cap U_j\to i\R$ with
$A_j-A_i=df_{ij}$ and $f_{ji}=-f_{ij}$.  Prove that
\[
c_{ijk}=f_{ij}+f_{jk}+f_{ki}
\]
is constant on every nonempty triple intersection.

Use the following good-cover form of the integral de Rham theorem as
theorem input:
\[
\frac{1}{2\pi i}\int_\Sigma h^*F\in\Z
\quad\text{for every smooth }h:\Sigma\to M
\]
from a closed oriented surface if and only if the $f_{ij}$ may be chosen
so that $c_{ijk}\in2\pi i\Z$.  Under this condition put
\[
g_{ij}=e^{f_{ij}}.
\]
Prove the cocycle identities for the $g_{ij}$, construct the associated
Hermitian line bundle, and show that the local operators $d+A_i$ define a
unitary connection of curvature $F$.  Conversely, compare local unitary
frames of a Hermitian line bundle with connection and recover the integral
period condition from its transition functions.
\end{exercise}

\begin{exercise}[Prequantum line bundles]
A \emph{prequantum line bundle} over $(M,\omega)$ is a Hermitian line
bundle $L\to M$ with a unitary connection $\nabla$ satisfying
\[
F_\nabla=-\frac{i}{\hbar}\omega.
\]
Prove that a prequantum line bundle exists if and only if
\[
\frac1{2\pi\hbar}\int_\Sigma f^*\omega\in\mathbb Z
\]
for every smooth map $f:\Sigma\to M$ from a closed oriented surface.
Apply the circle-cocycle construction above to $F=-i\omega/\hbar$,
identify its local connection forms with symplectic potentials, and prove
that
\[
A_i\longmapsto A_i+d\phi_i,
\qquad
f_{ij}\longmapsto f_{ij}+\phi_j-\phi_i
\]
produces an isomorphic prequantum line bundle with connection.  Show that
two prequantum line bundles with the same curvature differ by a flat
Hermitian line bundle, so curvature alone need not determine the
prequantization up to isomorphism.
\end{exercise}

\begin{exercise}[Deformation quantization]
A \emph{Poisson manifold} is a smooth manifold $M$ with a bracket on
$C^\infty(M)$ that is a Lie bracket and a derivation in each argument.
Equivalently, it has a bivector
$\Pi\in\Gamma(\Lambda^2TM)$.  Define the Schouten bracket to be the
unique graded Lie bracket of degree $-1$ on multivector fields extending
the vector-field bracket
and satisfying
\[
[A,B\wedge C]_{\rm Sch}
=[A,B]_{\rm Sch}\wedge C
+(-1)^{(p-1)q}B\wedge[A,C]_{\rm Sch}
\]
when $A$ has degree $p$ and $B$ has degree $q$.  Prove that the bivector
defines a Poisson bracket exactly when
\[
\{f,g\}=\Pi(df,dg),
\qquad [\Pi,\Pi]_{\rm Sch}=0.
\]

A \emph{differential star product} on a Poisson manifold is an associative
$\mathbb C[[\hbar]]$-bilinear product on
$C^\infty(M)[[\hbar]]$ of the form
\[
f\star g=fg+\sum_{r\geq1}\hbar^rC_r(f,g),
\]
where the $C_r$ are bidifferential operators, $1$ is a unit, and
\[
C_1(f,g)-C_1(g,f)=i\{f,g\}.
\]
Let $(E,\omega)$ be a symplectic vector space, let
$\Pi:E^*\to E$ be the inverse of $v\mapsto\omega(v,-)$, and write
$\Pi(k,\ell)=\ell(\Pi(k))$.  Let $D_\Pi$ be the constant bidifferential
operator characterized on decomposable functions by
\[
D_\Pi(f\otimes g)(x,y)=d_yg\bigl(\Pi(d_xf)\bigr).
\]
Define the Moyal product on $C^\infty(E)[[\hbar]]$ by
\[
f\star g
=m\circ\exp\!\left(\frac{i\hbar}{2}D_\Pi\right)(f\otimes g),
\]
where $m$ restricts a function on $E\times E$ to the diagonal.  Prove
associativity from the commutation of the induced constant
bidifferential operators on $E^3$, and prove
\[
f\star g=fg+\frac{i\hbar}{2}\{f,g\}+O(\hbar^2).
\]

For $k,\ell\in E^*$ put $e_k(x)=e^{ik(x)}$ and derive the Weyl relation
\[
e_k\star e_\ell
=e^{-i\hbar\Pi(k,\ell)/2}e_{k+\ell}.
\]
Check its multiplier identity directly, differentiate it to recover the
canonical commutation relations, and prove that every affine symplectic
transformation of $E$ preserves $\star$.  Call this intrinsic formal
algebra the \emph{Moyal algebra} of the symplectic vector space.
\end{exercise}

\begin{exercise}[Coadjoint-orbit spin quantization]
A \emph{unitary representation} of a topological group $G$ on a Hilbert
space $H$ is a strongly continuous homomorphism $G\to U(H)$.  Define
\[
SU(2)=\{U\in\operatorname{End}(\C^2):U^*U=\id,\ \det U=1\}.
\]
For a complex vector space $V$, set $\operatorname{Sym}^0(V)=\C$ and, for
$N\geq1$, define its $N$th \emph{symmetric power} by
\[
\operatorname{Sym}^N(V)
=V^{\otimes N}/
\langle w-\sigma w:w\in V^{\otimes N},\ \sigma\in S_N\rangle.
\]
Here $S_N$ is the permutation group acting by permuting the tensor factors.

$\sigma_1,\sigma_2,\sigma_3$ be normalized by
\[
\sigma_a\sigma_b
=\delta_{ab}I+i\sum_{c=1}^3\varepsilon_{abc}\sigma_c,
\qquad
\operatorname{Tr}(\sigma_a\sigma_b)=2\delta_{ab}.
\]
Let $SU(2)$ act by conjugation on the real vector space of traceless
Hermitian endomorphisms of $\C^2$.  Use
$x\mapsto x_1\sigma_1+x_2\sigma_2+x_3\sigma_3$ to identify this Euclidean

SU(2)\longrightarrow SO(3)
\]
is surjective with kernel $\{\pm\id\}$.

Let $\mathcal O_L$ be the coadjoint sphere of radius $L>0$ with the
Kirillov form developed above, and regard it as an $SU(2)$ orbit through
the spin covering.  Put

N=\frac{2L}{\hbar}.
\]
Prove that $\mathcal O_L$ is prequantizable exactly when
$N\in\mathbb Z_{>0}$.  Use as theorem input that $\mathcal O_L$ has an
$SU(2)$-invariant complex structure $J$, an endomorphism of its tangent
bundle satisfying $J^2=-\id$.  Let $T^{0,1}\mathcal O_L$ be the

V_N=\operatorname{Sym}^N(\C^2).
\]
Call this polarized-section space the \emph{geometric quantization} of
$\mathcal O_L$.
The action on $V_N$ is induced from the tensor-power action on
$(\C^2)^{\otimes N}$.

u_\theta=
\begin{pmatrix}e^{i\theta/2}&0\\0&e^{-i\theta/2}\end{pmatrix},
\qquad j=\frac N2,

prove that the weights of $V_N$ are $m=-j,-j+1,\ldots,j$ and derive the

\chi_j(u_\theta)

\qquad U_1(t)=2t,
\qquad U_{n+1}(t)=2tU_n(t)-U_{n-1}(t).

\qquad m=-j,-j+1,\ldots,j.

\end{exercise}

\begin{exercise}[Invariant Riemannian geometry]
A \emph{Riemannian metric} on an $n$-manifold $M$ is a smooth family of
positive-definite inner products $g_p$ on $T_pM$.  Its Riemannian density
is characterized by
\[
|\operatorname{vol}_g|_p(v_1\wedge\cdots\wedge v_n)
=\det\bigl(g_p(v_i,v_j)\bigr)^{1/2},
\]
and determines the Riemannian measure $d\operatorname{vol}_g$.  The
gradient is defined by
\[
g(\operatorname{grad}_g f,X)=Xf.
\]
The divergence is determined by
\[
\int_M Xf\,d\operatorname{vol}_g
=-\int_M f\,\operatorname{div}_gX\,d\operatorname{vol}_g
\]
for compactly supported $f$, and the Laplace--Beltrami operator is
$\Delta_g=\operatorname{div}_g\operatorname{grad}_g$.

An isometry $F:(M,g)\to(N,h)$ is a diffeomorphism satisfying
$h_{F(p)}(dF_pv,dF_pw)=g_p(v,w)$.
For a Lie group $G$ and $g\in G$, write $L_g(h)=gh$.  A Riemannian metric
on $G$ is \emph{left-invariant} when $L_g$ is an isometry for every $g$.

\begin{enumerate}[label=(\alph*)]
\item Prove that an isometry preserves the Riemannian density, gradient,
divergence, and Laplace--Beltrami operator.
\item Prove that a left-invariant Riemannian metric on a Lie group has
left-invariant Riemannian measure.
\item For the Euclidean metric, recover Lebesgue measure and
$\Delta=\sum_j\partial_j^2$.
\end{enumerate}
\end{exercise}

\begin{exercise}[Haar measure, convolution, and regular representations]
Every locally compact group below is equipped with a chosen nonzero left
Haar measure $\mu$.  Haar measure on a compact group is normalized by
$\mu(G)=1$.

Let $G$ be a locally compact group with left Haar measure $\mu$.  For
$f,g\in C_c(G)$, define
\[
(f*g)(x)=\int_G f(y)g(y^{-1}x)\,d\mu(y).
\]
The \emph{left regular representation} on $L^2(G)$ is
\[
(\lambda_s\xi)(t)=\xi(s^{-1}t).
\]

Identify Haar measure on $\R^n$, $\mathbb Z^n$, and $\mathbb T^n$. Construct
normalized Haar measure on a compact Lie group from a left-invariant metric.
Prove associativity of convolution, the $L^1$ convolution inequality, and
$\|f*g\|_2\leq\|f\|_1\|g\|_2$. Prove that $g\mapsto\lambda_g$ is a strongly
continuous unitary representation. For a finite-dimensional representation
of a compact group, average an inner product over $G$ to obtain an invariant
one.
\end{exercise}

\begin{exercise}[Pontryagin duality and Abelian Fourier analysis]
For a locally compact abelian group $G$, its \emph{Pontryagin dual}
$\widehat G$ is the group of continuous characters $\chi:G\to U(1)$, with
pointwise multiplication and the compact-open topology.
The \emph{double-dual map} sends
$x\in G$ to the character $\chi\mapsto\chi(x)$ of $\widehat G$.

Identify
\[
\widehat{\R^n}\cong\R^n,
\qquad
\widehat{\mathbb Z^n}\cong\mathbb T^n,
\qquad
\widehat{\mathbb T^n}\cong\mathbb Z^n,
\]
and verify the double-dual map in each case.  For a linear subspace
$W\subseteq\R^n$ and a full-rank
lattice $\Gamma\subseteq\R^n$, compute their annihilators and identify the
duals of $\R^n/W$ and $\R^n/\Gamma$.
\end{exercise}



\begin{exercise}[Fourier series]
On $\T=\R/2\pi\Z$, write
\[
dm_{\T}(x)=\frac{dx}{2\pi}
\]
for normalized Haar measure and define
\[
\widehat f(k)=\int_\T f(x)e^{-ikx}\,dm_{\T}(x),\qquad k\in\Z.
\]
The \emph{Fourier series} of $f$ is
$\sum_{k\in\Z}\widehat f(k)e^{ikx}$.

Prove that $(e^{ikx})_{k\in\Z}$ is a complete orthonormal family in
$L^2(\T)$ and derive
\[
f=\sum_{k\in\Z}\widehat f(k)e^{ikx},
\qquad
\|f\|_2^2=\sum_{k\in\Z}|\widehat f(k)|^2.
\]
\end{exercise}

\begin{exercise}[Euclidean Fourier inversion]
The \emph{Schwartz space} $\mathcal S(V)$ consists of the smooth functions
$f$ satisfying
\[
\sup_{q\in V}(1+\|q\|)^m\|d^rf(q)\|<\infty
\]
for all $m,r\geq0$, where the norm on the symmetric covariant tensors is
induced by the Euclidean structure.
Let $V$ be an $n$-dimensional real Euclidean space with Lebesgue measure
$dq$.  Equip $V^*$ with its dual Euclidean structure and the resulting
Lebesgue measure $dp$.  For $f\in\mathcal S(V)$, define
\[
(\mathcal F_\hbar f)(p)
=(2\pi\hbar)^{-n/2}\int_Ve^{-ip(q)/\hbar}f(q)\,dq,
\qquad \mathcal F=\mathcal F_1.
\]

Prove Fourier inversion on $\mathcal S(V)$ and prove that
$\mathcal F_\hbar$ extends uniquely to a unitary operator on $L^2(V)$.
For $v\in V$, prove
\[
\mathcal F_\hbar(-i\hbar d_v)\mathcal F_\hbar^{-1}\widehat f(p)
=p(v)\widehat f(p).
\]
\end{exercise}

\begin{exercise}[Free-particle dispersion]
Substitute the plane wave

\hbar\omega=\frac{\hbar^2\|k\|^2}{2m}.

Set $p=\hbar k$ and $E=\hbar\omega$.  Derive

E(p)=\frac{\|p\|^2}{2m},
\qquad
v_{\rm group}=\frac{p^\sharp}{m},
\]
where $p^\sharp\in V$ is the Euclidean-dual vector to $p$.
Deduce the time-of-flight displacement of a wave packet concentrated near
$p_0$ and the de Broglie relation between its measured wavelength and

\end{exercise}


\begin{exercise}[Bessel functions and radial Fourier analysis]
For $\alpha>-1$, the Bessel function of the first kind is
\[
J_\alpha(z)
=\sum_{m=0}^{\infty}
\frac{(-1)^m}{m!\,\Gamma(m+\alpha+1)}
\left(\frac z2\right)^{2m+\alpha},

with a branch of $z^\alpha$ fixed when $\alpha\notin\Z$.

Prove

z^2J_\alpha''(z)+zJ_\alpha'(z)
+(z^2-\alpha^2)J_\alpha(z)=0
\]
and the recurrence identities
\[
J_{\alpha-1}(z)+J_{\alpha+1}(z)=\frac{2\alpha}{z}J_\alpha(z),
\qquad
J_{\alpha-1}(z)-J_{\alpha+1}(z)=2J_\alpha'(z).
\]
Average the Euclidean Fourier kernel over spheres and express the Fourier
transform of a radial Schwartz function on $\R^n$ as a Hankel transform
with kernel $J_{n/2-1}$.  Derive the corresponding inversion identity.
\end{exercise}

\begin{exercise}[Bessel diffraction]

\begin{exercise}[Lattices and Poisson summation]
A \emph{lattice} in $\R^n$ is a subgroup
$\Gamma=A\mathbb Z^n$ for some invertible linear map $A:\R^n\to\R^n$. Its
covolume is $\vol(\Gamma)=|\det A|$, and a fundamental cell is
$F=A[0,1)^n$. Its \emph{reciprocal lattice} is the annihilator
\[
\Gamma^*=\{\xi\in\R^n:\xi\cdot\gamma\in2\pi\mathbb Z
\text{ for every }\gamma\in\Gamma\}.
\]

For $f\in\mathcal S(\R^n)$, its \emph{$\Gamma$-periodization} is
\[
P_\Gamma f(x)=\sum_{\gamma\in\Gamma}f(x+\gamma).
\]
\label{ex:poisson-summation}
Prove that $P_\Gamma f$ converges with all derivatives and compute its
Fourier coefficients. With the Fourier-transform normalization above, prove
\[
P_\Gamma f(x)=\frac{(2\pi)^{n/2}}{\vol(\Gamma)}
 \sum_{\xi\in\Gamma^*}\mathcal Ff(\xi)e^{i\xi\cdot x}.
\]
Set $x=0$ to obtain the Poisson summation formula. Derive the Jacobi theta
transformation by applying it to a Gaussian.
\end{exercise}

\begin{exercise}[Crystal diffraction and Bragg peaks]

Let $\Gamma_N$ be a finite rectangular portion of $\Gamma$. In the

\mathcal A_N(q)
=S_B(q)\sum_{\gamma\in\Gamma_N}e^{-iq\cdot\gamma}.

Use geometric sums and Poisson summation to show that the normalized

Show how zeros of $S_B$ produce missing reflections. Compare the peak

\end{exercise}

\begin{exercise}[Finite representations and Maschke's theorem]
For a group $G$, a representation on a vector space $V$ is a homomorphism
$\rho:G\to\operatorname{GL}(V)$.  A subspace is invariant when it is
preserved by every $\rho(g)$, and a nonzero representation is irreducible
when it has no proper nonzero invariant subspace.

Let $G$ be finite, let the characteristic of $k$ not divide $|G|$, and let
$V$ be a finite-dimensional representation.  Average an arbitrary
projection over $G$ to construct an invariant
complement to every invariant subspace. Deduce that every finite-dimensional
representation is a direct sum of irreducible representations.

Let $H$ be a finite-dimensional complex Hilbert space carrying a unitary
representation of $G$, and let $A\in\End(H)$ be self-adjoint and commute
with the action.  Prove that every spectral subspace of $A$ is invariant
and decomposes into irreducible representations.  If $A_s$ is a
norm-continuous family of such operators and a spectral cluster remains
isolated, use its spectral projection to prove that the multiplicity of
every irreducible type in the cluster is locally constant.  Deduce that a
change in these multiplicities requires a collision with another spectral
cluster.
\end{exercise}

\begin{exercise}[Schur's lemma]
An intertwiner $T:(V,\rho)\to(W,\sigma)$ between representations satisfies
$T\rho(g)=\sigma(g)T$.

Prove that a nonzero intertwiner between irreducible complex
finite-dimensional representations is an isomorphism and that every
endomorphism of an irreducible representation is scalar. Deduce uniqueness
of irreducible multiplicities in a decomposition.
\end{exercise}

\begin{exercise}[Characters and finite-group harmonic analysis]
The \emph{character} of a finite-dimensional representation $V$ is the class
function $\chi_V(g)=\Tr(\rho_V(g))$. For class functions on a finite group,
define
\[
\langle f,h\rangle_G=\frac1{|G|}\sum_{g\in G}\overline{f(g)}h(g).
\]

Use averaging of linear maps and Schur's lemma to prove orthonormality of the
irreducible characters. Prove that they form a complete orthonormal family
in the class functions and that
$\langle\chi_U,\chi_V\rangle_G=\dim\Hom_G(U,V)$. Deduce the multiplicity
formula and the character table constraints.
\end{exercise}

\begin{exercise}[One-parameter subgroups and the exponential map]

Prove existence and uniqueness of the one-parameter subgroup generated by
$X\in\mathfrak g$. For a closed matrix group, prove that it is
$t\mapsto e^{tX}$, identify the commutator bracket, and compute the first
nontrivial terms of the Baker--Campbell--Hausdorff formula.
\end{exercise}

\begin{exercise}[Compact convolution operators]
Let $G$ be compact with normalized Haar measure and $k\in C(G)$.  Prove
that
\[
(T_kf)(x)=\int_Gk(y^{-1}x)f(y)\,dy
\]
defines a compact operator on $L^2(G)$ by uniformly approximating its
continuous kernel by finite sums of product functions.  Prove that $T_k$
commutes with left translations and that
$k(g^{-1})=\overline{k(g)}$ makes $T_k$ self-adjoint.  Deduce that every
nonzero eigenspace of such a self-adjoint convolution operator is a
finite-dimensional invariant subspace of the left regular representation.
\end{exercise}

\begin{exercise}[Density of matrix coefficients]
For a unitary representation $\pi:G\to U(H_\pi)$, its matrix coefficients
are the functions $g\mapsto\langle y,\pi(g)x\rangle$.

Construct continuous convolution kernels that approximate the identity on
$C(G)$.  Apply the compact spectral theorem to their self-adjoint
symmetrizations and prove that the linear span of matrix coefficients of
finite-dimensional unitary representations is uniformly dense in $C(G)$
and dense in $L^2(G)$.
\end{exercise}

\begin{exercise}[Schur orthogonality]
For irreducible finite-dimensional unitary representations $\pi$ and
$\sigma$, prove that
\[
A\longmapsto\int_G\pi(g)A\sigma(g)^*\,dg
\]
is zero when $\pi\not\simeq\sigma$ and is
$A\mapsto\operatorname{tr}(A)I_{H_\pi}/\dim H_\pi$ when
$\pi=\sigma$.  Deduce that the coefficient map identifies
$H_\pi\otimes H_\pi^\vee$, after multiplication by
$\sqrt{\dim H_\pi}$, with an isometric subspace of $L^2(G)$, and that
distinct irreducible coefficient spaces are orthogonal, where
\[
C_\pi:H_\pi\otimes H_\pi^\vee\longrightarrow L^2(G),
\qquad
C_\pi(x\otimes\varphi)(g)=\varphi(\pi(g)x).
\]
\end{exercise}

\begin{exercise}[Peter--Weyl decomposition]
The unitary dual $\widehat G$ is the set of unitary-equivalence classes of
irreducible finite-dimensional unitary representations of $G$.

Use density and Schur orthogonality to prove
\[
L^2(G)\cong
\mathop{\widehat{\bigoplus}}_{\pi\in\widehat G}
H_\pi\otimes H_\pi^\vee,
\]
with the left regular representation acting as
$\pi\otimes I_{H_\pi^\vee}$ on the $\pi$-summand.  For
$f\in L^2(G)\subseteq L^1(G)$, define
$\widehat f(\pi)=\int_Gf(g)\pi(g)^*\,dg$ and prove
\[
f=\sum_{\pi\in\widehat G}
(\dim H_\pi)\operatorname{tr}\!\bigl(\widehat f(\pi)\pi(\,\cdot\,)\bigr),
\qquad
\|f\|_2^2=\sum_{\pi\in\widehat G}
(\dim H_\pi)\operatorname{tr}\!\bigl(\widehat f(\pi)^*\widehat f(\pi)\bigr),
\]
where the first sum converges in $L^2(G)$.  Prove uniform convergence on
$C(G)$ for the summability means supplied by the approximate identities in
the density exercise.
\end{exercise}

\begin{exercise}[Character inversion]
Let $G$ be compact with normalized Haar measure, and let $f$ be a finite
linear combination of irreducible characters.  For an irreducible unitary
representation $\pi$, set
\[
\widehat f(\pi)=\int_G f(s)\pi(s^{-1})\,ds.
\]
If $f$ is central, use Schur's lemma to write
$\widehat f(\pi)=c_\pi I_{d_\pi}$.  Derive
\[
f(e)=\sum_{\pi\in\widehat G}d_\pi\Tr(\widehat f(\pi))
=\sum_{\pi\in\widehat G}d_\pi^2c_\pi
\]
and prove that convolution by $f$ acts by the scalar $c_\pi$ on every
copy of $\pi$ in $L^2(G)$.
\end{exercise}

\begin{exercise}[Irreducible representations of $SU(2)$]

\begin{exercise}[Gauss hypergeometric functions]
Using the rising factorial defined in Spectral Theory, for
$c\notin\Z_{\leq0}$ the \emph{Gauss hypergeometric function} is
\[
{}_2F_1\!\left(\begin{matrix}a,b\\c\end{matrix};z\right)
=\sum_{m=0}^\infty
\frac{(a)_m(b)_m}{(c)_m}\frac{z^m}{m!}.
\]
It is interpreted as a finite sum when a numerator parameter is a
nonpositive integer and otherwise converges for $|z|<1$.
The Gauss hypergeometric operator is
\[
\mathcal L_{a,b,c}
=z(1-z)\frac{d^2}{dz^2}
+\bigl(c-(a+b+1)z\bigr)\frac{d}{dz}-ab.
\]

Prove that ${}_2F_1(a,b;c;z)$ is the solution of
$\mathcal L_{a,b,c}f=0$ analytic at $z=0$ with $f(0)=1$.  Derive Euler's
integral
\[
{}_2F_1(a,b;c;z)
=\frac{\Gamma(c)}{\Gamma(b)\Gamma(c-b)}
\int_0^1u^{b-1}(1-u)^{c-b-1}(1-zu)^{-a}\,du,
\]
and Gauss's summation
\[
{}_2F_1(a,b;c;1)
=\frac{\Gamma(c)\Gamma(c-a-b)}
{\Gamma(c-a)\Gamma(c-b)}.
\]
State the parameter conditions for the integral and the summation.
\end{exercise}

\begin{exercise}[Kummer, Whittaker, and Laguerre functions]
For $b\notin\Z_{\leq0}$, define the Kummer function
\[
M(a,b,z)={}_1F_1(a;b;z)
=\sum_{m=0}^\infty\frac{(a)_m}{(b)_m}\frac{z^m}{m!}.
\]
For $2\mu+1\notin\Z_{\leq0}$, define
\[
M_{\lambda,\mu}(z)
=e^{-z/2}z^{\mu+1/2}
M\!\left(\mu-\lambda+\tfrac12,2\mu+1,z\right),
\]
with a branch of $z^{\mu+1/2}$ fixed.  The Whittaker function
$W_{\lambda,\mu}$ is the solution of
\[
w''+\left(-\frac14+\frac{\lambda}{z}
+\frac{1/4-\mu^2}{z^2}\right)w=0
\]
normalized by
$W_{\lambda,\mu}(z)\sim e^{-z/2}z^\lambda$ as $z\to+\infty$.

For $\alpha>-1$ and $n\in\Z_{\geq0}$, define
\[
L_n^{(\alpha)}(x)
=\frac{x^{-\alpha}e^x}{n!}
\frac{d^n}{dx^n}\left(e^{-x}x^{n+\alpha}\right)
=\frac{(\alpha+1)_n}{n!}M(-n,\alpha+1,x).
\]
Write $L_n=L_n^{(0)}$.

Prove locally uniformly in $z$ that
\[
\lim_{B\to+\infty}
{}_2F_1\!\left(\begin{matrix}a,B\\c\end{matrix};\frac zB\right)
=M(a,c,z),
\]
and derive Kummer's equation
\[
zy''+(b-z)y'-ay=0.
\]
Apply the substitution
$w(z)=e^{-z/2}z^{\mu+1/2}y(z)$ to obtain the Whittaker equation with
$a=\mu-\lambda+\tfrac12$ and $b=2\mu+1$.

Derive the equality of the two displayed definitions of
$L_n^{(\alpha)}$.  Use the Rodrigues formula and integration by parts to
prove
\[
\int_0^\infty L_m^{(\alpha)}(x)L_n^{(\alpha)}(x)
x^\alpha e^{-x}\,dx
=\frac{\Gamma(n+\alpha+1)}{n!}\delta_{mn},
\]
and derive
\[
(n+1)L_{n+1}^{(\alpha)}(x)
=(2n+\alpha+1-x)L_n^{(\alpha)}(x)
-(n+\alpha)L_{n-1}^{(\alpha)}(x).
\]
Prove
\[
\sum_{n\geq0}L_n^{(\alpha)}(x)t^n
=(1-t)^{-\alpha-1}
\exp\!\left(-\frac{xt}{1-t}\right),
\qquad |t|<1,
\]
and obtain the corresponding coefficient contour integral around $t=0$.
\end{exercise}

\begin{exercise}[Jacobi polynomials and their spectral equation]
For $\alpha,\beta>-1$, the \emph{Jacobi polynomial} is
\[
P_n^{(\alpha,\beta)}(x)
=\frac{(\alpha+1)_n}{n!}
{}_2F_1\!\left(
\begin{matrix}-n,n+\alpha+\beta+1\\\alpha+1\end{matrix};
\frac{1-x}{2}
\right).

Prove

\left((1-x^2)\frac{d^2}{dx^2}
+\bigl(\beta-\alpha-(\alpha+\beta+2)x\bigr)\frac{d}{dx}\right)

=-n(n+\alpha+\beta+1)P_n^{(\alpha,\beta)}.

Derive Jacobi orthogonality on $[-1,1]$ with weight
$(1-x)^\alpha(1+x)^\beta$ and derive the three-term recurrence.
\end{exercise}

\begin{exercise}[Spherical harmonics and atomic orbitals]
Identify $S^2$ with $SO(3)/SO(2)$ and $L^2(S^2)$ with the right
$SO(2)$-invariant subspace of $L^2(SO(3))$. Use Peter--Weyl to obtain
\[
L^2(S^2)\cong\mathop{\widehat{\bigoplus}}_{\ell=0}^\infty V_\ell
\]
For normalized $J_3$-eigenstates $|\ell m\rangle$, define the Wigner
matrix coefficients

D^\ell_{mn}(g)=\langle \ell m|\pi_\ell(g)|\ell n\rangle.

Prove that the functions
$Y_{\ell m}\propto D^\ell_{m0}$, $-\ell\leq m\leq\ell$, form a complete
orthogonal family in $V_\ell$. For the unit round sphere, set
\[
\Delta_{S^2}
=\frac1{\sin\theta}\partial_\theta
  \bigl(\sin\theta\,\partial_\theta\bigr)
 +\frac1{\sin^2\theta}\partial_\phi^2.
\]
Derive
$\Delta_{S^2}Y_{\ell m}=-\ell(\ell+1)Y_{\ell m}$ and completeness of the
spherical harmonics.  Derive the Jacobi realization
\[
Y_{\ell m}(\theta,\phi)
=C_{\ell m}e^{im\phi}(\sin\theta)^{|m|}
P_{\ell-|m|}^{(|m|,|m|)}(\cos\theta)
\]
with $C_{\ell m}$ fixed by unit norm.  On
\[
L^2(\R^3,d^3x)
\cong L^2((0,\infty),r^2\,dr)\otimes L^2(S^2,d\Omega),
\]
consider the Coulomb Hamiltonian
\[
\widehat H_C=-\frac{\hbar^2}{2M}\Delta_{\R^3}-\frac{\kappa}{r},
\qquad M,\kappa>0.
\]
Separate $\widehat H_C$ in spherical coordinates and prove that its radial
operator on the $\ell$th angular sector is
\[
-\frac{\hbar^2}{2M}
\left(\frac{d^2}{dr^2}+\frac2r\frac d{dr}
-\frac{\ell(\ell+1)}{r^2}\right)-\frac\kappa r
\]
on $L^2((0,\infty),r^2\,dr)$.  Recover
$\psi_{n\ell m}(r,\theta,\phi)=R_{n\ell}(r)Y_{\ell m}(\theta,\phi)$.
Put
\[
a_0=\frac{\hbar^2}{M\kappa},
\qquad
\beta=\frac{\sqrt{-2ME}}{\hbar},
\qquad
\rho=2\beta r
\]
for $E<0$, and set $u(r)=rR(r)$.  Reduce the radial eigenvalue equation to
the Whittaker equation with
\[
\lambda=\frac1{a_0\beta},
\qquad
\mu=\ell+\frac12.
\]
Use regularity at the origin and square-integrability at infinity to prove
that
\[
\ell+1-\lambda=-n_r,
\qquad n_r\in\Z_{\geq0}.
\]
Writing $n=n_r+\ell+1$, deduce

E_n=-\frac{M\kappa^2}{2\hbar^2n^2},
\qquad
R_{n\ell}(r)
\propto e^{-r/(na_0)}
\left(\frac{2r}{na_0}\right)^\ell
L_{n-\ell-1}^{(2\ell+1)}\!\left(\frac{2r}{na_0}\right).
\]
Prove that the spatial multiplicity of $E_n$ is $n^2$ and derive the
spectral-line frequencies
\[
\omega_{n_i\to n_f}
=\frac{M\kappa^2}{2\hbar^3}
\left|\frac1{n_f^2}-\frac1{n_i^2}\right|.
\]
Determine the angular nodes of the $s$, $p$, and $d$ orbitals and relate them
to symmetry-enforced zeros in photoelectron angular distributions in the

nodal suppression with
\href{https://doi.org/10.1038/ncomms9287}
{Wei\ss{} et al., \emph{Nature Communications} \textbf{6} (2015)}, and
distinguish the orbital-symmetry prediction, plane-wave approximation,
measured intensity, and final-state scattering corrections.
\end{exercise}

\section{Harish--Chandra Theory}


\begin{exercise}[The hyperbolic plane and rank-one decompositions]
Let
\[
G=SL_2(\R),
\qquad
K=SO(2),
\]
and let $G$ act on the upper half-plane by fractional linear
transformations.  Define
\[
a_r=\begin{pmatrix}e^{r/2}&0\\0&e^{-r/2}\end{pmatrix},
\qquad
n_x=\begin{pmatrix}1&x\\0&1\end{pmatrix},
\]
with $r,x\in\R$.  Give the upper half-plane the metric
$ds^2=(dx^2+dy^2)/y^2$ and fix its curvature radius to be one.

Use the polar and QR decompositions of an element of $SL_2(\R)$ to prove
\[
K\times\R\times\R\longrightarrow G,
\qquad(k,r,x)\longmapsto ka_rn_x
\]
is a diffeomorphism and
\[
G=K\{a_r:r\geq0\}K.
\]
Prove that $K$ is the stabilizer of $i$ and that the orbit map identifies
$G/K$ isometrically with the hyperbolic plane.  Identify the two-point Weyl
group and prove that a $K$-bi-invariant function is determined by its value
on $a_r$, $r\geq0$.

Derive geodesic polar coordinates about $i$,
\[
ds^2=dr^2+\sinh^2r\,d\theta^2,
\qquad
d\operatorname{vol}_{\mathbb H^2}
=\sinh r\,dr\,d\theta.
\]
Normalize $dk$ to have total mass one and derive the Iwasawa and radial
integration formulas
\[
\int_G F(g)\,dg
=\int_K\int_\R\int_\R
F(ka_rn_x)e^r\,dx\,dr\,dk,
\]
\[
\int_{\mathbb H^2}f(z)\,d\operatorname{vol}(z)
=2\pi\int_0^\infty f(a_rK)\sinh r\,dr
\]
for radial $f$.
\end{exercise}

\begin{exercise}[Spherical convolution and the rank-one Gelfand pair]
For $K$-bi-invariant functions $f_1,f_2\in C_c(G)$, define
\[
(f_1*f_2)(g)=\int_G f_1(h)f_2(h^{-1}g)\,dh.
\]
Use the Cartan decomposition to prove that $g$ and $g^{-1}$ belong to the
same double coset $KgK$.  Apply the anti-involution $g\mapsto g^{-1}$ to
prove
\[
f_1*f_2=f_2*f_1.
\]
Thus $(G,K)$ is a \emph{Gelfand pair}.

A normalized continuous $K$-bi-invariant function $\varphi$ is
\emph{spherical} when
\[
\int_K\varphi(gkh)\,dk=\varphi(g)\varphi(h).
\]
Show that integration against $\varphi$ is a character of the commutative
convolution algebra, and prove the converse for characters represented by
continuous functions.  Deduce that spherical functions are the
distinguished special functions diagonalizing radial convolution.
\end{exercise}

\begin{exercise}[Poisson kernels and rank-one spherical functions]
For the sign convention
\[
\Delta_{\mathbb H^2}=y^2(\partial_x^2+\partial_y^2),
\]
derive its radial part
\[
\Delta_{\rm rad}=\frac{d^2}{dr^2}
 +\coth r\frac d{dr}.
\]

For $\lambda\in\R$, define
\[
\varphi_\lambda(r)
=\frac1{2\pi}\int_0^{2\pi}
(\cosh r-\sinh r\cos\theta)^{-1/2-i\lambda}\,d\theta.
\]
Interpret the integrand as a complex power of the hyperbolic Poisson kernel.
Prove that $\varphi_\lambda$ is real-valued, even in $\lambda$, normalized
at $r=0$, and satisfies the spherical product formula from the preceding
exercise.  Differentiate under the integral to prove
\[
\Delta_{\mathbb H^2}\varphi_\lambda
=-(\lambda^2+\tfrac14)\varphi_\lambda,
\qquad
\varphi_\lambda(eK)=1.
\]
Regarded as a $K$-bi-invariant function on $G$ and written in the radial
coordinate $a_rK$, it is determined by
\[
\varphi_\lambda''(r)+\coth r\,\varphi_\lambda'(r)
=-(\lambda^2+\tfrac14)\varphi_\lambda(r),
\qquad
\varphi_\lambda(0)=1,
\qquad
\varphi_\lambda'(0)=0.
\]
Show that the normalized radial eigenfunction satisfying
\[
\Delta\varphi_\lambda
=-(\lambda^2+\tfrac14)\varphi_\lambda,
\qquad \varphi_\lambda(0)=1,
\]
is the conical Legendre function
\[
\varphi_\lambda(r)=P_{-1/2+i\lambda}(\cosh r).
\]
\end{exercise}

\begin{exercise}[Finite-radius conical orthogonality]
For real $\lambda,\mu$, derive the Wronskian identity
\[
(\mu^2-\lambda^2)\int_0^R
\varphi_\lambda(r)\varphi_\mu(r)\sinh r\,dr
=\left[\sinh r\,
(\varphi_\mu\varphi_\lambda'-\varphi_\lambda\varphi_\mu')\right]_0^R.
\]
Impose the Dirichlet condition $\varphi_\lambda(R)=0$.  Use the radial
Sturm--Liouville operator to prove that its positive roots $\lambda_j$ are
real and simple and that the corresponding conical functions are
orthogonal in $L^2((0,R),\sinh r\,dr)$.  Compare the large-$r$ equation
with a constant-coefficient equation and show how the discrete parameters
become continuous as $R\to\infty$.
\end{exercise}

\begin{exercise}[Abel transforms and Euclidean Fourier reduction]
For a smooth compactly supported radial function $f$ on $\mathbb H^2$,
define its Abel transform by
\[
(\mathcal Af)(t)
=e^{t/2}\int_\R f(a_tn_xK)\,dx.
\]
Use the Iwasawa integration formula and the Poisson integral for
$\varphi_\lambda$ to prove
\[
\widetilde f(\lambda)
:=\int_{\mathbb H^2}f(z)\varphi_\lambda(z)\,
d\operatorname{vol}(z)
=\int_\R(\mathcal Af)(t)e^{-i\lambda t}\,dt.
\]
Prove that $\mathcal A$ intertwines radial convolution on $G/K$ with
ordinary convolution on $\R$.  Deduce injectivity of the spherical
transform from Euclidean Fourier inversion and recover the commutativity of
radial convolution.

For $f$ supported in a geodesic disk of radius $R$, determine the support
and parity properties of $\mathcal Af$.  Derive the rank-one formulas
\[
(\mathcal Af)(t)
=\sqrt2\int_{|t|}^\infty
\frac{f(r)\sinh r}{\sqrt{\cosh r-\cosh t}}\,dr,
\]
\[
f(r)=-\frac1{\pi\sqrt2}\int_r^\infty
\frac{(\mathcal Af)'(t)}{\sqrt{\cosh t-\cosh r}}\,dt,
\]
and use them to express $f(r)$ in terms of the inverse Euclidean Fourier
transform of $\widetilde f$.
\end{exercise}

\begin{exercise}[Harish--Chandra normalization]
For real $\lambda\ne0$, define $c(\lambda)$ by the asymptotic normalization
\[
\varphi_\lambda(r)
=c(\lambda)e^{(i\lambda-1/2)r}
+c(-\lambda)e^{(-i\lambda-1/2)r}
+o(e^{-r/2}).
\]

Apply endpoint asymptotics to the Poisson integral and evaluate the
resulting beta integrals to derive
\[
c(\lambda)=\frac{\Gamma(i\lambda)}
{\sqrt\pi\,\Gamma(\tfrac12+i\lambda)},
\qquad
|c(\lambda)|^{-2}=\pi\lambda\tanh(\pi\lambda).
\]
\end{exercise}

\begin{exercise}[Mehler--Fock inversion]
For the spherical transform defined by the Abel transform, use radial
integration to obtain
\[
\widetilde f(\lambda)
=2\pi\int_0^\infty
f(r)\varphi_\lambda(r)\sinh r\,dr.
\]

For $\Phi\in C_c^\infty((0,\infty)^2)$, define the diagonal
distribution by
\[
\iint\Phi(\lambda,\mu)\delta(\lambda-\mu)\,d\lambda\,d\mu
=\int_0^\infty\Phi(\lambda,\lambda)\,d\lambda.
\]
Interpret the following identity after pairing both sides with such test
functions:
truncate the radial integral at $R$, use the Harish--Chandra asymptotics in
the Wronskian identity, and apply the distributional limit of
$\sin(R(\lambda-\mu))/(\lambda-\mu)$ to prove
\[
2\pi\int_0^\infty
\varphi_\lambda(r)\varphi_\mu(r)\sinh r\,dr
=\frac{2\pi}{\lambda\tanh(\pi\lambda)}
\delta(\lambda-\mu).
\]
Deduce the Mehler--Fock inversion and Plancherel formulas
\[
f(r)=\frac1{2\pi^2}\int_0^\infty
\widetilde f(\lambda)P_{-1/2+i\lambda}(\cosh r)
|c(\lambda)|^{-2}\,d\lambda,
\]
\[
\|f\|_2^2=\frac1{2\pi^2}\int_0^\infty
|\widetilde f(\lambda)|^2|c(\lambda)|^{-2}\,d\lambda.
\]
\end{exercise}

\begin{exercise}[Associated Legendre functions and hyperbolic-disk modes]
In geodesic polar coordinates, derive the full Laplace--Beltrami operator

\Delta_{\mathbb H^2}
=\frac{\partial^2}{\partial r^2}
+\coth r\frac{\partial}{\partial r}
+\frac1{\sinh^2r}\frac{\partial^2}{\partial\theta^2}.
\]
For the intrinsic $SO(2)$ character $e^{i\ell\theta}$, derive the radial
equation

R''+\coth r\,R'
-\frac{\ell^2}{\sinh^2r}R
=-(\lambda^2+\tfrac14)R.
\]
Show that the solution regular at the origin is proportional to

P_{-1/2+i\lambda}^{-|\ell|}(\cosh r)
\]
and determine its normalization from
$R(r)\sim r^{|\ell|}$ as $r\downarrow0$.

For the geodesic disk $D_R$, prove that its Dirichlet eigenmodes are

e^{i\ell\theta}
P_{-1/2+i\lambda_{n\ell}}^{-|\ell|}(\cosh r),
\qquad
P_{-1/2+i\lambda_{n\ell}}^{-|\ell|}(\cosh R)=0.

Establish orthogonality for distinct pairs $(n,\ell)$ and identify the

\end{exercise}

\begin{exercise}[Harish--Chandra wave kernels and geodesic phase fronts]
Let $u_{tt}=v^2\Delta_{\mathbb H^2}u$ have radial initial data
$u(0,r)=f(r)$ and $u_t(0,r)=g(r)$ in $C_c^\infty$.  Put
$\omega_\lambda=v(\lambda^2+1/4)^{1/2}$.  Use Mehler--Fock inversion to
prove
\[
u(t,r)=\frac1{2\pi^2}\int_0^\infty
\left(
\cos(t\omega_\lambda)\widetilde f(\lambda)
+\frac{\sin(t\omega_\lambda)}{\omega_\lambda}\widetilde g(\lambda)
\right)
P_{-1/2+i\lambda}(\cosh r)|c(\lambda)|^{-2}\,d\lambda.
\]

For a point source at $z_0$, define the point distribution by
$\langle\delta_{z_0},\phi\rangle=\phi(z_0)$.  Take
$f=0$, $g=\delta_{z_0}$, use $G$-invariance to replace $r$ by the
hyperbolic distance $d(z,z_0)$, and derive the impulse kernel
\[
K_t(d)=\frac1{2\pi^2}\int_0^\infty
\frac{\sin(t\omega_\lambda)}{\omega_\lambda}
P_{-1/2+i\lambda}(\cosh d)
|c(\lambda)|^{-2}\,d\lambda.
\]
Use as theorem input finite propagation speed for the wave equation and
the one-dimensional stationary-phase estimate for a nondegenerate critical
point, including its uniform form for compact parameter sets.  Apply them
to show
that the leading phase fronts are the geodesic circles
$d(z,z_0)=vt$.  Derive the high-frequency spectral density from
$|c(\lambda)|^{-2}=\pi\lambda\tanh(\pi\lambda)$ and state the interior tail
present for the two-dimensional wave equation.
\end{exercise}

\section{Automorphic Forms}


\begin{exercise}[Lattice theta series]
A \emph{Euclidean lattice} $L\subset\R^d$ is a discrete subgroup for which
$\R^d/L$ is compact.  Its dual is
\[
L^*=\{y\in\R^d:\langle x,y\rangle\in\Z
\text{ for every }x\in L\}.
\]
For $\tau$ in the upper half-plane and $q=e^{2\pi i\tau}$, its theta series
is
\[
\Theta_L(\tau)=\sum_{x\in L}q^{\|x\|^2/2}.
\]
Its coefficient of $q^n$ counts lattice vectors of squared length $2n$.

Apply Exercise~\ref{ex:poisson-summation} to a Gaussian and prove
\[
\Theta_L(-1/\tau)
=\frac{(-i\tau)^{d/2}}{\vol(L)}\Theta_{L^*}(\tau),
\]
using the branch positive on the positive imaginary axis.  Show that
evenness gives $\Theta_L(\tau+1)=\Theta_L(\tau)$.  Explain why a general
lattice theta series transforms with its dual rather than being a scalar
modular form by itself.
\end{exercise}


\begin{exercise}[Modular forms, zeta functions, and Eisenstein series]
The modular group $SL_2(\Z)$ acts on the upper half-plane by
\[
\gamma\tau=\frac{a\tau+b}{c\tau+d},
\qquad
\gamma=\begin{pmatrix}a&b\\c&d\end{pmatrix},
\]
and is generated by $S:\tau\mapsto-1/\tau$ and
$T:\tau\mapsto\tau+1$.  A \emph{modular form of integral weight $k$} for
$SL_2(\Z)$ is a holomorphic function $f$ on the upper half-plane satisfying
\[
f(\gamma\tau)=(c\tau+d)^k f(\tau)
\]
and holomorphic at the cusp, meaning that its Fourier expansion
$f(\tau)=\sum_{n\geq0}a_nq^n$ has no negative powers.  It is a
\emph{cusp form} if $a_0=0$.  Write $M_k$ for the vector space of modular
forms of weight $k$.

For $\operatorname{Re}s>1$, define the Riemann zeta function by
\[
\zeta(s)=\sum_{n\geq1}n^{-s}.
\]
For a $d$-dimensional Euclidean lattice $L$, with $d\geq1$, and
$\operatorname{Re}s>d/2$, define its Epstein zeta function by
\[
Z_L(s)=\sum_{x\in L\setminus\{0\}}\|x\|^{-2s}.
\]
For $m\geq2$ and $\tau$ in the upper half-plane, define
\[
G_{2m}(\tau)
=\sum_{(c,d)\in\Z^2\setminus\{(0,0)\}}
\frac1{(c\tau+d)^{2m}},
\qquad
E_{2m}(\tau)=\frac{G_{2m}(\tau)}{2\zeta(2m)}.
\]

Put $\theta_L(t)=\Theta_L(it)$.  For $\operatorname{Re}s>d/2$, prove
\[
\pi^{-s}\Gamma(s)Z_L(s)
=\int_0^\infty\bigl(\theta_L(t)-1\bigr)t^{s-1}\,dt.
\]
Split the integral at $1$ and use Poisson duality to prove the meromorphic
continuation
\[
\begin{aligned}
\pi^{-s}\Gamma(s)Z_L(s)
={}&\int_1^\infty(\theta_L(t)-1)t^{s-1}\,dt\\
&+\frac1{\vol(L)}\int_1^\infty
(\theta_{L^*}(t)-1)t^{d/2-s-1}\,dt
+\frac1{\vol(L)(s-d/2)}-\frac1s.
\end{aligned}
\]
Deduce
\[
\pi^{-s}\Gamma(s)Z_L(s)
=\frac1{\vol(L)}
\pi^{-(d/2-s)}\Gamma(d/2-s)Z_{L^*}(d/2-s),
\]
$Z_L(0)=-1$, and
\[
\operatorname*{res}_{s=d/2}Z_L(s)
=\frac{\pi^{d/2}}{\vol(L)\Gamma(d/2)}.
\]
For $L=\Z$, use $Z_\Z(s)=2\zeta(2s)$ to derive the functional equation of
$\zeta$.  Use Fourier series and Parseval to prove
\[
\zeta(2)=\frac{\pi^2}{6},
\qquad
\zeta(4)=\frac{\pi^4}{90},
\]
and then deduce
\[
\zeta(-1)=-\frac1{12},
\qquad
\zeta(-3)=\frac1{120}.
\]
\end{exercise}

\begin{exercise}[Holomorphic Eisenstein series]
Prove absolute local uniform convergence of $G_{2m}$ and prove
\[
G_{2m}\!\left(\frac{a\tau+b}{c\tau+d}\right)
=(c\tau+d)^{2m}G_{2m}(\tau)
\]
for every
$\left(\begin{smallmatrix}a&b\\c&d\end{smallmatrix}\right)
\in SL_2(\Z)$.
Using Poisson summation in one lattice direction, derive
\[
G_{2m}(\tau)
=2\zeta(2m)
+\frac{2(2\pi i)^{2m}}{(2m-1)!}
\sum_{n\geq1}\sigma_{2m-1}(n)q^n,
\qquad
\sigma_r(n)=\sum_{d\mid n}d^r.
\]
Deduce that $E_{2m}$ is a modular form of weight $2m$ and that
\[
E_4(\tau)=1+240\sum_{n\geq1}\sigma_3(n)q^n,
\qquad
E_6(\tau)=1-504\sum_{n\geq1}\sigma_5(n)q^n.
\]
\end{exercise}

\begin{exercise}[Theta and eta functions]
Set
\[
\begin{aligned}
\vartheta_2(\tau)&=\sum_{n\in\Z}q^{(n+1/2)^2/2},\\
\vartheta_3(\tau)&=\sum_{n\in\Z}q^{n^2/2},\\
\vartheta_4(\tau)&=\sum_{n\in\Z}(-1)^nq^{n^2/2},
\end{aligned}
\qquad
\eta(\tau)=q^{1/24}\prod_{m\geq1}(1-q^m).
\]

Use Poisson summation on shifted and phase-twisted Gaussians to derive
\[

\quad

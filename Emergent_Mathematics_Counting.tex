\section{Graded Characters}

\begin{exercise}[Symmetric polynomials]
For $N\geq1$, let
\[
\Lambda_N=\C[x_1,\ldots,x_N]^{S_N}.
\]
Define
\[
e_r=\sum_{1\leq i_1<\cdots<i_r\leq N}
x_{i_1}\cdots x_{i_r},
\qquad
h_r=\sum_{1\leq i_1\leq\cdots\leq i_r\leq N}
x_{i_1}\cdots x_{i_r},
\]
\[
p_r=\sum_{i=1}^Nx_i^r,
\qquad
e_0=h_0=1.
\]
Set $e_r=0$ for $r>N$ and $h_r=0$ for $r<0$.

\begin{enumerate}[label=(\alph*)]
\item Prove that every element of $\Lambda_N$ is uniquely a polynomial in
$e_1,\ldots,e_N$.
\item Prove
\[
\sum_{r=0}^Ne_rt^r=\prod_{i=1}^N(1+x_it),
\qquad
\sum_{r\geq0}h_rt^r=\prod_{i=1}^N\frac1{1-x_it},
\]
and deduce
\[
\left(\sum_{r\geq0}h_rt^r\right)
\left(\sum_{r=0}^N(-1)^re_rt^r\right)=1,
\qquad
rh_r=\sum_{j=1}^rp_jh_{r-j}.
\]
\end{enumerate}
\end{exercise}

\begin{exercise}[Bosonic and fermionic trace generating functions]
For a finite-dimensional complex vector space $V$, define
\[
\operatorname{Sym}^mV
=V^{\otimes m}/\langle \xi-\sigma\xi:
\xi\in V^{\otimes m},\ \sigma\in S_m\rangle.
\]
An endomorphism $T$ induces $\operatorname{Sym}^mT$.  Using the exterior
powers and trace defined in Spectral Theory, define
\[
Z_B(T;z)=\sum_{m\geq0}\operatorname{tr}(\operatorname{Sym}^mT)z^m,
\qquad
Z_F(T;z)=\sum_{m\geq0}\operatorname{tr}(\Lambda^mT)z^m.
\]

An \emph{integer partition} of $n\geq0$ is a finite nonincreasing sequence
$\lambda=(\lambda_1,\ldots,\lambda_r)$ of positive integers with
$|\lambda|=\sum_i\lambda_i=n$.  Let $p(n)$ be the number of partitions,
with $p(0)=1$.

Let $V$ be a finite-dimensional complex vector space and $T\in\End(V)$.
Using the intrinsic determinant defined by the action on the highest
exterior power, prove
\[
Z_B(T;z)=\frac1{\det(I-zT)},
\qquad
Z_F(T;z)=\det(I+zT).
\]
The first identity holds for $|z|$ below the reciprocal spectral radius;
the second is a polynomial identity.  Prove them first for a direct sum of
generalized spectral subspaces and then by polynomial continuation, without
choosing vectors in those subspaces.
If $\lambda_1,\ldots,\lambda_N$ are the eigenvalues counted with algebraic
multiplicity, prove
\[
\operatorname{tr}(\operatorname{Sym}^mT)
=h_m(\lambda_1,\ldots,\lambda_N),
\qquad
\operatorname{tr}(\Lambda^mT)
=e_m(\lambda_1,\ldots,\lambda_N).
\]

Let the one-particle Hamiltonian have one-dimensional spectral subspaces at
energies $j\hbar\omega$, $j\geq1$.  Apply the identities to the finite
spectral truncation and then pass coefficientwise to the limit to derive
\[
\prod_{j\geq1}\frac1{1-q^j}
=\sum_{n\geq0}p(n)q^n,
\qquad
\prod_{j\geq1}(1+q^j)
=\sum_{n\geq0}p_{\rm dist}(n)q^n,
\]
where $p_{\rm dist}(n)$ counts partitions into distinct parts.  Construct
the multiplicity map separating powers of two and prove
\[
\prod_{j\geq1}(1+q^j)
=\prod_{j\geq1}\frac1{1-q^{2j-1}}.
\]
Interpret the coefficient of $q^n$ as the exact degeneracy at excitation
energy $n\hbar\omega$ for the corresponding bosonic or fermionic system.

For the bosonic system put $q=e^{-\beta_{\rm th}\hbar\omega}$, where
$\beta_{\rm th}=(k_BT)^{-1}$, and define
\[
Z(\beta_{\rm th})
=\prod_{j\geq1}\frac1{1-e^{-\beta_{\rm th}j\hbar\omega}},
\qquad
\Psi(\beta_{\rm th})=\log Z(\beta_{\rm th}).
\]
Define
\[
U=-\frac{d\Psi}{d\beta_{\rm th}}.
\]
For
\[
\mathcal S=k_B(\Psi+\beta_{\rm th}U),
\qquad
C=k_B\beta_{\rm th}^2\frac{d^2\Psi}{d\beta_{\rm th}^2},
\]
prove $d\mathcal S=k_B\beta_{\rm th}\,dU$ as $\beta_{\rm th}$ varies and
express $d^2\Psi/d\beta_{\rm th}^2$ as the energy variance.
\end{exercise}

\section{Modular and Moonshine Characters}

\begin{exercise}[Virasoro algebra and ordered descendants]
The Virasoro algebra has generators $L_n$, $n\in\Z$, and a central generator
$C$, with
\[
[L_m,L_n]=(m-n)L_{m+n}
+\frac1{12}(m^3-m)\delta_{m+n,0}C,
\qquad [C,L_n]=0.
\]
A module has central charge $c$ when $C$ acts by $cI$.
A highest-weight vector $|h\rangle$ satisfies
$L_0|h\rangle=h|h\rangle$ and $L_n|h\rangle=0$ for $n>0$.

Order the negative Virasoro modes by their indices.  Use the commutation
relations to rewrite every word as an ordered word.  Check every
three-letter overlap and use the Jacobi identity to prove that the rewriting
is confluent.  Filter by word length and deduce that the associated graded
algebra is the symmetric algebra on the negative modes.  Conclude that the
ordered monomials in the negative modes are linearly independent in the
universal highest-weight module.
\end{exercise}

\begin{exercise}[Verma and vacuum characters]
For $\tau$ in the upper half-plane put $q=e^{2\pi i\tau}$.  If $H_a$ is a
positive-energy module with finite-dimensional $L_0$ spectral subspaces,
define
\[
\chi_a(\tau)=\Tr_{H_a}q^{L_0-c/24}
\]
when the displayed operator is trace class; the same expression denotes
the formal graded trace otherwise.

Use the ordered descendants to prove that the Verma-module character is
\[
\chi_{c,h}^{\rm Verma}(\tau)
=q^{h-c/24}\prod_{m\geq1}(1-q^m)^{-1}.
\]
For the vacuum impose $L_{-1}|0\rangle=0$ and derive the corresponding
generic vacuum product.
\end{exercise}

\begin{exercise}[Null-vector quotients]
Let a singular vector of degree $r$ in a Verma module generate a submodule
isomorphic to the Verma module of highest weight $h+r$.  Prove that the
quotient character is
\[
\chi_{c,h}^{\rm Verma}-\chi_{c,h+r}^{\rm Verma}.
\]
For several singular vectors, prove that their submodules must be combined by
inclusion--exclusion, and show that independent subtraction is valid only
when the intersections of their descendant submodules have been accounted
for.
\end{exercise}

\begin{exercise}[Unitary modular character systems and Ising characters]
A finite character vector $\chi$ is a \emph{unitary modular character
system} when
\[
\chi(-1/\tau)=S\chi(\tau),
\qquad
\chi(\tau+1)=T\chi(\tau)
\]
for unitary matrices $S,T$.  For a matrix $M$ with nonnegative integral
entries define
\[
Z_M(\tau,\bar\tau)
=\sum_{a,b}\overline{\chi_a(\tau)}M_{ab}\chi_b(\tau).
\]
It is \emph{modular invariant} when
\[
S^*MS=M,
\qquad T^*MT=M.
\]

For the Ising character system with $c=1/2$ and weights
$0,1/2,1/16$, choose the square roots whose $q$-expansions have the
displayed positive leading coefficients and put
\[
\begin{aligned}
\chi_0&=\frac12\left(
 \sqrt{\frac{\vartheta_3}{\eta}}+
 \sqrt{\frac{\vartheta_4}{\eta}}\right),\\
\chi_{1/2}&=\frac12\left(
 \sqrt{\frac{\vartheta_3}{\eta}}-
 \sqrt{\frac{\vartheta_4}{\eta}}\right),\\
\chi_{1/16}&=\frac1{\sqrt2}\sqrt{\frac{\vartheta_2}{\eta}}.
\end{aligned}
\]
Use the theta and eta transformations developed in Automorphic Forms to
derive
\[
S=\frac12
\begin{pmatrix}
1&1&\sqrt2\\
1&1&-\sqrt2\\
\sqrt2&-\sqrt2&0
\end{pmatrix},
\qquad
T_{aa}=e^{2\pi i(h_a-c/24)}.
\]
Expand each character as $q^{h-c/24}$ times a power series, identify its
coefficients with exact descendant counts, and verify the modular
invariance of $\sum_a|\chi_a|^2$.
\end{exercise}

\begin{exercise}[Eta asymptotics and bosonic thermodynamics]
Use Exercise~\ref{ex:theta-eta-transformations} to prove
\[
\prod_{j\geq1}(1-e^{-tj})^{-1}
\sim\sqrt{\frac{t}{2\pi}}
\exp\!\left(\frac{\pi^2}{6t}\right)
\qquad(t\downarrow0).
\]
For $t=\beta_{\rm th}\hbar\omega$, differentiate the transformed
eta-product identity
to strengthen this to
\[
\Psi(\beta_{\rm th})
=\frac{\pi^2}{6t}+\frac12\log\frac{t}{2\pi}
-\frac{t}{24}+O(e^{-4\pi^2/t})
\]
and derive
\[
U=\hbar\omega\left(
\frac{\pi^2}{6t^2}-\frac1{2t}+\frac1{24}+o(1)\right),
\]
\[
C=k_B\left(\frac{\pi^2}{3t}-\frac12+o(1)\right),
\qquad
\mathcal S=k_B\left(
\frac{\pi^2}{3t}+\frac12\log\frac{t}{2\pi}-\frac12+o(1)\right).
\]
\end{exercise}

\begin{exercise}[Partition saddle]
Insert this estimate into Cauchy's coefficient integral, locate its positive
real saddle, prove that its equation is
\[
n=\frac{U}{\hbar\omega},
\]
and identify the leading saddle exponent with the Legendre transform
$\mathcal S/k_B$.  Expand the phase through quadratic order and derive the
major-arc contribution
\[
\frac1{4n\sqrt3}
\exp\!\left(\pi\sqrt{\frac{2n}{3}}\right).
\]
\end{exercise}

\begin{exercise}[Partition minor arcs]
On the coefficient circle through the positive saddle, use
\[
|1-re^{i\theta}|^2=(1-r)^2+4r\sin^2(\theta/2)
\]
together with the modular transformation at each rational cusp met by the
circle dissection to bound the complementary arcs by an exponentially smaller
quantity.  Deduce
\[
p(n)\sim\frac1{4n\sqrt3}
\exp\!\left(\pi\sqrt{\frac{2n}{3}}\right).
\]
\end{exercise}

\begin{exercise}[Ising state counts]
For the diagonal Ising system, define nonnegative integers $d_a(N)$ by
\[
\chi_a(\tau)
=q^{h_a-c/24}\sum_{N\geq0}d_a(N)q^N
\]
and define the resolved two-sided degeneracy
\[
\rho_{\rm Ising}(N,\bar N)
=\sum_{a\in\{0,1/2,1/16\}}d_a(N)d_a(\bar N).
\]
Use the explicit theta-product formulas and the $S$-transformation to
control each character in a complex neighborhood of the positive real
saddle.  Bound the complementary arcs as in the partition calculation and
prove, as $N,\bar N\to\infty$ with $N/\bar N$ in a fixed compact subset of
$(0,\infty)$,
\[
\log\rho_{\rm Ising}(N,\bar N)
\sim2\pi\sqrt{\frac{cN}{6}}
+2\pi\sqrt{\frac{c\bar N}{6}},
\qquad c=\frac12.
\]
Derive the resulting asymptotic number of states below a specified high
energy on a circle and the corresponding entropy.  State
the circumference, propagation speed, energy window, vacuum assumption,
sector resolution, saddle-point regime, and finite-size limitations required
when comparing these predictions with measured or transfer-matrix spectra.
\end{exercise}

\begin{exercise}[Lattice Fock characters]

For an even positive-definite lattice $L$ of rank $d$, let
$\varepsilon:L\times L\to\{\pm1\}$ satisfy
\[
\varepsilon(\lambda,0)=\varepsilon(0,\lambda)=1,
\qquad
\varepsilon(\lambda,\mu)\varepsilon(\lambda+\mu,\nu)
=\varepsilon(\mu,\nu)\varepsilon(\lambda,\mu+\nu),
\]
\[
\frac{\varepsilon(\lambda,\mu)}{\varepsilon(\mu,\lambda)}
=(-1)^{\langle\lambda,\mu\rangle}.
\]
Let $\C_\varepsilon[L]$ have symbols $e^\lambda$, $\lambda\in L$, with
$e^\lambda e^\mu=\varepsilon(\lambda,\mu)e^{\lambda+\mu}$, and define
\[
\mathcal F_L
=\operatorname{Sym}\!\left(
\bigoplus_{n\geq1}(L\otimes_\Z\C)t^{-n}\right)
\otimes\C_\varepsilon[L],
\]
where $\operatorname{Sym}(W)=\bigoplus_{r\geq0}\operatorname{Sym}^r(W)$.
Give $(L\otimes_\Z\C)t^{-n}$ degree $n$, give the element $e^\lambda$ of
the twisted group algebra degree $\|\lambda\|^2/2$, and let $L_0$ act by
total degree.

The standard lattice vertex-operator construction equips $\mathcal F_L$ with
a central-charge-$d$ vertex operator algebra structure.
\begin{enumerate}[label=(\alph*)]
\item Use the
symmetric-power trace formula and the definition of the lattice theta series
to prove
\[
\Tr_{\mathcal F_L}q^{L_0-d/24}
=\frac{\Theta_L(\tau)}{\eta(\tau)^d}.
\]
\item For the Leech lattice, use Exercise~\ref{ex:leech-theta-series} to derive
\[
\Tr_{\mathcal F_\Lambda}q^{L_0-1}
=\frac{\Theta_\Lambda}{\eta^{24}}=J+24.
\]
\item Show directly from the graded construction that the coefficient $24$ counts
the degree-one oscillator states and that
\[
196884=196560+324,
\qquad
324=24+\dim\operatorname{Sym}^2(\Lambda\otimes_\Z\C),
\]
splits the degree-two states into nonzero lattice momenta and oscillator
states.
\end{enumerate}
\end{exercise}

\begin{exercise}[McKay--Thompson series and monstrous moonshine]
Let $V=\bigoplus_{m\geq0}V_m$ be a positive-energy vertex operator algebra
with finite-dimensional graded pieces and central charge $c$, and let
$L_0|_{V_m}=m\id$.  Let $G$ act by grading-preserving automorphisms of $V$.
For $g\in G$, define its \emph{McKay--Thompson series} by
\[
T_g(\tau)=\Tr_V\!\left(gq^{L_0-c/24}\right).
\]
Let $\mathbb M$ denote the Monster sporadic simple group.

Take the Frenkel--Lepowsky--Meurman construction and
\href{https://doi.org/10.1007/BF01232032}{Borcherds's monstrous moonshine
theorem} as theorem input.  If $\theta$ lifts $\lambda\mapsto-\lambda$ to the
Leech-lattice vertex operator algebra and $V_\Lambda^{T,+}$ is the
$+1$ eigenspace in its $\theta$-twisted module, the orbifold
\[
V^\natural=V_\Lambda^+\oplus V_\Lambda^{T,+}
\]
is a central-charge-$24$ vertex operator algebra with
\[
\operatorname{Aut}(V^\natural)=\mathbb M,
\qquad
T_1(\tau)=J(\tau),
\]
and, for every $g\in\mathbb M$, the series $T_g$ is the normalized Hauptmodul
for the Conway--Norton genus-zero group $\Gamma_g$.  Deduce
\[
\dim V^\natural_0=1,
\qquad
\dim V^\natural_1=0,
\qquad
\dim V^\natural_2=196884,
\qquad
\dim V^\natural_3=21493760.
\]
If $\mathbf d$ denotes the indicated irreducible Monster representation of
dimension $d$, verify the first decompositions supplied by the Monster
character table,
\[
V^\natural_2\cong\mathbf1\oplus\mathbf{196883},
\qquad
V^\natural_3\cong
\mathbf1\oplus\mathbf{196883}\oplus\mathbf{21296876}.
\]
With $p=e^{2\pi i\sigma}$, use Borcherds's denominator identity
\[
p^{-1}\prod_{m>0,\ n\in\Z}
(1-p^mq^n)^{c(mn)}=J(\sigma)-J(\tau)
\]
as a formal product in the region $|p|<|q|<1$.  Compare coefficients through
order $p$ and derive the first replication identity
\[
\frac12\left(J(\tau)^2-\sum_{n\in\Z}c(n)q^{2n}\right)
-\sum_{n\in\Z}c(2n)q^n=196884.
\]
\end{exercise}

\begin{exercise}[Monster-resolved finite-size spectrum]
Let $\chi_\rho$ be the character of an irreducible Monster representation
$\rho$.  Use finite-group character orthogonality to prove that its
multiplicity in $V_m^\natural$ is
\[
\mu_\rho(m)=\frac1{|\mathbb M|}\sum_{g\in\mathbb M}
\overline{\chi_\rho(g)}\,[q^{m-1}]T_g(\tau).
\]
Deduce that the symmetry-resolved series
\[
Z_\rho(\tau)=\frac{\dim(\rho)}{|\mathbb M|}
\sum_{g\in\mathbb M}\overline{\chi_\rho(g)}T_g(\tau)
\]
has coefficient $\dim(\rho)\mu_\rho(m)$ at $q^{m-1}$.
Model an exactly Monster-symmetric chiral edge of circumference $\ell$ and
propagation speed $u$ by
\[
H=\frac{2\pi\hbar u}{\ell}(L_0-1).
\]
Derive the excitation energies
\[
E_m-E_0=\frac{2\pi\hbar u}{\ell}m
\]
and predict a missing level at $2\pi\hbar u/\ell$, followed by levels of
degeneracy $196884$ and $21493760$ at $4\pi\hbar u/\ell$ and
$6\pi\hbar u/\ell$.  Express the symmetry-resolved degeneracy at level $m$
as $\dim(\rho)\mu_\rho(m)$.  State the assumptions of exact Monster symmetry,
linear chiral dispersion, periodic boundary conditions, access to the twined
sectors, and the low-energy finite-size regime.  Explain why a
purely chiral central-charge-$24$ theory must be realized as an anomalous edge
or accompanied by a compensating opposite-chirality sector, and distinguish
the derived spectral prediction from evidence for a particular physical
realization.
\end{exercise}

\section{Transfer Matrices}

\begin{exercise}[Hard-hexagon transfer matrices]
Let $\mathcal R_N$ be a finite set of allowed width-$N$ configurations and
let $T_N(z)$ be an operator on the configuration space whose matrix element
between $a,b\in\mathcal R_N$ is the weight of adjoining $b$ to $a$.  For a
height-$M$ periodic lattice define
\[
Z_{N,M}(z)=\Tr(T_N(z)^M).
\]
When $T_N(z)$ has nonnegative entries, write $\lambda_N(z)$ for its
spectral radius.

Let
\[
\mathcal R_N
=\{a\in\{0,1\}^{\Z/N\Z}:a_i a_{i+1}=0\text{ for every }i\}.
\]
Two rows $a,b\in\mathcal R_N$ are compatible when
\[
a_i b_i=0,
\qquad a_i b_{i+1}=0
\quad(i\in\Z/N\Z).
\]
For $z\geq0$, with $|a|=\sum_i a_i$, define
\[
T_N(z)_{ab}
=\mathbf1_{\{a,b\text{ compatible}\}}
z^{(|a|+|b|)/2}.
\]

Identify the horizontal, vertical, and diagonal nearest-neighbor edges
encoded by the two compatibility conditions.  Prove
\[
\Tr(T_N(z)^M)
=\sum_I z^{|I|},
\]
where the sum is over independent vertex sets of the specified triangular
$N\times M$ torus.  Prove that every coefficient is an exact state count.
Let $0$ denote the empty row.  For $z>0$, prove
\[
(T_N(z)^2)_{ab}\geq T_N(z)_{a0}T_N(z)_{0b}>0.
\]
Apply the Perron--Frobenius theorem to show that $\lambda_N(z)$ is simple,
that every other spectral value has smaller modulus, and that
\[
\lim_{M\to\infty}\frac1{NM}\log Z_{N,M}(z)
=\frac1N\log\lambda_N(z).
\]
Put $\eta=\log z=\beta_{\rm th}\mu$ and
\[
\Psi_{N,M}(\eta)=\log Z_{N,M}(e^\eta).
\]
Give the independent sets the probability
$\mathbb P_\eta(I)=e^{\eta|I|}/Z_{N,M}(e^\eta)$ and write
$\mathbb E_\eta$ for expectation with respect to this measure.  For a
square-integrable random variable $Y$, write
$\operatorname{Var}_\eta(Y)=\mathbb E_\eta|Y-\mathbb E_\eta Y|^2$.
Set $A=\partial_\eta\Psi_{N,M}$ and prove
\[
A=\mathbb E_\eta|I|,
\qquad
\partial_\eta A=\operatorname{Var}_\eta(|I|),
\]
and derive the mean occupied fraction and isothermal finite-size
susceptibility
\[
\rho_{N,M}=\frac{A}{NM},
\qquad
\chi_{N,M}=\frac{\partial\rho_{N,M}}{\partial\mu}
=\frac{\beta_{\rm th}}{NM}\operatorname{Var}_\eta(|I|).
\]
State the lattice
geometry, boundary conditions, and finite $N,M$ limitations required when
these quantities are compared with adsorption or exclusion-process data.
\end{exercise}

\begin{exercise}[The planar Ising model and Kramers--Wannier duality]
Let $G=(V,E)$ be a finite connected graph with a fixed planar embedding.
For couplings $K_e>0$, define
\[
Z_G^{\mathrm f}(K)
=\sum_{s:V\to\{-1,1\}}
\exp\!\left(\sum_{e=uv}K_es_us_v\right).
\]
Let $G^*$ be the planar dual, fix the spin of its outer-face vertex to
$+1$, and denote the corresponding partition function by
$Z_{G^*}^{+}(K^*)$.  Let $\mathcal E(G)$ be the family of edge subsets in
which every vertex has even degree.  The dual couplings are determined by
\[
e^{-2K_{e^*}^*}=\tanh K_e.
\]

Expand each edge factor as
\[
e^{K_es_us_v}=\cosh K_e(1+s_us_v\tanh K_e)
\]
and prove
\[
Z_G^{\mathrm f}(K)
=2^{|V|}\prod_{e\in E}\cosh K_e
\sum_{\gamma\in\mathcal E(G)}
\prod_{e\in\gamma}\tanh K_e.
\]
Identify $\mathcal E(G)$ with the domain walls of dual spin configurations
whose outer-face spin is fixed, and deduce the exact planar duality
\[
Z_G^{\mathrm f}(K)
=2^{|V|}\prod_{e\in E}\bigl(\cosh K_e\,e^{-K_{e^*}^*}\bigr)
Z_{G^*}^{+}(K^*).
\]
For horizontal and vertical square-lattice couplings $K_h,K_v$, show that
the duality interchanges the directions and that its fixed locus is
\[
\sinh(2K_h)\sinh(2K_v)=1.
\]
Assuming the infinite-volume ferromagnetic model has a single transition
and that duality interchanges its high- and low-temperature phases, deduce
in the isotropic case
\[
K_c=\frac12\log(1+\sqrt2).
\]
State separately the exact finite-graph identity, the infinite-volume
assumptions, and the conclusion about the critical temperature.
\end{exercise}

\begin{exercise}[Ising star--triangle transformation]
For $K_1,K_2,K_3>0$ and boundary spins $s_1,s_2,s_3$, prove that there are
unique $A>0$ and real $L_{12},L_{23},L_{31}$ satisfying
\[
\sum_{s_0=\pm1}
e^{K_1s_0s_1+K_2s_0s_2+K_3s_0s_3}
=A e^{L_{12}s_1s_2+L_{23}s_2s_3+L_{31}s_3s_1}.
\]
Put
\[
\begin{aligned}
w_0&=2\cosh(K_1+K_2+K_3),&
w_1&=2\cosh(-K_1+K_2+K_3),\\
w_2&=2\cosh(K_1-K_2+K_3),&
w_3&=2\cosh(K_1+K_2-K_3).
\end{aligned}
\]
Derive
\[
A=(w_0w_1w_2w_3)^{1/4},
\]
\[
e^{4L_{12}}=\frac{w_0w_3}{w_1w_2},\qquad
e^{4L_{23}}=\frac{w_0w_1}{w_2w_3},\qquad
e^{4L_{31}}=\frac{w_0w_2}{w_1w_3}.
\]
Show that replacing every star of one sublattice of the honeycomb model
by a triangle changes the partition function only by the product of the
local factors $A$.  In the isotropic case derive
\[
e^{4L}=\frac{\cosh(3K)}{\cosh K}.
\]
Combine this transformation with planar duality and, under the
single-transition hypothesis of the preceding exercise, obtain
\[
K_c^{\rm tri}=\frac14\log3,
\qquad
K_c^{\rm hex}=\frac12\log(2+\sqrt3).
\]
Identify the coupling normalization and lattice geometry needed when these
critical temperatures are compared with an Ising magnet or a programmable
spin array.
\end{exercise}

\begin{exercise}[Ising transfer matrices and free fermions]
Let $\mathcal S_N=\{-1,1\}^{\Z/N\Z}$ and $K_h,K_v>0$.  For row
configurations $s,t\in\mathcal S_N$ define
\[
T_N^{\rm Is}(K_h,K_v)_{s,t}
=\exp\!\left[
\frac{K_h}{2}\sum_i(s_is_{i+1}+t_it_{i+1})
+K_v\sum_i s_it_i\right].
\]
Let $R_N$ be cyclic row translation.  On
$\mathcal H_N=\ell^2(\mathcal S_N)$ let
\[
(Z_jf)(s)=s_jf(s),
\qquad
(X_jf)(s)=f(s^{(j)}),
\qquad
Y_j=iX_jZ_j,
\]
where $s^{(j)}$ is obtained by reversing $s_j$.  Put
\[
P=\prod_{j=1}^NX_j,
\qquad
c_{2j-1}=\left(\prod_{\ell<j}X_\ell\right)Z_j,
\qquad
c_{2j}=\left(\prod_{\ell<j}X_\ell\right)Y_j.
\]
Let $K_v^*>0$ satisfy $e^{-2K_v^*}=\tanh K_v$, and set
\[
\mathcal K_+=\left\{\frac{(2m+1)\pi}{N}:0\leq m<N\right\},
\qquad
\mathcal K_-=\left\{\frac{2m\pi}{N}:0\leq m<N\right\}.
\]
Write $a_x$ and $a_t$ for the spatial and transfer-step lattice spacings
and $L=Na_x$ for the circumference.

\begin{enumerate}[label=(\alph*)]
\item Verify
\[
R_NT_N^{\rm Is}(K_h,K_v)
=T_N^{\rm Is}(K_h,K_v)R_N.
\]
\item Prove
\[
T_N^{\rm Is}(K_h,K_v)
=(2\sinh(2K_v))^{N/2}
e^{\frac{K_h}{2}\sum_jZ_jZ_{j+1}}
e^{K_v^*\sum_jX_j}

\section{Determinantal Ensembles}

\begin{exercise}[Hermite Gaussian moments]
Let $\mathbb P$ be the probability measure on $\R$ with density
$(2\pi)^{-1/2}e^{-x^2/2}$, let $X(x)=x$, and write
$\mathbb Ef(X)=\int_\R f(x)\,d\mathbb P(x)$ whenever the integral exists.

For this $X$, prove
\[
\mathbb E[e^{tX}]=e^{t^2/2}.
\]
Define the probabilists' Hermite polynomials by
\[
e^{xt-t^2/2}
=\sum_{n\geq0}\operatorname{He}_n(x)\frac{t^n}{n!}
\]
and prove
\[
\mathbb E[\operatorname{He}_m(X)\operatorname{He}_n(X)]
=n!\delta_{mn},
\qquad
\mathbb E[X^{2n}]=\frac{(2n)!}{2^nn!}.
\]
Relate $\operatorname{He}_n$ to the oscillator Hermite functions developed
in Spectral Theory.
\end{exercise}

\begin{exercise}[Dyson ensembles and unitary orbit Jacobians]
For $\mathbb F=\R,\C,\mathbb H$, put
$\beta_{\rm D}=\dim_\R\mathbb F\in\{1,2,4\}$.  The Gaussian Dyson ensemble on the
self-adjoint operators on $\mathbb F^N$ has density
\[
d\mathbb P_{N,\beta_{\rm D}}(H)
=C_{N,\beta_{\rm D}}\exp\!\left[-\frac{\beta_{\rm D}N}{4\sigma^2}
\Tr(H^2)\right]dH
\]
for the invariant Hilbert--Schmidt volume.  For
$\lambda=(\lambda_1,\ldots,\lambda_N)$ define
\[
\Delta(\lambda)=\det[\lambda_i^{j-1}]_{i,j=1}^N.
\]

Use the exterior-power determinant to prove
\[
\Delta(\lambda)=\prod_{i<j}(\lambda_j-\lambda_i).
\]
Compute the Jacobian of the unitary conjugation-orbit map on the regular
spectral locus, including the stabilizer and permutation factors, and derive
\[
p_{N,2}(\lambda)
=\frac1{Z_{N,2}}
e^{-N\sum_i\lambda_i^2/(2\sigma^2)}
|\Delta(\lambda)|^2.
\]
\end{exercise}

\begin{exercise}[Dyson level repulsion]
For the real and quaternionic classes, show that each off-diagonal spectral
direction has real multiplicity $\beta_{\rm D}$ and deduce

p_{N,\beta_{\rm D}}(\lambda)
=\frac1{Z_{N,\beta_{\rm D}}}
e^{-\beta_{\rm D}N\sum_i\lambda_i^2/(4\sigma^2)}
|\Delta(\lambda)|^{\beta_{\rm D}}.

For $2\times2$ GOE and GUE, separate the trace from the traceless operator,

P_1(s)=\frac\pi2s e^{-\pi s^2/4},
\qquad
P_2(s)=\frac{32}{\pi^2}s^2e^{-4s^2/\pi}.
\]
Identify the exponent $\beta_{\rm D}$ as the small-gap repulsion prediction and
state the symmetry-sector separation required in spectral data.
\end{exercise}

\begin{exercise}[Fredholm determinants and trace identities]
Let $A$ be a positive compact operator on a Hilbert space.  It is
\emph{trace class} when its nonzero spectral values, repeated with
multiplicity, satisfy $\sum_j\lambda_j(A)<\infty$.  Define
\[
\det(I+zA)=\prod_j(1+z\lambda_j(A)),
\qquad
\Tr(A^m)=\sum_j\lambda_j(A)^m.
\]

Prove absolute local uniform convergence of the defining product and
\[
\det(I-zA)
=\exp\!\left(-\sum_{m\geq1}\frac{z^m}{m}\Tr(A^m)\right)
\qquad(|z|<\|A\|^{-1}).
\]
Prove that the determinant is entire, identify its zeros, and show that its
finite-rank specialization agrees with the exterior-power identity from the
opening section.
\end{exercise}

\begin{exercise}[Point processes and determinantal statistics]
Let $X$ be a locally compact second-countable Hausdorff space and let $\mu$
be a Radon measure on $X$.  A simple point process is a random
locally finite counting measure $\Xi=\sum_i\delta_{x_i}$ with distinct
points almost surely.  When its $k$th factorial moment measure is absolutely
continuous with respect to $\mu^{\otimes k}$, its density $\rho_k$ is
defined, for every nonnegative compactly supported measurable function $f$,
by
\[
\mathbb E\sum_{i_1,\ldots,i_k}^{\ne}
f(x_{i_1},\ldots,x_{i_k})
=\int_{X^k}f\rho_k\,d\mu^{\otimes k}.
\]
A \emph{determinantal point process} with measurable Hermitian kernel $K$
is one for which
\[
\rho_k(x_1,\ldots,x_k)=\det[K(x_i,x_j)]_{i,j=1}^k.
\]
For the existence and counting formulas below, the integral operator defined
by $K$ is a locally trace-class positive contraction on $L^2(X,\mu)$.

For a relatively compact measurable set $B$, put $N_B=\Xi(B)$ and
$K_B=\mathbf1_BK\mathbf1_B$.  Use factorial moments and the Fredholm series
to prove
\[
\mathbb E[z^{N_B}]=\det(I+(z-1)K_B),
\qquad
\mathbb P(N_B=0)=\det(I-K_B).
\]
Random variables are independent when their joint law is the product of
their marginal laws.
Prove that $N_B$ is distributed as a sum of independent Bernoulli variables
whose parameters are the eigenvalues of $K_B$.
\end{exercise}

\begin{exercise}[Unitary polynomial ensembles and Christoffel--Darboux kernels]
Let $\mu$ be a Radon measure on $\R$ such that $w(x)d\mu(x)$ has infinite
support and finite moments, and let $P_j$ be its monic orthogonal
polynomials, normalized by
\[
\int P_j(x)P_k(x)w(x)d\mu(x)=H_j\delta_{jk},
\qquad H_j>0.
\]
The associated unitary polynomial ensemble has density, with respect to
$\mu^{\otimes N}$,
\[
\mathfrak p_N(x_1,\ldots,x_N)
=\frac1{N!\prod_{j=0}^{N-1}H_j}
\Delta(x)^2\prod_{i=1}^Nw(x_i)
\]
and kernel
\[
K_N(x,y)=\sqrt{w(x)w(y)}
\sum_{j=0}^{N-1}\frac{P_j(x)P_j(y)}{H_j}.
\]

Prove Andr\'eief's identity
\[
\int_{X^N}\det[f_i(x_j)]\det[g_i(x_j)]\prod_jd\mu(x_j)
=N!\det\!\left[\int_Xf_i g_j\,d\mu\right].
\]
Deduce that the unitary polynomial ensemble is determinantal with kernel
$K_N$.  Derive the Christoffel--Darboux formula from the three-term
recurrence and prove
\[
\int K_N(x,z)K_N(z,y)d\mu(z)=K_N(x,y),
\qquad
\int K_N(x,x)d\mu(x)=N.
\]
\end{exercise}

\begin{exercise}[Hermite steepest descent]
Write $\psi_j^{\rm H}$ for the normalized oscillator Hermite functions

Derive a contour integral for the normalized Hermite functions from their

\begin{exercise}[GUE bulk statistics]
For $E\in(-2\sigma,2\sigma)$, define
\[
\rho_{\rm sc}(E)=\frac1{2\pi\sigma^2}\sqrt{4\sigma^2-E^2},
\qquad
K_{\sin}(u,v)=\frac{\sin\pi(u-v)}{\pi(u-v)}.
\]
Define
\[
\phi_{j,N}(x)
=\left(\frac{N}{2\sigma^2}\right)^{1/4}
\psi_j^{\rm H}\!\left(\sqrt{\frac{N}{2\sigma^2}}x\right),
\qquad
K_N^{\rm GUE}(x,y)=\sum_{j=0}^{N-1}\phi_{j,N}(x)\phi_{j,N}(y).
\]

Insert the bulk Hermite asymptotics into the Christoffel--Darboux identity
and prove
\[
\frac{1}{N\rho_{\rm sc}(E)}
K_N^{\rm GUE}\!\left(
E+\frac{u}{N\rho_{\rm sc}(E)},
E+\frac{v}{N\rho_{\rm sc}(E)}\right)
\longrightarrow K_{\sin}(u,v)
\]
locally uniformly.  For the limiting sine process prove

\operatorname{Var}N_{[0,L]}
=\int_0^LK_{\sin}(u,u)\,du
-\int_0^L\!\int_0^L|K_{\sin}(u,v)|^2\,du\,dv
=\frac1{\pi^2}\log L+O(1).
\]
State the unfolding, symmetry-sector separation, and finite-window
assumptions required when testing the spacing and number-variance predictions
with resonance or Hamiltonian spectra.
\end{exercise}

\begin{exercise}[GUE edge statistics]
Define

K_{\rm Ai}(u,v)
=\frac{\operatorname{Ai}(u)\operatorname{Ai}'(v)

\qquad
F_2(s)=\det(I-K_{\rm Ai})_{L^2(s,\infty)},

Insert the turning-point asymptotics into the Christoffel--Darboux identity

2\sigma+\frac{\sigma v}{N^{2/3}}\right)
\longrightarrow K_{\rm Ai}(u,v).

Use the exponentially weighted bounds to pass to the Fredholm series and

\end{exercise}

\begin{exercise}[Painlev\'e form of the Tracy--Widom law]
The Hastings--McLeod function is the real pole-free solution $q$ of
\[
q''(s)=s q(s)+2q(s)^3,
\qquad q(s)\sim\operatorname{Ai}(s)\quad(s\to+\infty).
\]

Let $K_s$ be the Airy-kernel operator on $L^2((s,\infty))$ and define
\[
Q_s=(I-K_s)^{-1}\operatorname{Ai},
\qquad
P_s=(I-K_s)^{-1}\operatorname{Ai}'.
\]
With the inner product on $L^2((s,\infty))$, put
\[
q(s)=Q_s(s),\qquad p(s)=P_s(s),
\qquad
u(s)=\langle Q_s,\operatorname{Ai}\rangle,
\qquad
v(s)=\langle P_s,\operatorname{Ai}\rangle.
\]
Use the integrable form of $K_{\rm Ai}$, the Airy equation, and
differentiation of the moving endpoint to prove
\[
q'=p-qu,
\qquad
p'=sq+pu-2qv,
\qquad
u'=-q^2,
\qquad
v'=-pq.
\]
Prove that $u^2-2v-q^2$ is constant and vanishes as $s\to+\infty$.
Deduce
\[
q''=sq+2q^3,
\qquad
q(s)\sim\operatorname{Ai}(s)\quad(s\to+\infty).
\]
Use the positive integral representation
\[
K_{\rm Ai}(x,y)=\int_0^\infty
\operatorname{Ai}(x+t)\operatorname{Ai}(y+t)\,dt
\]
to prove that $I-K_s$ is invertible for every real $s$.  Conclude that the
resolvent construction gives the real pole-free Hastings--McLeod solution.
For the resolvent $R_s=K_s(I-K_s)^{-1}$, prove
\[
\frac d{ds}\log\det(I-K_s)=R_s(s,s)=u(s)
=\int_s^\infty q(x)^2\,dx.
\]
Conclude that the Fredholm and Painlev\'e definitions agree:
\[
F_2(s)
=\exp\!\left[-\int_s^\infty(x-s)q(x)^2\,dx\right].
\]
Use the Airy asymptotic of the Hastings--McLeod solution to derive
\[
1-F_2(s)
\sim\frac{e^{-4s^{3/2}/3}}{16\pi s^{3/2}}
\quad(s\to+\infty).
\]
State how exceedance counts for centered and edge-scaled extreme eigenvalues
or resonance levels test this right-tail prediction, including the
finite-size range over which the asymptotic is used.
\end{exercise}

\section{Schur Measures}

\begin{exercise}[Schur polynomials and identities]
An integer partition is a finite nonincreasing sequence of positive
integers.  Its size is the sum of its parts, and its length is the number
of nonzero parts.

For a partition $\lambda$ of length at most $N$, define
\[
s_\lambda(x_1,\ldots,x_N)
=\det\!\left[h_{\lambda_i-i+j}\right]_{i,j=1}^N.
\]

Derive the alternant formula
\[
s_\lambda(x)
=\frac{\det[x_i^{\lambda_j+N-j}]_{i,j=1}^N}
{\det[x_i^{N-j}]_{i,j=1}^N}
\]
from the fundamental symmetric identities.  Prove
$s_{(r)}=h_r$ and $s_{(1^r)}=e_r$.
\end{exercise}

\begin{exercise}[RSK, last-passage values, and the finite Schur measure]
A semistandard Young tableau has weakly increasing rows and strictly
increasing columns.  To insert an integer $x$, replace the first entry
strictly larger than $x$ in the first row and insert the replaced entry in
the next row; append when no such entry exists.  For
$W=(W_{ij})\in M_{m,n}(\Z_{\geq0})$, form the lexicographically ordered
two-line word containing $W_{ij}$ copies of $(i,j)$, insert the lower
letters, and record the upper letters in the created boxes.  The resulting
pair $(P,Q)$ of tableaux of common shape $\lambda$ is the RSK image of
$W$.

For nonnegative weights $W=(W_{ij})$, define
\[
G(m,n)=\max_{\pi:(1,1)\nearrow(m,n)}
\sum_{(i,j)\in\pi}W_{ij}.
\]
For parameters $\alpha_i,\gamma_j\geq0$ with
$\alpha_i\gamma_j<1$, the geometric model has independent weights
\[
\mathbb P(W_{ij}=k)
=(1-\alpha_i\gamma_j)(\alpha_i\gamma_j)^k,
\qquad k\in\Z_{\geq0}.
\]

Prove that row insertion is reversible and hence that RSK is a bijection.
Show that the length of the first row of its shape is the maximal weight of
an up-right path:
\[
\lambda_1=G(m,n).
\]
For variables $\alpha=(\alpha_1,\ldots,\alpha_m)$, prove the tableau formula
\[
s_\lambda(\alpha)=\sum_T\prod_i\alpha_i^{m_i(T)}.
\]
Use the contents of the two RSK tableaux and the geometric weight law to
derive
\[
\mathbb P(\lambda)
=\prod_{i,j}(1-\alpha_i\gamma_j)
s_\lambda(\alpha)s_\lambda(\gamma)
\]
and the Cauchy identity
\[
\sum_\lambda s_\lambda(\alpha)s_\lambda(\gamma)
=\prod_{i,j}(1-\alpha_i\gamma_j)^{-1}.
\]
\end{exercise}

\begin{exercise}[Interlacing partitions and Schur paths]
For partitions $\mu$ and $\lambda$, write $\mu\prec\lambda$ when
\[
\lambda_1\geq\mu_1\geq\lambda_2\geq\mu_2\geq\cdots.
\]

Prove the branching formula
\[
s_\lambda(\alpha_1,\ldots,\alpha_m)
=\sum_{\varnothing=\lambda^{(0)}\prec\cdots\prec\lambda^{(m)}=\lambda}
\prod_{k=1}^m
\alpha_k^{|\lambda^{(k)}|-|\lambda^{(k-1)}|}.
\]
Encode each interlacing chain by nonintersecting paths on the integer lattice.
For a finite directed acyclic graph with compatible sources $u_i$ and sinks
$v_i$, let $Z(u_i,v_j)$ be the weighted sum of paths from $u_i$ to $v_j$.
Use the first intersection and tail exchange as a sign-reversing involution
to prove
\[
\sum_{\substack{(\pi_1,\ldots,\pi_r)\\
\text{vertex-disjoint}}}
\prod_i\operatorname{wt}(\pi_i)
=\det[Z(u_i,v_j)]_{i,j=1}^r.
\]
Apply this identity to the interlacing-chain encoding of $s_\lambda$.
\end{exercise}

\begin{exercise}[Finite Schur kernels]
For the preceding finite alphabets, set
\[
\Phi(z)=\frac{\prod_j(1-\gamma_j/z)}
{\prod_i(1-\alpha_i z)}.
\]
Choose concentric contours with
\[
\max_j\gamma_j<|w|<|z|<\min_i\alpha_i^{-1}.
\]
For $r,s\in\Z+\tfrac12$, define
\[
K_{\alpha,\gamma}(r,s)
=\frac1{(2\pi i)^2}\oint\!\oint
\frac{\Phi(z)}{\Phi(w)}\frac{1}{z-w}
z^{-r-1/2}w^{s-1/2}\,dz\,dw.
\]

Encode $\lambda$ by
\[
\mathcal X(\lambda)
=\{\lambda_i-i+\tfrac12:i\geq1\}.
\]
Glue the two nonintersecting path ensembles associated with
$s_\lambda(\alpha)$ and $s_\lambda(\gamma)$ along their common boundary
$\lambda$.  Apply the Andr\'eief identity successively to prove that the
occupation correlations of the glued ensemble are determinants.  Evaluate
the resulting one-particle propagators by coefficient extraction and derive
the kernel $K_{\alpha,\gamma}$.  Deduce the exact finite Fredholm formula
\[
\mathbb P(G(m,n)\leq L)
=\det(I-K_{\alpha,\gamma})_{
\ell^2(\{L+\tfrac12,L+\tfrac32,\ldots\})}.
\]
State how repeated last-passage or growth experiments with the specified
geometric weights test this distribution without invoking an asymptotic
limit.
\end{exercise}

\begin{exercise}[Exponential last-passage percolation]
Let $E_{ij}$ be independent rate-one exponential random variables and define

G_{\exp}(m,n)=\max_{\pi:(1,1)\nearrow(m,n)}

\]

The variables $E_{ij}$ are independent when their joint law is the product
of their marginal laws.
For random variables $Y_\varepsilon,Y$, write $Y_\varepsilon\Longrightarrow Y$
when their laws converge against bounded continuous test functions.
Set $\alpha_i=\gamma_j=e^{-\varepsilon/2}$ in the $n\times n$ geometric
model and prove that $\varepsilon W_{ij}$ converges in distribution to a
rate-one exponential variable as $\varepsilon\downarrow0$.  Take this limit
in the finite Schur Fredholm formula and identify the resulting finite-$n$
kernel with the Laguerre Christoffel--Darboux kernel.
\end{exercise}

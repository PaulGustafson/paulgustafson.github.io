\section{Unitary Ensembles}

\begin{exercise}[Unitary symmetry and spectral measures]
On the real inner-product space of Hermitian endomorphisms of a
finite-dimensional Hermitian space $W$, let
$U(W)$ act by conjugation.  For a real confining function $\mathcal V$, prove
that
\[
 \dd\mathbb P(H)=Z^{-1}e^{-\Tr(\mathcal V(H))}\dd H
\]
is invariant whenever it is integrable.  Disintegrate this measure along
conjugacy orbits and identify the unordered eigenvalue map without choosing
eigenvectors.
\end{exercise}

\begin{exercise}[Weyl integration and eigenvalue repulsion]
Compute the normal Jacobian of the conjugation orbit through a simple-spectrum
Hermitian operator and prove the Weyl integration formula
\[
 \int F(H)\dd H=C_n\int_{\R^n}F(\lambda)
 \prod_{i<j}(\lambda_i-\lambda_j)^2\prod_i\dd\lambda_i
\]
for conjugation-invariant $F$.  Determine $C_n$ from a Gaussian integral.
Show that the squared Vandermonde forces quadratic repulsion at collisions and
rewrite the eigenvalue density as a determinant using monic polynomials
orthogonal for $e^{-\mathcal V(x)}\dd x$.
\end{exercise}

\begin{exercise}[Orthogonal-polynomial and Christoffel--Darboux kernels]
Let $p_k$ be the monic orthogonal polynomials for
$e^{-\mathcal V(x)}\dd x$, assumed positive with all moments, and let
$h_k=\norm{p_k}^2$.  Derive the three-term recurrence and
prove the Christoffel--Darboux identity
\[
 \sum_{k=0}^{n-1}\frac{p_k(x)p_k(y)}{h_k}
 =\frac{p_n(x)p_{n-1}(y)-p_{n-1}(x)p_n(y)}{h_{n-1}(x-y)}.
\]
For $\phi_k=p_ke^{-\mathcal V/2}/\sqrt{h_k}$ define
$K_n(x,y)=\sum_{k<n}\phi_k(x)\phi_k(y)$.  Apply the Andreief determinant
identity to prove that every $r$-point eigenvalue correlation is
$\det(K_n(x_i,x_j))_{i,j\le r}$ and that $K_n$ is an orthogonal-projection
kernel.
\end{exercise}

\begin{exercise}[Hermite, sine, and Airy kernels]
For $\mathcal V(x)=x^2$, identify $p_k$ with the Hermite polynomials already developed
for the oscillator and compute their normalization.  Use steepest descent in
their contour representation to derive the bulk and soft-edge limits
\[
 \frac1{\rho_n(E)}K_n\!\left(E+\frac u{\rho_n(E)},
 E+\frac v{\rho_n(E)}\right)\longrightarrow
 \frac{\sin\pi(u-v)}{\pi(u-v)},
\]
\[
 \frac1{c_n}K_n\!\left(a_n+\frac u{c_n},a_n+\frac v{c_n}\right)
 \longrightarrow K_{\rm Ai}(u,v)
 =\frac{\operatorname{Ai}(u)\operatorname{Ai}'(v)
 -\operatorname{Ai}'(u)\operatorname{Ai}(v)}{u-v}.
\]
Define
$\operatorname{Ai}(u)=(2\pi\ii)^{-1}\int_{\mathcal C}
e^{t^3/3-ut}\dd t$ on the standard Airy contour and derive both
real asymptotic regimes and its differential equation.
\end{exercise}

\begin{exercise}[Fredholm determinants and quantum-chaotic spectra]
Define the singular values of a compact operator from the spectral theorem
for $|A|=(A^*A)^{1/2}$ and call $A$ trace class when their sum is finite.
With no positivity or self-adjointness assumption, define
\[
 \det(1+A)=\sum_{r\ge0}\Tr(\wedge^rA)
\]
and prove absolute convergence, multiplicativity, and trace-norm continuity.
For a determinantal process prove the gap formula
$\mathbb P(N_D=0)=\det(1-1_DK1_D)$.  In the sine-kernel limit derive
$R_2(s)=1-(\sin\pi s/(\pi s))^2=\pi^2s^2/3+O(s^4)$; at the edge define
$F_2(s)=\det(1-K_{\rm Ai})_{L^2(s,\infty)}$ and derive its Painlev\'e-II
representation.  For a quantum system whose unfolded spectrum has no
antiunitary symmetry, formulate the predictions of quadratic level repulsion,
the sine-kernel two-point curve, and the $F_2$ largest-level law, separating
the random-matrix modeling hypothesis from the measured spacings.
\end{exercise}

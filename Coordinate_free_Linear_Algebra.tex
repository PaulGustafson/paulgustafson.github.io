\section{Algebra}

Linear algebra studies linear maps between vector spaces. A \textbf{vector
space} $V$ over a field $k$ is a set equipped with vector addition and scalar
multiplication operations satisfying straightforward compatibility conditions.

A \textbf{linear map} $f: V \to U$ between $k$-vector spaces is a map of underlying sets such that $f(\alpha v + \beta w) = \alpha f(v) + \beta f(w)$ for all $\alpha, \beta \in k$ and $v, w \in V$.

\begin{exercise}[Small string vibrations obey superposition]
\label{ex:wave-superposition}\leavevmode\par
\textit{Derive the wave equation starting from a taut string and show that
its solutions form a vector space. Show that time evolution from initial data
to the state at time $t$ is a linear map.}
\end{exercise}

The \textbf{free $k$-vector space on a set $X$}, denoted $kX$, is the set of finitely supported set-theoretical functions $X \to k$ with pointwise addition and scalar multiplication. The \textbf{direct sum} vector space $V \oplus W$ is the set $V \times W$ with pointwise addition and scalar multiplication defined by $\alpha (v \oplus w) = \alpha v \oplus \alpha w$. The \textbf{quotient} of $V$ by a subspace $W$, denoted $V / W$, is the set of equivalence classes of $V$ modulo the relation $x \sim y$ iff $x - y \in W$. The \textbf{tensor product} $V \otimes W$ is the vector space $k(V \times W) / N$ where $N$ is the subspace generated by the obvious linearity relations on $V$ and $W$.

An \textbf{isomorphism} of vector spaces $V\cong W$ consists of linear maps $f:V\to W$ and $g:W\to V$ such that $f\circ g=1_W$ and $g\circ f=1_V$.

\begin{exercise}[Universal constructions]\leavevmode\par
Give the universal properties of free spaces, direct sums, quotients, and
tensor products, and use them to prove the three isomorphism theorems and
$k(X\sqcup Y)\cong kX\oplus kY$, $k(X\times Y)\cong kX\otimes kY$.
\end{exercise}

The \textbf{dual space} $V^*$ is the space of linear maps from a vector space $V$ to its ground field $k$. The evaluation map $\ev_V : V^* \otimes V \to k$ is the linear extension of the evaluation morphism $f \otimes x \mapsto f(x)$. We say that $V$ is \textbf{finite dimensional} if there is a co-evaluation morphism $\coev_V: k \to V \otimes V^*$ satisfying the snake equations $(1_{V} \otimes \ev_V) \circ (\coev_V \otimes 1_V) = 1_V$
and
$(\ev_V \otimes 1_{V^*}) \circ (1_{V^*} \otimes \coev_{V}) = 1_{V^*}$. We define the \textbf{dimension} $\dim(V) = \trace(1_V)$, where $\trace(f) : \End(V) \to k$ is given by $f \mapsto (\ev_V \circ s_{V, V^*} \circ (f \otimes 1_{V^*}) \circ \coev_V)(1)$ where $s_{X, Y} : X \otimes Y \to Y \otimes X$ is the symmetric braiding $x \otimes y \mapsto y \otimes x$.



\begin{exercise}[Trace and dimension identities]\label{ex:linear-trace-dimension}\leavevmode\par
Prove the following for finite-dimensional vector spaces $V,W$ and linear maps $f,g:V\to V$.
    \begin{itemize}
        \item $\trace(f g) = \trace(g f)$
        \item $\trace(f \oplus g) = \trace(f) + \trace(g)$
        \item $\trace(f \otimes g) = \trace(f) \trace(g)$
        \item $\dim(V) \in \mathbf{N}$
        \item $V \cong W$ iff $\dim(V) = \dim(W)$
    \end{itemize}
\end{exercise}


Define the tensor algebra and \textbf{exterior algebra of $V$} by
\[
T(V)=\bigoplus_{k\geq 0}V^{\otimes k},
\qquad
\Lambda V=T(V)/\langle v\otimes v:v\in V\rangle.
\]
The space $\Lambda^kV$ is the degree-$k$ component of $\Lambda V$.

\begin{exercise}[The top exterior power is one-dimensional]
    Show that the highest graded component of $\Lambda V$ is one-dimensional.
\end{exercise}

The \textbf{determinant} $\det(f)$ is the coefficient of the linear map on the highest-graded component of $\Lambda V$ given by $v_1 \wedge \cdots \wedge v_{\dim(V)} \mapsto f(v_1) \wedge \cdots f(v_{\dim(V)})$.

\begin{exercise}[Determinant identities and volume]\leavevmode\par
Prove the following:
\begin{itemize}
    \item $\det(f \oplus g) = \det(f) \det(g)$
    \item If $V = \mathbb{R}^n$, then $\det(f)$ is the signed volume of $f(I^n)$ where $I = [0,1]$.
    \item Cramer's rule
\end{itemize}
\end{exercise}

For $f\in\End(V)$, the \textbf{characteristic polynomial} is the polynomial $\lambda \mapsto \det(f - \lambda 1_V)$.

A scalar $\lambda\in k$ is an \textbf{eigenvalue} of $f$ if $\ker(f-\lambda1_V)\neq0$. This kernel is its \textbf{eigenspace}; its dimension is the geometric multiplicity of $\lambda$.

\begin{exercise}[Characteristic polynomial and Jordan form]\leavevmode\par
Let $f\in\End(V)$ have split characteristic polynomial. Prove the
Cayley--Hamilton theorem and Jordan normal form, and derive the algebraic and
geometric multiplicity inequality together with the eigenvalue formulas for
$\trace(f)$ and $\det(f)$.
\end{exercise}

\begin{exercise}[Critical damping is a Jordan block]\leavevmode\par
\textit{Use Jordan normal form to solve
$m\ddot x+b\dot x+\kappa x=0$ in all three damping regimes, and show that
critical damping is exactly $b^2=4m\kappa$, where a nontrivial Jordan block
produces $t e^{-bt/(2m)}$.}
\end{exercise}

A \textbf{Hermitian inner product} on a complex vector space $V$ is a positive-definite sesquilinear form $\langle\mathord{\cdot},\mathord{\cdot}\rangle$, conjugate-linear in the first variable and linear in the second. It induces the norm $\|v\|=\sqrt{\langle v,v\rangle}$. For a linear map $A:V\to W$ between finite-dimensional Hermitian spaces, an \textbf{adjoint} is a map $A^*:W\to V$ satisfying
\[
\langle Av,w\rangle=\langle v,A^*w\rangle.
\]
An endomorphism $A$ is \textbf{self-adjoint} if $A=A^*$, \textbf{normal} if $A^*A=AA^*$, \textbf{positive} if $\langle v,Av\rangle\geq0$ for every $v$, and \textbf{unitary} if $A^*A=AA^*=1$. Equip $\End(V)$ with the induced operator norm.

\begin{exercise}[Finite-dimensional spectral theorem]\label{ex:finite-spectral-theorem}\leavevmode\par
\textit{Develop the finite-dimensional spectral theorem: prove existence of
adjoints and positive square roots, and prove that every normal operator has
an orthogonal spectral decomposition}
\[
A=\sum_\lambda\lambda p_\lambda,
\qquad \sum_\lambda p_\lambda=1.
\]
\textit{Deduce the singular-value decomposition}
\[
T=U\Sigma V^*.
\]
\end{exercise}

\begin{exercise}[The spectral decomposition produces normal modes]
\label{ex:vibrating-chain}\leavevmode\par
\textit{For a fixed-end chain of $N$ equal masses and springs, set
$q_0=q_{N+1}=0$ and define}
\[
(Kq)_j=\frac{\kappa}{m}(2q_j-q_{j-1}-q_{j+1}).
\]
\textit{Find the spectral decomposition of $K$ and use it to determine all
normal modes of $\ddot q+Kq=0$, their frequencies, and the conserved energy}
\[
\mathcal E(q,\dot q)
=\frac m2\|\dot q\|^2+\frac m2\langle q,Kq\rangle.
\]
\end{exercise}

\subsection{Group representations}

An action of a group $G$ on a set $X$ assigns to each $g\in G$ a bijection $x\mapsto gx$ such that $1x=x$ and $(gh)x=g(hx)$. A \textbf{representation} of $G$ on a vector space $V$ is a homomorphism
\[
\rho:G\longrightarrow\operatorname{GL}(V).
\]
A subspace $W\subseteq V$ is \textbf{$G$-invariant} if $\rho(g)W\subseteq W$ for every $g\in G$. Restriction gives the \textbf{subrepresentation} $W$, and $\rho(g)(v+W)=\rho(g)v+W$ gives the \textbf{quotient representation} $V/W$. A nonzero representation is \textbf{irreducible} if its only invariant subspaces are $0$ and $V$.

For representations $V$ and $W$, the direct sum, tensor product, and dual representations are defined by
\[
\begin{aligned}
g(v,w)&=(gv,gw),\\
g(v\otimes w)&=gv\otimes gw,\\
(g\phi)(v)&=\phi(g^{-1}v),\qquad \phi\in V^*.
\end{aligned}
\]
An \textbf{intertwiner} $T:V\to W$ is a linear map satisfying $T\rho_V(g)=\rho_W(g)T$ for every $g$. Write $\operatorname{Hom}_G(V,W)$ for the space of intertwiners and
\[
V^G=\{v\in V:\rho(g)v=v\text{ for every }g\in G\}
\]
for the fixed-point space.

\begin{exercise}[Averaging produces invariant projections and inner products]\leavevmode\par
\textit{Let $G$ be finite and let $V$ be a complex representation. Define}
\[
P_G=\frac1{|G|}\sum_{g\in G}\rho(g).
\]
\textit{Prove that $P_G$ is the projection of $V$ onto $V^G$. Prove also that averaging any Hermitian inner product over $G$ produces a $G$-invariant Hermitian inner product, so every finite-dimensional complex representation of $G$ is equivalent to a unitary representation.}
\end{exercise}

\subsection{Maschke's theorem and Schur's lemma}

\begin{exercise}[Maschke's theorem]\leavevmode\par
\textit{Let $W\subseteq V$ be $G$-invariant and let $P:V\to W$ be any linear projection. Define}
\[
\widetilde P
=\frac1{|G|}\sum_{g\in G}\rho(g)P\rho(g)^{-1}.
\]
\textit{Prove that $\widetilde P$ is $G$-equivariant and has image $W$. Deduce that $\ker\widetilde P$ is a $G$-invariant complement of $W$. If $G$ is finite and the characteristic of $k$ does not divide $|G|$, prove that every finite-dimensional $k$-representation of $G$ is a direct sum of irreducible representations.}
\end{exercise}

\begin{exercise}[Schur's lemma]\leavevmode\par
\textit{Let $V$ and $W$ be irreducible finite-dimensional complex representations. Prove that every nonzero intertwiner $T:V\to W$ is an isomorphism and that}
\[
\End_G(V)=\mathbb C1_V.
\]
\textit{Let $\mathbb C[G]$ act on $V$ by the linear extension of $\rho$. Deduce that every element of the center $Z(\mathbb C[G])$ acts by a scalar on each irreducible representation.}
\end{exercise}

\subsection{Characters and orthogonality}

The \textbf{character} of a finite-dimensional representation $V$ is the function
\[
\chi_V(g)=\trace(\rho_V(g)).
\]
For class functions $\chi,\psi:G\to\mathbb C$, define
\[
\langle\chi,\psi\rangle_G
=\frac1{|G|}\sum_{g\in G}\overline{\chi(g)}\psi(g).
\]

\begin{exercise}[Great Orthogonality Theorem]\leavevmode\par
\textit{For irreducible unitary representations $\Gamma^\alpha$ and
$\Gamma^\beta$ of dimensions $d_\alpha,d_\beta$ of a finite group $G$,
prove}
\[
\sum_{g\in G}
\Gamma^\alpha(g)_{ij}
\overline{\Gamma^\beta(g)_{k\ell}}
=\frac{|G|}{d_\alpha}
\delta_{\alpha\beta}\delta_{ik}\delta_{j\ell}.
\]
\textit{Develop its character consequences, including}
\[
m_\alpha=\langle\chi_\alpha,\chi_V\rangle_G.
\]
\textit{Deduce that characters classify finite-dimensional complex
representations.}
\end{exercise}

For an irreducible character $\chi_\alpha$, define the \textbf{symmetry projection}
\[
P_\alpha
=\frac{d_\alpha}{|G|}
\sum_{g\in G}\overline{\chi_\alpha(g)}\rho(g).
\]

\begin{exercise}[Symmetry projections]\leavevmode\par
\textit{Prove that the operators $P_\alpha$ are pairwise orthogonal idempotents and that $P_\alpha V$ is the $\alpha$-isotypic component of $V$. Interpret $P_\alpha$ as constructing symmetry-adapted linear combinations of atomic orbitals or displacement vectors.}
\end{exercise}

\subsection{Molecular symmetry}

The \textbf{point group} of a molecule is the finite group $G\subseteq O(3)$ of spatial symmetries preserving its nuclear configuration, including atomic species. For a molecule with $N$ atoms, $G$ acts on the real displacement space
\[
V_{\mathrm{disp}}=\bigoplus_{a=1}^N\mathbb R^3_a
\cong\mathbb R^{3N}
\]
by permuting atoms and applying the spatial linear action to each displacement. The three-dimensional translational representation carries the polar-vector action $R_g$, while the rotational representation carries the axial-vector action $\det(R_g)R_g$; denote them by $V_{\mathrm{trans}}$ and $V_{\mathrm{rot}}$. We complexify these real representations when applying character theory. For a nonlinear molecule, the vibrational representation is the class
\[
[V_{\mathrm{vib}}]
=[V_{\mathrm{disp}}]-[V_{\mathrm{trans}}]-[V_{\mathrm{rot}}]
\]
in the representation ring.

\begin{exercise}[Character of molecular displacements]\leavevmode\par
\textit{For $g\in G$, prove that}
\[
\chi_{\mathrm{disp}}(g)
=\sum_{\substack{a\\g(a)=a}}\trace(R_g),
\]
\textit{where $R_g:\mathbb R^3\to\mathbb R^3$ is the spatial action and the sum is over atoms fixed by $g$. Prove that $\dim V_{\mathrm{vib}}=3N-6$ for a nonlinear molecule, and use character orthogonality to express the multiplicity of every irreducible representation in $V_{\mathrm{vib}}$.}
\end{exercise}

For water, take the $z$-axis along the $C_2$ axis and the molecule in the $yz$-plane. Its point group is
\[
C_{2v}=\{E,C_2,\sigma_{xz},\sigma_{yz}\}.
\]
\begin{exercise}[Water has three symmetry-classified vibrations]\leavevmode\par
\textit{Derive the $C_{2v}$ character table and use it to show that water's
vibrational representation is}
\[
V_{\mathrm{vib}}\cong2A_1\oplus B_2.
\]
\textit{Identify and construct the three normal modes.}
\end{exercise}

\subsection{Spectroscopic selection rules}

Let $V_i$ and $V_f$ be the symmetry types of initial and final states. The electric-dipole operator transforms as the vector representation $V_\mu=V_{\mathrm{trans}}$. A transition amplitude
\[
\langle\psi_f,\mu\psi_i\rangle
\]
is the matrix coefficient used below.

\begin{exercise}[Symmetry determines electric-dipole transitions]\leavevmode\par
\textit{Derive the electric-dipole selection rule}
\[
1\subseteq V_f^*\otimes V_\mu\otimes V_i.
\]
\textit{Apply it to show that all three fundamental vibrations of water are
infrared active.}
\end{exercise}

\begin{exercise}[All fundamental vibrations of water are Raman active]\leavevmode\par
\textit{Let the polarizability tensor carry $\operatorname{Sym}^2(V_\mu)$. Prove that a vibrational mode is Raman active precisely when its irreducible representation occurs in this symmetric square. Prove that}
\[
\operatorname{Sym}^2(V_\mu)
\cong3A_1\oplus A_2\oplus B_1\oplus B_2
\]
\textit{for $C_{2v}$. Deduce that all three fundamental vibrational modes of water are Raman active.}
\end{exercise}

\subsection{Lie groups, infinitesimal symmetry, and spin}

The molecular point groups above are discrete subgroups of $O(3)$. Define the
group of orientation-preserving spatial rotations by
\[
\operatorname{SO}(3)
=\{R\in M_3(\mathbb R):R^TR=1,\ \det R=1\}.
\]

A \textbf{matrix Lie group} is a subgroup $G\subseteq\operatorname{GL}(n,\mathbb C)$ that is closed in the subspace topology. Its \textbf{Lie algebra} is
\[
\mathfrak g=\{X\in M_n(\mathbb C):e^{tX}\in G\text{ for all }t\in\mathbb R\},
\]
where $e^{tX}=\sum_{k\geq0}(tX)^k/k!$. The Lie bracket is $[X,Y]=XY-YX$. An abstract \textbf{Lie algebra} over $k$ is a $k$-vector space with a bilinear bracket that is skew-symmetric and satisfies the Jacobi identity. A \textbf{Lie algebra representation} on $V$ is a linear map $\rho:\mathfrak g\to\mathfrak{gl}(V)$ with $\rho([X,Y])=[\rho(X),\rho(Y)]$.

For $g\in G$ and $X,Y\in\mathfrak g$, the adjoint actions are
\[
\operatorname{Ad}_g(X)=gXg^{-1},
\qquad
\operatorname{ad}_X(Y)=[X,Y].
\]

\begin{exercise}[Classical matrix Lie algebras]\leavevmode\par
\textit{Prove that the Lie algebras of $\operatorname{GL}(n,\mathbb C)$, $\operatorname{SL}(n,\mathbb C)$, $\operatorname{O}(n)$, $\operatorname{SO}(n)$, $\operatorname{U}(n)$, and $\operatorname{SU}(n)$ are respectively}
\[
\mathfrak{gl}(n,\mathbb C),
\quad
\mathfrak{sl}(n,\mathbb C)=\{X:\trace(X)=0\},
\]
\[
\mathfrak{o}(n)=\{X:X^T=-X\},
\quad
\mathfrak{u}(n)=\{X:X^*=-X\},
\quad
\mathfrak{su}(n)=\mathfrak{u}(n)\cap\mathfrak{sl}(n,\mathbb C).
\]
\textit{Verify that each is a Lie algebra under $[X,Y]=XY-YX$.}
\end{exercise}

\begin{exercise}[One-parameter subgroups and the bracket]\leavevmode\par
\textit{Prove that $t\mapsto e^{tX}$ is a smooth group homomorphism $\mathbb R\to G$ whenever $X\in\mathfrak g$, and that every smooth homomorphism $\mathbb R\to G$ arises uniquely in this way. Prove that}
\[
[X,Y]
=\left.\frac{d}{dt}\right|_{t=0}
\left.\frac{d}{ds}\right|_{s=0}
e^{tX}e^{sY}e^{-tX}e^{-sY},
\]
\textit{and deduce $\operatorname{Ad}_{e^{tX}}=e^{t\operatorname{ad}_X}$.}
\end{exercise}

\begin{exercise}[Differentiating group representations]\leavevmode\par
\textit{Let $\rho:G\to\operatorname{GL}(V)$ be a smooth finite-dimensional representation of a matrix Lie group. Prove that}
\[
\dot\rho(X)
=\left.\frac{d}{dt}\right|_{t=0}\rho(e^{tX})
\]
\textit{is a Lie algebra representation $\mathfrak g\to\mathfrak{gl}(V)$, and that $\rho(e^{tX})=e^{t\dot\rho(X)}$. If $G$ is connected, prove that a subspace is $G$-invariant if and only if it is $\mathfrak g$-invariant.}
\end{exercise}

\begin{exercise}[Rotations, angular momentum, and spin]\leavevmode\par
\textit{Using the Pauli matrices}
\[
\sigma_1=\begin{pmatrix}0&1\\1&0\end{pmatrix},
\qquad
\sigma_2=\begin{pmatrix}0&-i\\i&0\end{pmatrix},
\qquad
\sigma_3=\begin{pmatrix}1&0\\0&-1\end{pmatrix}.
\]
\textit{Identify $\mathfrak{su}(2)\cong\mathfrak{so}(3)$, construct the
double cover $\operatorname{SU}(2)\to\operatorname{SO}(3)$, and classify the
irreducible spin representations $\rho_j:SU(2)\to U(V_j)$. For
$J_a=i\dot\rho_j(-i\sigma_a/2)$, derive
$\dim V_j=2j+1$ and $J^2=j(j+1)1_{V_j}$, including the distinction between
integer and half-integer spin.}
\end{exercise}

A \textbf{quantum number} is an eigenvalue used to label a joint eigenspace
of commuting observables.

\begin{exercise}[Energy and symmetry produce quantum numbers]\leavevmode\par
\textit{Let $H$ be a self-adjoint operator on a finite-dimensional Hermitian
space $V$ with a commuting $\operatorname{SU}(2)$ action. Derive the joint
quantum-number decomposition}
\[
|E,j,m,\alpha\rangle
\]
\textit{for $H,J^2,J_3$, including the allowed values and the degeneracy
forced by spin $j$.}
\end{exercise}

\begin{exercise}[Hidden symmetry determines the hydrogen spectrum]
\label{ex:hydrogen-hidden-symmetry}\leavevmode\par
\textit{With $p=-i\hbar\nabla$, $r=|x|$, and $L=x\times p$, consider the
hydrogen Hamiltonian and Runge--Lenz operator}
\[
H=\frac{p^2}{2\mu}-\frac{\kappa}{r},
\qquad
A=\frac{1}{2\mu}(p\times L-L\times p)-\kappa\frac{x}{r}.
\]
\textit{Use the hidden $\mathfrak{so}(4)$ symmetry to derive the bound-state
spectrum and degeneracies}
\[
E_n=-\frac{\mu\kappa^2}{2\hbar^2n^2},
\qquad
n\geq1,
\qquad \dim W_n=n^2,
\]
\textit{and the angular-momentum decomposition}
\[
W_n\cong\bigoplus_{\ell=0}^{n-1}V_\ell.
\]
\textit{Recover the quantum numbers and transition frequencies, and compare
them with hydrogen spectroscopy \cite{kramida-hydrogen} and Pauli's
algebraic derivation \cite{pauli-hydrogen}.}
\end{exercise}

\subsection{Monoidal and braided categories}

A $\mathbb C$-linear category is a category whose morphism sets
\[
\operatorname{Hom}_{\mathcal C}(x,y)
\]
are complex vector spaces and whose composition is bilinear.

A $\mathbb C$-linear monoidal category is a $\mathbb C$-linear category equipped with a tensor product $\otimes$ that is bilinear on morphisms and is associative with tensor unit $1$, up to coherent natural isomorphism.

Let $\mathcal C$ be such a category with finite direct sums. An object $x$ is \textbf{simple} if $\End(x)\cong\mathbb C$, and $\mathcal C$ is \textbf{semisimple} if every object is a finite direct sum of simple objects. Assume that $1$ is simple, that there are finitely many isomorphism classes of simple objects, and that $\mathcal C$ is \textbf{rigid}: every $x$ has a dual $x^*$ with evaluation and coevaluation maps satisfying the snake identities.

The definitions of duality, trace, and dimension above extend verbatim to $\mathcal C$. Write $d_x=\trace(1_x)$.
Assume that the categorical dimensions of nonzero objects are positive real numbers.

\begin{exercise}[Categorical trace and dimension]\leavevmode\par
\textit{Repeat the first three parts of Exercise~\ref{ex:linear-trace-dimension} in $\mathcal C$.}
\end{exercise}

For $x,y,z\in\mathcal C$, write
\[
F^{xyz}:(x\otimes y)\otimes z\longrightarrow x\otimes(y\otimes z)
\]
for the associator. A \textbf{braiding} is a family of isomorphisms
\[
c_{x,y}:x\otimes y\longrightarrow y\otimes x.
\]
For $f:x\to x'$ and $g:y\to y'$, it satisfies $(g\otimes f)c_{x,y}=c_{x',y'}(f\otimes g)$.
The associator satisfies the pentagon identity, and the associator and braiding satisfy the hexagon identities. Suppressing associators, the latter are
\[
\begin{aligned}
c_{x,y\otimes z}&=(1_y\otimes c_{x,z})(c_{x,y}\otimes1_z),\\
c_{x\otimes y,z}&=(c_{x,z}\otimes1_y)(1_x\otimes c_{y,z}).
\end{aligned}
\]

The \textbf{monodromy} of $x$ and $y$ is the double braiding
\[
M_{x,y}=c_{y,x}c_{x,y}:x\otimes y\longrightarrow x\otimes y.
\]
Thus $c_{x,y}$ is one crossing, while $M_{x,y}$ is a full braid. If $x\otimes y$ is simple, identify $M_{x,y}$ with its scalar in $\mathbb C^\times=\mathbb C\setminus\{0\}$. Decompositions of tensor products of simple objects are \textbf{fusion rules}, and the spaces $\operatorname{Hom}_{\mathcal C}(z,x\otimes y)$ are \textbf{fusion spaces}.

An object $x$ is \textbf{invertible} if $x\otimes x^*\cong1$.

\begin{exercise}[The $\nu=1/3$ anyonic phase advances by $2\pi/3$]\leavevmode\par
\textit{Assume that every simple object is invertible and that their
isomorphism classes form $\mathbb Z/m$, generated by $a$. Classify its
monodromy, writing $M_{r,s}=M_{a^r,a^s}$, and show that, for some
$k\bmod m$,}
\[
M_{r,s}=\exp\left(\frac{2\pi i krs}{m}\right).
\]
\textit{For $k=1$ and $m=3$, derive the $2\pi/3$ phase advance per enclosed
anyon and compare it with \cite{nakamura-anyons}.}
\end{exercise}

Now suppose that $\mathcal C$ has two simple objects $1,\tau$ satisfying the fusion rule
\[
\tau\otimes\tau\cong1\oplus\tau.
\]

\begin{exercise}[Fibonacci categorical dimension]\leavevmode\par
\textit{Derive the golden-ratio dimension of $\tau$ and the Fibonacci growth
of its fusion spaces.}
\end{exercise}

\begin{exercise}[Fibonacci anyons braid non-Abelianly]\leavevmode\par
\textit{On $V=\operatorname{Hom}_{\mathcal C}(\tau,\tau^{\otimes3})$, use
the fusion basis, set $\varphi=(1+\sqrt5)/2$, and use the matrices
\cite{bonderson}}
\[
F=
\begin{pmatrix}
\varphi^{-1}&\varphi^{-1/2}\\
\varphi^{-1/2}&-\varphi^{-1}
\end{pmatrix},
\qquad
R=
\begin{pmatrix}
e^{-4\pi i/5}&0\\
0&e^{3\pi i/5}
\end{pmatrix}.
\]
\textit{Show that $b_1\mapsto R$ and $b_2\mapsto F^{-1}RF$ define a
non-Abelian unitary representation of $B_3$. Relate
these fusion and braiding data to the engineered
superconducting-processor realization in \cite{xu-fibonacci} and the related
trapped-ion realization in \cite{iqbal-anyons}; distinguish these simulations
of non-Abelian topological order from evidence for naturally occurring
Fibonacci quasiparticles.}
\end{exercise}

\section{Geometry}

\subsection{Differential forms and cohomology}

A \textbf{smooth manifold} is a topological space equipped with an atlas whose transition maps are smooth. A smooth map $f:M\to N$ has a tangent map $df:TM\to TN$. The \textbf{tangent bundle} $TM$ is the union of the tangent spaces, the \textbf{cotangent bundle} is $T^*M$, and a \textbf{vector field} is a smooth section of $TM$.

For an $n$-dimensional real vector space $V$, its \textbf{determinant line}
is $\det(V^*)=\Lambda^nV^*$.
An \textbf{orientation} of $V$ is a choice of one of the two connected
components of $\det(V^*)\setminus\{0\}$; the elements of the chosen component
are positive. An orientation of an $n$-manifold $M$ is a smoothly varying
choice of orientation of its tangent spaces.

The exterior powers $\Lambda^kT_p^*M$ are those defined in the Algebra section. To define integration in a coordinate-free way, we will integrate differential forms over chains. The space of \textbf{differential $n$-forms} is $\Omega^n(M)=\Gamma(\Lambda^nT^*M)$. On an oriented $n$-manifold, a \textbf{volume form} is a nowhere-vanishing element of $\Omega^n(M)$ that is positive with respect to the orientation. The exterior derivative $d:\Omega^n(M)\to\Omega^{n+1}(M)$ extends the differential of smooth functions and satisfies
\[
d(\alpha\wedge\beta)=d\alpha\wedge\beta+(-1)^p\alpha\wedge d\beta
\]
for $\alpha\in\Omega^p(M)$. A \textbf{singular $n$-simplex} is a smooth map $\sigma:\Delta^n \to M$ where $\Delta^n$ is the standard $n$-simplex. The collection of singular $n$-simplices in $M$ freely generates the abelian group $C_n(M)$ of \textbf{singular $n$-chains} in $M$. The \textbf{boundary map} $\partial : C_n(M) \to C_{n-1}(M)$ is defined on each simplex $\sigma:\Delta^n \to M$ by
$$\partial\sigma=\sum_{j=0}^{n}(-1)^j\sigma\circ f_j,$$ where $f_j : \Delta^{n-1} \to \Delta^n$ is the inclusion of the face opposite the $j$-th vertex of $\Delta^n$.

A \textbf{chain complex} $(A_n, d_n)_n$ is a sequence of abelian groups $A_n$ and homomorphisms $d_n : A_n \to A_{n-1}$ such that $d_{n-1} \circ d_n = 0$ for all $n$. The $n$-th \textbf{homology group} $H_n$ of the chain complex is
$$H_n = \ker(d_{n}) / \im(d_{n+1}).$$
The prototypical example is singular homology which corresponds to the chain complex of singular chains with the boundary map defined above.

Cohomology and cochains are defined similarly, using superscripts instead of subscripts and boundary maps $d^n : A^n \to A^{n+1}$ increase degree instead of decrease it. Singular cohomology corresponds to the cochain complex given by dualizing the singular chain complex above.

The \textbf{de Rham cohomology} of $M$ is
\[
H_{\mathrm{dR}}^n(M)
=
\ker(d\colon\Omega^n(M)\to\Omega^{n+1}(M))/
\operatorname{im}(d\colon\Omega^{n-1}(M)\to\Omega^n(M)).
\]

For a smooth map $f:N\to M$, the \textbf{pullback} of $\alpha\in\Omega^p(M)$ is
\[
(f^*\alpha)_x(v_1,\ldots,v_p)
=\alpha_{f(x)}(df_xv_1,\ldots,df_xv_p).
\]

\begin{exercise}[de Rham theorem]\leavevmode\par
\textit{Prove that integration over smooth singular cycles induces a natural isomorphism between de Rham cohomology and singular cohomology with real coefficients.}
\end{exercise}

\begin{exercise}[Nontrivial cohomology produces the Aharonov--Bohm phase]\leavevmode\par
\textit{Outside an idealized solenoid, let}
\[
M=\mathbb R^3\setminus\{(0,0,z):z\in\mathbb R\},
\qquad
d\theta=\frac{-y\,dx+x\,dy}{x^2+y^2},
\qquad
A=\frac{\Phi}{2\pi}d\theta.
\]
\textit{For a particle of charge $q$, use de Rham cohomology to derive the
gauge-invariant phase}
\[
\exp\!\left(\frac{iq}{\hbar}\int_\gamma A\right)
\]
\textit{and the Aharonov--Bohm interference shift $q\Phi/\hbar$
\cite{aharonov-bohm}.}
\end{exercise}

\subsection{Hodge theory and electromagnetism}

A \textbf{pseudo-Riemannian metric} is a smooth nondegenerate symmetric section $g\in\Gamma(T^*M\otimes T^*M)$. On an oriented pseudo-Riemannian $n$-manifold, the metric induces a bilinear form $\langle\cdot,\cdot\rangle_g$ on differential forms. A \textbf{metric volume form} $\operatorname{vol}_g$ is a positive volume form satisfying
\[
\left|\langle\operatorname{vol}_g,\operatorname{vol}_g\rangle_g\right|=1.
\]
The \textbf{Hodge star} is an operator $\star:\Omega^p(M)\to\Omega^{n-p}(M)$ satisfying
\[
\alpha\wedge\star\beta=\langle\alpha,\beta\rangle_g\operatorname{vol}_g.
\]

\begin{exercise}[Metric volume and Hodge star]\leavevmode\par
\textit{Prove that an oriented pseudo-Riemannian manifold has a unique metric volume form and a unique Hodge-star operator. Determine $\star^2$ in terms of the degree and signature of the metric.}
\end{exercise}

On an oriented Lorentzian four-manifold, the electromagnetic field strength is
a two-form $F$ and the electric four-current is a three-form $J$.

\begin{exercise}[Maxwell's equations imply charge conservation]\leavevmode\par
\textit{From Maxwell's equations}
\[
dF=0,
\qquad
d\star F=J.
\]
\textit{Derive charge conservation in differential and integral form. On
Minkowski spacetime, write}
\[
F=B-dt\wedge E,
\qquad
J=\rho\,\operatorname{vol}_3-dt\wedge\star_3j,
\]
\textit{and recover the classical Maxwell equations and continuity equation.}
\end{exercise}

\begin{exercise}[Gauge invariance forces charge conservation]\leavevmode\par
\textit{For $F=dA$, consider}
\[
S[A]
=-\frac12\int_M F\wedge\star F+\int_M A\wedge J.
\]
\textit{Derive Maxwell's equations and prove that gauge invariance is
equivalent to charge conservation, assuming vanishing boundary terms.}
\end{exercise}

\subsection{Bundles, K-theory, and band topology}

A \textbf{smooth vector bundle} $p:E\to M$ is a smooth family of vector spaces locally isomorphic to a product $U\times V\to U$. For a smooth map $f:N\to M$, its \textbf{pullback bundle} is
\[
f^*E=\{(x,e)\in N\times E:f(x)=p(e)\}\longrightarrow N.
\]

For a compact Hausdorff space $X$, let $\operatorname{Vect}(X)$ be the category of finite-rank complex vector bundles with continuous local trivializations over $X$, with the usual direct sum, tensor product, dual, and pullback constructions. The \textbf{Grothendieck group} $K^0(X)$ is generated by isomorphism classes $[E]$ subject to
\[
[E\oplus F]=[E]+[F].
\]
Define multiplication by $[E][F]=[E\otimes F]$. For a continuous map $f:X\to Y$ between compact Hausdorff spaces, define pullback by
\[
f^*:K^0(Y)\longrightarrow K^0(X),
\qquad
[E]\longmapsto[f^*E].
\]

\begin{exercise}[Grothendieck group of vector bundles]\leavevmode\par
\textit{Prove that $(f\circ g)^*E\cong g^*f^*E$ and that pullback preserves direct sums, tensor products, and duals. Let $X$ and $Y$ be compact Hausdorff spaces and let $f:X\to Y$ be continuous. Prove that direct sum and tensor product induce a commutative ring structure on $K^0(X)$ and that $f^*$ is a ring homomorphism. Prove that}
\[
K^0(\mathrm{pt})\cong\mathbb Z
\]
\textit{via the rank map.}
\end{exercise}

Fix a basepoint $x_0\in X$, regarded as a map $x_0:\mathrm{pt}\to X$. The \textbf{reduced K-theory} of the pointed space $(X,x_0)$ is
\[
\widetilde K^0(X)
=\ker\!\left(K^0(X)\xrightarrow{x_0^*}K^0(\mathrm{pt})\right).
\]
For a bundle $E$ of rank $r$, its \textbf{reduced class} is
\[
[E]-[\underline{\mathbb C}^{\,r}],
\]
where $\underline{\mathbb C}^{\,r}=X\times\mathbb C^r$ is the trivial rank-$r$ bundle. Write $K^n(X)$ for the generalized cohomology groups extending $K^0(X)$.

\begin{exercise}[Stable equivalence and Bott periodicity]\leavevmode\par
\textit{Let $E$ and $F$ be complex vector bundles over a compact Hausdorff space $X$. Prove that $[E]=[F]$ in $K^0(X)$ if and only if there is a vector bundle $G$ such that}
\[
E\oplus G\cong F\oplus G.
\]
\textit{Deduce that the reduced class of $E$ is unchanged after replacing $E$ by $E\oplus\underline{\mathbb C}^{\,n}$. Prove Bott periodicity}
\[
K^{n+2}(X)\cong K^n(X)
\]
\textit{and deduce that complex K-theory is naturally $\mathbb Z/2$-graded.}
\end{exercise}

For $0\leq r\leq N$, identify the complex Grassmannian $\operatorname{Gr}_r(\mathbb C^N)$ with the space of rank-$r$ orthogonal projections in $M_N(\mathbb C)$. Its \textbf{tautological bundle} has fiber $\operatorname{im}P$ over $P$. Let $BU$ denote the stable complex Grassmannian obtained from finite Grassmannians by adjoining trivial summands.

\begin{exercise}[Stable classification of complex vector bundles]\label{ex:stable-bundle-classification}\leavevmode\par
\textit{Classify stable complex vector bundles over a connected finite CW
complex $X$ by projection-valued maps, proving}
\[
K^0(X)\cong[X,\mathbb Z\times BU],
\qquad
\widetilde K^0(X)\cong[X,BU]_*.
\]
\end{exercise}

For the gapped families below, assume that $X$ is connected, so vector-bundle ranks are constant. Let $V$ be a finite-dimensional Hermitian vector space as in Algebra. Exercise~\ref{ex:finite-spectral-theorem} supplies the spectral decomposition used below. A continuous family is a map
\[
H:X\longrightarrow\End(V)
\]
and write $H_x=H(x)$. It is \textbf{gapped} if every $H_x$ is self-adjoint and invertible. Given a positively oriented contour $\Gamma$ enclosing the negative spectra and excluding the positive spectra, define
\[
P_x^-=\frac{1}{2\pi i}\int_\Gamma(z-H_x)^{-1}\,dz,
\qquad
E_x^-=\operatorname{im}(P_x^-).
\]

\begin{exercise}[Gapped families define stable K-theory classes]\leavevmode\par
\textit{Show that the negative spectral subspaces of a gapped family form a
vector bundle and that its class in $K^0(X)$ is invariant under
gap-preserving homotopy. Show that}
\[
[E_H^-]-[\underline{\mathbb C}^{\,\operatorname{rank}E_H^-}]
\in\widetilde K^0(X)
\]
\textit{is invariant under adjoining constant gapped summands, and that every
reduced class is represented by a flattened family $H_x=1-2P_x$.}
\end{exercise}

\begin{exercise}[K-theory classifies class A band insulators]\label{ex:class-a-insulators}\leavevmode\par
\textit{A class A band model over a compact connected parameter space $X$
has a smooth gapped Hamiltonian}
\[
H:X\longrightarrow\End(V)
\]
\textit{whose negative-energy bundle consists of occupied states; for a
crystal, $X=\mathbb T^d$. Prove that
stable equivalence classes are classified by $\widetilde K^0(X)$
\cite{kitaev-periodic}, and compute the strong phases}
\[
\widetilde K^0(S^d)
\cong
\begin{cases}
\mathbb Z,&d\text{ even},\\
0,&d\text{ odd}.
\end{cases}
\]
\textit{Compute the classifications over $\mathbb T^d$ for $d=1,2,3$ and
distinguish strong from weak invariants.}
\end{exercise}

\subsection{Symplectic geometry}

For a vector field $X$, contraction is the map $\iota_X:\Omega^p(M)\to\Omega^{p-1}(M)$ defined by
\[
(\iota_X\alpha)(X_2,\ldots,X_p)=\alpha(X,X_2,\ldots,X_p).
\]
Define the Lie derivative of forms by Cartan's formula
$\mathcal{L}_X=d\iota_X+\iota_Xd$.

A \textbf{symplectic form} is a two-form $\omega\in\Omega^2(M)$ such that $d\omega=0$ and $X\mapsto\iota_X\omega$ is a bundle isomorphism $TM\to T^*M$. Write $C^\infty(M)$ for the smooth real-valued functions on $M$. For $H\in C^\infty(M)$, define $X_H$ by
\[
\iota_{X_H}\omega=dH.
\]

Let $\pi:T^*Q\to Q$ be the cotangent bundle of a configuration manifold. Its \textbf{tautological one-form} is
\[
\theta_{(q,p)}(\xi)=p\bigl(d\pi_{(q,p)}\xi\bigr),
\]
and its \textbf{canonical symplectic form} is $\omega_0=-d\theta$.

\begin{exercise}[The symplectic form produces Hamilton's equations]\leavevmode\par
\textit{For}
\[
H(q,p)=\frac{1}{2m}\sum_{i=1}^np_i^2+V(q),
\]
\textit{derive Hamilton's equations intrinsically from $\omega_0$, recover
Newton's equation, and solve the harmonic-oscillator flow.}
\end{exercise}

\begin{exercise}[Magnetic symplectic curvature produces the Lorentz force]\leavevmode\par
\textit{For a particle of charge $e$ in a closed magnetic field
$B\in\Omega^2(\mathbb R^3)$, derive the Lorentz force from}
\[
\omega_B=\omega_0-e\pi^*B
\qquad\text{and}\qquad
H(q,p)=\frac{|p|^2}{2m}+e\phi(q),
\]
\textit{and relate the gauge freedom for $B=dA$ to the Aharonov--Bohm
potential.}
\end{exercise}

\begin{exercise}[Liouville's theorem]\leavevmode\par
\textit{Prove that every Hamiltonian flow preserves the symplectic volume}
\[
\frac{\omega^n}{n!}.
\]
\end{exercise}

\begin{exercise}[Hamiltonian phase volume produces ideal-gas thermodynamics]\leavevmode\par
\textit{For Liouville measure $d\mu_L=\omega^n/n!$, show that Hamiltonian
evolution preserves the fine-grained Gibbs entropy}
\[
S_G[\rho]
=-k_B\int_M\rho\log\rho\,d\mu_L.
\]
\textit{For $N$ indistinguishable noninteracting particles in a box, divide
phase volume by $N!h^{3N}$ and use}
\[
H(q,p)=\sum_{a=1}^N\frac{|p_a|^2}{2m}.
\]
\textit{Derive the Sackur--Tetrode entropy and the ideal-gas laws}
\[
E=\frac32Nk_BT,
\qquad
PV=Nk_BT.
\]
\textit{Derive the adiabatic and sound-speed laws, including, for
$\varrho=mN/V$,}
\[
c=\sqrt{\frac{5P}{3\varrho}}.
\]
\textit{Reconcile the entropy increase in free expansion with conservation
of fine-grained Gibbs entropy.}
\end{exercise}

For $f,g\in C^\infty(M)$, define
\[
\{f,g\}=\omega(X_f,X_g).
\]

\begin{exercise}[Hamiltonian symmetry produces conservation laws]\leavevmode\par
\textit{Prove that a Hamiltonian $f$ generates a symmetry of $H$ exactly
when $f$ is conserved by the flow of $H$, and distinguish this statement
from Noether's variational theorem \cite{noether}.}
\end{exercise}

\begin{exercise}[Central forces and angular momentum]\leavevmode\par
\textit{On $T^*\mathbb R^3$, let}
\[
H(q,p)=\frac{|p|^2}{2m}+V(|q|),
\qquad
L=q\times p.
\]
\textit{Derive the angular-momentum Poisson algebra and show that rotational
symmetry confines each nonradial trajectory to a fixed plane.}
\end{exercise}

\subsection{Connections and curvature}

An \textbf{affine connection} is a map $\nabla : \Gamma(TM) \times \Gamma(TM) \to \Gamma(TM)$ such that
\[
\nabla_{fX}Y=f\nabla_XY,
\qquad
\nabla_X(fY)=X(f)Y+f\nabla_XY.
\]
For a covector field $\alpha$, set
\[
(\nabla_X\alpha)(Y)=X(\alpha(Y))-\alpha(\nabla_XY).
\]
Define the covariant derivative of general tensor fields by the product rule and by requiring it to commute with contractions.

A connection is \textbf{torsion-free and metric-compatible} when
\[
\nabla_XY-\nabla_YX=[X,Y],
\qquad
\nabla g=0.
\]
Such a connection is called a \textbf{Levi-Civita connection}.

\begin{exercise}[Levi-Civita connection]\leavevmode\par
\textit{Prove that every pseudo-Riemannian manifold admits a unique Levi-Civita connection. Derive the coordinate-free Koszul formula.}
\end{exercise}

An affinely parametrized curve $\gamma:I\to M$ is a \textbf{geodesic} if $\nabla_{\dot\gamma}\dot\gamma=0$. The Lie derivative of the metric is
\[
(\mathcal L_Xg)(Y,Z)
=X(g(Y,Z))-g([X,Y],Z)-g(Y,[X,Z]),
\]
and $X$ is a \textbf{Killing vector field} if $\mathcal L_Xg=0$.

Let $(M,g)$ be a pseudo-Riemannian manifold with Levi-Civita connection $\nabla$.

\begin{exercise}[Killing fields and conserved quantities]\leavevmode\par
\textit{Let $X$ be a Killing vector field and let $\gamma$ be an affinely parametrized geodesic. Prove that}
\[
\frac{d}{dt}g(\dot\gamma,\dot\gamma)=0,
\qquad
\frac{d}{dt}g(X,\dot\gamma)=0.
\]
\textit{Deduce that $g(X,\dot\gamma)$ is constant along $\gamma$.}
\end{exercise}

The \textbf{curvature} of $\nabla$ is the $\End(TM)$-valued $2$-form $F_\nabla$ determined by
\[
R(X,Y)Z
:=F_\nabla(X,Y)Z
=\nabla_X\nabla_YZ-\nabla_Y\nabla_XZ-\nabla_{[X,Y]}Z.
\]

\begin{exercise}[Curvature as a tensor]\leavevmode\par
\textit{For $f\in C^\infty(M)$ and vector fields $X,Y,Z$, prove that}
\[
R(fX,Y)Z=fR(X,Y)Z,
\qquad
R(X,fY)Z=fR(X,Y)Z,
\]
\[
R(X,Y)(fZ)=fR(X,Y)Z.
\]
\textit{Prove that $R(X,Y)=-R(Y,X)$ and deduce that}
\[
R\in\Gamma\!\left(\Lambda^2T^*M\otimes T^*M\otimes TM\right).
\]
\end{exercise}

\begin{exercise}[Geodesic deviation]
\textit{Let $\Gamma:(-\varepsilon,\varepsilon)\times I\to M$ be a smooth
variation through affinely parametrized geodesics, and define the vector
fields along $\Gamma$ by}
\[
T=\frac{\partial\Gamma}{\partial t},
\qquad
J=\frac{\partial\Gamma}{\partial s}.
\]
\textit{Prove that $[T,J]=0$ and that}
\[
\nabla_T\nabla_TJ=R(T,J)T.
\]
\end{exercise}

\begin{exercise}[Euler characteristic is categorical dimension]\leavevmode\par
\textit{Show that the categorical trace on bounded complexes of
finite-dimensional vector spaces is the Lefschetz number and that categorical
dimension is Euler characteristic. Apply this to a finite cellular chain
complex for the surface in the next exercise.}
\end{exercise}

\begin{exercise}[Gauss--Bonnet theorem]\leavevmode\par
\textit{Let $(M,g)$ be a compact oriented Riemannian surface with piecewise
smooth boundary. With Gaussian curvature $K$, signed geodesic curvature
$k_g$, and signed exterior angles $\theta_j$, prove} \cite{do-carmo}
\[
\int_MK\operatorname{vol}_g
+\int_{\partial M}k_g\,ds
+\sum_j\theta_j
=2\pi\chi(M).
\]
\textit{Derive its consequences for round spheres, positively curved tori,
and closed hyperbolic surfaces.}
\end{exercise}

The trace used below is the ordinary trace defined in the Algebra section. Define the \textbf{Ricci tensor}, \textbf{scalar curvature}, and \textbf{Einstein tensor} by
\[
\operatorname{Ric}(X,Y)=\operatorname{tr}\bigl(Z\mapsto R(Z,X)Y\bigr),
\qquad
S=\operatorname{tr}_g(\operatorname{Ric}),
\qquad
G=\operatorname{Ric}-\frac12Sg.
\]
For a symmetric covariant two-tensor $A$, its \textbf{divergence} is the covector field
\[
(\operatorname{div}A)(Y)
=\operatorname{tr}_g\bigl((X,Z)\mapsto(\nabla_XA)(Z,Y)\bigr).
\]

\begin{exercise}[Contracted Bianchi identity and conservation]\leavevmode\par
\textit{Assume the second Bianchi identity}
\[
(\nabla_XR)(Y,Z)+(\nabla_YR)(Z,X)+(\nabla_ZR)(X,Y)=0.
\]
\textit{Contract this identity to prove that}
\[
\operatorname{div}G=0.
\]
\textit{Deduce that if $T$ is a symmetric covariant two-tensor satisfying}
\[
G+\Lambda g=\kappa T
\]
\textit{for constants $\Lambda,\kappa$ with $\kappa\ne0$, then $\operatorname{div}T=0$.}
\end{exercise}

Let $E\to M$ be a real or complex vector bundle and write $\Omega^0(M,E)=\Gamma(E)$. A \textbf{connection} on $E$ is a map
\[
\nabla:\Omega^0(M,E)\longrightarrow\Omega^1(M,E)
\]
linear over the ground field and satisfying $\nabla(fs)=df\otimes s+f\nabla s$. For $\alpha\in\Omega^p(M)$ and an $E$-valued form $\eta$, extend it by
\[
\nabla(\alpha\wedge\eta)
=d\alpha\wedge\eta+(-1)^p\alpha\wedge\nabla\eta.
\]
Define the curvature by
$F_\nabla=\nabla^2\in\Omega^2(M,\End E)$, and equip $\End E$-valued forms
with the induced connection.

\begin{exercise}[Bundle curvature agrees with Riemann curvature and satisfies Bianchi]
\textit{For $E=TM$, prove that $F_\nabla=\nabla^2$ agrees with the
Riemann-curvature formula above.}
\textit{Let $\nabla$ be a connection on $E\to M$. Prove that}
\[
\nabla F_\nabla=0.
\]
\end{exercise}

\begin{exercise}[Berry phase as holonomy]\leavevmode\par
\textit{Let $H:X\to\End(V)$ be a smooth family of self-adjoint operators with
a simple isolated eigenvalue $\lambda_x$, and set}
\[
L=\{(x,v)\in X\times V:H_xv=\lambda_xv\}.
\]
\textit{For a local unit eigenvector $\psi$, define the Berry connection and
Berry curvature by}
\[
\mathcal A=i\langle\psi,d\psi\rangle,
\qquad
\mathcal F=d\mathcal A.
\]
\textit{Prove that $L\to X$ is a line bundle and that, for a closed curve
$\gamma$, the gauge-independent Berry phase is its holonomy}
\[
\exp\left(i\int_\gamma\mathcal A\right)
\].
\end{exercise}

Let $P$ be an invariant symmetric $k$-linear polynomial on the endomorphism
algebra of a fiber of $E$, where invariant means invariant under simultaneous
conjugation. Define
\[
P(F_\nabla)
=P(F_\nabla,\ldots,F_\nabla)
\in\Omega^{2k}(M).
\]

\begin{exercise}[Chern--Weil theorem]\leavevmode\par
\textit{Prove that}
\[
dP(F_\nabla)=0.
\]
\textit{For two connections $\nabla^0,\nabla^1$ on $E$, prove that}
\[
[P(F_{\nabla^0})]
=[P(F_{\nabla^1})]
\in H_{\mathrm{dR}}^{2k}(M).
\]
\textit{Relate this construction to the characteristic forms of
\cite{chern-simons}.}
\end{exercise}

\begin{exercise}[Chern character]\label{ex:chern-character}\leavevmode\par
\textit{Define the Chern character form by}
\[
\operatorname{ch}(E,\nabla)
=\operatorname{tr}\exp\left(\frac{iF_\nabla}{2\pi}\right).
\]
\textit{For compact $M$, prove that it defines a ring homomorphism}
\[
\operatorname{ch}:K^0(M)\longrightarrow H_{\mathrm{dR}}^{\mathrm{even}}(M),
\]
\textit{For the two-dimensional class A insulator of
Exercise~\ref{ex:class-a-insulators}, identify the phase with}
\[
C_1(E_H^-)
=\frac{i}{2\pi}\int_{\mathbb T^2}
\operatorname{tr}\bigl(P\,dP\wedge dP\bigr)\in\mathbb Z,
\]
\textit{and show that a nonzero value obstructs a global smooth Bloch frame.}
\end{exercise}

\subsection{Yang--Mills theory and the Standard Model}

Let $G$ be a Lie group with Lie algebra $\mathfrak g$. A \textbf{principal $G$-bundle} is a smooth free right $G$-manifold $P$ with quotient $M=P/G$ such that the projection $P\to M$ is a locally trivial fiber bundle. For a finite-dimensional representation $\rho:G\to\operatorname{GL}(V)$, the \textbf{associated bundle} is
\[
E=P\times_G V=(P\times V)/\sim,
\qquad
(p\cdot g,v)\sim(p,\rho(g)^{-1}v).
\]
A \textbf{principal connection} is a $\mathfrak g$-valued one-form $A$ on $P$ that is $G$-equivariant and reproduces the generators of the fundamental vector fields. In a local trivialization it is represented by an ordinary form $A\in\Omega^1(U,\mathfrak g)$, with curvature
\[
F_A=dA+A\wedge A\in\Omega^2(U,\mathfrak g).
\]
\begin{exercise}[Isospin motivates Yang--Mills connections]\leavevmode\par
\textit{Treat the proton and neutron as a nucleon doublet carrying the
defining representation of $\operatorname{SU}(2)$. Gauge this symmetry and
derive}
\[
A\longmapsto A^g=g^{-1}Ag+g^{-1}dg.
\]
\textit{Show that the resulting connection differentiates associated-bundle
fields and has covariant curvature, and relate this construction to
\cite{yang-mills}.}
\end{exercise}

\begin{exercise}[Yang--Mills action]\leavevmode\par
\textit{On an oriented Riemannian four-manifold, consider the Yang--Mills
action}
\[
S(A)=\int_M\operatorname{Tr}(F_A\wedge\star F_A).
\]
\textit{Prove gauge invariance, derive}
\[
d_A\star F_A=0,
\]
\textit{and recover source-free Maxwell theory in the abelian case.}
\end{exercise}

\begin{exercise}[Standard Model representations and symmetry breaking]\leavevmode\par
\textit{Take the Standard Model gauge group to be}
\[
G_{\mathrm{SM}}
=\operatorname{SU}(3)_C\times\operatorname{SU}(2)_L
\times\operatorname{U}(1)_Y.
\]
\textit{Write $(\mathbf n,\mathbf m)_Y$ for the representation of dimension
$n$ under $\operatorname{SU}(3)_C$, dimension $m$ under
$\operatorname{SU}(2)_L$, and weak hypercharge $Y$. Take one generation of
Weyl fermion multiplets to be}
\[
Q_L=(\mathbf 3,\mathbf 2)_{1/6},
\qquad
u_R=(\mathbf 3,\mathbf 1)_{2/3},
\qquad
d_R=(\mathbf 3,\mathbf 1)_{-1/3},
\]
\[
L_L=(\mathbf 1,\mathbf 2)_{-1/2},
\qquad
e_R=(\mathbf 1,\mathbf 1)_{-1},
\]
\textit{and define $Q=T_3+Y$. Let $\phi$ be a section of the Higgs bundle
associated to $(\mathbf 1,\mathbf 2)_{1/2}$, with}
\[
V(\phi)=-\mu^2|\phi|^2+\lambda|\phi|^4,
\qquad
D\phi=d\phi+\tfrac{i}{2}g\,W^a\sigma_a\phi+\tfrac{i}{2}g'B\phi,
\]
\textit{where $\mu^2,\lambda>0$ and $g,g'$ are coupling constants. From
these data derive the fermion charges,
the symmetry breaking
$\operatorname{SU}(2)_L\times\operatorname{U}(1)_Y
\to\operatorname{U}(1)_{\mathrm{EM}}$, the $W$ and $Z$ masses, the massless
photon, and cancellation of the hypercharge anomalies.}
\end{exercise}

\subsection{Chern--Simons theory and TQFT}

\begin{exercise}[Chern--Simons transgression]\leavevmode\par
\textit{For two connections on $E$, set}
\[
\nabla^t=(1-t)\nabla^0+t\nabla^1,
\qquad
A=\nabla^1-\nabla^0,
\qquad
F_t=F_{\nabla^t},
\]
\textit{where $A\in\Omega^1(M,\End E)$, and define}
\[
\operatorname{CS}_P(\nabla^0,\nabla^1)
=k\int_0^1P(A,F_t,\ldots,F_t)\,dt.
\]
\textit{Prove that}
\[
P(F_{\nabla^1})-P(F_{\nabla^0})
=d\,\operatorname{CS}_P(\nabla^0,\nabla^1).
\]
\end{exercise}

Let $\mathfrak g\subseteq M_N(\mathbb C)$ be a matrix Lie algebra whose trace pairing is nondegenerate. For $A\in\Omega^1(M,\mathfrak g)$, wedge products include matrix multiplication. As above, write
\[
F_A=dA+A\wedge A
\]
and define
\[
\operatorname{CS}(A)
=\operatorname{tr}\left(
A\wedge dA+\frac23A\wedge A\wedge A
\right).
\]

\begin{exercise}[Chern--Simons form]\leavevmode\par
\textit{Prove that}
\[
d\,\operatorname{CS}(A)
=\operatorname{tr}(F_A\wedge F_A).
\]
\end{exercise}

For a closed oriented three-manifold $M$, define
\[
S_{\mathrm{CS}}(A)=\int_M\operatorname{CS}(A).
\]

\begin{exercise}[Chern--Simons equation]\leavevmode\par
\textit{For $\alpha\in\Omega^1(M,\mathfrak g)$, set $A_t=A+t\alpha$. Prove that}
\[
\left.\frac{d}{dt}\right|_{t=0}S_{\mathrm{CS}}(A_t)
=2\int_M\operatorname{tr}(\alpha\wedge F_A).
\]
\textit{Deduce that the critical points satisfy $F_A=0$.}
\end{exercise}

For $k\in\mathbb R$, set $S_k=(k/4\pi)S_{\mathrm{CS}}$.

\begin{exercise}[Gauge transformation]\leavevmode\par
\textit{Let $G$ be a compact, connected, simple, and simply connected matrix
Lie group with Lie algebra $\mathfrak g$. For a smooth map $g:M\to G$, define}
\[
A^g=g^{-1}Ag+g^{-1}dg.
\]
\textit{Normalize the invariant trace so that}
\[
\nu(g)=\frac{1}{24\pi^2}\int_M
\operatorname{tr}\bigl((g^{-1}dg)^3\bigr)\in\mathbb Z.
\]
\textit{Prove that}
\[
\operatorname{CS}(A^g)-\operatorname{CS}(A)
=-d\,\operatorname{tr}(dg\,g^{-1}\wedge A)
-\frac13\operatorname{tr}\bigl((g^{-1}dg)^3\bigr).
\]
\textit{Deduce that $\exp(iS_k)$ is invariant under all gauge transformations when $k\in\mathbb Z$.}
\end{exercise}

Now restrict to $2+1$ spacetime dimensions. Let $M$ be a closed oriented three-manifold and let $\ell>0$, so the cosmological constant is $\Lambda=-\ell^{-2}$. After identifying the coframe representation with $\mathfrak{sl}(2,\mathbb R)$, let
\[
e,\omega\in\Omega^1(M,\mathfrak{sl}(2,\mathbb R)),
\qquad
A^\pm=\omega\pm\frac1\ell e.
\]
The pair $(A^+,A^-)$ has gauge Lie algebra $\mathfrak{sl}(2,\mathbb R)\oplus\mathfrak{sl}(2,\mathbb R)$.
Write
\[
d_\omega e=de+\omega\wedge e+e\wedge\omega,
\qquad
F_\omega=d\omega+\omega\wedge\omega.
\]

\begin{exercise}[Flat connections encode three-dimensional gravity]\leavevmode\par
\textit{Prove that $F_{A^+}=F_{A^-}=0$ is equivalent to}
\[
d_\omega e=0,
\qquad
F_\omega+\frac1{\ell^2}e\wedge e=0.
\]
\textit{Assuming that $e:TM\to\mathfrak{sl}(2,\mathbb R)$ is fiberwise invertible, identify these equations with the torsion-free condition and the Einstein equation with constant negative curvature.}
\end{exercise}

\begin{exercise}[Three-dimensional gravity is a Chern--Simons theory]\leavevmode\par
\textit{For the first-order Einstein--Hilbert functional}
\[
S_{\mathrm{EH}}(e,\omega)
=\int_M\operatorname{tr}\left(
e\wedge F_\omega+\frac{1}{3\ell^2}e\wedge e\wedge e
\right).
\]
\textit{Prove that, up to normalization and an exact term, it equals}
\[
S_{\mathrm{CS}}(A^+)-S_{\mathrm{CS}}(A^-)
\]
\cite{achucarro-townsend,witten-gravity}.
\end{exercise}

For a connection $A$ and a finite-dimensional representation $V$ of its gauge group, parallel transport around an oriented closed curve $\gamma$ gives the holonomy $\operatorname{Hol}_\gamma(A)$. Define the \textbf{Wilson loop}
\[
W_V(\gamma,A)
=\operatorname{tr}_V\!\left(\operatorname{Hol}_\gamma(A)\right).
\]
For a framed oriented link $L=\gamma_1\sqcup\cdots\sqcup\gamma_r$ labeled by $V_1,\ldots,V_r$, set
\[
W_L(A)=\prod_{j=1}^rW_{V_j}(\gamma_j,A).
\]
For a framed link, define the formal Chern--Simons expectation value \cite{witten-jones}
\[
\langle W_L\rangle_k
=
\frac{\int e^{iS_k(A)}W_L(A)\,\mathcal DA}
{\int e^{iS_k(A)}\,\mathcal DA}
\]
\begin{exercise}[Wilson loops and the Jones polynomial]\leavevmode\par
\textit{For normalized $SU(2)$ theory at level $k$, set}
\[
q=\exp\left(\frac{2\pi i}{k+2}\right).
\]
\textit{Assume the normalized Wilson-loop expectations satisfy}
\[
q^{-1}\langle W_{L_+}\rangle_k
-q\langle W_{L_-}\rangle_k
=\left(q^{1/2}-q^{-1/2}\right)\langle W_{L_0}\rangle_k
\]
\textit{and $\langle W_{\bigcirc}\rangle_k=1$. Show that they determine the
Jones polynomial and account for framing dependence.}
\end{exercise}

\begin{exercise}[Reshetikhin--Turaev TQFT]\leavevmode\par
\textit{Prove that a modular braided fusion category defines a
Reshetikhin--Turaev TQFT}
\[
Z:\operatorname{Bord}_3^{\mathrm{or}}
\longrightarrow\operatorname{Vect}_{\mathbb C}.
\]
\textit{Here $\operatorname{Bord}_3^{\mathrm{or}}$ is the category of closed
oriented surfaces and oriented three-dimensional bordisms. Relate its Wilson
lines to the $F$- and $R$-matrices and, for
$SU(2)_k$, to the Chern--Simons skein relation
\cite{reshetikhin-turaev}.}
\end{exercise}

\section{Analysis}

\subsection{Norms, operator exponentials, and waves}

A \textbf{norm} on a complex vector space $H$ is a function $\|\mathord{\cdot}\|:H\to[0,\infty)$ that is definite, homogeneous, and satisfies the triangle inequality. A normed space is a \textbf{Banach space} if every Cauchy sequence converges. A \textbf{Hilbert space} is a complex vector space with a Hermitian inner product $\langle\mathord{\cdot},\mathord{\cdot}\rangle$ whose norm $\|x\|=\sqrt{\langle x,x\rangle}$ is complete. A linear map $A:H\to K$ is \textbf{bounded} if
\[
\|A\|=\sup_{\|x\|=1}\|Ax\|<\infty.
\]
Write $\mathcal B(H,K)$ for the normed space of bounded operators and $\mathcal B(H)=\mathcal B(H,H)$.

For $A\in\mathcal B(H)$, define the operator exponential by the formal series
\[
e^A=\sum_{n=0}^{\infty}\frac{A^n}{n!}
\]
whenever the series converges in operator norm.

\begin{exercise}[Cauchy--Schwarz inequality]\leavevmode\par
\textit{Prove that every inner-product space satisfies}
\[
|\langle x,y\rangle|\leq\|x\|\|y\|.
\]
\textit{Deduce the triangle inequality for the induced norm. Prove that $\mathcal B(H,K)$ is complete when $K$ is complete and that the exponential series converges in $\mathcal B(H)$ for every bounded $A$.}
\end{exercise}

\begin{exercise}[Bounded-operator exponentials]\leavevmode\par
\textit{Prove that $e^{tA}$ is differentiable in operator norm and that}
\[
\frac{d}{dt}e^{tA}=Ae^{tA}=e^{tA}A.
\]
\textit{Prove the semigroup law $e^{(s+t)A}=e^{sA}e^{tA}$ and deduce that
$e^{tA}$ is invertible with inverse $e^{-tA}$. If $A$ and $B$ commute, prove
that $e^{A+B}=e^Ae^B$.}
\end{exercise}

\begin{exercise}[Fourier modes solve the fixed-end wave equation]
\label{ex:fixed-end-wave}\leavevmode\par
\textit{Solve the fixed-end wave equation from
Exercise~\ref{ex:wave-superposition} by Fourier series, prove conservation
of wave energy, and recover its normal-mode frequencies as the continuum
limit of Exercise~\ref{ex:vibrating-chain}.}
\end{exercise}

\subsection{Hilbert-space operator theory}

An \textbf{adjoint} of $A\in\mathcal B(H,K)$ is an operator $A^*\in\mathcal B(K,H)$ satisfying
\[
\langle Ax,y\rangle=\langle x,A^*y\rangle.
\]
An operator $A\in\mathcal B(H)$ is \textbf{self-adjoint} if $A=A^*$, \textbf{normal} if $A^*A=AA^*$, \textbf{positive} if $\langle x,Ax\rangle\geq0$ for every $x$, and \textbf{unitary} if $A^*A=AA^*=1$.

\begin{exercise}[Adjoints of bounded operators]\leavevmode\par
\textit{Prove that every bounded operator between Hilbert spaces has a unique adjoint and establish the identities}
\[
(AB)^*=B^*A^*,
\qquad
\|A^*A\|=\|A\|^2.
\]
\end{exercise}

\subsection{Compact operators and the Fredholm alternative}

A bounded operator $K:H\to L$ between Hilbert spaces is \textbf{compact} if
the image of the closed unit ball has compact closure. Write
$\mathcal K(H,L)$ for the compact operators and
$\mathcal K(H)=\mathcal K(H,H)$.

\begin{exercise}[Compact operators]\leavevmode\par
\textit{Give the sequential characterization of compact operators and prove
that they form an adjoint-closed norm-closed operator ideal generated densely
by finite-rank operators.}
\end{exercise}

A bounded operator $T:H\to L$ is \textbf{Fredholm} if $\ker T$ and $L/\operatorname{im}T$ are finite dimensional and $\operatorname{im}T$ is closed. Its \textbf{Fredholm index} is
\[
\operatorname{ind}(T)
=\dim\ker T-\dim(L/\operatorname{im}T).
\]

\begin{exercise}[Fredholm alternative]\leavevmode\par
\textit{For $K\in\mathcal K(H)$ and $\lambda\ne0$, prove the Fredholm
alternative \cite{conway-functional}: $\lambda1-K$ has index zero,}
\[
\dim\ker(\lambda1-K)
=\dim\ker(\overline\lambda1-K^*)<\infty.
\]
\textit{Characterize solvability of $(\lambda1-K)x=y$ and deduce the
that every nonzero spectral value of a compact operator is an eigenvalue of
finite multiplicity, with zero the only possible accumulation point.}
\end{exercise}

\subsection{Quantum states and uncertainty}

A finite-dimensional \textbf{quantum state} is a positive operator $\rho$ with $\trace(\rho)=1$. A unit vector $\psi$ defines the pure state $\rho_\psi=|\psi\rangle\langle\psi|$. For a self-adjoint observable $A$, define
\[
\langle A\rangle_\rho=\trace(\rho A),
\qquad
\operatorname{Var}_\rho(A)
=\trace\!\left(\rho(A-\langle A\rangle_\rho1)^2\right).
\]
For $\rho=\rho_\psi$, use the subscripts $\psi$ instead of $\rho$.

\begin{exercise}[Robertson uncertainty inequality]\leavevmode\par
\textit{For self-adjoint $A,B\in\mathcal B(H)$ and a unit vector $\psi$, prove} \cite{robertson}
\[
\operatorname{Var}_\psi(A)\operatorname{Var}_\psi(B)
\geq\frac14\left|\langle\psi,[A,B]\psi\rangle\right|^2,
\qquad [A,B]=AB-BA.
\]
\textit{Determine the equality condition.}
\end{exercise}

\begin{exercise}[Commuting observables are conserved]\leavevmode\par
\textit{Let $A,H$ be self-adjoint and write $U_t=e^{-itH}$. Prove that
$U_t$ is unitary and that}
\[
\frac{d}{dt}\langle U_t\psi,A U_t\psi\rangle
=-i\langle U_t\psi,[H,A]U_t\psi\rangle.
\]
\textit{Deduce that $[H,A]=0$ if and only if the expectation of $A$ is constant under this evolution for every initial vector.}
\end{exercise}

\begin{exercise}[No-cloning theorem]\label{ex:no-cloning}\leavevmode\par
\textit{Let $b\in H$ be a unit vector and suppose that a unitary $C:H\otimes H\to H\otimes H$ satisfies $C(v\otimes b)=v\otimes v$ and $C(w\otimes b)=w\otimes w$ for unit vectors $v,w$. Prove that} \cite{wootters-zurek}
\[
\langle v,w\rangle=\langle v,w\rangle^2.
\]
\textit{Deduce that no unitary map clones every unit vector.}
\end{exercise}

\subsection{Schmidt decomposition and entanglement}

For finite-dimensional Hilbert spaces $H_A,H_B$, the \textbf{partial trace}
over $H_B$ is the linear map
\[
\operatorname{Tr}_{H_B}:\End(H_A\otimes H_B)\longrightarrow\End(H_A)
\]
characterized by
\[
\operatorname{Tr}\!\left(A\operatorname{Tr}_{H_B}(X)\right)
=\operatorname{Tr}\!\left((A\otimes1_{H_B})X\right)
\qquad(A\in\End(H_A)).
\]

For finite-dimensional Hilbert spaces $H_A,H_B$ and $\psi\in H_A\otimes H_B$, define
\[
T_\psi:H_A^*\to H_B,
\qquad
T_\psi(\phi)=(\phi\otimes1)(\psi).
\]
A \textbf{Schmidt decomposition} of $\psi$ is an expression
\[
\psi=\sum_{i=1}^r s_i a_i\otimes b_i.
\]
where $(a_i)$ and $(b_i)$ are orthonormal families and $s_i>0$; $r$ is the \textbf{Schmidt rank}. For a unit vector $\psi$, define the reduced density operators by
\[
\rho_A=\operatorname{Tr}_{H_B}(|\psi\rangle\langle\psi|),
\qquad
\rho_B=\operatorname{Tr}_{H_A}(|\psi\rangle\langle\psi|).
\]

\begin{exercise}[Schmidt decomposition and entanglement]\leavevmode\par
\textit{Use singular-value decomposition to derive the Schmidt decomposition,
characterize separability, and compute the entanglement entropy}
\[
S(\psi)=-\operatorname{Tr}(\rho_A\log\rho_A)
=-\sum_i s_i^2\log s_i^2,
\]
\textit{where the $s_i$ are the Schmidt coefficients.}
\end{exercise}

\subsection{CHSH, Gisin, and Tsirelson}

\begin{exercise}[Every entangled pure two-qubit state violates CHSH]\leavevmode\par
\textit{Let $A_0,A_1$ and $B_0,B_1$ be self-adjoint operators with
eigenvalues in $\{-1,1\}$, and define the CHSH operator}
\[
\mathcal C=A_0\otimes(B_0+B_1)+A_1\otimes(B_0-B_1).
\]
\textit{Establish the classical CHSH bound $2$. For the two-qubit state}
\[
\psi_\theta=\cos\theta\,|00\rangle+\sin\theta\,|11\rangle,
\qquad 0\leq\theta\leq\frac\pi4,
\]
\textit{derive the maximal CHSH value} \cite{gisin}
\[
2\sqrt{1+\sin^2(2\theta)}.
\]
\textit{Conclude that every entangled pure two-qubit state violates
CHSH.}
\end{exercise}

A \textbf{bipartite binary quantum correlation matrix} has entries
\[
c_{ij}=\langle\psi,(A_i\otimes B_j)\psi\rangle,
\]
where $\psi$ is a unit vector. The operators $A_i$ and $B_j$ are self-adjoint unitaries on finite-dimensional Hilbert spaces.

For a real matrix $M=(M_{ij})$, define
\[
\beta_{\mathrm C}(M)
=\max_{\varepsilon_i,\delta_j\in\{-1,1\}}
\left|\sum_{i,j}M_{ij}\varepsilon_i\delta_j\right|
\]
and
\[
\beta_{\mathrm Q}(M)
=\sup_c\left|\sum_{i,j}M_{ij}c_{ij}\right|,
\]
where the supremum is over bipartite binary quantum correlation matrices.

\begin{exercise}[Tsirelson's representation theorem and bound]\leavevmode\par
\textit{Prove Tsirelson's representation theorem \cite{tsirelson}:}
\[
c_{ij}=\langle x_i,y_j\rangle.
\]
\textit{Derive the sharp CHSH bound $2\sqrt2$.}
\end{exercise}

\subsection{Grothendieck inequality}

The \textbf{real Grothendieck constant} $K_G^{\mathbb R}$ is the least constant for which
\[
\beta_{\mathrm Q}(M)\leq K_G^{\mathbb R}\beta_{\mathrm C}(M)
\]
holds for every real matrix $M$.

\begin{exercise}[Grothendieck's inequality]\leavevmode\par
\textit{Prove that $K_G^{\mathbb R}<\infty$
\cite{grothendieck,pisier-grothendieck}. Use}
\[
M_{\mathrm{CHSH}}=
\begin{pmatrix}1&1\\1&-1\end{pmatrix},
\]
\textit{to prove $K_G^{\mathbb R}\geq\sqrt2$ and interpret the inequality
as a dimension-independent bound on quantum advantage.}
\end{exercise}

\subsection{Von Neumann algebras and subfactors}

For $\mathcal S\subseteq\mathcal B(H)$, its \textbf{commutant} is
\[
\mathcal S'=\{T\in\mathcal B(H):Tx=xT\text{ for every }x\in\mathcal S\}.
\]
A \textbf{von Neumann algebra} is a unital $*$-subalgebra $M\subseteq\mathcal B(H)$ satisfying $M=M''$. A net $(T_\lambda)$ converges to $T$ in the \textbf{strong operator topology} if $T_\lambda\xi\to T\xi$ for every $\xi\in H$, and in the \textbf{weak operator topology} if
\[
\langle T_\lambda\xi,\eta\rangle\longrightarrow\langle T\xi,\eta\rangle
\]
for every $\xi,\eta\in H$.

\begin{exercise}[Bicommutant theorem]\leavevmode\par
\textit{Let $A\subseteq\mathcal B(H)$ be a unital $*$-subalgebra. Prove that}
\[
A''
=\overline A^{\,\mathrm{SOT}}
=\overline A^{\,\mathrm{WOT}}.
\]
\end{exercise}

The \textbf{center} of a von Neumann algebra is $Z(M)=M\cap M'$, and $M$ is a \textbf{factor} if $Z(M)=\mathbb C1$.

For a linear map $\Phi:M\to N$ between von Neumann algebras, its $n$th
\textbf{matrix amplification} is
\[
\Phi_n([x_{ij}])=[\Phi(x_{ij})].
\]
The map $\Phi$ is \textbf{completely positive} if every $\Phi_n$ is positive,
and it is \textbf{normal} if it preserves suprema of bounded increasing nets
of positive elements. A \textbf{normal $*$-representation} of $M$ on a Hilbert
space $K$ is a normal unital $*$-homomorphism $\pi:M\to\mathcal B(K)$. A
\textbf{Heisenberg-picture quantum channel} is a normal unital completely
positive map between von Neumann algebras.

\begin{exercise}[Normal CP maps have Stinespring dilations]
\leavevmode\par
\textit{Let $M$ be a von Neumann algebra and let
$\Phi:M\to\mathcal B(H)$ be normal and completely positive. Prove
Stinespring's theorem \cite{stinespring}: there are a Hilbert space $K$, a
normal representation $\pi:M\to\mathcal B(K)$, and $V:H\to K$ such that}
\[
\Phi(x)=V^*\pi(x)V,
\qquad
V^*V=\Phi(1).
\]
\textit{Deduce that every normal channel on $\mathcal B(H_0)$ has the
environmental form}
\[
\Phi(x)=V^*(x\otimes1_E)V
\]
\textit{and, in finite dimensions, the Kraus and partial-trace forms}
\[
\Phi_*(\rho)=\sum_\alpha K_\alpha\rho K_\alpha^*
=\operatorname{Tr}_E(V\rho V^*).
\]
\textit{Conclude that no quantum channel clones every pure state.}
\end{exercise}

A positive functional $\psi$ is \textbf{normal} if it preserves suprema of
bounded increasing nets of positive elements and \textbf{faithful} if
$\psi(a^*a)=0$ implies $a=0$. A \textbf{type $\mathrm{II}_1$ factor} is an
infinite-dimensional factor with a faithful normal tracial state
\[
\tau:M\longrightarrow\mathbb C,
\qquad
\tau(ab)=\tau(ba),
\qquad
\tau(1)=1.
\]

For projections $p,q\in M$, write $p\sim q$ if there is $v\in M$ such that
\[
v^*v=p,
\qquad
vv^*=q.
\]
Write $p\precsim q$ if $p$ is equivalent to a projection $r$ satisfying $rq=r$.

\begin{exercise}[Projection dimension]\leavevmode\par
\textit{Let $M$ be a type $\mathrm{II}_1$ factor with normalized trace $\tau$. Prove that}
\[
p\sim q\quad\Longrightarrow\quad\tau(p)=\tau(q),
\]
\textit{and prove the comparison theorem}
\[
p\precsim q
\quad\Longleftrightarrow\quad
\tau(p)\leq\tau(q).
\]
\textit{Deduce that $\tau(p)=\tau(q)$ if and only if $p\sim q$.}
\end{exercise}

A von Neumann algebra is \textbf{finite} if $v^*v=1$ implies $vv^*=1$. Let $N\subseteq M$ be finite von Neumann algebras with a common normalized trace $\tau$. A \textbf{trace-preserving conditional expectation} is a normal unital completely positive map $E_N:M\to N$ such that
\[
\tau\circ E_N=\tau,
\qquad
E_N(n_1xn_2)=n_1E_N(x)n_2
\]
for $x\in M$ and $n_1,n_2\in N$. Let $L^2(M,\tau)$ be the completion of $M$ for
\[
\langle x,y\rangle_\tau=\tau(y^*x).
\]
Let $M$ act on $L^2(M,\tau)$ by left multiplication.

\begin{exercise}[Conditional expectation and Jones projection]\leavevmode\par
\textit{Prove that there is a unique trace-preserving conditional expectation $E_N:M\to N$. Prove that $E_N$ extends to the orthogonal projection}
\[
e_N:L^2(M,\tau)\longrightarrow L^2(N,\tau).
\]
\textit{Prove that}
\[
e_N=e_N^*=e_N^2,
\qquad
e_Nxe_N=E_N(x)e_N
\]
\textit{for every $x\in M$.}
\end{exercise}

The \textbf{basic construction} is
\[
M_1=\langle M,e_N\rangle''
\subseteq\mathcal B(L^2(M,\tau)),
\]
the double commutant of the algebra generated by $M$ and $e_N$. For a projection $p=[p_{ij}]\in M_k(N)$ acting by left multiplication, define
\[
\dim_N\!\left(pL^2(N,\tau)^k\right)=\sum_{i=1}^k\tau(p_{ii}).
\]
An inclusion $N\subseteq M$ has \textbf{finite Jones index} if $L^2(M,\tau)$ is isomorphic as a right $N$-module to such a module; then
\[
[M:N]=\dim_NL^2(M,\tau).
\]

\begin{exercise}[Jones basic construction]\leavevmode\par
\textit{Let $N\subseteq M$ be an inclusion of type $\mathrm{II}_1$ factors of finite Jones index. Prove that $M_1$ is a finite factor and that its normalized trace satisfies} \cite{jones-sunder}
\[
\tau_1(xe_Ny)
=\frac{1}{[M:N]}\tau(xy)
\]
\textit{for $x,y\in M$.}
\end{exercise}

The \textbf{Jones tower} is the sequence obtained by iterating the basic construction:
\[
N=M_{-1}\subset M=M_0\subset M_1\subset M_2\subset\cdots.
\]
Let $e_i$ be the Jones projection used to construct $M_i$ from $M_{i-1}$, and set
\[
\delta=\sqrt{[M:N]}.
\]
The \textbf{Temperley--Lieb algebra} with loop parameter $\delta$ is generated by $U_i$ subject to
\[
U_i^2=\delta U_i,
\qquad
U_iU_j=U_jU_i\quad(|i-j|\geq2),
\qquad
U_iU_{i\pm1}U_i=U_i.
\]

\begin{exercise}[Temperley--Lieb relations]\leavevmode\par
\textit{Prove that the Jones projections satisfy} \cite{temperley-lieb,jones-index}
\[
e_i^2=e_i=e_i^*,
\qquad
e_ie_j=e_je_i\quad(|i-j|\geq2),
\]
\[
e_ie_{i\pm1}e_i=\delta^{-2}e_i.
\]
\textit{Deduce that $U_i\mapsto\delta e_i$ defines a representation of the Temperley--Lieb algebra.}
\end{exercise}

\begin{exercise}[Braid representation]\leavevmode\par
\textit{Let $q\in\mathbb C^\times$ satisfy $\delta=q+q^{-1}$ and define
$\sigma_i=q\,1-U_i$. Prove that $\sigma_i$ is invertible with inverse
$q^{-1}1-U_i$, and prove that}
\[
\sigma_i\sigma_{i+1}\sigma_i
=\sigma_{i+1}\sigma_i\sigma_{i+1},
\]
\[
\sigma_i\sigma_j=\sigma_j\sigma_i
\qquad
(|i-j|\geq2).
\]
\textit{Deduce that the $\sigma_i$ define a representation of the braid group} \cite{jones-polynomial}.
\end{exercise}

A \textbf{Markov trace} on the Temperley--Lieb tower is a compatible family of positive normalized traces satisfying
\[
\operatorname{tr}(xU_i)=\delta^{-1}\operatorname{tr}(x)
\]
whenever $x$ belongs to the algebra generated by the preceding generators.

\begin{exercise}[Positivity quantizes subfactor indices below four]\leavevmode\par
\textit{Define}
\[
P_0(\delta)=1,
\qquad
P_1(\delta)=\delta,
\qquad
P_{n+1}(\delta)=\delta P_n(\delta)-P_{n-1}(\delta).
\]
\textit{Prove that the normalized traces on the Jones tower restrict to a
Markov trace. Prove that, when $\sin\theta\ne0$,}
\[
P_n(2\cos\theta)
=\frac{\sin((n+1)\theta)}{\sin\theta}.
\]
\textit{Use positivity of the Markov trace to prove that if $0<\delta<2$, then}
\[
\delta=2\cos\left(\frac{\pi}{m}\right)
\]
\textit{for some integer $m\geq3$.}
\end{exercise}

\begin{exercise}[Jones index theorem]\leavevmode\par
\textit{Let $N\subseteq M$ be an inclusion of type $\mathrm{II}_1$ factors of finite Jones index. Prove that} \cite{jones-index}
\[
[M:N]
\in
\left\{
4\cos^2\left(\frac{\pi}{n}\right):n\geq3
\right\}
\cup
[4,\infty).
\]
\textit{Relate the discrete values to the $SU(2)_k$ braid representations
and Chern--Simons Wilson-loop invariants constructed above.}
\end{exercise}

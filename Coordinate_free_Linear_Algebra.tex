\section{Algebra}

Linear algebra studies linear maps between vector spaces. A \textbf{vector space} $V$ over a field $k$ is a set equipped with vector addition and scalar multiplication operations satisfying straightforward compatibility conditions. A \textbf{linear map} $f: V \to U$ between $k$-vector spaces is a map of underlying sets such that $f(\alpha v + \beta w) = \alpha f(v) + \beta f(w)$ for all $\alpha, \beta \in k$ and $v, w \in V$. Examples include differential operators, group representations, coefficient matrices of linear systems of equations, and local approximations of globally nonlinear functions \cite{halmos,hoffman-kunze}.

The \textbf{free $k$-vector space on a set $X$}, denoted $kX$, is the set of finitely supported set-theoretical functions $X \to k$ with pointwise addition and scalar multiplication. The \textbf{direct sum} vector space $V \oplus W$ is the set $V \times W$ with pointwise addition and scalar multiplication defined by $\alpha (v \oplus w) = \alpha v \oplus \alpha w$. The \textbf{quotient} of $V$ by a subspace $W$, denoted $V / W$, is the set of equivalence classes of $V$ modulo the relation $x \sim y$ iff $x - y \in W$. The \textbf{tensor product} $V \otimes W$ is the vector space $k(V \times W) / N$ where $N$ is the subspace generated by the obvious linearity relations on $V$ and $W$.

\begin{exercise} An \textbf{isomorphism} of vector spaces $V \cong W$ consists of a pair of linear maps $f : V \to W$ and $g : W \to V$ such that $f \circ g = 1_W$ and $g \circ f = 1_V$. Prove the following:
\begin{itemize}
\item $k(X \sqcup Y) \cong kX \oplus kY$
\item $k(X \times Y) \cong kX \otimes kY$
\item The first isomorphism theorem for vector spaces (and second and third if you want).
\item Find universal properties determining up to canonical isomorphism the free $k$-vector space, direct sum, quotient, tensor product, and dual space (defined below), respectively.
\end{itemize}
\end{exercise}

The \textbf{dual space} $V^*$ is the space of linear maps from a vector space $V$ to its ground field $k$. The evaluation map $\ev_V : V^* \otimes V \to k$ is the linear extension of the evaluation morphism $f \otimes x \mapsto f(x)$. We say that $V$ is \textbf{finite dimensional} if there is a co-evaluation morphism $\coev_V: k \to V \otimes V^*$ satisfying the snake equations $(1_{V} \otimes \ev_V) \circ (\coev_V \otimes 1_V) = 1_V$
and
$(\ev_V \otimes 1_{V^*}) \circ (1_{V^*} \otimes \coev_{V}) = 1_{V^*}$. We define the \textbf{dimension} $\dim(V) = \trace(1_V)$, where $\trace(f) : \End(V) \to k$ is given by $f \mapsto (\ev_V \circ s_{V, V^*} \circ (f \otimes 1_{V^*}) \circ \coev_V)(1)$ where $s_{X, Y} : X \otimes Y \to Y \otimes X$ is the symmetric braiding $x \otimes y \mapsto y \otimes x$ \cite{ponto-shulman}.



\begin{exercise} Prove the following for finite-dimensional vector spaces $V, W$ and linear maps $f, g : V \to V$.
    \begin{itemize}
        \item $\trace(f g) = \trace(g f)$
        \item $\trace(f \oplus g) = \trace(f) + \trace(g)$
        \item $\trace(f \otimes g) = \trace(f) \trace(g)$
        \item $\dim(V) \in \mathbf{N}$
        \item $V \cong W$ iff $\dim(V) = \dim(W)$
    \end{itemize}
\end{exercise}


The simplest linear maps are scalar multiples of the identity. Although most linear maps are not of this form, it is useful to decompose them into sums of scalar maps when possible. One condition under which we can decompose a linear map into a direct sum of scalars is if the map is self-adjoint. Suppose $V$ is finite dimensional over $\mathbb{C}$. A Hermitian inner product $\langle \cdot, \cdot \rangle: V \times V \to \mathbb{C}$ is a positive-definite sesquilinear form (linear in the first variable, anti-linear in the second variable, and conjugate symmetric). Write $|v|=\sqrt{\langle v,v\rangle}$. For a linear map $f:V\to W$ between Hermitian spaces, the adjoint $f^*:W\to V$ is characterized by $\langle fv,w\rangle=\langle v,f^*w\rangle$. An endomorphism $f$ is \textbf{self-adjoint} if $f=f^*$. An isomorphism $f$ is \textbf{unitary} if it preserves inner products, equivalently if $f^*=f^{-1}$.

For maps $f:V\to V'$ and $g:W\to W'$, define $(f\otimes g)(v\otimes w)=f(v)\otimes g(w)$. The tensor product of Hermitian spaces has inner product
\[
\langle v_1\otimes w_1,v_2\otimes w_2\rangle
=\langle v_1,v_2\rangle\langle w_1,w_2\rangle.
\]
The Cauchy--Schwarz inequality is $|\langle v,w\rangle|\leq |v||w|$, with equality iff $v$ and $w$ are linearly dependent.

\begin{exercise}[No-cloning theorem]
\textit{Let $V$ be a finite-dimensional complex inner-product space, let $b\in V$ be a unit vector, and suppose that $C:V\otimes V\to V\otimes V$ is unitary. Prove that if} \cite{wootters-zurek}
\[
C(v\otimes b)=v\otimes v,
\qquad
C(w\otimes b)=w\otimes w
\]
\textit{for unit vectors $v,w\in V$, then}
\[
\langle v,w\rangle=\langle v,w\rangle^2.
\]
\textit{Deduce that either $\langle v,w\rangle=0$ or $v=w$. Conclude that no unitary map can satisfy $C(v\otimes b)=v\otimes v$ for every unit vector $v\in V$.}
\end{exercise}

A scalar $\lambda \in k$ is an \textbf{eigenvalue} for a linear map $f : V \to V$ if there exists a nonzero vector $v \in V$ for which $fv = \lambda v$. The \textbf{eigenspace} of $\lambda$ is
\[
\ker(f-\lambda 1_V),
\]
whose nonzero elements are eigenvectors; its dimension is called the \textbf{geometric multiplicity} of $\lambda$.

\begin{exercise}
Prove that any self-adjoint linear map $f$ on a finite-dimensional complex vector space $V$ is diagonalizable:
$$f = \sum \lambda_i p_i,$$
where $\sum p_i = 1_V$ and $p_i p_j = \delta_{ij} p_i$. Use this fact to prove the existence of the singular value decomposition for an arbitrary linear map between complex finite-dimensional vector spaces.
\end{exercise}

For $\psi\in V\otimes W$, define
\[
T_\psi:V^*\to W,
\qquad
T_\psi(\phi)=(\phi\otimes1_W)(\psi).
\]
The map $j_V:V\to V^*$ given by $j_V(v)=\langle\mathord{\cdot},v\rangle$ is conjugate-linear. Equip $V^*$ with the Hermitian inner product determined by $\langle j_V(v),j_V(w)\rangle=\langle w,v\rangle$. With $T_\psi^*:W\to V^*$ denoting the adjoint, define
\[
\rho_V=j_V^{-1}T_\psi^*T_\psi j_V,
\qquad
\rho_W=T_\psi T_\psi^*.
\]
Write $U(V)$ for the group of unitary endomorphisms of $V$, acting with $U(W)$ on $V\otimes W$ by tensor products. A nonzero tensor is \textbf{decomposable} if it has the form $v\otimes w$. The rank of an endomorphism is the dimension of its image.

\begin{exercise}[Schmidt decomposition and reduced endomorphisms]\leavevmode\par
\textit{Let $V$ and $W$ be finite-dimensional Hermitian vector spaces and let $0\neq\psi\in V\otimes W$. Prove that there exist orthonormal families $(v_i)$ in $V$ and $(w_i)$ in $W$, and positive scalars $s_i$, such that} \cite{nielsen-chuang}
\[
\psi=\sum_{i=1}^r s_i v_i\otimes w_i.
\]
\textit{Prove the following:}
\begin{itemize}
\item The nonzero eigenvalues of $\rho_V$ and $\rho_W$ are the numbers $s_i^2$, with the same multiplicities.
\item $\psi$ is decomposable iff exactly one $s_i$ is nonzero.
\item $\rho_V$ has rank one iff $\rho_W$ has rank one iff $\psi$ is decomposable.
\item The multiset $(s_i)$ is invariant under the action of $U(V)\times U(W)$.
\end{itemize}
\end{exercise}

For a unit vector $\psi\in V$ and a self-adjoint $A\in\End(V)$, define
\[
m_\psi(A)=\langle\psi,A\psi\rangle,
\qquad
\Delta_\psi(A)=\left|\left(A-m_\psi(A)1_V\right)\psi\right|.
\]

\begin{exercise}[Uncertainty principle]
\textit{Let $A,B\in\End(V)$ be self-adjoint and let $\psi\in V$ be a unit vector. Prove that} \cite{robertson}
\[
\Delta_\psi(A)\Delta_\psi(B)
\geq\frac{1}{2}\left|\left\langle\psi,[A,B]\psi\right\rangle\right|,
\qquad
[A,B]=AB-BA.
\]
\textit{Determine the condition for equality from the equality condition in the Cauchy--Schwarz inequality.}
\end{exercise}

For $K\in\End(V)$, define $e^K=\sum_{n\geq0}K^n/n!$. If $H$ is self-adjoint, write $U_t=e^{-itH}$.

\begin{exercise}[Conservation from commutators]\leavevmode\par
\textit{Let $A,H\in\End(V)$ be self-adjoint, let $U_t=e^{-itH}$, and let $\psi\in V$. Prove that} \cite{nielsen-chuang}
\[
\frac{d}{dt}\left\langle U_t\psi,A\,U_t\psi\right\rangle
=-i\left\langle U_t\psi,[H,A]U_t\psi\right\rangle.
\]
\textit{Deduce that $[H,A]=0$ implies that $\langle U_t\psi,A\,U_t\psi\rangle$ is independent of $t$ for every $\psi\in V$. Prove the converse.}
\end{exercise}

We can solve for the eigenvalues of a linear operator on a finite-dimensional vector space via the determinant. Define the tensor algebra and \textbf{exterior algebra of $V$} by \cite{lang}
\[
T(V)=\bigoplus_{k\geq 0}V^{\otimes k},
\qquad
\Lambda V=T(V)/\langle v\otimes v:v\in V\rangle.
\]
The space $\Lambda^kV$ is the degree-$k$ component of $\Lambda V$.

\begin{exercise}
    Show that the highest graded component of $\Lambda V$ is one-dimensional.
\end{exercise}

The \textbf{determinant} $\det(f)$ is the coefficient of the linear map on the highest-graded component of $\Lambda V$ given by $v_1 \wedge \cdots \wedge v_{\dim(V)} \mapsto f(v_1) \wedge \cdots f(v_{\dim(V)})$.

For an $n$-dimensional real vector space $V$, its \textbf{determinant line} is $\det(V^*)=\Lambda^nV^*$. An \textbf{orientation} is a choice of one of the two connected components of $\det(V^*)\setminus\{0\}$; the elements of the chosen component are positive.

\begin{exercise} Prove the following:
\begin{itemize}
    \item $\det(f \oplus g) = \det(f) \det(g)$
    \item If $V = \mathbb{R}^n$, then $\det(f)$ is the signed volume of $f(I^n)$ where $I = [0,1]$.
    \item Cramer's rule
\end{itemize}
\end{exercise}

For $f\in\End(V)$, the \textbf{characteristic polynomial} is the polynomial $\lambda \mapsto \det(f - \lambda 1_V)$.

\begin{exercise}
    Assume that the characteristic polynomial of $f$ splits over $k$. Prove the following for a linear operator $f$ on a finite-dimensional vector space $V$:
    \begin{itemize}

        \item Every eigenvalue of $f$ is a root of its characteristic polynomial.
        \item The geometric multiplicity of each eigenvalue is at most equal to its \textbf{algebraic multiplicity} (i.e. multiplicity as a root of the characteristic polynomial)
        \item $\trace(f) = \sum_i \lambda_i$ and $\det(f) = \prod_i \lambda_i$, where each eigenvalue $\lambda_i$ is repeated according to its algebraic multiplicity
        \item The operator $f$ satisfies its own characteristic polynomial (replacing $\lambda$ with $f$)
        \item The existence of the Jordan normal form for a linear operator on a finite-dimensional complex vector space
    \end{itemize}
\end{exercise}

Let $\mathcal C$ be a $\mathbb C$-linear monoidal category with tensor product $\otimes$, tensor unit $1$, and finite direct sums. Write $\operatorname{Hom}_{\mathcal C}(x,y)$ for its complex vector space of morphisms from $x$ to $y$. An object $x$ is \textbf{simple} if $\End(x)\cong\mathbb C$, and $\mathcal C$ is \textbf{semisimple} if every object is a finite direct sum of simple objects. Assume that $1$ is simple, that there are finitely many isomorphism classes of simple objects, and that $\mathcal C$ is \textbf{rigid}: every $x$ has a dual $x^*$ with evaluation and coevaluation maps satisfying the snake identities \cite{etingof-tensor}.

The definitions of duality, trace, and dimension above extend verbatim to $\mathcal C$. Write $d_x=\trace(1_x)$. Then
\[
d_{x\oplus y}=d_x+d_y,
\qquad
d_{x\otimes y}=d_xd_y.
\]
Assume that the categorical dimensions of nonzero objects are positive real numbers.

For $x,y,z\in\mathcal C$, write
\[
F^{xyz}:(x\otimes y)\otimes z\longrightarrow x\otimes(y\otimes z)
\]
for the associator. A \textbf{braiding} is a family of isomorphisms
\[
c_{x,y}:x\otimes y\longrightarrow y\otimes x.
\]
For $f:x\to x'$ and $g:y\to y'$, it satisfies $(g\otimes f)c_{x,y}=c_{x',y'}(f\otimes g)$.
The associator satisfies the pentagon identity, and the associator and braiding satisfy the hexagon identities. Suppressing associators, the latter are
\[
\begin{aligned}
c_{x,y\otimes z}&=(1_y\otimes c_{x,z})(c_{x,y}\otimes1_z),\\
c_{x\otimes y,z}&=(c_{x,z}\otimes1_y)(1_x\otimes c_{y,z}).
\end{aligned}
\]

The \textbf{monodromy} of $x$ and $y$ is the double braiding
\[
M_{x,y}=c_{y,x}c_{x,y}:x\otimes y\longrightarrow x\otimes y.
\]
Thus $c_{x,y}$ is one crossing, while $M_{x,y}$ is a full braid. If $x\otimes y$ is simple, identify $M_{x,y}$ with its scalar in $\mathbb C^\times=\mathbb C\setminus\{0\}$. Decompositions of tensor products of simple objects are \textbf{fusion rules}, and the spaces $\operatorname{Hom}_{\mathcal C}(z,x\otimes y)$ are \textbf{fusion spaces}.

An object $x$ is \textbf{invertible} if $x\otimes x^*\cong1$. Assume first that every simple object is invertible and that their isomorphism classes form the cyclic group
\[
\{1,a,a^2,\ldots,a^{m-1}\},
\qquad
a^r\otimes a^s\cong a^{r+s\pmod m}.
\]
Write $U(1)=\{z\in\mathbb C:|z|=1\}$ and $M_{r,s}=M_{a^r,a^s}\in U(1)$.

\begin{exercise}[Abelian braiding phase]\leavevmode\par
\textit{Use the hexagon identities to prove that}
\[
M_{r+s,t}=M_{r,t}M_{s,t},
\qquad
M_{r,s+t}=M_{r,s}M_{r,t}.
\]
\textit{Deduce that there is an integer $k$ modulo $m$ such that}
\[
M_{r,s}=\exp\left(\frac{2\pi i krs}{m}\right).
\]
\textit{For $k=1$, prove that the monodromy of $a$ about $N$ copies of $a$ is multiplication by}
\[
\exp\left(\frac{2\pi iN}{m}\right).
\]
\textit{In particular, for $m=3$, increasing $N$ by one changes the phase by $2\pi/3$.}
\end{exercise}

The phase $2\pi/3$ is the value observed by interferometry in the $\nu=1/3$ fractional quantum Hall state \cite{nakamura-anyons}.

Now suppose that $\mathcal C$ has two simple objects $1,\tau$ satisfying the fusion rule
\[
\tau\otimes\tau\cong1\oplus\tau.
\]

\begin{exercise}[Fibonacci categorical dimension]\leavevmode\par
\textit{Let $d_\tau=\trace(1_\tau)$. Prove that} \cite{etingof-tensor}
\[
d_\tau^2=1+d_\tau
\]
\textit{and, using positivity, that}
\[
d_\tau=\frac{1+\sqrt5}{2}.
\]
\textit{Let}
\[
V_n=\operatorname{Hom}_{\mathcal C}(1,\tau^{\otimes n}).
\]
\textit{Show that the dimensions of these fusion spaces satisfy the Fibonacci recurrence and determine their asymptotic growth rate.}
\end{exercise}

Let
\[
V=\operatorname{Hom}_{\mathcal C}(\tau,\tau^{\otimes3}).
\]
This fusion space is two-dimensional, with basis indexed by the intermediate objects $1,\tau$ in $(\tau\otimes\tau)\otimes\tau$. Set $\varphi=(1+\sqrt5)/2$ and choose \cite{bonderson}
\[
F=
\begin{pmatrix}
\varphi^{-1}&\varphi^{-1/2}\\
\varphi^{-1/2}&-\varphi^{-1}
\end{pmatrix},
\qquad
R=
\begin{pmatrix}
e^{-4\pi i/5}&0\\
0&e^{3\pi i/5}
\end{pmatrix}.
\]
Define $\sigma_1=R$ and $\sigma_2=F^{-1}RF$. Let
\[
B_3=\langle b_1,b_2\mid b_1b_2b_1=b_2b_1b_2\rangle.
\]

\begin{exercise}[Non-Abelian braid representation]\leavevmode\par
\textit{Working exactly, verify that $F$ and $R$ are unitary and prove that}
\[
\sigma_1\sigma_2\sigma_1=\sigma_2\sigma_1\sigma_2.
\]
\textit{Prove also that $\sigma_1\sigma_2\neq\sigma_2\sigma_1$. Deduce that $b_i\mapsto\sigma_i$ defines a non-Abelian representation}
\[
B_3\longrightarrow U(V).
\]
\end{exercise}

Such Fibonacci fusion and braiding data have been implemented on superconducting quantum processors \cite{xu-fibonacci}.\footnote{This is an engineered quantum simulation of Fibonacci topological order, not evidence that naturally occurring Fibonacci quasiparticles have been conclusively observed. A related trapped-ion realization of non-Abelian topological order is reported in \cite{iqbal-anyons}.}

\section{Geometry}
A \textbf{smooth manifold} is a topological space that is locally diffeomorphic to Euclidean space $\mathbb{R}^n$. A \textbf{smooth vector bundle} $p : E \to M$ consists of a smooth surjection $p$ between smooth manifolds $E$ (the total space) and $M$ (the base space) such that all fibers are isomorphic to a single vector space $V$, and smooth local trivializations exist (locally, the fiber bundle looks like a product space). For a smooth map $f:N\to M$, the \textbf{pullback bundle} is
\[
f^*E=\{(x,e)\in N\times E:f(x)=p(e)\}\longrightarrow N.
\]
A prototypical example is the \textbf{tangent bundle} $TM$ which consists of all tangent spaces of points of a given manifold. The \textbf{cotangent bundle} $T^*M$ is the dual bundle of $TM$. A \textbf{vector field} $X \in \Gamma(TM)$ is a smooth section of the tangent bundle (i.e. a smooth map $s: M \to TM$ such that $p \circ s = 1_M$) \cite{spivak}. An orientation of $M$ is a smoothly varying choice of orientation of each tangent space, using the determinant lines defined in the Algebra section.

\begin{exercise}
Write down precise definitions for the following:
\begin{itemize}
\item smooth manifold, smooth map, vector bundle, tangent bundle.
\item bundle map, pullback, direct sum, tensor product, and dual bundle
\item derivative of a smooth function between manifolds
\end{itemize}
\end{exercise}

For a compact Hausdorff space $X$, let $\operatorname{Vect}(X)$ be the category of finite-rank complex vector bundles with continuous local trivializations over $X$. It has direct sums, tensor products, duals, and pullbacks, and is an additive symmetric monoidal category with duals \cite{hatcher-k}. The \textbf{Grothendieck group} $K^0(X)$ is generated by isomorphism classes $[E]$ subject to
\[
[E\oplus F]=[E]+[F].
\]
Tensor product defines $[E][F]=[E\otimes F]$, making $K^0(X)$ a commutative ring. For a continuous map $f:X\to Y$ between compact Hausdorff spaces, pullback defines
\[
f^*:K^0(Y)\longrightarrow K^0(X),
\qquad
[E]\longmapsto[f^*E].
\]

\begin{exercise}[Grothendieck group of vector bundles]\leavevmode\par
\textit{Let $X$ and $Y$ be compact Hausdorff spaces and let $f:X\to Y$ be continuous. Prove that direct sum and tensor product induce a commutative ring structure on $K^0(X)$ and that $f^*$ is a ring homomorphism. Prove that}
\[
K^0(\mathrm{pt})\cong\mathbb Z
\]
\textit{via the rank map.}
\end{exercise}

Fix a basepoint $x_0\in X$, regarded as a map $x_0:\mathrm{pt}\to X$. The \textbf{reduced K-theory} of the pointed space $(X,x_0)$ is
\[
\widetilde K^0(X)
=\ker\!\left(K^0(X)\xrightarrow{x_0^*}K^0(\mathrm{pt})\right).
\]
For a bundle $E$ of rank $r$, its \textbf{reduced class} is
\[
[E]-[\underline{\mathbb C}^{\,r}],
\]
where $\underline{\mathbb C}^{\,r}=X\times\mathbb C^r$ is the trivial rank-$r$ bundle. Topological K-theory extends $K^0$ to generalized cohomology groups $K^n(X)$. Bott periodicity gives
\[
K^{n+2}(X)\cong K^n(X),
\]
so complex K-theory is naturally $\mathbb Z/2$-graded.

\begin{exercise}[Stable equivalence]\leavevmode\par
\textit{Let $E$ and $F$ be complex vector bundles over a compact Hausdorff space $X$. Prove that $[E]=[F]$ in $K^0(X)$ if and only if there is a vector bundle $G$ such that}
\[
E\oplus G\cong F\oplus G.
\]
\textit{Deduce that the reduced class of $E$ is unchanged after replacing $E$ by $E\oplus\underline{\mathbb C}^{\,n}$.}
\end{exercise}

For the gapped families below, assume that $X$ is connected, so vector-bundle ranks are constant. Let $V$ be a finite-dimensional Hermitian vector space. A continuous family is a map
\[
H:X\longrightarrow\End(V)
\]
and we write $H_x=H(x)$. It is \textbf{gapped} if every $H_x$ is self-adjoint and invertible. Compactness gives numbers $\delta,M>0$ such that every eigenvalue of every $H_x$ lies in $[-M,-\delta]\cup[\delta,M]$. Choose a positively oriented contour $\Gamma$ enclosing $[-M,-\delta]$ and excluding $[\delta,M]$, and define
\[
P_x^-=\frac{1}{2\pi i}\int_\Gamma(z-H_x)^{-1}\,dz,
\qquad
E_x^-=\operatorname{im}(P_x^-).
\]

\begin{exercise}[Negative spectral bundle]\leavevmode\par
\textit{Let $H$ be a gapped family. Prove that}
\[
(P_x^-)^2=P_x^-,
\qquad
(P_x^-)^*=P_x^-,
\]
\textit{that $x\mapsto P_x^-$ is continuous, and that}
\[
E_H^-=\coprod_{x\in X}E_x^-
\]
\textit{is a complex vector bundle over $X$. Prove that its rank is locally constant.}
\end{exercise}

For a gapped family $H:X\times[0,1]\to\End(V)$, let $E_j^-$ be the negative spectral bundle of $x\mapsto H(x,j)$, and let $i_j:X\to X\times[0,1]$ be given by $i_j(x)=(x,j)$.

\begin{exercise}[Gap-preserving homotopy]\leavevmode\par
\textit{Prove that the negative spectral subspaces of $H$ form a vector bundle $E^-\to X\times[0,1]$. Prove that the endpoint pullbacks of any vector bundle over $X\times[0,1]$ are isomorphic, and apply this to $i_0^*E^-$ and $i_1^*E^-$. Deduce that}
\[
[E_0^-]=[E_1^-]\in K^0(X).
\]
\end{exercise}

Let $Q\in\End(W)$ be a fixed invertible self-adjoint endomorphism of a finite-dimensional Hermitian space $W$, let $W^-$ be its negative eigenspace, and write $\underline{W^-}=X\times W^-$.

\begin{exercise}[Stabilization]\leavevmode\par
\textit{For a gapped family $H$, prove that}
\[
E_{H\oplus Q}^-\cong E_H^-\oplus\underline{W^-}.
\]
\textit{Deduce that}
\[
[E_H^-]-[\underline{\mathbb C}^{\,\operatorname{rank}E_H^-}]
\in\widetilde K^0(X)
\]
\textit{is unchanged by adjoining a constant gapped summand.}
\end{exercise}

Let $E\subseteq X\times V$ be a subbundle and let $P_x$ be the orthogonal projection onto $E_x$.

\begin{exercise}[Flattened family]\leavevmode\par
\textit{Define $H_x=1_V-2P_x$. Prove that $x\mapsto H_x$ is a gapped family, that}
\[
H_x^2=1_V,
\]
\textit{and that its negative spectral bundle is $E$. Deduce that every subbundle of a trivial bundle arises as the negative spectral bundle of a gapped family.}
\end{exercise}

Thus a finite-dimensional gapped family determines a class in $\widetilde K^0(X)$ that is invariant under gap-preserving homotopy and stabilization. Conversely, every subbundle of a trivial bundle occurs as a negative spectral bundle.

In a finite-dimensional gapped family, the negative spectral subspaces form a vector bundle. Gap-preserving deformation preserves its stable isomorphism class, so the associated reduced K-theory class is a topological invariant. This construction underlies the classification of noninteracting topological insulators and superconductors under specified symmetry assumptions \cite{kitaev-periodic}.

The exterior powers $\Lambda^kT_p^*M$ are those defined in the Algebra section. To define integration in a coordinate-free way, we will integrate differential forms over chains. The space of \textbf{differential $n$-forms} is $\Omega^n(M)=\Gamma(\Lambda^nT^*M)$. On an oriented $n$-manifold, a \textbf{volume form} is a nowhere-vanishing element of $\Omega^n(M)$ that is positive with respect to the orientation. The exterior derivative $d:\Omega^n(M)\to\Omega^{n+1}(M)$ extends the differential of smooth functions and satisfies \cite{bott-tu}
\[
d(\alpha\wedge\beta)=d\alpha\wedge\beta+(-1)^p\alpha\wedge d\beta
\]
for $\alpha\in\Omega^p(M)$. A \textbf{singular $n$-simplex} is a smooth map $\sigma:\Delta^n \to M$ where $\Delta^n$ is the standard $n$-simplex. The collection of singular $n$-simplices in $M$ freely generates the abelian group $C_n(M)$ of \textbf{singular $n$-chains} in $M$. The \textbf{boundary map} $\partial : C_n(M) \to C_{n-1}(M)$ is defined on each simplex $\sigma:\Delta^n \to M$ by
$$\partial\sigma=\sum_{j=0}^{n}(-1)^j\sigma\circ f_j,$$ where $f_j : \Delta^{n-1} \to \Delta^n$ is the inclusion of the face opposite the $j$-th vertex of $\Delta^n$.

A \textbf{chain complex} $(A_n, d_n)_n$ is a sequence of abelian groups $A_n$ and homomorphisms $d_n : A_n \to A_{n-1}$ such that $d_{n-1} \circ d_n = 0$ for all $n$. The $n$-th \textbf{homology group} $H_n$ of the chain complex is
$$H_n = \ker(d_{n}) / \im(d_{n+1}).$$
The prototypical example is singular homology which corresponds to the chain complex of singular chains with the boundary map defined above.

Cohomology and cochains are defined similarly, using superscripts instead of subscripts and boundary maps $d^n : A^n \to A^{n+1}$ increase degree instead of decrease it. Singular cohomology corresponds to the cochain complex given by dualizing the singular chain complex above.

The \textbf{de Rham cohomology} of $M$ is
\[
H_{\mathrm{dR}}^n(M)
=
\ker(d\colon\Omega^n(M)\to\Omega^{n+1}(M))/
\operatorname{im}(d\colon\Omega^{n-1}(M)\to\Omega^n(M)).
\]

\begin{exercise}
Prove the following:
\begin{itemize}
\item Show that $d^2=0$, so differential forms on a manifold $M$ form a cochain complex.
\item Define the integral of an $n$-form on an $n$-chain. Prove that $\int_c d \omega = \int_{\partial c} \omega$ for any $n$-chain $c$ and differential $n-1$-form $\omega$.

\item Singular cohomology with real coefficients and de Rham cohomology are naturally isomorphic.

\item Bounded chain complexes of finite-dimensional vector spaces form a symmetric monoidal abelian category with duals, the trace of an endomorphism is its Lefschetz number, and the dimension of a chain complex is its Euler characteristic.
\end{itemize}
\end{exercise}

For a vector field $X$, contraction is the map $\iota_X:\Omega^p(M)\to\Omega^{p-1}(M)$ defined by
\[
(\iota_X\alpha)(X_2,\ldots,X_p)=\alpha(X,X_2,\ldots,X_p).
\]
The Lie derivative of forms is determined by Cartan's formula $\mathcal{L}_X=d\iota_X+\iota_Xd$.

A \textbf{symplectic form} is a two-form $\omega\in\Omega^2(M)$ such that $d\omega=0$ and $X\mapsto\iota_X\omega$ is a bundle isomorphism $TM\to T^*M$. Write $C^\infty(M)$ for the smooth real-valued functions on $M$. For $H\in C^\infty(M)$, define $X_H$ by
\[
\iota_{X_H}\omega=dH.
\]

\begin{exercise}[Liouville's theorem]\leavevmode\par
\textit{Let $(M,\omega)$ be $2n$-dimensional with $\omega$ symplectic. Prove that $\omega^n/n!$ is a volume form. For $H\in C^\infty(M)$, prove using Cartan's formula that} \cite{arnold}
\[
\mathcal{L}_{X_H}\omega=0.
\]
\textit{Deduce that $\mathcal{L}_{X_H}\omega^n=0$ and hence that the local flow of $X_H$ preserves the volume form}
\[
\frac{\omega^n}{n!}.
\]
\end{exercise}

For $f,g\in C^\infty(M)$, define
\[
\{f,g\}=\omega(X_f,X_g).
\]

The following is a Hamiltonian symmetry--conservation statement, not the full variational theorem of \cite{noether}.

\begin{exercise}[Hamiltonian Noether theorem]\leavevmode\par
\textit{Let $(M,\omega)$ be symplectic. Prove that} \cite{arnold}
\[
X_g(f)=\{f,g\}.
\]
\textit{Let $\Phi_t$ be the local flow of $X_f$. Prove that the following are equivalent:}
\begin{itemize}
\item $H\circ\Phi_t=H$ whenever both sides are defined.
\item $X_f(H)=0$.
\item $\{H,f\}=0$.
\item $f$ is constant along the integral curves of $X_H$.
\end{itemize}
\end{exercise}

A \textbf{pseudo-Riemannian metric} is a smooth nondegenerate symmetric section $g\in\Gamma(T^*M\otimes T^*M)$. On an oriented pseudo-Riemannian $n$-manifold, the metric induces a bilinear form $\langle\cdot,\cdot\rangle_g$ on differential forms. The \textbf{metric volume form} $\operatorname{vol}_g$ is the unique positive volume form satisfying
\[
\left|\langle\operatorname{vol}_g,\operatorname{vol}_g\rangle_g\right|=1.
\]
The Hodge star $\star$ is determined by
\[
\alpha\wedge\star\beta=\langle\alpha,\beta\rangle_g\operatorname{vol}_g.
\]

\begin{exercise}[Charge conservation from Maxwell's equations]\leavevmode\par
\textit{Let $M$ be an oriented four-dimensional pseudo-Riemannian manifold. Suppose that $F\in\Omega^2(M)$ and $J\in\Omega^3(M)$ satisfy} \cite{frankel}
\[
dF=0,
\qquad
d\star F=J.
\]
\textit{Prove that $dJ=0$. If $C$ is an oriented compact four-dimensional chain, deduce that}
\[
\int_{\partial C}J=0.
\]
\textit{If the three-dimensional pieces of its boundary satisfy}
\[
\partial C=\Sigma_1-\Sigma_0+B,
\]
\textit{prove that}
\[
\int_{\Sigma_1}J-\int_{\Sigma_0}J=-\int_BJ.
\]
\end{exercise}

For a closed form $A$, write $[A]$ for its de Rham cohomology class. The next exercise is a modern de Rham formulation; the original paper does not use this language \cite{aharonov-bohm,frankel}.

\begin{exercise}[Aharonov--Bohm cohomological obstruction]
\textit{Let $M$ be a compact smooth manifold and let $A\in\Omega^1(M)$ satisfy $dA=0$. Prove the following:}
\begin{itemize}
\item For every contractible open set $U\subseteq M$, there exists $\lambda\in C^\infty(U)$ such that $A|_U=d\lambda$.
\item For every smooth closed curve $\gamma$, the number $\int_\gamma A$ depends only on the homology class of $\gamma$.
\item There exists a global $\lambda\in C^\infty(M)$ with $A=d\lambda$ iff $[A]=0\in H_{\mathrm{dR}}^1(M)$.
\item If $[A]\neq0$, then there exists a smooth closed curve $\gamma$ such that $\int_\gamma A\neq0$.
\end{itemize}
\end{exercise}

An \textbf{affine connection} is a map $\nabla : \Gamma(TM) \times \Gamma(TM) \to \Gamma(TM)$ such that
\[
\nabla_{fX}Y=f\nabla_XY,
\qquad
\nabla_X(fY)=X(f)Y+f\nabla_XY.
\]
For a covector field $\alpha$, set
\[
(\nabla_X\alpha)(Y)=X(\alpha(Y))-\alpha(\nabla_XY).
\]
The connection extends uniquely to all tensor fields by the product rule and by commuting with contractions; this defines the \textbf{covariant derivative} of a tensor field.

The \textbf{Levi-Civita connection} of a pseudo-Riemannian metric is the unique affine connection satisfying
\[
\nabla_XY-\nabla_YX=[X,Y],
\qquad
\nabla g=0.
\]
An affinely parametrized curve $\gamma:I\to M$ is a \textbf{geodesic} if $\nabla_{\dot\gamma}\dot\gamma=0$. The Lie derivative of the metric is
\[
(\mathcal L_Xg)(Y,Z)
=X(g(Y,Z))-g([X,Y],Z)-g(Y,[X,Z]),
\]
and $X$ is a \textbf{Killing vector field} if $\mathcal L_Xg=0$.

Let $(M,g)$ be a pseudo-Riemannian manifold with Levi-Civita connection $\nabla$. Curvature governs the relative acceleration of geodesics, while symmetries of $g$ determine quantities preserved along them \cite{oneill,wald}.

\begin{exercise}[Killing fields and conserved quantities]\leavevmode\par
\textit{Let $X$ be a Killing vector field and let $\gamma$ be an affinely parametrized geodesic. Prove that}
\[
\frac{d}{dt}g(\dot\gamma,\dot\gamma)=0,
\qquad
\frac{d}{dt}g(X,\dot\gamma)=0.
\]
\textit{Deduce that $g(X,\dot\gamma)$ is constant along $\gamma$.}
\end{exercise}

The \textbf{curvature} of $\nabla$ is the map
\[
R(X,Y)Z
=\nabla_X\nabla_YZ-\nabla_Y\nabla_XZ-\nabla_{[X,Y]}Z.
\]

\begin{exercise}[Curvature as a tensor]\leavevmode\par
\textit{For $f\in C^\infty(M)$ and vector fields $X,Y,Z$, prove that}
\[
R(fX,Y)Z=fR(X,Y)Z,
\qquad
R(X,fY)Z=fR(X,Y)Z,
\]
\[
R(X,Y)(fZ)=fR(X,Y)Z.
\]
\textit{Verify explicitly that every derivative of $f$ cancels. Prove that $R(X,Y)=-R(Y,X)$ and deduce that}
\[
R\in\Gamma\!\left(\Lambda^2T^*M\otimes T^*M\otimes TM\right).
\]
\end{exercise}

Let $\Gamma:(-\varepsilon,\varepsilon)\times I\to M$ be a smooth variation through affinely parametrized geodesics, and define the vector fields along $\Gamma$ by
\[
T=\frac{\partial\Gamma}{\partial t},
\qquad
J=\frac{\partial\Gamma}{\partial s}.
\]
Then $[T,J]=0$.

\begin{exercise}[Geodesic deviation]
\textit{For a variation $\Gamma$ as above, prove that}
\[
\nabla_T\nabla_TJ=R(T,J)T.
\]
\end{exercise}

Thus curvature determines the relative acceleration of nearby geodesics.

The trace used below is the ordinary trace defined in the Algebra section. Define the \textbf{Ricci tensor}, \textbf{scalar curvature}, and \textbf{Einstein tensor} by
\[
\operatorname{Ric}(X,Y)=\operatorname{tr}\bigl(Z\mapsto R(Z,X)Y\bigr),
\qquad
S=\operatorname{tr}_g(\operatorname{Ric}),
\qquad
G=\operatorname{Ric}-\frac12Sg.
\]
For a symmetric covariant two-tensor $A$, its \textbf{divergence} is the covector field
\[
(\operatorname{div}A)(Y)
=\operatorname{tr}_g\bigl((X,Z)\mapsto(\nabla_XA)(Z,Y)\bigr).
\]

\begin{exercise}[Contracted Bianchi identity and conservation]\leavevmode\par
\textit{Assume the second Bianchi identity}
\[
(\nabla_XR)(Y,Z)+(\nabla_YR)(Z,X)+(\nabla_ZR)(X,Y)=0.
\]
\textit{Contract this identity to prove that}
\[
\operatorname{div}G=0.
\]
\textit{Deduce that if $T$ is a symmetric covariant two-tensor satisfying}
\[
G+\Lambda g=\kappa T
\]
\textit{for constants $\Lambda,\kappa$ with $\kappa\ne0$, then $\operatorname{div}T=0$.}
\end{exercise}

Connections and curvature also determine characteristic classes. In odd dimensions, their transgression forms define Chern--Simons functionals.

Let $E\to M$ be a complex vector bundle and write $\Omega^0(M,E)=\Gamma(E)$. A \textbf{connection} on $E$ is a $\mathbb C$-linear map
\[
\nabla:\Omega^0(M,E)\longrightarrow\Omega^1(M,E)
\]
satisfying $\nabla(fs)=df\otimes s+f\nabla s$. For $\alpha\in\Omega^p(M)$ and an $E$-valued form $\eta$, extend it by
\[
\nabla(\alpha\wedge\eta)
=d\alpha\wedge\eta+(-1)^p\alpha\wedge\nabla\eta.
\]
Its \textbf{curvature} is
\[
F_\nabla=\nabla^2\in\Omega^2(M,\End E).
\]
The connection induces one on $\End E$-valued forms.

\begin{exercise}[Bianchi identity]
\textit{Let $\nabla$ be a connection on $E\to M$. Prove that}
\[
\nabla F_\nabla=0.
\]
\end{exercise}

Let $P$ be an invariant symmetric $k$-linear polynomial on the endomorphism algebra of a fiber of $E$, where invariant means invariant under simultaneous conjugation. Define \cite{chern-simons}
\[
P(F_\nabla)
=P(F_\nabla,\ldots,F_\nabla)
\in\Omega^{2k}(M).
\]

\begin{exercise}[Chern--Weil theorem]\leavevmode\par
\textit{Prove that}
\[
dP(F_\nabla)=0.
\]
\textit{For two connections $\nabla^0,\nabla^1$ on $E$, prove that}
\[
[P(F_{\nabla^0})]
=[P(F_{\nabla^1})]
\in H_{\mathrm{dR}}^{2k}(M).
\]
\end{exercise}

Define the \textbf{Chern character form} by
\[
\operatorname{ch}(E,\nabla)
=\operatorname{tr}\exp\left(\frac{iF_\nabla}{2\pi}\right).
\]
When induced connections are understood, suppress them from the notation.

\begin{exercise}[Chern character]\leavevmode\par
\textit{Prove that $d\,\operatorname{ch}(E,\nabla)=0$ and that its de Rham class is independent of $\nabla$. For a second complex vector bundle $F$ with connection, prove that}
\[
\operatorname{ch}(E\oplus F)
=\operatorname{ch}(E)+\operatorname{ch}(F),
\]
\[
\operatorname{ch}(E\otimes F)
=\operatorname{ch}(E)\wedge\operatorname{ch}(F).
\]
\textit{If $M$ is compact, deduce that the Chern character defines a ring homomorphism}
\[
\operatorname{ch}:K^0(M)\longrightarrow H_{\mathrm{dR}}^{\mathrm{even}}(M),
\qquad
H_{\mathrm{dR}}^{\mathrm{even}}(M)=\bigoplus_{j\geq0}H_{\mathrm{dR}}^{2j}(M).
\]
\end{exercise}

For two connections on $E$, set
\[
\nabla^t=(1-t)\nabla^0+t\nabla^1,
\qquad
A=\nabla^1-\nabla^0,
\qquad
F_t=F_{\nabla^t},
\]
where $A\in\Omega^1(M,\End E)$, and define
\[
\operatorname{CS}_P(\nabla^0,\nabla^1)
=k\int_0^1P(A,F_t,\ldots,F_t)\,dt.
\]
Thus $\operatorname{CS}_P(\nabla^0,\nabla^1)\in\Omega^{2k-1}(M)$.

\begin{exercise}[Chern--Simons transgression]\leavevmode\par
\textit{Prove that}
\[
P(F_{\nabla^1})-P(F_{\nabla^0})
=d\,\operatorname{CS}_P(\nabla^0,\nabla^1).
\]
\end{exercise}

Let $\mathfrak g\subseteq M_N(\mathbb C)$ be a matrix Lie algebra whose trace pairing is nondegenerate. For $A\in\Omega^1(M,\mathfrak g)$, wedge products include matrix multiplication. Define
\[
F_A=dA+A\wedge A
\]
and
\[
\operatorname{CS}(A)
=\operatorname{tr}\left(
A\wedge dA+\frac23A\wedge A\wedge A
\right).
\]

\begin{exercise}[Chern--Simons form]\leavevmode\par
\textit{Prove that}
\[
d\,\operatorname{CS}(A)
=\operatorname{tr}(F_A\wedge F_A).
\]
\end{exercise}

For a closed oriented three-manifold $M$, define
\[
S_{\mathrm{CS}}(A)=\int_M\operatorname{CS}(A).
\]

\begin{exercise}[Chern--Simons equation]\leavevmode\par
\textit{For $\alpha\in\Omega^1(M,\mathfrak g)$, set $A_t=A+t\alpha$. Prove that}
\[
\left.\frac{d}{dt}\right|_{t=0}S_{\mathrm{CS}}(A_t)
=2\int_M\operatorname{tr}(\alpha\wedge F_A).
\]
\textit{Deduce that the critical points satisfy $F_A=0$.}
\end{exercise}

Let $G$ be a matrix Lie group with Lie algebra $\mathfrak g$. For a smooth map $g:M\to G$, define
\[
A^g=g^{-1}Ag+g^{-1}dg.
\]
Assume that $G$ is compact, connected, simple, and simply connected, and normalize the invariant trace so that
\[
\nu(g)=\frac{1}{24\pi^2}\int_M
\operatorname{tr}\bigl((g^{-1}dg)^3\bigr)\in\mathbb Z.
\]
For $k\in\mathbb R$, set $S_k=(k/4\pi)S_{\mathrm{CS}}$.

\begin{exercise}[Gauge transformation]\leavevmode\par
\textit{Prove that}
\[
\operatorname{CS}(A^g)-\operatorname{CS}(A)
=-d\,\operatorname{tr}(dg\,g^{-1}\wedge A)
-\frac13\operatorname{tr}\bigl((g^{-1}dg)^3\bigr).
\]
\textit{Deduce that $\exp(iS_k)$ is invariant under all gauge transformations when $k\in\mathbb Z$.}
\end{exercise}

Now restrict to $2+1$ spacetime dimensions. Let $M$ be a closed oriented three-manifold and let $\ell>0$, so the cosmological constant is $\Lambda=-\ell^{-2}$. After identifying the coframe representation with $\mathfrak{sl}(2,\mathbb R)$, let
\[
e,\omega\in\Omega^1(M,\mathfrak{sl}(2,\mathbb R)),
\qquad
A^\pm=\omega\pm\frac1\ell e.
\]
The pair $(A^+,A^-)$ has gauge Lie algebra $\mathfrak{sl}(2,\mathbb R)\oplus\mathfrak{sl}(2,\mathbb R)$.
Write
\[
d_\omega e=de+\omega\wedge e+e\wedge\omega,
\qquad
F_\omega=d\omega+\omega\wedge\omega.
\]

\begin{exercise}[Three-dimensional gravity]\leavevmode\par
\textit{Prove that $F_{A^+}=F_{A^-}=0$ is equivalent to}
\[
d_\omega e=0,
\qquad
F_\omega+\frac1{\ell^2}e\wedge e=0.
\]
\textit{Assuming that $e:TM\to\mathfrak{sl}(2,\mathbb R)$ is fiberwise invertible, identify these equations with the torsion-free condition and the Einstein equation with constant negative curvature.}
\end{exercise}

Define the first-order three-dimensional Einstein--Hilbert functional, up to its conventional overall coefficient, by
\[
S_{\mathrm{EH}}(e,\omega)
=\int_M\operatorname{tr}\left(
e\wedge F_\omega+\frac{1}{3\ell^2}e\wedge e\wedge e
\right).
\]

\begin{exercise}[Gravity as Chern--Simons theory]\leavevmode\par
\textit{Prove that, up to an overall normalization and an exact term,}
\[
S_{\mathrm{CS}}(A^+)-S_{\mathrm{CS}}(A^-)
\]
\textit{equals $S_{\mathrm{EH}}(e,\omega)$} \cite{achucarro-townsend,witten-gravity}.
\end{exercise}

This is a local Lie-algebra formulation of gravity in three spacetime dimensions. Changing between Lorentzian and Euclidean signatures changes the relevant real form and reality conditions. It is an analytic continuation, not a statement that four-dimensional gravity is obtained from Chern--Simons theory by Wick rotation.

For a connection $A$ and a finite-dimensional representation $V$ of its gauge group, parallel transport around an oriented closed curve $\gamma$ gives the holonomy $\operatorname{Hol}_\gamma(A)$. Define the \textbf{Wilson loop}
\[
W_V(\gamma,A)
=\operatorname{tr}_V\!\left(\operatorname{Hol}_\gamma(A)\right).
\]
For a framed oriented link $L=\gamma_1\sqcup\cdots\sqcup\gamma_r$ labeled by $V_1,\ldots,V_r$, set
\[
W_L(A)=\prod_{j=1}^rW_{V_j}(\gamma_j,A).
\]
At the level of this document, the Chern--Simons expectation value
\[
\langle W_L\rangle_k
=
\frac{\int e^{iS_k(A)}W_L(A)\,\mathcal DA}
{\int e^{iS_k(A)}\,\mathcal DA}
\]
is a formal path-integral expression, and its value depends on the chosen framing \cite{witten-jones}. For normalized $SU(2)$ theory at integer level $k\geq1$, set
\[
q=\exp\left(\frac{2\pi i}{k+2}\right).
\]

\begin{exercise}[Wilson loops and the Jones polynomial]\leavevmode\par
\textit{Let $L_+$, $L_-$, and $L_0$ differ only by a positive crossing, a negative crossing, and the oriented smoothing. Assume that, after the standard framing normalization, the Wilson-loop expectation values satisfy}
\[
q^{-1}\langle W_{L_+}\rangle_k
-q\langle W_{L_-}\rangle_k
=\left(q^{1/2}-q^{-1/2}\right)\langle W_{L_0}\rangle_k
\]
\textit{and $\langle W_{\bigcirc}\rangle_k=1$. Prove that these relations determine the Jones polynomial of an oriented link.}
\end{exercise}

Chern--Simons theory relates the three parts of the document. Its Wilson lines carry braid-group actions governed algebraically by braided tensor categories, while the resulting link invariants are realized analytically through Temperley--Lieb algebras and subfactors \cite{reshetikhin-turaev}. These connections are indicated here, not proved as a full equivalence.

\section{Analysis}

A \textbf{unital $C^*$-algebra} is a complex unital Banach algebra $A$ with an involution $a\mapsto a^*$ satisfying
\[
\|a^*a\|=\|a\|^2.
\]
Its positive cone is $A_+=\{a^*a:a\in A\}$. A \textbf{state} is a linear map $\varphi:A\to\mathbb C$ such that \cite{paulsen}
\[
\varphi(a)\geq0\quad(a\in A_+),
\qquad
\varphi(1)=1.
\]

\begin{exercise}
\textit{Let $\varphi$ be a positive linear functional on a unital $C^*$-algebra $A$. Prove that}
\[
|\varphi(b^*a)|^2
\leq
\varphi(a^*a)\varphi(b^*b).
\]
\textit{Deduce that $\|\varphi\|=\varphi(1)$.}
\end{exercise}

For a state $\varphi$, define
\[
\langle a,b\rangle_\varphi=\varphi(b^*a),
\qquad
N_\varphi=\{a\in A:\varphi(a^*a)=0\},
\]
and let $H_\varphi$ be the Hilbert-space completion of $A/N_\varphi$. For a Hilbert space $H$, write $\mathcal B(H)$ for the $C^*$-algebra of bounded operators on $H$. A vector $\Omega$ is \textbf{cyclic} for a representation $\pi$ if the span of $\pi(A)\Omega$ is dense.

\begin{exercise}[GNS construction]\leavevmode\par
\textit{Let $\varphi$ be a state on a unital $C^*$-algebra $A$. Prove that left multiplication induces a unital $*$-representation}
\[
\pi_\varphi:A\longrightarrow\mathcal B(H_\varphi)
\]
\textit{with a cyclic vector $\Omega_\varphi$ satisfying}
\[
\varphi(a)
=\langle\pi_\varphi(a)\Omega_\varphi,\Omega_\varphi\rangle_\varphi.
\]
\textit{Prove that $(H_\varphi,\pi_\varphi,\Omega_\varphi)$ is unique up to unitary equivalence.}
\end{exercise}

For a linear map $\Phi:A\to B$ between unital $C^*$-algebras, its $n$-th \textbf{amplification} is
\[
\Phi_n:M_n(A)\longrightarrow M_n(B),
\qquad
[a_{ij}]\longmapsto[\Phi(a_{ij})].
\]
The map $\Phi$ is \textbf{completely positive} if every $\Phi_n$ is positive. Its \textbf{completely bounded norm} is
\[
\|\Phi\|_{\mathrm{cb}}=\sup_{n\geq1}\|\Phi_n\|.
\]

\begin{exercise}
\textit{For $n\geq2$, let}
\[
T:M_n(\mathbb C)\longrightarrow M_n(\mathbb C),
\qquad
T(a)=a^{\mathsf T}.
\]
\textit{Prove that $T$ is positive but not completely positive, and prove that}
\[
\|T\|_{\mathrm{cb}}=n.
\]
\end{exercise}

For the standard matrix units $E_{ij}\in M_n(\mathbb C)$ and a linear map $\Phi:M_n(\mathbb C)\to M_m(\mathbb C)$, define the \textbf{Choi matrix}
\[
C_\Phi=\sum_{i,j=1}^n E_{ij}\otimes\Phi(E_{ij}).
\]

\begin{exercise}[Choi theorem]\leavevmode\par
\textit{Let $\Phi:M_n(\mathbb C)\to M_m(\mathbb C)$ be linear. Prove that the following are equivalent} \cite{choi}:
\begin{itemize}
\item $\Phi$ is completely positive.
\item $C_\Phi\geq0$.
\item There are linear maps $V_r:\mathbb C^m\to\mathbb C^n$ such that
\[
\Phi(a)=\sum_rV_r^*aV_r.
\]
\end{itemize}
\end{exercise}

\begin{exercise}[Stinespring dilation]\leavevmode\par
\textit{Let $\Phi:A\to\mathcal B(H)$ be completely positive. Prove that there are a Hilbert space $K$, a $*$-representation $\pi:A\to\mathcal B(K)$, and a bounded operator $V:H\to K$ such that} \cite{stinespring}
\[
\Phi(a)=V^*\pi(a)V.
\]
\textit{Prove that $V$ is an isometry when $\Phi$ is unital. Call the dilation minimal if the span of $\pi(A)VH$ is dense in $K$, and prove that a minimal dilation is unique up to unitary equivalence.}
\end{exercise}

For a unital completely positive map $\Phi:A\to B$, define its \textbf{multiplicative domain} by
\[
\operatorname{MD}(\Phi)
=
\left\{
a\in A:
\begin{array}{l}
\Phi(a^*a)=\Phi(a)^*\Phi(a),\\
\Phi(aa^*)=\Phi(a)\Phi(a)^*
\end{array}
\right\}.
\]

\begin{exercise}[Multiplicative domain]\leavevmode\par
\textit{Let $\Phi:A\to B$ be unital and completely positive. Prove the Kadison--Schwarz inequality}
\[
\Phi(a)^*\Phi(a)\leq\Phi(a^*a).
\]
\textit{Prove that $\operatorname{MD}(\Phi)$ is a $C^*$-subalgebra of $A$ and that, for $a\in\operatorname{MD}(\Phi)$ and $b\in A$,}
\[
\Phi(ab)=\Phi(a)\Phi(b),
\qquad
\Phi(ba)=\Phi(b)\Phi(a).
\]
\end{exercise}

For $\mathcal S\subseteq\mathcal B(H)$, its \textbf{commutant} is
\[
\mathcal S'=\{T\in\mathcal B(H):Tx=xT\text{ for every }x\in\mathcal S\}.
\]
A \textbf{von Neumann algebra} is a unital $*$-subalgebra $M\subseteq\mathcal B(H)$ satisfying $M=M''$ \cite{takesaki}. A net $(T_\lambda)$ converges to $T$ in the \textbf{strong operator topology} if $T_\lambda\xi\to T\xi$ for every $\xi\in H$, and in the \textbf{weak operator topology} if
\[
\langle T_\lambda\xi,\eta\rangle\longrightarrow\langle T\xi,\eta\rangle
\]
for every $\xi,\eta\in H$.

\begin{exercise}[Bicommutant theorem]\leavevmode\par
\textit{Let $A\subseteq\mathcal B(H)$ be a unital $*$-subalgebra. Prove that}
\[
A''
=\overline A^{\,\mathrm{SOT}}
=\overline A^{\,\mathrm{WOT}}.
\]
\end{exercise}

The \textbf{center} of a von Neumann algebra is $Z(M)=M\cap M'$, and $M$ is a \textbf{factor} if $Z(M)=\mathbb C1$. A positive functional or positive map is \textbf{normal} if it preserves suprema of bounded increasing nets of positive elements, and a positive functional $\psi$ is \textbf{faithful} if $\psi(a^*a)=0$ implies $a=0$. A \textbf{type $\mathrm{II}_1$ factor} is an infinite-dimensional factor with a faithful normal tracial state
\[
\tau:M\longrightarrow\mathbb C,
\qquad
\tau(ab)=\tau(ba),
\qquad
\tau(1)=1.
\]
The normalized trace is an analytic analogue of the categorical trace in Algebra, not an identification with it.

For projections $p,q\in M$, write $p\sim q$ if there is $v\in M$ such that
\[
v^*v=p,
\qquad
vv^*=q.
\]
Write $p\precsim q$ if $p$ is equivalent to a projection $r$ satisfying $rq=r$.

\begin{exercise}[Projection dimension]\leavevmode\par
\textit{Let $M$ be a type $\mathrm{II}_1$ factor with normalized trace $\tau$. Prove that}
\[
p\sim q\quad\Longrightarrow\quad\tau(p)=\tau(q),
\]
\textit{and prove the comparison theorem}
\[
p\precsim q
\quad\Longleftrightarrow\quad
\tau(p)\leq\tau(q).
\]
\textit{Deduce that $\tau(p)=\tau(q)$ if and only if $p\sim q$.}
\end{exercise}

A von Neumann algebra is \textbf{finite} if $v^*v=1$ implies $vv^*=1$. Let $N\subseteq M$ be finite von Neumann algebras with a common normalized trace $\tau$. A \textbf{trace-preserving conditional expectation} is a normal unital completely positive map $E_N:M\to N$ such that
\[
\tau\circ E_N=\tau,
\qquad
E_N(n_1xn_2)=n_1E_N(x)n_2
\]
for $x\in M$ and $n_1,n_2\in N$. Let $L^2(M,\tau)$ be the completion of $M$ for
\[
\langle x,y\rangle_\tau=\tau(y^*x).
\]
Let $M$ act on $L^2(M,\tau)$ by left multiplication.

\begin{exercise}
\textit{Prove that $E_N$ extends to the orthogonal projection}
\[
e_N:L^2(M,\tau)\longrightarrow L^2(N,\tau).
\]
\textit{Prove that}
\[
e_N=e_N^*=e_N^2,
\qquad
e_Nxe_N=E_N(x)e_N
\]
\textit{for every $x\in M$.}
\end{exercise}

The \textbf{basic construction} is
\[
M_1=\langle M,e_N\rangle''
\subseteq\mathcal B(L^2(M,\tau)),
\]
the double commutant of the algebra generated by $M$ and $e_N$. For a projection $p=[p_{ij}]\in M_k(N)$ acting by left multiplication, define
\[
\dim_N\!\left(pL^2(N,\tau)^k\right)=\sum_{i=1}^k\tau(p_{ii}).
\]
An inclusion $N\subseteq M$ has \textbf{finite Jones index} if $L^2(M,\tau)$ is isomorphic as a right $N$-module to such a module; then
\[
[M:N]=\dim_NL^2(M,\tau).
\]

\begin{exercise}[Basic construction]\leavevmode\par
\textit{Let $N\subseteq M$ be an inclusion of type $\mathrm{II}_1$ factors of finite Jones index. Prove that $M_1$ is a finite factor and that its normalized trace satisfies} \cite{jones-sunder}
\[
\tau_1(xe_Ny)
=\frac{1}{[M:N]}\tau(xy)
\]
\textit{for $x,y\in M$.}
\end{exercise}

Iterating the basic construction gives the \textbf{Jones tower}
\[
N=M_{-1}\subset M=M_0\subset M_1\subset M_2\subset\cdots.
\]
Let $e_i$ be the Jones projection used to construct $M_i$ from $M_{i-1}$, and set
\[
\delta=\sqrt{[M:N]}.
\]
The \textbf{Temperley--Lieb algebra} with loop parameter $\delta$ is generated by $U_i$ subject to
\[
U_i^2=\delta U_i,
\qquad
U_iU_j=U_jU_i\quad(|i-j|\geq2),
\qquad
U_iU_{i\pm1}U_i=U_i.
\]

\begin{exercise}[Temperley--Lieb relations]\leavevmode\par
\textit{Prove that the Jones projections satisfy} \cite{temperley-lieb,jones-index}
\[
e_i^2=e_i=e_i^*,
\qquad
e_ie_j=e_je_i\quad(|i-j|\geq2),
\]
\[
e_ie_{i\pm1}e_i=\delta^{-2}e_i.
\]
\textit{Deduce that $U_i\mapsto\delta e_i$ defines a representation of the Temperley--Lieb algebra.}
\end{exercise}

Let $q\in\mathbb C^\times$ satisfy $\delta=q+q^{-1}$ and define
\[
\sigma_i=q\,1-U_i.
\]
These operators reconnect the Temperley--Lieb construction with the braid groups introduced in Algebra.

\begin{exercise}[Braid representation]\leavevmode\par
\textit{Prove that $\sigma_i$ is invertible with inverse $q^{-1}1-U_i$, and prove that}
\[
\sigma_i\sigma_{i+1}\sigma_i
=\sigma_{i+1}\sigma_i\sigma_{i+1},
\]
\[
\sigma_i\sigma_j=\sigma_j\sigma_i
\qquad
(|i-j|\geq2).
\]
\textit{Deduce that the $\sigma_i$ define a representation of the braid group} \cite{jones-polynomial}.
\end{exercise}

The compatible normalized traces on the Jones tower restrict to a positive \textbf{Markov trace} on the Temperley--Lieb algebras satisfying
\[
\operatorname{tr}(xU_i)=\delta^{-1}\operatorname{tr}(x)
\]
whenever $x$ belongs to the algebra generated by the preceding generators. Define
\[
P_0(\delta)=1,
\qquad
P_1(\delta)=\delta,
\qquad
P_{n+1}(\delta)=\delta P_n(\delta)-P_{n-1}(\delta).
\]

\begin{exercise}
\textit{Prove that, when $\sin\theta\ne0$,}
\[
P_n(2\cos\theta)
=\frac{\sin((n+1)\theta)}{\sin\theta}.
\]
\textit{Use positivity of the Markov trace to prove that if $0<\delta<2$, then}
\[
\delta=2\cos\left(\frac{\pi}{m}\right)
\]
\textit{for some integer $m\geq3$.}
\end{exercise}

\begin{exercise}[Jones index theorem]\leavevmode\par
\textit{Let $N\subseteq M$ be an inclusion of type $\mathrm{II}_1$ factors of finite Jones index. Prove that} \cite{jones-index,jones-sunder}
\[
[M:N]
\in
\left\{
4\cos^2\left(\frac{\pi}{n}\right):n\geq3
\right\}
\cup
[4,\infty).
\]
\end{exercise}

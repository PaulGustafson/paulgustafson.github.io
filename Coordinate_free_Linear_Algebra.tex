\section{Algebra}

\subsection{Linear algebra}

Fix a field $k$. A \textbf{$k$-vector space} is an abelian group $(V,+)$
with a scalar multiplication $k\times V\to V$ satisfying
\[
\begin{gathered}
\alpha(v+w)=\alpha v+\alpha w,\qquad
(\alpha+\beta)v=\alpha v+\beta v,\\
(\alpha\beta)v=\alpha(\beta v),\qquad 1v=v
\end{gathered}
\]
for $\alpha,\beta\in k$ and $v,w\in V$. A \textbf{linear map}
$f:V\to U$ between $k$-vector spaces
satisfies
\[
f(\alpha v+\beta w)=\alpha f(v)+\beta f(w)
\]
for $\alpha,\beta\in k$ and $v,w\in V$.

\begin{exercise}[Small string vibrations obey superposition]
\label{ex:wave-superposition}\leavevmode\par
\textit{Derive the wave equation starting from a taut string and show that
its solutions form a vector space. Show that time evolution from initial data
to the state at time $t$ is a linear map.}
\end{exercise}

The \textbf{free $k$-vector space on a set $X$}, denoted $kX$, is the set of finitely supported set-theoretical functions $X \to k$ with pointwise addition and scalar multiplication. The \textbf{direct sum} vector space $V \oplus W$ is the set $V \times W$ with pointwise addition and scalar multiplication defined by $\alpha (v \oplus w) = \alpha v \oplus \alpha w$. The \textbf{quotient} of $V$ by a subspace $W$, denoted $V / W$, is the set of equivalence classes of $V$ modulo the relation $x \sim y$ iff $x - y \in W$. The \textbf{tensor product} $V\otimes W$ is $k(V\times W)/N$, where $N$ is spanned by
\[
\begin{aligned}
(v+v',w)&-(v,w)-(v',w),&
(v,w+w')&-(v,w)-(v,w'),\\
(\lambda v,w)&-\lambda(v,w),&
(v,\lambda w)&-\lambda(v,w)
\end{aligned}
\]
for $v,v'\in V$, $w,w'\in W$, and $\lambda\in k$.

An \textbf{isomorphism} of vector spaces $V\cong W$ consists of linear maps $f:V\to W$ and $g:W\to V$ such that $f\circ g=1_W$ and $g\circ f=1_V$.

\begin{exercise}[Universal constructions]\leavevmode\par
Give the universal properties of free spaces, direct sums, quotients, and
tensor products, and use them to prove the three isomorphism theorems and
$k(X\sqcup Y)\cong kX\oplus kY$, $k(X\times Y)\cong kX\otimes kY$.
\end{exercise}

The \textbf{dual space} $V^*$ is the space of linear maps from a vector space $V$ to its ground field $k$. The evaluation map $\ev_V : V^* \otimes V \to k$ is the linear extension of the evaluation morphism $f \otimes x \mapsto f(x)$. We say that $V$ is \textbf{finite dimensional} if there is a co-evaluation morphism $\coev_V: k \to V \otimes V^*$ satisfying the snake equations $(1_{V} \otimes \ev_V) \circ (\coev_V \otimes 1_V) = 1_V$
and
$(\ev_V \otimes 1_{V^*}) \circ (1_{V^*} \otimes \coev_{V}) = 1_{V^*}$. We define the \textbf{dimension} $\dim(V) = \trace(1_V)$, where $\trace : \End(V) \to k$ is given by $f \mapsto (\ev_V \circ s_{V, V^*} \circ (f \otimes 1_{V^*}) \circ \coev_V)(1)$ where $s_{X, Y} : X \otimes Y \to Y \otimes X$ is the symmetric braiding $x \otimes y \mapsto y \otimes x$.



\begin{exercise}[Trace and dimension identities]\label{ex:linear-trace-dimension}\leavevmode\par
Prove the following for finite-dimensional vector spaces $V,W$.
    \begin{itemize}
        \item If $f:V\to W$ and $g:W\to V$, then
        $\trace(gf)=\trace(fg)$.
        \item If $f\in\End(V)$ and $g\in\End(W)$, then
        $\trace(f \oplus g) = \trace(f) + \trace(g)$.
        \item If $f\in\End(V)$ and $g\in\End(W)$, then
        $\trace(f \otimes g) = \trace(f) \trace(g)$.
        \item If $\operatorname{char}(k)=0$, then $\dim(V)\in\mathbf{N}\subset k$
        \item If $\operatorname{char}(k)=0$, then $V\cong W$ iff $\dim(V)=\dim(W)$
    \end{itemize}
\end{exercise}


Define the tensor algebra and \textbf{exterior algebra of $V$} by
\[
T(V)=\bigoplus_{k\geq 0}V^{\otimes k},
\qquad
\Lambda V=T(V)/\langle v\otimes v:v\in V\rangle.
\]
The space $\Lambda^kV$ is the degree-$k$ component of $\Lambda V$.

\begin{exercise}[The top exterior power is one-dimensional]
    Let $V$ be finite-dimensional with $n=\dim(V)$. Show that
    $\Lambda^nV$ is one-dimensional and that $\Lambda^kV=0$ for all $k>n$.
\end{exercise}

The \textbf{determinant} $\det(f)$ is the coefficient of the linear map on the highest-graded component of $\Lambda V$ given by $v_1 \wedge \cdots \wedge v_{\dim(V)} \mapsto f(v_1) \wedge \cdots \wedge f(v_{\dim(V)})$.

\begin{exercise}[Determinant identities and volume]\leavevmode\par
Prove the following:
\begin{itemize}
    \item $\det(f \oplus g) = \det(f) \det(g)$
    \item If $V = \mathbb{R}^n$, then $\det(f)$ is the signed volume of $f(I^n)$ where $I = [0,1]$.
    \item Cramer's rule
\end{itemize}
\end{exercise}

For $f\in\End(V)$, the \textbf{characteristic polynomial} is the polynomial $\lambda \mapsto \det(f - \lambda 1_V)$.

A scalar $\lambda\in k$ is an \textbf{eigenvalue} of $f$ if $\ker(f-\lambda1_V)\neq0$. This kernel is its \textbf{eigenspace}; its dimension is the geometric multiplicity of $\lambda$.

\begin{exercise}[Characteristic polynomial and Jordan form]\leavevmode\par
Let $f\in\End(V)$ have split characteristic polynomial. Prove the
Cayley--Hamilton theorem and Jordan normal form, and derive the algebraic and
geometric multiplicity inequality together with the eigenvalue formulas for
$\trace(f)$ and $\det(f)$.
\end{exercise}

\begin{exercise}[Critical damping is a Jordan block]\leavevmode\par
\textit{Use Jordan normal form to solve
$m\ddot x+b\dot x+\kappa x=0$ in all three damping regimes, and show that
critical damping is exactly $b^2=4m\kappa$, where a nontrivial Jordan block
produces $t e^{-bt/(2m)}$.}
\end{exercise}

A \textbf{Hermitian inner product} on a complex vector space $V$ is a positive-definite sesquilinear form $\langle\mathord{\cdot},\mathord{\cdot}\rangle$, conjugate-linear in the first variable and linear in the second. It induces the norm $\|v\|=\sqrt{\langle v,v\rangle}$. For a linear map $A:V\to W$ between finite-dimensional Hermitian spaces, an \textbf{adjoint} is a map $A^*:W\to V$ satisfying
\[
\langle Av,w\rangle=\langle v,A^*w\rangle.
\]
An endomorphism $A$ is \textbf{self-adjoint} if $A=A^*$, \textbf{normal} if $A^*A=AA^*$, \textbf{positive} if $\langle v,Av\rangle\geq0$ for every $v$, and \textbf{unitary} if $A^*A=AA^*=1$. Equip $\End(V)$ with the induced operator norm.

\begin{exercise}[Finite-dimensional spectral theorem]\label{ex:finite-spectral-theorem}\leavevmode\par
\textit{Develop the finite-dimensional spectral theorem: prove existence of
adjoints and positive square roots, and prove that every normal operator has
an orthogonal spectral decomposition}
\[
A=\sum_\lambda\lambda p_\lambda,
\qquad \sum_\lambda p_\lambda=1.
\]
\textit{Deduce the singular-value decomposition}
\[
T=U\Sigma V^*.
\]
\end{exercise}

\begin{exercise}[The spectral decomposition produces normal modes]
\label{ex:vibrating-chain}\leavevmode\par
\textit{For a fixed-end chain of $N$ equal masses and springs, set
$q_0=q_{N+1}=0$ and define}
\[
(Kq)_j=\frac{\kappa}{m}(2q_j-q_{j-1}-q_{j+1}).
\]
\textit{Find the spectral decomposition of $K$ and use it to determine all
normal modes of $\ddot q+Kq=0$, their frequencies, and the conserved energy}
\[
\mathcal E(q,\dot q)
=\frac m2\|\dot q\|^2+\frac m2\langle q,Kq\rangle.
\]
\end{exercise}

\begin{exercise}[Fourier modes solve the fixed-end wave equation]
\label{ex:fixed-end-wave}\leavevmode\par
\textit{Solve the fixed-end wave equation from
Exercise~\ref{ex:wave-superposition} by Fourier series, prove conservation
of wave energy, and recover its normal-mode frequencies as the continuum
limit of Exercise~\ref{ex:vibrating-chain}.}
\end{exercise}

\subsection{Group representations}

An action of a group $G$ on a set $X$ assigns to each $g\in G$ a bijection $x\mapsto gx$ such that $1x=x$ and $(gh)x=g(hx)$. A \textbf{representation} of $G$ on a vector space $V$ is a homomorphism
\[
\rho:G\longrightarrow\operatorname{GL}(V).
\]
A subspace $W\subseteq V$ is \textbf{$G$-invariant} if $\rho(g)W\subseteq W$ for every $g\in G$. Restriction gives the \textbf{subrepresentation} $W$, and $\rho(g)(v+W)=\rho(g)v+W$ gives the \textbf{quotient representation} $V/W$. A nonzero representation is \textbf{irreducible} if its only invariant subspaces are $0$ and $V$.

For representations $V$ and $W$, the direct sum, tensor product, and dual representations are defined by
\[
\begin{aligned}
g(v,w)&=(gv,gw),\\
g(v\otimes w)&=gv\otimes gw,\\
(g\phi)(v)&=\phi(g^{-1}v),\qquad \phi\in V^*.
\end{aligned}
\]
An \textbf{intertwiner} $T:V\to W$ is a linear map satisfying $T\rho_V(g)=\rho_W(g)T$ for every $g$. Write $\operatorname{Hom}_G(V,W)$ for the space of intertwiners and
\[
V^G=\{v\in V:\rho(g)v=v\text{ for every }g\in G\}
\]
for the fixed-point space.

\begin{exercise}[Averaging produces invariants]\leavevmode\par
\textit{Let $G$ be finite and let $V$ be a complex representation. Define}
\[
P_G=\frac1{|G|}\sum_{g\in G}\rho(g).
\]
\textit{Prove that $P_G$ is the projection of $V$ onto $V^G$. Prove also that averaging any Hermitian inner product over $G$ produces a $G$-invariant Hermitian inner product, so every finite-dimensional complex representation of $G$ is equivalent to a unitary representation.}
\end{exercise}

\subsection{Maschke's theorem and Schur's lemma}

\begin{exercise}[Maschke's theorem]\leavevmode\par
\textit{Let $W\subseteq V$ be $G$-invariant and let $P:V\to W$ be any linear projection. Define}
\[
\widetilde P
=\frac1{|G|}\sum_{g\in G}\rho(g)P\rho(g)^{-1}.
\]
\textit{Prove that $\widetilde P$ is $G$-equivariant and has image $W$. Deduce that $\ker\widetilde P$ is a $G$-invariant complement of $W$. If $G$ is finite and the characteristic of $k$ does not divide $|G|$, prove that every finite-dimensional $k$-representation of $G$ is a direct sum of irreducible representations.}
\end{exercise}

\begin{exercise}[Schur's lemma]\leavevmode\par
\textit{Let $V$ and $W$ be irreducible finite-dimensional complex representations. Prove that every nonzero intertwiner $T:V\to W$ is an isomorphism and that}
\[
\End_G(V)=\mathbb C1_V.
\]
\textit{Let $\mathbb C[G]$ act on $V$ by the linear extension of $\rho$. Deduce that every element of the center $Z(\mathbb C[G])$ acts by a scalar on each irreducible representation.}
\end{exercise}

\subsection{Characters and orthogonality}

The \textbf{character} of a finite-dimensional representation $V$ is the function
\[
\chi_V(g)=\trace(\rho_V(g)).
\]
For class functions $\chi,\psi:G\to\mathbb C$, define
\[
\langle\chi,\psi\rangle_G
=\frac1{|G|}\sum_{g\in G}\overline{\chi(g)}\psi(g).
\]

\begin{exercise}[Great Orthogonality Theorem]\leavevmode\par
\textit{For irreducible unitary representations $\Gamma^\alpha$ and
$\Gamma^\beta$ of dimensions $d_\alpha,d_\beta$ of a finite group $G$,
prove}
\[
\sum_{g\in G}
\Gamma^\alpha(g)_{ij}
\overline{\Gamma^\beta(g)_{k\ell}}
=\frac{|G|}{d_\alpha}
\delta_{\alpha\beta}\delta_{ik}\delta_{j\ell}.
\]
\textit{Develop its character consequences, including}
\[
m_\alpha=\langle\chi_\alpha,\chi_V\rangle_G.
\]
\textit{Deduce that characters classify finite-dimensional complex
representations.}
\end{exercise}

For an irreducible character $\chi_\alpha$, define the \textbf{symmetry projection}
\[
P_\alpha
=\frac{d_\alpha}{|G|}
\sum_{g\in G}\overline{\chi_\alpha(g)}\rho(g).
\]

\begin{exercise}[Symmetry projections]\leavevmode\par
\textit{Prove that the operators $P_\alpha$ are pairwise orthogonal idempotents and that $P_\alpha V$ is the $\alpha$-isotypic component of $V$. Interpret $P_\alpha$ as constructing symmetry-adapted linear combinations of atomic orbitals or displacement vectors.}
\end{exercise}

\subsection{Molecular symmetry}

The \textbf{point group} of a molecule is the finite group $G\subseteq O(3)$ of spatial symmetries preserving its nuclear configuration, including atomic species. For a molecule with $N$ atoms, $G$ acts on the real displacement space
\[
V_{\mathrm{disp}}=\bigoplus_{a=1}^N\mathbb R^3_a
\cong\mathbb R^{3N}
\]
by permuting atoms and applying the spatial linear action to each displacement. The three-dimensional translational representation carries the polar-vector action $R_g$, while the rotational representation carries the axial-vector action $\det(R_g)R_g$; denote them by $V_{\mathrm{trans}}$ and $V_{\mathrm{rot}}$. We complexify these real representations when applying character theory. For a nonlinear molecule, the vibrational representation is the class
\[
[V_{\mathrm{vib}}]
=[V_{\mathrm{disp}}]-[V_{\mathrm{trans}}]-[V_{\mathrm{rot}}]
\]
in the representation ring.

\begin{exercise}[Character of molecular displacements]\leavevmode\par
\textit{For $g\in G$, prove that}
\[
\chi_{\mathrm{disp}}(g)
=\sum_{\substack{a\\g(a)=a}}\trace(R_g),
\]
\textit{where $R_g:\mathbb R^3\to\mathbb R^3$ is the spatial action and the sum is over atoms fixed by $g$. Prove that $\dim V_{\mathrm{vib}}=3N-6$ for a nonlinear molecule, and use character orthogonality to express the multiplicity of every irreducible representation in $V_{\mathrm{vib}}$.}
\end{exercise}

For water, take the $z$-axis along the $C_2$ axis and the molecule in the $yz$-plane. Its point group is
\[
C_{2v}=\{E,C_2,\sigma_{xz},\sigma_{yz}\}.
\]
\begin{exercise}[Water has three symmetry-classified vibrations]\leavevmode\par
\textit{Derive the $C_{2v}$ character table and use it to show that water's
vibrational representation is}
\[
V_{\mathrm{vib}}\cong2A_1\oplus B_2.
\]
\textit{Identify and construct the three normal modes.}
\end{exercise}

Let $V_i$ and $V_f$ be the symmetry types of initial and final states. The electric-dipole operator transforms as the vector representation $V_\mu=V_{\mathrm{trans}}$. A transition amplitude

\langle\psi_f,\mu\psi_i\rangle

is the matrix coefficient used below.

\begin{exercise}[Symmetry determines electric-dipole transitions]

\textit{Apply it to show that all three fundamental vibrations of water are

\qquad
\nu_1=3656.65\ \mathrm{cm}^{-1},
\qquad

\textit{for the bend, symmetric stretch, and asymmetric stretch

\end{exercise}

\begin{exercise}[Water's vibrations are Raman active]\leavevmode\par
\textit{Let the polarizability tensor carry $\operatorname{Sym}^2(V_\mu)$. Prove that a vibrational mode is Raman active precisely when its irreducible representation occurs in this symmetric square. Prove that}

\operatorname{Sym}^2(V_\mu)

\subsection{Lie groups, infinitesimal symmetry, and spin}

The molecular point groups above are discrete subgroups of $O(3)$. Define the
group of orientation-preserving spatial rotations by
\[
\operatorname{SO}(3)
=\{R\in M_3(\mathbb R):R^TR=1,\ \det R=1\}.
\]

A \textbf{matrix Lie group} is a subgroup $G\subseteq\operatorname{GL}(n,\mathbb C)$ that is closed in the subspace topology. Its \textbf{Lie algebra} is
\[
\mathfrak g=\{X\in M_n(\mathbb C):e^{tX}\in G\text{ for all }t\in\mathbb R\},
\]
where $e^{tX}=\sum_{k\geq0}(tX)^k/k!$. The Lie bracket is $[X,Y]=XY-YX$. An abstract \textbf{Lie algebra} over $k$ is a $k$-vector space with a bilinear bracket that is skew-symmetric and satisfies the Jacobi identity. A \textbf{Lie algebra representation} on $V$ is a linear map $\rho:\mathfrak g\to\mathfrak{gl}(V)$ with $\rho([X,Y])=[\rho(X),\rho(Y)]$.

For $g\in G$ and $X,Y\in\mathfrak g$, the adjoint actions are
\[
\operatorname{Ad}_g(X)=gXg^{-1},
\qquad
\operatorname{ad}_X(Y)=[X,Y].
\]

\begin{exercise}[Classical matrix Lie algebras]\leavevmode\par
\textit{Prove that the Lie algebras of $\operatorname{GL}(n,\mathbb C)$, $\operatorname{SL}(n,\mathbb C)$, $\operatorname{O}(n)$, $\operatorname{SO}(n)$, $\operatorname{U}(n)$, and $\operatorname{SU}(n)$ are respectively}
\[
\mathfrak{gl}(n,\mathbb C),
\quad
\mathfrak{sl}(n,\mathbb C)=\{X:\trace(X)=0\},
\]
\[
\mathfrak{o}(n)=\{X:X^T=-X\},
\quad
\mathfrak{u}(n)=\{X:X^*=-X\},
\quad
\mathfrak{su}(n)=\mathfrak{u}(n)\cap\mathfrak{sl}(n,\mathbb C).
\]
\textit{Verify that each is a Lie algebra under $[X,Y]=XY-YX$.}
\end{exercise}

\begin{exercise}[One-parameter subgroups and the bracket]\leavevmode\par
\textit{Prove that $t\mapsto e^{tX}$ is a smooth group homomorphism $\mathbb R\to G$ whenever $X\in\mathfrak g$, and that every smooth homomorphism $\mathbb R\to G$ arises uniquely in this way. Prove that}
\[
[X,Y]
=\left.\frac{d}{dt}\right|_{t=0}
\left.\frac{d}{ds}\right|_{s=0}
e^{tX}e^{sY}e^{-tX}e^{-sY},
\]
\textit{and deduce $\operatorname{Ad}_{e^{tX}}=e^{t\operatorname{ad}_X}$.}
\end{exercise}

\begin{exercise}[Differentiating group representations]\leavevmode\par
\textit{Let $\rho:G\to\operatorname{GL}(V)$ be a smooth finite-dimensional representation of a matrix Lie group. Prove that}
\[
\dot\rho(X)
=\left.\frac{d}{dt}\right|_{t=0}\rho(e^{tX})
\]
\textit{is a Lie algebra representation $\mathfrak g\to\mathfrak{gl}(V)$, and that $\rho(e^{tX})=e^{t\dot\rho(X)}$. If $G$ is connected, prove that a subspace is $G$-invariant if and only if it is $\mathfrak g$-invariant.}
\end{exercise}

\begin{exercise}[Rotations, angular momentum, and spin]\leavevmode\par
\textit{Using the Pauli matrices}
\[
\sigma_1=\begin{pmatrix}0&1\\1&0\end{pmatrix},
\qquad
\sigma_2=\begin{pmatrix}0&-i\\i&0\end{pmatrix},
\qquad
\sigma_3=\begin{pmatrix}1&0\\0&-1\end{pmatrix}.
\]
\textit{Identify $\mathfrak{su}(2)\cong\mathfrak{so}(3)$, construct the
double cover $\operatorname{SU}(2)\to\operatorname{SO}(3)$, and classify the
irreducible spin representations $\rho_j:SU(2)\to U(V_j)$. For
$J_a=i\dot\rho_j(-i\sigma_a/2)$, derive
$\dim V_j=2j+1$ and $J^2=j(j+1)1_{V_j}$, including the distinction between
integer and half-integer spin.}
\end{exercise}

A \textbf{quantum number} is an eigenvalue used to label a joint eigenspace
of commuting observables.

\begin{exercise}[Energy and symmetry produce quantum numbers]\leavevmode\par
\textit{Let $H$ be a self-adjoint operator on a finite-dimensional Hermitian
space $V$ with a commuting $\operatorname{SU}(2)$ action. Derive the joint
quantum-number decomposition}
\[
|E,j,m,\alpha\rangle
\]
\textit{for $H,J^2,J_3$, including the allowed values and the degeneracy
forced by spin $j$.}
\end{exercise}

\begin{exercise}[Hidden symmetry gives the hydrogen spectrum]
\label{ex:hydrogen-hidden-symmetry}\leavevmode\par
\textit{With $p=-i\hbar\nabla$, $r=|x|$, and $L=x\times p$, consider the
hydrogen Hamiltonian and Runge--Lenz operator}
\[
H=\frac{p^2}{2\mu}-\frac{\kappa}{r},
\qquad
A=\frac{1}{2\mu}(p\times L-L\times p)-\kappa\frac{x}{r}.
\]
\textit{Use the hidden $\mathfrak{so}(4)$ symmetry to derive the bound-state
spectrum and degeneracies}
\[
E_n=-\frac{\mu\kappa^2}{2\hbar^2n^2},
\qquad
n\geq1,
\qquad \dim W_n=n^2,
\]
\textit{and the angular-momentum decomposition}
\[
W_n\cong\bigoplus_{\ell=0}^{n-1}V_\ell.
\]
\textit{Recover the quantum numbers and transition frequencies, and compare
them with hydrogen spectroscopy \cite{kramida-hydrogen} and Pauli's
algebraic derivation \cite{pauli-hydrogen}.}
\end{exercise}

\subsection{Monoidal and braided categories}

A $\mathbb C$-linear category is a category whose morphism sets
\[
\operatorname{Hom}_{\mathcal C}(x,y)
\]
are complex vector spaces and whose composition is bilinear.

A $\mathbb C$-linear monoidal category is a $\mathbb C$-linear category equipped with a tensor product $\otimes$ that is bilinear on morphisms and is associative with tensor unit $1$, up to coherent natural isomorphism.

Let $\mathcal C$ be such a category with finite direct sums. An object $x$ is \textbf{simple} if $\End(x)\cong\mathbb C$, and $\mathcal C$ is \textbf{semisimple} if every object is a finite direct sum of simple objects. Assume that $1$ is simple, that there are finitely many isomorphism classes of simple objects, and that $\mathcal C$ is \textbf{rigid}: every $x$ has a dual $x^*$ with evaluation and coevaluation maps satisfying the snake identities.

A \textbf{pivotal structure} on $\mathcal C$ is a monoidal natural
isomorphism $j_x:x\to x^{**}$. It defines the right and left traces of
$f:x\to x$ by
\[
\begin{aligned}
\trace_R(f)
&=\ev_{x^*}(j_x\otimes1_{x^*})(f\otimes1_{x^*})\coev_x,\\
\trace_L(f)
&=\ev_x(1_{x^*}\otimes f)(1_{x^*}\otimes j_x^{-1})\coev_{x^*}.
\end{aligned}
\]
The pivotal structure is \textbf{spherical} if
$\trace_L(f)=\trace_R(f)$ for every endomorphism $f$. Assume that
$\mathcal C$ has a spherical pivotal structure, write the common trace as
$\trace(f)$, and set $d_x=\trace(1_x)$. Assume that the categorical
dimensions of nonzero objects are positive real numbers.

For $x,y,z\in\mathcal C$, write
\[
F^{xyz}:(x\otimes y)\otimes z\longrightarrow x\otimes(y\otimes z)
\]
for the associator. A \textbf{braiding} is a family of isomorphisms
\[
c_{x,y}:x\otimes y\longrightarrow y\otimes x.
\]
For $f:x\to x'$ and $g:y\to y'$, it satisfies $(g\otimes f)c_{x,y}=c_{x',y'}(f\otimes g)$.
The associator satisfies the pentagon identity, and the associator and braiding satisfy the hexagon identities. Suppressing associators, the latter are
\[
\begin{aligned}
c_{x,y\otimes z}&=(1_y\otimes c_{x,z})(c_{x,y}\otimes1_z),\\
c_{x\otimes y,z}&=(c_{x,z}\otimes1_y)(1_x\otimes c_{y,z}).
\end{aligned}
\]

The \textbf{monodromy} of $x$ and $y$ is the double braiding
\[
M_{x,y}=c_{y,x}c_{x,y}:x\otimes y\longrightarrow x\otimes y.
\]
Thus $c_{x,y}$ is one crossing, while $M_{x,y}$ is a full braid. If $x\otimes y$ is simple, identify $M_{x,y}$ with its scalar in $\mathbb C^\times=\mathbb C\setminus\{0\}$. Decompositions of tensor products of simple objects are \textbf{fusion rules}, and the spaces $\operatorname{Hom}_{\mathcal C}(z,x\otimes y)$ are \textbf{fusion spaces}.

An object $x$ is \textbf{invertible} if $x\otimes x^*\cong1$.

\begin{exercise}[The $\nu=1/3$ anyonic phase advances by $2\pi/3$]
\label{ex:anyonic-monodromy}\leavevmode\par
\textit{Assume that every simple object is invertible and that their
isomorphism classes form $\mathbb Z/m$, generated by $a$. Classify its
monodromy, writing $M_{r,s}=M_{a^r,a^s}$, and show that, for some
$k\bmod m$,}
\[
M_{r,s}=\exp\left(\frac{2\pi i krs}{m}\right).
\]
\textit{For $k=1$ and $m=3$, derive the $2\pi/3$ phase advance per enclosed
anyon and compare it with \cite{nakamura-anyons}.}
\end{exercise}

\begin{exercise}[Reshetikhin--Turaev TQFT]
\label{ex:rt-tqft}\leavevmode\par
\textit{A ribbon fusion category is a braided fusion category with a
dual-compatible twist $\theta$ satisfying
$\theta_{x\otimes y}=M_{x,y}(\theta_x\otimes\theta_y)$. For simple objects
$x_i$, define $S_{ij}=\trace(M_{x_i,x_j})$ and call the category modular when
$S$ is invertible. Prove that every modular ribbon fusion category $\mathcal C$
defines a symmetric monoidal functor} \cite{reshetikhin-turaev}
\[
Z_{\mathcal C}:\widehat{\operatorname{Bord}}_3^{\mathcal C}
\longrightarrow\operatorname{Vect}_{\mathbb C},
\]
\textit{where $\widehat{\operatorname{Bord}}_3^{\mathcal C}$ is the
anomaly-corrected category of oriented three-dimensional bordisms with
$\mathcal C$-colored ribbon graphs.}
\end{exercise}

\section{Geometry}

\subsection{Differential forms and cohomology}

A \textbf{smooth $n$-manifold} is a Hausdorff, second-countable topological
space equipped with a maximal atlas of charts to open subsets of
$\mathbb R^n$ whose transition maps are smooth. A smooth map $f:M\to N$ has
a tangent map $df:TM\to TN$. The \textbf{tangent bundle} $TM$ is the union
of the tangent spaces, the \textbf{cotangent bundle} is $T^*M$, and a
\textbf{vector field} is a smooth section of $TM$.

For an $n$-dimensional real vector space $V$, its \textbf{determinant line}
is $\det(V^*)=\Lambda^nV^*$.
An \textbf{orientation} of $V$ is a choice of one of the two connected
components of $\det(V^*)\setminus\{0\}$; the elements of the chosen component
are positive. An orientation of an $n$-manifold $M$ is a smoothly varying
choice of orientation of its tangent spaces.

The exterior powers $\Lambda^kT_p^*M$ are those defined in the Algebra section. The space of \textbf{differential $n$-forms} is $\Omega^n(M)=\Gamma(\Lambda^nT^*M)$. On an oriented $n$-manifold, a \textbf{volume form} is a nowhere-vanishing element of $\Omega^n(M)$ that is positive with respect to the orientation. The exterior derivative $d:\Omega^n(M)\to\Omega^{n+1}(M)$ extends the differential of smooth functions and satisfies
\[
d(\alpha\wedge\beta)=d\alpha\wedge\beta+(-1)^p\alpha\wedge d\beta
\]
for $\alpha\in\Omega^p(M)$. A \textbf{singular $n$-simplex} is a smooth map $\sigma:\Delta^n \to M$ where $\Delta^n$ is the standard $n$-simplex. The collection of singular $n$-simplices in $M$ freely generates the abelian group $C_n(M)$ of \textbf{singular $n$-chains} in $M$. The \textbf{boundary map} $\partial : C_n(M) \to C_{n-1}(M)$ is defined on each simplex $\sigma:\Delta^n \to M$ by
$$\partial\sigma=\sum_{j=0}^{n}(-1)^j\sigma\circ f_j,$$ where $f_j : \Delta^{n-1} \to \Delta^n$ is the inclusion of the face opposite the $j$-th vertex of $\Delta^n$.

A \textbf{chain complex} $(A_n, d_n)_n$ is a sequence of abelian groups $A_n$ and homomorphisms $d_n : A_n \to A_{n-1}$ such that $d_{n-1} \circ d_n = 0$ for all $n$. The $n$-th \textbf{homology group} $H_n$ of the chain complex is
$$H_n = \ker(d_{n}) / \im(d_{n+1}).$$
The \textbf{singular homology} $H_n(M)$ is the homology of
$(C_\bullet(M),\partial)$.

A \textbf{cochain complex} $(A^n,d^n)_n$ is a sequence of abelian groups and
homomorphisms $d^n:A^n\to A^{n+1}$ satisfying $d^{n+1}d^n=0$. Its
\textbf{$n$th cohomology group} is
\[
H^n(A^\bullet)=\ker d^n/\operatorname{im}d^{n-1}.
\]
For an abelian group $G$, the \textbf{singular cochains} are
\[
C^n(M;G)=\operatorname{Hom}(C_n(M),G),
\qquad
(d\varphi)(c)=\varphi(\partial c),
\]
and their cohomology is the \textbf{singular cohomology} $H^n(M;G)$.

The \textbf{de Rham cohomology} of $M$ is
\[
H_{\mathrm{dR}}^n(M)
=
\ker(d\colon\Omega^n(M)\to\Omega^{n+1}(M))/
\operatorname{im}(d\colon\Omega^{n-1}(M)\to\Omega^n(M)).
\]

For a smooth map $f:N\to M$, the \textbf{pullback} of $\alpha\in\Omega^p(M)$ is
\[
(f^*\alpha)_x(v_1,\ldots,v_p)
=\alpha_{f(x)}(df_xv_1,\ldots,df_xv_p).
\]

\begin{exercise}[de Rham theorem]\leavevmode\par
\textit{Prove that integration over smooth singular cycles induces a natural isomorphism between de Rham cohomology and singular cohomology with real coefficients.}
\end{exercise}

\begin{exercise}[Cohomology produces the Aharonov--Bohm phase]
\label{ex:aharonov-bohm}\leavevmode\par
\textit{Outside an idealized solenoid, let}
\[
M=\mathbb R^3\setminus\{(0,0,z):z\in\mathbb R\},
\qquad
d\theta=\frac{-y\,dx+x\,dy}{x^2+y^2},
\qquad
A=\frac{\Phi}{2\pi}d\theta.
\]
\textit{For a particle of charge $q$, use de Rham cohomology to derive the
gauge-invariant phase}
\[
\exp\!\left(\frac{iq}{\hbar}\int_\gamma A\right)
\]
\textit{and the Aharonov--Bohm interference shift $q\Phi/\hbar$
\cite{aharonov-bohm}. Explain how the shielded-flux electron-holography
experiment of Tonomura et al. isolates this holonomy, and compare its measured
relative phase shift with the prediction \cite{tonomura-ab}.}
\end{exercise}

\subsection{Hodge theory and electromagnetism}

A \textbf{pseudo-Riemannian metric} is a smooth nondegenerate symmetric section $g\in\Gamma(T^*M\otimes T^*M)$. On an oriented pseudo-Riemannian $n$-manifold, the metric induces a bilinear form $\langle\cdot,\cdot\rangle_g$ on differential forms. A \textbf{metric volume form} $\operatorname{vol}_g$ is a positive volume form satisfying
\[
\left|\langle\operatorname{vol}_g,\operatorname{vol}_g\rangle_g\right|=1.
\]
The \textbf{Hodge star} is an operator $\star:\Omega^p(M)\to\Omega^{n-p}(M)$ satisfying
\[
\alpha\wedge\star\beta=\langle\alpha,\beta\rangle_g\operatorname{vol}_g.
\]

\begin{exercise}[Metric volume and Hodge star]\leavevmode\par
\textit{Prove that an oriented pseudo-Riemannian manifold has a unique metric volume form and a unique Hodge-star operator. Determine $\star^2$ in terms of the degree and signature of the metric.}
\end{exercise}

On an oriented Lorentzian four-manifold, the electromagnetic field strength is
a two-form $F$ and the electric four-current is a three-form $J$.

\begin{exercise}[Maxwell's equations imply charge conservation]\leavevmode\par
\textit{From Maxwell's equations}
\[
dF=0,
\qquad
d\star F=J.
\]
\textit{Derive charge conservation in differential and integral form. On
Minkowski spacetime, write}
\[
F=B-dt\wedge E,
\qquad
J=\rho\,\operatorname{vol}_3-dt\wedge\star_3j,
\]
\textit{and recover the classical Maxwell equations and continuity equation.}
\end{exercise}

\begin{exercise}[Gauge invariance forces charge conservation]\leavevmode\par
\textit{For $F=dA$, consider}
\[
S[A]
=-\frac12\int_M F\wedge\star F+\int_M A\wedge J.
\]
\textit{Derive Maxwell's equations and prove that gauge invariance is
equivalent to charge conservation, assuming vanishing boundary terms.}
\end{exercise}

\subsection{Vector bundles and connections}

A \textbf{smooth vector bundle} $p:E\to M$ is a smooth family of vector spaces locally isomorphic to a product $U\times V\to U$. For a smooth map $f:N\to M$, its \textbf{pullback bundle} is
\[
f^*E=\{(x,e)\in N\times E:f(x)=p(e)\}\longrightarrow N.
\]

Write $\Omega^0(M,E)=\Gamma(E)$. A \textbf{connection} on $E$ is a
ground-field linear map
\[
\nabla:\Omega^0(M,E)\longrightarrow\Omega^1(M,E)
\]
satisfying $\nabla(fs)=df\otimes s+f\nabla s$. Extend it to $E$-valued forms
by, for $\alpha\in\Omega^p(M)$,
\[
\nabla(\alpha\wedge\eta)
=d\alpha\wedge\eta+(-1)^p\alpha\wedge\nabla\eta,
\]
and define its \textbf{curvature} by
$F_\nabla=\nabla^2\in\Omega^2(M,\End E)$. Equip $\End E$-valued forms
with the induced connection.

\begin{exercise}[Bianchi identity]\leavevmode\par
\textit{Prove that every vector-bundle connection satisfies
$\nabla F_\nabla=0$.}
\end{exercise}

\subsection{K-theory}

For a compact Hausdorff space $X$, let $\operatorname{Vect}(X)$ be the category of finite-rank complex vector bundles with continuous local trivializations over $X$, equipped with direct sum, tensor product, dual, and pullback. The \textbf{Grothendieck group} $K^0(X)$ is generated by isomorphism classes $[E]$ subject to
\[
[E\oplus F]=[E]+[F].
\]
Define multiplication by $[E][F]=[E\otimes F]$. For a continuous map $f:X\to Y$ between compact Hausdorff spaces, define pullback by
\[
f^*:K^0(Y)\longrightarrow K^0(X),
\qquad
[E]\longmapsto[f^*E].
\]

\begin{exercise}[Grothendieck group of vector bundles]\leavevmode\par
\textit{Prove that $(f\circ g)^*E\cong g^*f^*E$ and that pullback preserves direct sums, tensor products, and duals. Let $X$ and $Y$ be compact Hausdorff spaces and let $f:X\to Y$ be continuous. Prove that direct sum and tensor product induce a commutative ring structure on $K^0(X)$ and that $f^*$ is a ring homomorphism. Prove that}
\[
K^0(\mathrm{pt})\cong\mathbb Z
\]
\textit{via the rank map.}
\end{exercise}

Fix a basepoint $x_0\in X$, regarded as a map $x_0:\mathrm{pt}\to X$. The \textbf{reduced K-theory} of the pointed space $(X,x_0)$ is
\[
\widetilde K^0(X)
=\ker\!\left(K^0(X)\xrightarrow{x_0^*}K^0(\mathrm{pt})\right).
\]
For a bundle $E$ of rank $r$, its \textbf{reduced class} is
\[
[E]-[\underline{\mathbb C}^{\,r}],
\]
where $\underline{\mathbb C}^{\,r}=X\times\mathbb C^r$ is the trivial rank-$r$ bundle. Write $K^n(X)$ for the generalized cohomology groups extending $K^0(X)$.

\begin{exercise}[Stable equivalence and Bott periodicity]\leavevmode\par
\textit{Let $E$ and $F$ be complex vector bundles over a compact Hausdorff space $X$. Prove that $[E]=[F]$ in $K^0(X)$ if and only if there is a vector bundle $G$ such that}
\[
E\oplus G\cong F\oplus G.
\]
\textit{Deduce that the reduced class of $E$ is unchanged after replacing $E$ by $E\oplus\underline{\mathbb C}^{\,n}$. Prove Bott periodicity}
\[
K^{n+2}(X)\cong K^n(X)
\]
\textit{and deduce that complex K-theory is naturally $\mathbb Z/2$-graded.}
\end{exercise}

For $0\leq r\leq N$, identify the complex Grassmannian $\operatorname{Gr}_r(\mathbb C^N)$ with the space of rank-$r$ orthogonal projections in $M_N(\mathbb C)$. Its \textbf{tautological bundle} has fiber $\operatorname{im}P$ over $P$. Let $BU$ denote the stable complex Grassmannian obtained from finite Grassmannians by adjoining trivial summands.

\begin{exercise}[Stable classification of complex vector bundles]\label{ex:stable-bundle-classification}\leavevmode\par
\textit{Classify stable complex vector bundles over a connected finite CW
complex $X$ by projection-valued maps, proving}
\[
K^0(X)\cong[X,\mathbb Z\times BU],
\qquad
\widetilde K^0(X)\cong[X,BU]_*.
\]
\end{exercise}

\begin{exercise}[K\"unneth theorem computes torus K-theory]
\label{ex:kunneth-k-theory}\leavevmode\par
\textit{For finite CW complexes $X$ and $Y$ with torsion-free K-groups,
prove that external product induces the graded isomorphism} \cite{atiyah-k-theory}
\[
K^*(X)\mathbin{\widehat\otimes}K^*(Y)
\cong K^*(X\times Y).
\]
\textit{Here $\widehat\otimes$ denotes graded tensor product. Use Bott
periodicity to deduce}
\[
K^*(\mathbb T^d)
\cong\Lambda_{\mathbb Z}(\eta_1,\ldots,\eta_d),
\qquad |\eta_j|=1.
\]
\end{exercise}

\subsection{Chern classes and the Chern character}

Let $P$ be an invariant symmetric $k$-linear polynomial on the endomorphism
algebra of a fiber of $E$, where invariant means invariant under simultaneous
conjugation. Define
\[
P(F_\nabla)
=P(F_\nabla,\ldots,F_\nabla)
\in\Omega^{2k}(M).
\]

\begin{exercise}[Chern--Weil theory produces Chern classes]
\label{ex:chern-weil}\leavevmode\par
\textit{Prove that $P(F_\nabla)$ is closed and that its de Rham class is
independent of the connection. Apply this to}
\[
\det\left(1+\frac{iF_\nabla}{2\pi}\right)
=1+c_1(E,\nabla)+c_2(E,\nabla)+\cdots
\]
\textit{to define the Chern classes of $E$} \cite{chern-simons}.
\end{exercise}

\begin{exercise}[Chern character]\label{ex:chern-character}\leavevmode\par
\textit{Define}
\[
\operatorname{ch}(E,\nabla)
=\operatorname{tr}\exp\left(\frac{iF_\nabla}{2\pi}\right).
\]
\textit{For compact $M$, prove that it induces a ring homomorphism}
\[
\operatorname{ch}:K^0(M)\longrightarrow H_{\mathrm{dR}}^{\mathrm{even}}(M).
\]
\end{exercise}

\subsection{Band topology}

For the gapped families below, assume that $X$ is compact and connected, so
spectra are uniformly bounded and vector-bundle ranks are constant. Let $V$
be a finite-dimensional Hermitian vector space.
Exercise~\ref{ex:finite-spectral-theorem} supplies the spectral decomposition
used below. A continuous family is a map
\[
H:X\longrightarrow\End(V)
\]
and write $H_x=H(x)$. It is \textbf{gapped} if every $H_x$ is self-adjoint and invertible. Given a positively oriented contour $\Gamma$ enclosing the negative spectra and excluding the positive spectra, define
\[
P_x^-=\frac{1}{2\pi i}\int_\Gamma(z-H_x)^{-1}\,dz,
\qquad
E_x^-=\operatorname{im}(P_x^-).
\]

\begin{exercise}[Gapped families define stable K-theory classes]\leavevmode\par
\textit{Show that the negative spectral subspaces of a gapped family form a
vector bundle and that its class in $K^0(X)$ is invariant under
gap-preserving homotopy. Show that}
\[
[E_H^-]-[\underline{\mathbb C}^{\,\operatorname{rank}E_H^-}]
\in\widetilde K^0(X)
\]
\textit{is invariant under adjoining constant gapped summands, and that every
reduced class is represented by a flattened family $H_x=1-2P_x$.}
\end{exercise}

\begin{exercise}[Berry phase as holonomy]\leavevmode\par
\textit{Let $H:X\to\End(V)$ be a smooth self-adjoint family with a simple
isolated eigenvalue $\lambda_x$, and let $L\to X$ be its eigenline bundle.
For a local unit eigenvector $\psi$, define}
\[
\mathcal A=i\langle\psi,d\psi\rangle,
\qquad
\mathcal F=d\mathcal A.
\]
\textit{Prove that these local forms define a connection and curvature on
$L$ and that the Berry phase around a closed curve $\gamma$ is its holonomy
$\exp(i\int_\gamma\mathcal A)$. For $H_n=n\cdot\sigma$, $n\in S^2$, prove
that the phases of the two eigenlines around a loop enclosing oriented solid
angle $\Omega$ are $\gamma_\pm=\mp\Omega/2$ modulo $2\pi$, and compare this
prediction with neutron spin rotation \cite{bitter-dubbers}.}
\end{exercise}

\begin{exercise}[K-theory classifies class A band insulators]\label{ex:class-a-insulators}\leavevmode\par
\textit{A class A band model over a compact connected parameter space $X$
has a smooth gapped Hamiltonian}
\[
H:X\longrightarrow\End(V)
\]
\textit{whose negative-energy bundle consists of occupied states; for a
crystal, $X=\mathbb T^d$. Prove that
stable equivalence classes are classified by $\widetilde K^0(X)$
\cite{kitaev-periodic}, and compute the strong phases}
\[
\widetilde K^0(S^d)
\cong
\begin{cases}
\mathbb Z,&d\text{ even},\\
0,&d\text{ odd}.
\end{cases}
\]
\textit{Use Exercise~\ref{ex:kunneth-k-theory} to compute the classifications
over $\mathbb T^d$ for $d=1,2,3$ and distinguish strong from weak invariants.}
\textit{For $d=2$, use Exercise~\ref{ex:chern-weil} to identify the integer
invariant with the first Chern number}
\[
C_1(E_H^-)
=\frac{i}{2\pi}\int_{\mathbb T^2}
\operatorname{tr}\bigl(P^-\,dP^-\wedge dP^-\bigr),
\]
\textit{and prove that a nonzero value obstructs a global smooth frame of
occupied states. Assuming the TKNN formula
$\sigma_{xy}=C_1(E_H^-)e^2/h$ \cite{tknn}, deduce the quantized Hall resistance and
compare its plateaus and geometry independence with the original observation
\cite{von-klitzing-qhe}.}
\end{exercise}

\subsection{Symplectic geometry}

For a vector field $X$, contraction is the map $\iota_X:\Omega^p(M)\to\Omega^{p-1}(M)$ defined by
\[
(\iota_X\alpha)(X_2,\ldots,X_p)=\alpha(X,X_2,\ldots,X_p).
\]
Define the Lie derivative of forms by Cartan's formula
$\mathcal{L}_X=d\iota_X+\iota_Xd$.

A \textbf{symplectic form} is a two-form $\omega\in\Omega^2(M)$ such that $d\omega=0$ and $X\mapsto\iota_X\omega$ is a bundle isomorphism $TM\to T^*M$. Write $C^\infty(M)$ for the smooth real-valued functions on $M$. For $H\in C^\infty(M)$, define $X_H$ by
\[
\iota_{X_H}\omega=dH.
\]

Let $\pi:T^*Q\to Q$ be the cotangent bundle of a configuration manifold. Its \textbf{tautological one-form} is
\[
\theta_{(q,p)}(\xi)=p\bigl(d\pi_{(q,p)}\xi\bigr),
\]
and its \textbf{canonical symplectic form} is $\omega_0=-d\theta$.

\begin{exercise}[The symplectic form produces Hamilton's equations]\leavevmode\par
\textit{For}
\[
H(q,p)=\frac{1}{2m}\sum_{i=1}^np_i^2+V(q),
\]
\textit{derive Hamilton's equations intrinsically from $\omega_0$, recover
Newton's equation, and solve the harmonic-oscillator flow.}
\end{exercise}

\begin{exercise}[Magnetic curvature produces the Lorentz force]\leavevmode\par
\textit{For a particle of charge $e$ in a closed magnetic field
$B\in\Omega^2(\mathbb R^3)$, derive the Lorentz force from}
\[
\omega_B=\omega_0-e\pi^*B
\qquad\text{and}\qquad
H(q,p)=\frac{|p|^2}{2m}+e\phi(q),
\]
\textit{and relate the gauge freedom for $B=dA$ to the Aharonov--Bohm
potential.}
\end{exercise}

\begin{exercise}[Liouville's theorem]\leavevmode\par
\textit{Prove that every Hamiltonian flow preserves the symplectic volume
$\omega^n/n!$.}
\end{exercise}

\begin{exercise}[Phase volume produces ideal-gas thermodynamics]\leavevmode\par
\textit{For Liouville measure $d\mu_L=\omega^n/n!$, show that Hamiltonian
evolution preserves the fine-grained Gibbs entropy}
\[
S_G[\rho]
=-k_B\int_M\rho\log\rho\,d\mu_L.
\]
\textit{For $N$ indistinguishable noninteracting particles in a box, divide
phase volume by $N!h^{3N}$ and use}
\[
H(q,p)=\sum_{a=1}^N\frac{|p_a|^2}{2m}.
\]
\textit{Derive the Sackur--Tetrode entropy and the ideal-gas laws}
\[
E=\frac32Nk_BT,
\qquad
PV=Nk_BT.
\]
\textit{Derive the adiabatic and sound-speed laws, including, for
$\varrho=mN/V$,}
\[
c=\sqrt{\frac{5P}{3\varrho}}.
\]
\textit{Reconcile the entropy increase in free expansion with conservation
of fine-grained Gibbs entropy.}
\end{exercise}

For $f,g\in C^\infty(M)$, define
\[
\{f,g\}=\omega(X_f,X_g).
\]

\begin{exercise}[Hamiltonian symmetry produces conservation laws]\leavevmode\par
\textit{Prove that a Hamiltonian $f$ generates a symmetry of $H$ exactly
when $f$ is conserved by the flow of $H$, and distinguish this statement
from Noether's variational theorem \cite{noether}.}
\end{exercise}

\begin{exercise}[Central forces and angular momentum]\leavevmode\par
\textit{On $T^*\mathbb R^3$, let}
\[
H(q,p)=\frac{|p|^2}{2m}+V(|q|),
\qquad
L=q\times p.
\]
\textit{Derive the angular-momentum Poisson algebra and show that rotational
symmetry confines each nonradial trajectory to a fixed plane.}
\end{exercise}

\subsection{Riemannian geometry}

An \textbf{affine connection} is a connection on $TM$; write
$\nabla_XY=(\nabla Y)(X)$.
For a covector field $\alpha$, set
\[
(\nabla_X\alpha)(Y)=X(\alpha(Y))-\alpha(\nabla_XY).
\]
Define the covariant derivative of general tensor fields by the product rule and by requiring it to commute with contractions.

An affine connection is \textbf{torsion-free and metric-compatible} when
\[
\nabla_XY-\nabla_YX=[X,Y],
\qquad
\nabla g=0.
\]
Such a connection is called a \textbf{Levi-Civita connection}.

\begin{exercise}[Levi-Civita connection]\leavevmode\par
\textit{Prove that every pseudo-Riemannian manifold admits a unique Levi-Civita connection. Derive the coordinate-free Koszul formula.}
\end{exercise}

An affinely parametrized curve $\gamma:I\to M$ is a \textbf{geodesic} if $\nabla_{\dot\gamma}\dot\gamma=0$. The Lie derivative of the metric is
\[
(\mathcal L_Xg)(Y,Z)
=X(g(Y,Z))-g([X,Y],Z)-g(Y,[X,Z]),
\]
and $X$ is a \textbf{Killing vector field} if $\mathcal L_Xg=0$.

Let $(M,g)$ be a pseudo-Riemannian manifold with Levi-Civita connection $\nabla$.

\begin{exercise}[Killing fields and conserved quantities]\leavevmode\par
\textit{Let $X$ be a Killing vector field and let $\gamma$ be an affinely parametrized geodesic. Prove that}
\[
\frac{d}{dt}g(\dot\gamma,\dot\gamma)=0,
\qquad
\frac{d}{dt}g(X,\dot\gamma)=0.
\]
\textit{Deduce that $g(X,\dot\gamma)$ is constant along $\gamma$.}
\end{exercise}

The \textbf{curvature} of $\nabla$ is the $\End(TM)$-valued $2$-form $F_\nabla$ determined by
\[
R(X,Y)Z
:=F_\nabla(X,Y)Z
=\nabla_X\nabla_YZ-\nabla_Y\nabla_XZ-\nabla_{[X,Y]}Z.
\]

\begin{exercise}[Curvature as a tensor]\leavevmode\par
\textit{For $f\in C^\infty(M)$ and vector fields $X,Y,Z$, prove that}
\[
R(fX,Y)Z=fR(X,Y)Z,
\qquad
R(X,fY)Z=fR(X,Y)Z,
\]
\[
R(X,Y)(fZ)=fR(X,Y)Z.
\]
\textit{Prove that $R(X,Y)=-R(Y,X)$ and deduce that}
\[
R\in\Gamma\!\left(\Lambda^2T^*M\otimes T^*M\otimes TM\right).
\]
\end{exercise}

\begin{exercise}[Geodesic deviation]
\textit{Let $\Gamma:(-\varepsilon,\varepsilon)\times I\to M$ be a smooth
variation through affinely parametrized geodesics, and define the vector
fields along $\Gamma$ by}
\[
T=\frac{\partial\Gamma}{\partial t},
\qquad
J=\frac{\partial\Gamma}{\partial s}.
\]
\textit{Prove that $[T,J]=0$ and that}
\[
\nabla_T\nabla_TJ=R(T,J)T.
\]
\end{exercise}

\begin{exercise}[Geodesic deviation produces interferometer strain]
\leavevmode\par
\textit{Consider the weak plane-wave metric}
\[
g=-dt^2+\bigl(1+h_+(t-z)\bigr)dx^2
+\bigl(1-h_+(t-z)\bigr)dy^2+dz^2,
\qquad |h_+|\ll1.
\]
\textit{Use geodesic deviation to show, to first order, that freely falling
mirrors separated along the $x$- and $y$-axes acquire opposite fractional
length changes $h_+/2$ and $-h_+/2$. Deduce that an equal-arm Michelson
interferometer measures differential strain $h_+$, and compare this prediction
with the peak strain of order $10^{-21}$ observed for GW150914
\cite{ligo-gw150914}.}
\end{exercise}

\begin{exercise}[Gauss--Bonnet theorem]\leavevmode\par
\textit{Let $(M,g)$ be a compact oriented Riemannian surface with piecewise
smooth boundary. With Gaussian curvature $K$, signed geodesic curvature
$k_g$, and signed exterior angles $\theta_j$, prove} \cite{do-carmo}
\[
\int_MK\operatorname{vol}_g
+\int_{\partial M}k_g\,ds
+\sum_j\theta_j
=2\pi\chi(M).
\]
\textit{Derive its consequences for round spheres, positively curved tori,
and closed hyperbolic surfaces.}
\end{exercise}

The trace used below is the ordinary trace defined in the Algebra section. Define the \textbf{Ricci tensor}, \textbf{scalar curvature}, and \textbf{Einstein tensor} by
\[
\operatorname{Ric}(X,Y)=\operatorname{tr}\bigl(Z\mapsto R(Z,X)Y\bigr),
\qquad
S=\operatorname{tr}_g(\operatorname{Ric}),
\qquad
G=\operatorname{Ric}-\frac12Sg.
\]
For a symmetric covariant two-tensor $A$, its \textbf{divergence} is the covector field
\[
(\operatorname{div}A)(Y)
=\operatorname{tr}_g\bigl((X,Z)\mapsto(\nabla_XA)(Z,Y)\bigr).
\]

\begin{exercise}[Contracted Bianchi identity and conservation]\leavevmode\par
\textit{Assume the second Bianchi identity}
\[
(\nabla_XR)(Y,Z)+(\nabla_YR)(Z,X)+(\nabla_ZR)(X,Y)=0.
\]
\textit{Contract this identity to prove that}
\[
\operatorname{div}G=0.
\]
\textit{Deduce that if $T$ is a symmetric covariant two-tensor satisfying}
\[
G+\Lambda g=\kappa T
\]
\textit{for constants $\Lambda,\kappa$ with $\kappa\ne0$, then $\operatorname{div}T=0$.}
\end{exercise}

\subsection{Yang--Mills theory and the Standard Model}

Let $G$ be a Lie group with Lie algebra $\mathfrak g$. A \textbf{principal $G$-bundle} is a smooth free right $G$-manifold $P$ with quotient $M=P/G$ such that the projection $P\to M$ is a locally trivial fiber bundle. For a finite-dimensional representation $\rho:G\to\operatorname{GL}(V)$, the \textbf{associated bundle} is
\[
E=P\times_G V=(P\times V)/\sim,
\qquad
(p\cdot g,v)\sim(p,\rho(g)^{-1}v).
\]
A \textbf{principal connection} is a $\mathfrak g$-valued one-form $A$ on $P$ that is $G$-equivariant and reproduces the generators of the fundamental vector fields. In a local trivialization it is represented by an ordinary form $A\in\Omega^1(U,\mathfrak g)$, with curvature
\[
F_A=dA+A\wedge A\in\Omega^2(U,\mathfrak g).
\]
\begin{exercise}[Isospin motivates Yang--Mills connections]\leavevmode\par
\textit{Treat the proton and neutron as a nucleon doublet carrying the
defining representation of $\operatorname{SU}(2)$. Gauge this symmetry and
derive}
\[
A\longmapsto A^g=g^{-1}Ag+g^{-1}dg.
\]
\textit{Show that the resulting connection differentiates associated-bundle
fields and has covariant curvature, and relate this construction to
\cite{yang-mills}.}
\end{exercise}

\begin{exercise}[Yang--Mills action]\leavevmode\par
\textit{On an oriented Riemannian four-manifold, consider the Yang--Mills
action}
\[
S(A)=\int_M\operatorname{Tr}(F_A\wedge\star F_A).
\]
\textit{Prove gauge invariance, derive}
\[
d_A\star F_A=0,
\]
\textit{and recover source-free Maxwell theory in the abelian case.}
\end{exercise}

\begin{exercise}[Standard Model symmetry breaking]\leavevmode\par
\textit{Take the Standard Model gauge group to be}
\[
G_{\mathrm{SM}}
=\operatorname{SU}(3)_C\times\operatorname{SU}(2)_L
\times\operatorname{U}(1)_Y.
\]
\textit{Write $(\mathbf n,\mathbf m)_Y$ for the representation of dimension
$n$ under $\operatorname{SU}(3)_C$, dimension $m$ under
$\operatorname{SU}(2)_L$, and weak hypercharge $Y$. Take one generation of
Weyl fermion multiplets to be}
\[
Q_L=(\mathbf 3,\mathbf 2)_{1/6},
\qquad
u_R=(\mathbf 3,\mathbf 1)_{2/3},
\qquad
d_R=(\mathbf 3,\mathbf 1)_{-1/3},
\]
\[
L_L=(\mathbf 1,\mathbf 2)_{-1/2},
\qquad
e_R=(\mathbf 1,\mathbf 1)_{-1},
\]
\textit{and define $Q=T_3+Y$. Let $\phi$ be a section of the Higgs bundle
associated to $(\mathbf 1,\mathbf 2)_{1/2}$, with}
\[
V(\phi)=-\mu^2|\phi|^2+\lambda|\phi|^4,
\qquad
D\phi=d\phi+\tfrac{i}{2}g\,W^a\sigma_a\phi+\tfrac{i}{2}g'B\phi,
\]
\textit{where $\mu^2,\lambda>0$ and $g,g'$ are coupling constants. From
these data derive the fermion charges,
the symmetry breaking
$\operatorname{SU}(2)_L\times\operatorname{U}(1)_Y
\to\operatorname{U}(1)_{\mathrm{EM}}$, the $W$ and $Z$ masses, the massless
photon, and cancellation of the hypercharge anomalies. Identify the
$W^\pm\to e^\pm\nu$ missing-transverse-energy and
$Z^0\to\ell^+\ell^-$ invariant-mass signatures permitted by these
representations, and compare them with the original UA1 observations
\cite{ua1-w,ua1-z}.}
\end{exercise}

\subsection{Chern--Simons theory and TQFT}

\begin{exercise}[Chern--Simons transgression]\leavevmode\par
\textit{For two connections on $E$, set}
\[
\nabla^t=(1-t)\nabla^0+t\nabla^1,
\qquad
A=\nabla^1-\nabla^0,
\qquad
F_t=F_{\nabla^t},
\]
\textit{where $A\in\Omega^1(M,\End E)$, and define}
\[
\operatorname{CS}_P(\nabla^0,\nabla^1)
=k\int_0^1P(A,F_t,\ldots,F_t)\,dt.
\]
\textit{Prove that}
\[
P(F_{\nabla^1})-P(F_{\nabla^0})
=d\,\operatorname{CS}_P(\nabla^0,\nabla^1).
\]
\end{exercise}

Let $\mathfrak g\subseteq M_N(\mathbb C)$ be a matrix Lie algebra whose trace pairing is nondegenerate. For $A\in\Omega^1(M,\mathfrak g)$, wedge products include matrix multiplication. As above, write
\[
F_A=dA+A\wedge A
\]
and define
\[
\operatorname{CS}(A)
=\operatorname{tr}\left(
A\wedge dA+\frac23A\wedge A\wedge A
\right).
\]

\begin{exercise}[Chern--Simons form]\leavevmode\par
\textit{Prove that}
\[
d\,\operatorname{CS}(A)
=\operatorname{tr}(F_A\wedge F_A).
\]
\end{exercise}

For a closed oriented three-manifold $M$, define
\[
S_{\mathrm{CS}}(A)=\int_M\operatorname{CS}(A).
\]

\begin{exercise}[Chern--Simons equation]\leavevmode\par
\textit{For $\alpha\in\Omega^1(M,\mathfrak g)$, set $A_t=A+t\alpha$. Prove that}
\[
\left.\frac{d}{dt}\right|_{t=0}S_{\mathrm{CS}}(A_t)
=2\int_M\operatorname{tr}(\alpha\wedge F_A).
\]
\textit{Deduce that the critical points satisfy $F_A=0$.}
\end{exercise}

For $k\in\mathbb R$, set $S_k=(k/4\pi)S_{\mathrm{CS}}$.

\begin{exercise}[Gauge transformation]\leavevmode\par
\textit{Let $G$ be a compact, connected, simple, and simply connected matrix
Lie group with Lie algebra $\mathfrak g$. For a smooth map $g:M\to G$, define}
\[
A^g=g^{-1}Ag+g^{-1}dg.
\]
\textit{Normalize the invariant trace so that}
\[
\nu(g)=\frac{1}{24\pi^2}\int_M
\operatorname{tr}\bigl((g^{-1}dg)^3\bigr)\in\mathbb Z.
\]
\textit{Prove that}
\[
\operatorname{CS}(A^g)-\operatorname{CS}(A)
=-d\,\operatorname{tr}(dg\,g^{-1}\wedge A)
-\frac13\operatorname{tr}\bigl((g^{-1}dg)^3\bigr).
\]
\textit{Deduce that $\exp(iS_k)$ is invariant under all gauge transformations when $k\in\mathbb Z$.}
\end{exercise}

Now restrict to $2+1$ spacetime dimensions. Let $M$ be a closed oriented three-manifold and let $\ell>0$, so the cosmological constant is $\Lambda=-\ell^{-2}$. After identifying the coframe representation with $\mathfrak{sl}(2,\mathbb R)$, let
\[
e,\omega\in\Omega^1(M,\mathfrak{sl}(2,\mathbb R)),
\qquad
A^\pm=\omega\pm\frac1\ell e.
\]
The pair $(A^+,A^-)$ has gauge Lie algebra $\mathfrak{sl}(2,\mathbb R)\oplus\mathfrak{sl}(2,\mathbb R)$.
Write
\[
d_\omega e=de+\omega\wedge e+e\wedge\omega,
\qquad
F_\omega=d\omega+\omega\wedge\omega.
\]

\begin{exercise}[Flat connections encode three-dimensional gravity]\leavevmode\par
\textit{Prove that $F_{A^+}=F_{A^-}=0$ is equivalent to}
\[
d_\omega e=0,
\qquad
F_\omega+\frac1{\ell^2}e\wedge e=0.
\]
\textit{Assuming that $e:TM\to\mathfrak{sl}(2,\mathbb R)$ is fiberwise invertible, identify these equations with the torsion-free condition and the Einstein equation with constant negative curvature.}
\end{exercise}

\begin{exercise}[Three-dimensional gravity is a Chern--Simons theory]\leavevmode\par
\textit{For the first-order Einstein--Hilbert functional}
\[
S_{\mathrm{EH}}(e,\omega)
=\int_M\operatorname{tr}\left(
e\wedge F_\omega+\frac{1}{3\ell^2}e\wedge e\wedge e
\right).
\]
\textit{Prove that, up to normalization and an exact term, it equals}
\[
S_{\mathrm{CS}}(A^+)-S_{\mathrm{CS}}(A^-)
\]
\cite{achucarro-townsend,witten-gravity}.
\end{exercise}

For a connection $A$ and a finite-dimensional representation $V$ of its gauge group, parallel transport around an oriented closed curve $\gamma$ gives the holonomy $\operatorname{Hol}_\gamma(A)$. Define the \textbf{Wilson loop}
\[
W_V(\gamma,A)
=\operatorname{tr}_V\!\left(\operatorname{Hol}_\gamma(A)\right).
\]
For a framed oriented link $L=\gamma_1\sqcup\cdots\sqcup\gamma_r$ labeled by $V_1,\ldots,V_r$, set
\[
W_L(A)=\prod_{j=1}^rW_{V_j}(\gamma_j,A).
\]
For a framed link, define the formal Chern--Simons expectation value \cite{witten-jones}
\[
\langle W_L\rangle_k
=
\frac{\int e^{iS_k(A)}W_L(A)\,\mathcal DA}
{\int e^{iS_k(A)}\,\mathcal DA}.
\]
\begin{exercise}[Wilson loops and the Jones polynomial]\leavevmode\par
\textit{For normalized $SU(2)$ theory at level $k$, set}
\[
q=\exp\left(\frac{2\pi i}{k+2}\right).
\]
\textit{Assume the normalized Wilson-loop expectations satisfy}
\[
q^{-1}\langle W_{L_+}\rangle_k
-q\langle W_{L_-}\rangle_k
=\left(q^{1/2}-q^{-1/2}\right)\langle W_{L_0}\rangle_k
\]
\textit{and $\langle W_{\bigcirc}\rangle_k=1$. Show that they determine the
Jones polynomial, account for framing dependence, and agree with the
Reshetikhin--Turaev invariant of Exercise~\ref{ex:rt-tqft}.}
\end{exercise}

\section{Analysis}

\subsection{Operators on Hilbert space}

A \textbf{Hilbert space} is a complete complex inner-product space. Write
$\mathcal B(H,K)$ for the bounded operators $H\to K$ with
\[
\|T\|=\sup_{\|x\|=1}\|Tx\|,
\qquad
\mathcal B(H)=\mathcal B(H,H).
\]
A \textbf{$C^*$-algebra} is a complex Banach $*$-algebra $A$ satisfying
$\|a^*a\|=\|a\|^2$. An element $a\in A$ is \textbf{positive}, written
$a\geq0$, if $a=b^*b$ for some $b\in A$. The algebra $M_n(A)$ has its standard
$C^*$-algebra structure.

\begin{exercise}[Hilbert-space operators form a $C^*$-algebra]\leavevmode\par
\textit{Prove the Cauchy--Schwarz inequality, construct the adjoint of every
$T\in\mathcal B(H,K)$, and prove}
\[
(ST)^*=T^*S^*,
\qquad
\|T^*T\|=\|T\|^2.
\]
\textit{Deduce that $\mathcal B(H)$ is a $C^*$-algebra and that
$M_n(\mathcal B(H))\cong\mathcal B(H^{\oplus n})$ isometrically as a
$*$-algebra.}
\end{exercise}

\subsection{Operator K-theory}

For a unital $C^*$-algebra $A$, let
$M_\infty(A)=\bigcup_{n\geq1}M_n(A)$ under $x\mapsto x\oplus0$. Projections
$p,q\in M_\infty(A)$ are \textbf{stably Murray--von Neumann equivalent} if,
after adjoining zero blocks, there is $v\in M_\infty(A)$ with
\[
v^*v=p,
\qquad
vv^*=q.
\]
Let $V(A)$ be the resulting set of equivalence classes with addition induced
by direct sum. Define $K_0(A)$ as the Grothendieck group of $V(A)$.

Let $U_\infty(A)=\bigcup_{n\geq1}U(M_n(A))$ under $u\mapsto u\oplus1$, and
define $K_1(A)=\pi_0(U_\infty(A))$. For a nonunital $C^*$-algebra $A$, let
$\widetilde A$ be its unitization with quotient
$\epsilon:\widetilde A\to\mathbb C$, and define
\[
K_i(A)=
\ker\!\left(K_i(\widetilde A)\xrightarrow{\epsilon_*}K_i(\mathbb C)\right),
\qquad i=0,1.
\]

\begin{exercise}[Stable projections and unitaries define K-groups]
\leavevmode\par
\textit{Prove that direct sum makes $V(A)$ a commutative monoid, that $K_1(A)$
is an abelian group, and that $K_0$ and $K_1$ are functorial for
$*$-homomorphisms. Prove the stable isomorphisms}
\[
K_i(M_n(A))\cong K_i(A)
\]
\textit{and compute}
\[
K_0(\mathbb C)\cong\mathbb Z,
\qquad
K_1(\mathbb C)=0.
\]
\end{exercise}

An operator $K\in\mathcal B(H,K')$ is \textbf{compact} if the image of its
closed unit ball has compact closure. Write $\mathcal K(H)$ for the compact
operators on $H$.

\begin{exercise}[Compact operators carry integer-valued K-theory]
\leavevmode\par
\textit{For an infinite-dimensional separable Hilbert space $H$, prove that
$\mathcal K(H)$ is the norm-closed ideal generated by finite-rank operators.
Classify projections in $M_\infty(\mathcal K(H))$ by rank and prove}
\[
K_0(\mathcal K(H))\cong\mathbb Z,
\qquad
K_1(\mathcal K(H))=0.
\]
\end{exercise}

For compact Hausdorff $X$, write $C(X)$ for the unital $C^*$-algebra of
continuous complex-valued functions on $X$ with pointwise operations.

\begin{exercise}[Operator K-theory recovers vector bundles]
\leavevmode\par
\textit{For compact Hausdorff $X$, associate to
$p\in M_n(C(X))$ the bundle with fiber $\operatorname{im}p(x)$. Prove the
Serre--Swan equivalence between complex vector bundles over $X$ and finitely
generated projective $C(X)$-modules, where projective means isomorphic to a
direct summand of $C(X)^n$ \cite{swan-vector-bundles}. Deduce the natural
isomorphism}
\[
K_0(C(X))\cong K^0(X).
\]
\textit{Identify this description with the projection-valued classification
of Exercise~\ref{ex:stable-bundle-classification}.}
\end{exercise}

\subsection{Fredholm operators and index theory}

An operator $T:H\to K'$ is \textbf{Fredholm} if its kernel and cokernel are
finite dimensional and its image is closed; define
\[
\operatorname{ind}(T)=\dim\ker T-\dim\operatorname{coker}T.
\]

\begin{exercise}[The Calkin boundary map is the Fredholm index]
\label{ex:calkin-index}
\leavevmode\par
\textit{Let $H$ be infinite-dimensional and separable and set
$\mathcal Q(H)=\mathcal B(H)/\mathcal K(H)$. Prove that $T\in\mathcal B(H)$
is Fredholm exactly when its image $q(T)\in\mathcal Q(H)$ is invertible.
Construct the six-term K-theory sequence of}
\cite{blackadar-k-theory}
\[
0\longrightarrow\mathcal K(H)
\longrightarrow\mathcal B(H)
\xrightarrow{q}\mathcal Q(H)
\longrightarrow0
\]
\textit{Compute $K_i(\mathcal B(H))=0$, use polar retraction to regard
$q(T)$ as a $K_1$-class, and prove, with the index convention for the
boundary map,}
\[
\partial:K_1(\mathcal Q(H))
\xrightarrow{\ \cong\ }K_0(\mathcal K(H))\cong\mathbb Z,
\qquad
\partial[q(T)]
=[P_{\ker T}]-[P_{\ker T^*}]
=\operatorname{ind}(T).
\]
\textit{Deduce that the Fredholm index is locally constant, additive under
composition, and invariant under compact perturbations.}
\end{exercise}

Let $\operatorname{Fred}(H)$ denote the Fredholm operators on $H$, with the
operator-norm topology.

\begin{exercise}[Fredholm families represent K-theory]\leavevmode\par
\textit{For an infinite-dimensional separable Hilbert space $H$ and compact
Hausdorff space $X$, use Exercise~\ref{ex:calkin-index} to construct the index
bundle of a norm-continuous family $X\to\operatorname{Fred}(H)$ and prove the
Atiyah--J\"anich isomorphism}
\cite{atiyah-k-theory}
\[
[X,\operatorname{Fred}(H)]\cong K^0(X).
\]
\end{exercise}

Let $M$ be a closed smooth manifold and let $E,F\to M$ be complex vector
bundles. An order-$m$ \textbf{differential operator}
$D:\Gamma(E)\to\Gamma(F)$ is locally of the form
$D=\sum_{|\alpha|\leq m}a_\alpha(x)\partial^\alpha$. Its \textbf{principal
symbol} is
\[
\sigma_D(x,\xi)=\sum_{|\alpha|=m}a_\alpha(x)\xi^\alpha:
E_x\longrightarrow F_x.
\]
The operator is \textbf{elliptic} if $\sigma_D(x,\xi)$ is invertible for
$\xi\ne0$. Write $K_c^0(Y)=\widetilde K^0(Y^+)$ for compactly supported
K-theory. The \textbf{symbol class} of an elliptic operator is
\[
[\sigma_D]=[\pi^*E,\pi^*F,\sigma_D]\in K_c^0(T^*M).
\]
Define $\operatorname{ind}_{\mathrm{top}}:K_c^0(T^*M)\to\mathbb Z$ by the
K-theoretic pushforward to a point.

\begin{exercise}[Atiyah--Singer index theorem]
\label{ex:atiyah-singer}\leavevmode\par
\textit{Prove that an elliptic operator $D$ has a Fredholm extension and that}
\cite{atiyah-singer-index}
\[
\operatorname{ind}(D)
=\dim\ker D-\dim\ker D^*
=\operatorname{ind}_{\mathrm{top}}[\sigma_D].
\]
\textit{For a Riemannian metric on $M$, apply this to the de Rham operator
$d+d^*:\Omega^{\mathrm{even}}(M)\to\Omega^{\mathrm{odd}}(M)$ and recover
$\operatorname{ind}(d+d^*)=\chi(M)$.}
\end{exercise}

\begin{exercise}[Instanton charge produces the chiral anomaly]\leavevmode\par
\textit{Let $A$ be a finite-action anti-self-dual $\operatorname{SU}(N)$
connection on $\mathbb R^4$ extending to a Hermitian bundle $E\to S^4$.
For skew-Hermitian curvature $F_A$ and the fundamental trace, define its
topological charge by}
\[
Q(A)=\frac{1}{8\pi^2}\int_{S^4}\operatorname{tr}(F_A\wedge F_A)
=\langle c_2(E),[S^4]\rangle.
\]
\textit{Prove that $Q(A)\in\mathbb Z$. Let $S^\pm$ be the half-spinor bundles
and $D_A^\pm$ the coupled chiral Dirac operators. Use
Exercise~\ref{ex:atiyah-singer} to prove}
\[
\dim\ker D_A^- - \dim\ker D_A^+=Q(A).
\]
\textit{For $N_f$ massless Dirac fermions in the fundamental representation,
deduce the integrated chiral anomaly}
\cite{fujikawa-anomaly}
\[
\Delta Q_5=\int_{\mathbb R^4}d(\star J_5)=2N_fQ(A).
\]
\end{exercise}

\subsection{Operator spaces and completely bounded maps}

A \textbf{concrete operator space} is a closed linear subspace
$E\subseteq\mathcal B(H,K)$. Its $n$th matrix level is
\[
M_n(E)\subseteq\mathcal B(H^{\oplus n},K^{\oplus n})
\]
with the inherited operator norm. For a linear map $\phi:E\to F$, define
\[
\phi_n([x_{ij}])=[\phi(x_{ij})],
\qquad
\|\phi\|_{\mathrm{cb}}=\sup_{n\geq1}\|\phi_n\|.
\]
The map is a \textbf{complete contraction}, \textbf{complete isometry}, or
\textbf{completely bounded} when every $\phi_n$ is contractive, isometric, or
uniformly bounded, respectively. Write $\operatorname{CB}(E,F)$ for the
completely bounded maps.

\begin{exercise}[Ruan's axioms characterize concrete operator spaces]
\leavevmode\par
\textit{Show that concrete matrix norms satisfy}
\[
\|x\oplus y\|=\max\{\|x\|,\|y\|\},
\qquad
\|\alpha x\beta\|\leq\|\alpha\|\,\|x\|\,\|\beta\|.
\]
\textit{Conversely, prove that a vector space with matrix norms satisfying
these axioms admits a completely isometric embedding into
$\mathcal B(H)$ \cite{ruan-operator-spaces}.}
\end{exercise}

The \textbf{row} and \textbf{column operator spaces} are
\[
R_n=M_{1,n}\subseteq\mathcal B(\mathbb C^n,\mathbb C),
\qquad
C_n=M_{n,1}\subseteq\mathcal B(\mathbb C,\mathbb C^n).
\]

\begin{exercise}[Matrix levels detect noncommutativity]
\label{ex:row-column}\leavevmode\par
\textit{For the identity maps between the underlying Hilbert spaces of
$R_n$ and $C_n$, prove}
\[
\|1:R_n\to C_n\|_{\mathrm{cb}}
=\|1:C_n\to R_n\|_{\mathrm{cb}}=\sqrt n.
\]
\textit{For transposition $t:M_n\to M_n$, prove}
\[
\|t\|=1,
\qquad
\|t\|_{\mathrm{cb}}=n.
\]
\textit{Explain why Banach-space norm alone forgets both distinctions.}
\end{exercise}

\begin{exercise}[Smith's lemma stabilizes maps into matrices]\leavevmode\par
\textit{For an operator space $E$ and a linear map $\phi:E\to M_n$, prove}
\cite{smith-cb}
\[
\|\phi\|_{\mathrm{cb}}=\|\phi_n\|.
\]
\textit{Deduce that complete boundedness is a finite computation whenever the
codomain is a fixed matrix algebra.}
\end{exercise}

\subsection{Operator-space duality and Haagerup tensoring}

The \textbf{operator-space dual} of $E$ is the Banach dual $E^*$ with matrix
norms determined by
\[
M_n(E^*)=\operatorname{CB}(E,M_n)
\]
isometrically. Rectangular spaces $M_{p,q}(E)$ carry the norms inherited from
$\mathcal B(H^{\oplus q},K^{\oplus p})$. For
$u=\sum_{k=1}^r x_k\otimes y_k\in E\otimes F$, define
\[
\|u\|_h
=\inf\left\{
\|[x_1\ \cdots\ x_r]\|_{M_{1,r}(E)}
\|[y_1\ \cdots\ y_r]^{\mathsf T}\|_{M_{r,1}(F)}
\right\}.
\]
The completion is the \textbf{Haagerup tensor product} $E\otimes_hF$.

\begin{exercise}[Rows and columns give Schatten norms]
\leavevmode\par
\textit{Under $e_i\otimes e_j\mapsto E_{ij}$, prove the complete isometries}
\[
C_n\otimes_hR_n\cong M_n,
\qquad
R_n\otimes_hC_n\cong S_1^n,
\]
\textit{where $M_n$ has operator norm and $S_1^n$ has trace norm. Interpret
the reversal of the tensor factors as the distinction between largest
singular value and total singular mass.}
\end{exercise}

\begin{exercise}[Trace norm governs state discrimination]
\leavevmode\par
\textit{Let $\rho_0,\rho_1\in M_n$ be positive with trace one, and let
$p_0,p_1$ be probabilities with $p_0+p_1=1$. A binary measurement is a pair
$M_0,M_1\geq0$ with $M_0+M_1=1$. Define
$\|x\|_1=\trace\sqrt{x^*x}$. For}
\[
p_{\mathrm{succ}}
=p_0\trace(M_0\rho_0)+p_1\trace(M_1\rho_1),
\]
\textit{prove the Helstrom formula \cite{helstrom-detection}}
\[
\max_{(M_0,M_1)}p_{\mathrm{succ}}
=\frac12\left(1+\|p_0\rho_0-p_1\rho_1\|_1\right).
\]
\textit{Determine an optimal measurement from the positive spectral
subspace of $p_0\rho_0-p_1\rho_1$, and use
$R_n\otimes_hC_n\cong S_1^n$ to interpret distinguishability as a Haagerup
tensor norm.}
\end{exercise}

For a bilinear map $u:E\times F\to G$, define
\[
u_n(x,y)_{ij}=\sum_{k=1}^n u(x_{ik},y_{kj}),
\qquad
\|u\|_{\mathrm{mb}}=\sup_n\|u_n\|.
\]
Such a map is \textbf{multiplicatively bounded} if this supremum is finite.

\begin{exercise}[Haagerup tensoring linearizes products]
\leavevmode\par
\textit{Prove that $u$ is multiplicatively bounded exactly when its
linearization extends to a completely bounded map}
\[
\widetilde u:E\otimes_hF\longrightarrow G,
\qquad
\widetilde u(x\otimes y)=u(x,y),
\]
\textit{and prove $\|\widetilde u\|_{\mathrm{cb}}=\|u\|_{\mathrm{mb}}$.
Apply this to multiplication in a $C^*$-algebra.}
\end{exercise}

\subsection{Operator systems and completely positive maps}

An \textbf{operator system} is a self-adjoint unital subspace
$S\subseteq\mathcal B(H)$. A linear map $\Phi:S\to T$ is \textbf{positive}
if $\Phi(S_+)\subseteq T_+$, and \textbf{completely positive} if every
$\Phi_n:M_n(S)\to M_n(T)$ is positive.

\begin{exercise}[Partial transpose detects entanglement]
\leavevmode\par
\textit{A state in $M_m\otimes M_n$ is a positive element $\rho$ with
$\trace\rho=1$. Call it separable if
$\rho=\sum_k p_k\sigma_k\otimes\tau_k$ for states $\sigma_k,\tau_k$ and
probabilities $p_k$. Prove that transposition $t:M_n\to M_n$ is positive and
that every separable state has positive partial transpose}
\cite{peres-ppt}
\[
\rho^\Gamma=(1\otimes t)(\rho)\geq0.
\]
\textit{For
$\Omega_2=(|00\rangle+|11\rangle)/\sqrt2$, compute the spectrum of
$(|\Omega_2\rangle\langle\Omega_2|)^\Gamma$ and deduce that $t$ is not
$2$-positive. Assume the Peres--Horodecki theorem that positivity of the
partial transpose is also sufficient for separability in
$2\times2$ and $2\times3$ systems \cite{horodecki-ppt}. Apply it to}
\[
\rho_p=p|\psi^-\rangle\langle\psi^-|+(1-p)\frac{1_4}{4},
\qquad
|\psi^-\rangle=\frac{|01\rangle-|10\rangle}{\sqrt2},
\]
\textit{and prove that $\rho_p$ is separable exactly when $p\leq1/3$.}
\end{exercise}

For $\Phi:M_n\to M_m$, its \textbf{Choi matrix} is
\[
C_\Phi=\sum_{i,j=1}^n E_{ij}\otimes\Phi(E_{ij})\in M_n\otimes M_m.
\]

\begin{exercise}[Choi matrices classify completely positive maps]
\leavevmode\par
\textit{Prove that the following are equivalent \cite{choi-cp}:}
\[
\Phi\text{ is completely positive},
\qquad
C_\Phi\geq0,
\qquad
\Phi(x)=\sum_\alpha V_\alpha xV_\alpha^*.
\]
\textit{Determine the least number of Kraus operators from
$\operatorname{rank}C_\Phi$ and express trace preservation and unitality as
partial-trace conditions on $C_\Phi$.}
\end{exercise}

\begin{exercise}[Choi states enable process tomography]
\leavevmode\par
\textit{Let
$\Omega_n=n^{-1/2}\sum_{i=1}^n e_i\otimes e_i$ and let
$\Phi:M_n\to M_m$ be completely positive and trace preserving. Define}
\[
\rho_\Phi=(1\otimes\Phi)(|\Omega_n\rangle\langle\Omega_n|).
\]
\textit{Prove}
\[
\rho_\Phi=\frac1nC_\Phi,
\qquad
\Phi(x)=n\,\operatorname{Tr}_1
\!\left((x^{\mathsf T}\otimes1)\rho_\Phi\right),
\qquad
\operatorname{Tr}_2\rho_\Phi=\frac1n1.
\]
\textit{Here $\operatorname{Tr}_1$ and $\operatorname{Tr}_2$ are the partial
traces over the first and second tensor factors. Call a bipartite probe
$\sigma\in M_n\otimes M_n$ full
operator-Schmidt rank if
$\sigma=\sum_{k=1}^{n^2}A_k\otimes B_k$ with both families linearly
independent. Prove that $(1\otimes\Phi)(\sigma)$ determines every linear map
$\Phi:M_n\to M_m$ exactly when $\sigma$ has full operator-Schmidt rank.
Construct a separable probe with this property, and compare the entangled-
and separable-probe process reconstructions reported in the photonic
experiment of Altepeter et al. \cite{altepeter-tomography}.}
\end{exercise}

\begin{exercise}[Stinespring dilation turns positivity into geometry]
\leavevmode\par
\textit{For a completely positive map
$\Phi:A\to\mathcal B(H)$ from a unital $C^*$-algebra, construct a
$*$-representation $\pi:A\to\mathcal B(K)$ and $V:H\to K$ such that}
\cite{stinespring}
\[
\Phi(a)=V^*\pi(a)V,
\qquad
V^*V=\Phi(1).
\]
\textit{Prove uniqueness of a minimal dilation up to unitary equivalence.}
\end{exercise}

For $E\subseteq\mathcal B(H,K)$, define its \textbf{Paulsen operator system}
\[
\mathcal S(E)=
\left\{
\begin{pmatrix}
\lambda1_K&x\\
y^*&\mu1_H
\end{pmatrix}:
\lambda,\mu\in\mathbb C,\ x,y\in E
\right\}
\subseteq\mathcal B(K\oplus H).
\]

\begin{exercise}[The $2\times2$ trick converts contractions into positivity]
\leavevmode\par
\textit{For $\phi:E\to\mathcal B(H_1,H_2)$, prove that $\phi$ is a complete
contraction exactly when the unital map}
\[
\begin{pmatrix}\lambda1_K&x\\y^*&\mu1_H\end{pmatrix}
\longmapsto
\begin{pmatrix}\lambda1_{H_2}&\phi(x)\\
\phi(y)^*&\mu1_{H_1}\end{pmatrix}
\]
\textit{is completely positive. Use this operator-system encoding to derive
the completely contractive extension theorem.}
\end{exercise}

\begin{exercise}[Wittstock factorizes completely bounded maps]
\leavevmode\par
\textit{For a completely bounded map
$\phi:A\to\mathcal B(H)$, prove that there are a representation
$\pi:A\to\mathcal B(K)$ and operators $V,W:H\to K$ such that}
\cite{wittstock}
\[
\phi(a)=V^*\pi(a)W,
\qquad
\|\phi\|_{\mathrm{cb}}
=\inf\{\|V\|\|W\|:\phi(\mathord{\cdot})=V^*\pi(\mathord{\cdot})W\}.
\]
\textit{Deduce that every completely bounded map is a linear combination of
completely positive maps.}
\end{exercise}

\subsection{Quantum channels and matrix-level distinguishability}

For $x\in M_n$, write $\|x\|_1=\trace\sqrt{x^*x}$. A
\textbf{state} is $\rho\in M_n$ with $\rho\geq0$ and $\trace\rho=1$. The
\textbf{partial trace} $\operatorname{Tr}_B:M_n\otimes M_m\to M_n$ is
characterized by
\[
\trace\!\left(a\,\operatorname{Tr}_B x\right)
=\trace\!\left((a\otimes1)x\right).
\]
A \textbf{Schr\"odinger-picture quantum channel} is a completely positive
trace-preserving map $\Phi:M_n\to M_m$. Its
\textbf{Heisenberg adjoint} $\Phi^*:M_m\to M_n$ is determined by
\[
\trace(\Phi(\rho)a)=\trace(\rho\Phi^*(a)).
\]

\begin{exercise}[Schmidt spectra measure entanglement]
\leavevmode\par
\textit{For $\psi\in H_A\otimes H_B$, apply singular-value decomposition to
$T_\psi:H_A^*\to H_B$, where $T_\psi(f)=(f\otimes1)(\psi)$, and obtain}
\[
\psi=\sum_{i=1}^r s_i a_i\otimes b_i.
\]
\textit{Characterize product vectors by $r=1$, identify $s_i^2$ as the
nonzero eigenvalues of both reduced states, and compute}
\[
S(\psi)=-\sum_i s_i^2\log s_i^2.
\]
\end{exercise}

\begin{exercise}[Channels are isometries with an environment]\leavevmode\par
\textit{Combine Choi's theorem and Stinespring dilation to prove that every
channel has equivalent forms}
\[
\Phi(\rho)=\sum_\alpha K_\alpha\rho K_\alpha^*
=\operatorname{Tr}_E(V\rho V^*),
\qquad
V^*V=1.
\]
\textit{Describe the Choi--Jamio{\l}kowski correspondence and determine
which bipartite states correspond to channels.}
\end{exercise}

For a linear map $\Psi:M_n\to M_m$, define its \textbf{diamond norm} by
\[
\|\Psi\|_\diamond
=\sup_{k\geq1}\|\Psi\otimes1_{M_k}\|_{1\to1}.
\]

\begin{exercise}[The diamond norm is the operational cb norm]
\leavevmode\par
\textit{Prove the stabilization and duality identities}
\cite{watrous-cb-norms}
\[
\|\Psi\|_\diamond
=\|\Psi\otimes1_{M_n}\|_{1\to1},
\qquad
\|\Psi\|_\diamond=\|\Psi^*\|_{\mathrm{cb}}.
\]
\textit{For equally likely channels $\Phi_0,\Phi_1$, prove that the optimal
one-use discrimination probability, allowing an entangled ancilla, is}
\[
p_{\mathrm{succ}}
=\frac12+\frac14\|\Phi_0-\Phi_1\|_\diamond.
\]
\end{exercise}

\begin{exercise}[Entanglement improves process discrimination]
\leavevmode\par
\textit{Assume the Piani--Watrous theorem that every entangled bipartite
state improves the success probability in some channel-discrimination task
\cite{piani-watrous}. Explain why this is an operational manifestation of
matrix amplification. Reconstruct the subchannels and unsteerable benchmark
in the photonic experiment of Sun et al., and compare its measured
steering-assisted success probabilities with that benchmark
\cite{sun-subchannel}.}
\end{exercise}

\begin{exercise}[No channel universally clones pure states]
\label{ex:no-cloning}\leavevmode\par
\textit{Suppose a channel $C$ and fixed blank state $\sigma$ satisfy
$C(\rho\otimes\sigma)=\rho\otimes\rho$ for every pure state $\rho$. Use
contractivity of trace distance, or monotonicity of fidelity, to obtain a
contradiction from two nonorthogonal states. Recover the unitary inner-product
argument of Wootters and Zurek \cite{wootters-zurek}.}
\end{exercise}

\subsection{Nonlocal correlations and Grothendieck inequalities}

\begin{exercise}[Every entangled pure two-qubit state violates CHSH]\leavevmode\par
\textit{Let $A_0,A_1$ and $B_0,B_1$ be self-adjoint operators with
eigenvalues in $\{-1,1\}$, and define the CHSH operator}
\[
\mathcal C=A_0\otimes(B_0+B_1)+A_1\otimes(B_0-B_1).
\]
\textit{Establish the classical CHSH bound $2$. For the two-qubit state}
\[
\psi_\theta=\cos\theta\,|00\rangle+\sin\theta\,|11\rangle,
\qquad 0\leq\theta\leq\frac\pi4,
\]
\textit{derive the maximal CHSH value} \cite{gisin}
\[
2\sqrt{1+\sin^2(2\theta)}.
\]
\textit{Conclude that every entangled pure two-qubit state violates
CHSH. Relate the four observables to spacelike-separated measurement settings
and compare the quantum violation with a loophole-free Bell experiment using
electron spins separated by $1.3$ kilometers \cite{hensen-bell}.}
\end{exercise}

A \textbf{bipartite binary quantum correlation matrix} has entries
\[
c_{ij}=\langle\psi,(A_i\otimes B_j)\psi\rangle,
\]
where $\psi$ is a unit vector. The operators $A_i$ and $B_j$ are self-adjoint unitaries on finite-dimensional Hilbert spaces.

For a real matrix $M=(M_{ij})$, define
\[
\beta_{\mathrm C}(M)
=\max_{\varepsilon_i,\delta_j\in\{-1,1\}}
\left|\sum_{i,j}M_{ij}\varepsilon_i\delta_j\right|
\]
and
\[
\beta_{\mathrm Q}(M)
=\sup_c\left|\sum_{i,j}M_{ij}c_{ij}\right|,
\]
where the supremum is over bipartite binary quantum correlation matrices.

\begin{exercise}[Tsirelson turns quantum correlations into vectors]
\leavevmode\par
\textit{Prove that every binary quantum correlation has a representation}
\cite{tsirelson}
\[
c_{ij}=\langle x_i,y_j\rangle_{\mathbb R},
\qquad
\|x_i\|,\|y_j\|\leq1.
\]
\textit{Prove the converse using finite-dimensional Clifford
representations and deduce the sharp CHSH value $2\sqrt2$.}
\end{exercise}

The \textbf{real Grothendieck constant} $K_G^{\mathbb R}$ is the least constant for which
\[
\beta_{\mathrm Q}(M)\leq K_G^{\mathbb R}\beta_{\mathrm C}(M)
\]
holds for every real matrix $M$.

\begin{exercise}[Grothendieck bounds quantum advantage uniformly]
\leavevmode\par
\textit{Prove that $K_G^{\mathbb R}<\infty$ \cite{grothendieck}. Use}
\[
M_{\mathrm{CHSH}}=
\begin{pmatrix}1&1\\1&-1\end{pmatrix},
\]
\textit{to prove $K_G^{\mathbb R}\geq\sqrt2$ and interpret the inequality as
a dimension-independent comparison of scalar and Hilbert-space strategies.}
\end{exercise}

For a bilinear form $u:A\times B\to\mathbb C$, define
$T_u:A\to B^*$ by $T_u(a)(b)=u(a,b)$. The form is
\textbf{jointly completely bounded} when $T_u$ is completely bounded, with
$\|u\|_{\mathrm{jcb}}=\|T_u\|_{\mathrm{cb}}$. A \textbf{state} on a unital
$C^*$-algebra is a positive functional of norm one.

\begin{exercise}[The operator-space Grothendieck inequality]
\leavevmode\par
\textit{Prove that for every jointly completely bounded
$u:A\times B\to\mathbb C$ there are states $f_1,f_2$ on $A$ and $g_1,g_2$
on $B$ such that} \cite{haagerup-musat}
\[
|u(a,b)|\leq\|u\|_{\mathrm{jcb}}
\left(
f_1(aa^*)^{1/2}g_1(b^*b)^{1/2}
+f_2(a^*a)^{1/2}g_2(bb^*)^{1/2}
\right).
\]
\textit{Deduce a factorization of $T_u$ through a direct sum of row and
column Hilbert operator spaces, and recover the scalar Grothendieck pattern
on commutative subalgebras.}
\end{exercise}

\subsection{Dual operator spaces and von Neumann algebras}

For $\mathcal S\subseteq\mathcal B(H)$, its \textbf{commutant} is
\[
\mathcal S'=\{T\in\mathcal B(H):Tx=xT\text{ for every }x\in\mathcal S\}.
\]
A \textbf{von Neumann algebra} is a unital $*$-subalgebra $M\subseteq\mathcal B(H)$ satisfying $M=M''$. A net $(T_\lambda)$ converges to $T$ in the \textbf{strong operator topology} if $T_\lambda\xi\to T\xi$ for every $\xi\in H$, and in the \textbf{weak operator topology} if
\[
\langle T_\lambda\xi,\eta\rangle\longrightarrow\langle T\xi,\eta\rangle
\]
for every $\xi,\eta\in H$.

\begin{exercise}[Bicommutants are dual operator algebras]\leavevmode\par
\textit{Let $A\subseteq\mathcal B(H)$ be a unital $*$-subalgebra. Prove that}
\[
A''
=\overline A^{\,\mathrm{SOT}}
=\overline A^{\,\mathrm{WOT}}.
\]
\textit{Identify the canonical predual of $A''$ as a quotient of the
trace-class operators.}
\end{exercise}

An operator space $E$ is a \textbf{dual operator space} with predual $E_*$ if
$E=E_*^*$ completely isometrically. A map between dual operator spaces is
\textbf{normal} if it is weak-$*$ continuous. The \textbf{center} of a von
Neumann algebra is $Z(M)=M\cap M'$, and $M$ is a \textbf{factor} if
$Z(M)=\mathbb C1$. A \textbf{normal channel} is a normal unital completely
positive map.

\begin{exercise}[Normal CP maps have Stinespring dilations]
\leavevmode\par
\textit{Let $M$ be a von Neumann algebra and let
$\Phi:M\to\mathcal B(H)$ be normal and completely positive. Prove
Stinespring's theorem \cite{stinespring}: there are a Hilbert space $K$, a
normal representation $\pi:M\to\mathcal B(K)$, and $V:H\to K$ such that}
\[
\Phi(x)=V^*\pi(x)V,
\qquad
V^*V=\Phi(1).
\]
\textit{Deduce that every normal channel on $\mathcal B(H_0)$ has the form}
\[
\Phi(x)=V^*(x\otimes1_E)V
\]
\textit{and that its preadjoint is a Schr\"odinger-picture channel.}
\end{exercise}

A positive functional $\psi$ is \textbf{normal} if it preserves suprema of
bounded increasing nets of positive elements and \textbf{faithful} if
$\psi(a^*a)=0$ implies $a=0$. A \textbf{type $\mathrm{II}_1$ factor} is an
infinite-dimensional factor with a faithful normal tracial state
\[
\tau:M\longrightarrow\mathbb C,
\qquad
\tau(ab)=\tau(ba),
\qquad
\tau(1)=1.
\]

For projections $p,q\in M$, write $p\sim q$ if there is $v\in M$ such that
\[
v^*v=p,
\qquad
vv^*=q.
\]
Write $p\precsim q$ if $p$ is equivalent to a projection $r$ satisfying $rq=r$.

\begin{exercise}[Projection dimension]\leavevmode\par
\textit{Let $M$ be a type $\mathrm{II}_1$ factor with normalized trace $\tau$. Prove that}
\[
p\sim q\quad\Longrightarrow\quad\tau(p)=\tau(q),
\]
\textit{and prove the comparison theorem}
\[
p\precsim q
\quad\Longleftrightarrow\quad
\tau(p)\leq\tau(q).
\]
\textit{Deduce that $\tau(p)=\tau(q)$ if and only if $p\sim q$.}
\end{exercise}

\subsection{Conditional expectations and subfactors}

A von Neumann algebra is \textbf{finite} if $v^*v=1$ implies $vv^*=1$. Let
$N\subseteq M$ be finite von Neumann algebras with a common faithful normal
tracial state $\tau$. A \textbf{trace-preserving conditional expectation} is
a normal unital completely positive map $E_N:M\to N$ such that
\[
\tau\circ E_N=\tau,
\qquad
E_N(n_1xn_2)=n_1E_N(x)n_2
\]
for $x\in M$ and $n_1,n_2\in N$. Let $L^2(M,\tau)$ be the completion of $M$ for
\[
\langle x,y\rangle_\tau=\tau(y^*x).
\]
Let $M$ act on $L^2(M,\tau)$ by left multiplication.

\begin{exercise}[Norm-one projections are conditional expectations]
\leavevmode\par
\textit{Let $A$ be a unital $C^*$-algebra, $B\subseteq A$ a unital
$C^*$-subalgebra, and $P:A\to B$ a linear projection with $\|P\|=1$. Prove
Tomiyama's theorem \cite{tomiyama}: $P$ is positive, completely positive, and}
\[
P(b_1ab_2)=b_1P(a)b_2.
\]
\end{exercise}

\begin{exercise}[Conditional expectations give Jones projections]
\leavevmode\par
\textit{Prove that there is a unique trace-preserving conditional
expectation $E_N:M\to N$ and that it extends to the orthogonal projection}
\[
e_N:L^2(M,\tau)\longrightarrow L^2(N,\tau).
\]
\textit{Prove that}
\[
e_N=e_N^*=e_N^2,
\qquad
e_Nxe_N=E_N(x)e_N
\]
\textit{for every $x\in M$.}
\end{exercise}

The \textbf{basic construction} is
\[
M_1=\langle M,e_N\rangle''
\subseteq\mathcal B(L^2(M,\tau)),
\]
the double commutant of the algebra generated by $M$ and $e_N$. For a projection $p=[p_{ij}]\in M_k(N)$ acting by left multiplication, define
\[
\dim_N\!\left(pL^2(N,\tau)^k\right)=\sum_{i=1}^k\tau(p_{ii}).
\]
An inclusion $N\subseteq M$ has \textbf{finite Jones index} if $L^2(M,\tau)$ is isomorphic as a right $N$-module to such a module; then
\[
[M:N]=\dim_NL^2(M,\tau).
\]

\begin{exercise}[The Pimsner--Popa bound detects finite index]
\leavevmode\par
\textit{For an inclusion $N\subseteq M$ of type $\mathrm{II}_1$ factors
with trace-preserving expectation $E_N$, prove that finite index is
equivalent to the existence of $\lambda>0$ satisfying}
\cite{pimsner-popa}
\[
E_N(x)\geq\lambda x\qquad(x\in M_+).
\]
\textit{Prove that the largest such $\lambda$ is $[M:N]^{-1}$ and interpret
this order inequality as a quantitative strengthening of complete
positivity.}
\end{exercise}

\begin{exercise}[Jones basic construction]\leavevmode\par
\textit{Let $N\subseteq M$ be an inclusion of type $\mathrm{II}_1$ factors of finite Jones index. Prove that $M_1$ is a finite factor and that its normalized trace satisfies} \cite{jones-sunder}
\[
\tau_1(xe_Ny)
=\frac{1}{[M:N]}\tau(xy)
\]
\textit{for $x,y\in M$.}
\end{exercise}

The \textbf{Jones tower} is the sequence obtained by iterating the basic construction:
\[
N=M_{-1}\subset M=M_0\subset M_1\subset M_2\subset\cdots.
\]
Let $e_i$ be the Jones projection used to construct $M_i$ from $M_{i-1}$, and set
\[
\delta=\sqrt{[M:N]}.
\]
The \textbf{Temperley--Lieb algebra} with loop parameter $\delta$ is generated by $U_i$ subject to
\[
U_i^2=\delta U_i,
\qquad
U_iU_j=U_jU_i\quad(|i-j|\geq2),
\qquad
U_iU_{i\pm1}U_i=U_i.
\]

\begin{exercise}[Temperley--Lieb relations]\leavevmode\par
\textit{Prove that the Jones projections satisfy} \cite{temperley-lieb,jones-index}
\[
e_i^2=e_i=e_i^*,
\qquad
e_ie_j=e_je_i\quad(|i-j|\geq2),
\]
\[
e_ie_{i\pm1}e_i=\delta^{-2}e_i.
\]
\textit{Deduce that $U_i\mapsto\delta e_i$ defines a representation of the Temperley--Lieb algebra.}
\end{exercise}

\begin{exercise}[Braid representation]\leavevmode\par
\textit{Let $q\in\mathbb C^\times$ satisfy $\delta=q+q^{-1}$ and define
$\sigma_i=q\,1-U_i$. Prove that $\sigma_i$ is invertible with inverse
$q^{-1}1-U_i$, and prove that}
\[
\sigma_i\sigma_{i+1}\sigma_i
=\sigma_{i+1}\sigma_i\sigma_{i+1},
\]
\[
\sigma_i\sigma_j=\sigma_j\sigma_i
\qquad
(|i-j|\geq2).
\]
\textit{Deduce that the $\sigma_i$ define a representation of the braid group} \cite{jones-polynomial}.
\end{exercise}

A \textbf{Markov trace} on the Temperley--Lieb tower is a compatible family of positive normalized traces satisfying
\[
\operatorname{tr}(xU_i)=\delta^{-1}\operatorname{tr}(x)
\]
whenever $x$ belongs to the algebra generated by the preceding generators.

\begin{exercise}[Positivity quantizes subfactor indices below four]\leavevmode\par
\textit{Define}
\[
P_0(\delta)=1,
\qquad
P_1(\delta)=\delta,
\qquad
P_{n+1}(\delta)=\delta P_n(\delta)-P_{n-1}(\delta).
\]
\textit{Prove that the normalized traces on the Jones tower restrict to a
Markov trace. Prove that, when $\sin\theta\ne0$,}
\[
P_n(2\cos\theta)
=\frac{\sin((n+1)\theta)}{\sin\theta}.
\]
\textit{Use positivity of the Markov trace to prove that if $0<\delta<2$, then}
\[
\delta=2\cos\left(\frac{\pi}{m}\right)
\]
\textit{for some integer $m\geq3$.}
\end{exercise}

\begin{exercise}[Jones index theorem]
\label{ex:jones-index}\leavevmode\par
\textit{Let $N\subseteq M$ be an inclusion of type $\mathrm{II}_1$ factors of finite Jones index. Prove that} \cite{jones-index}
\[
[M:N]
\in
\left\{
4\cos^2\left(\frac{\pi}{n}\right):n\geq3
\right\}
\cup
[4,\infty).
\]
\textit{Relate the discrete values to the $SU(2)_k$ braid representations
and Chern--Simons Wilson-loop invariants constructed above.}
\end{exercise}

\section{Tensor categories and generalized cohomology}

\subsection{Tensor products, duality, and trace}

Write $\operatorname{Vect}_{\mathbb C}^{\mathrm{fd}}$ for finite-dimensional
complex vector spaces, $\operatorname{Rep}_{\mathbb C}^{\mathrm{fd}}(G)$ for
finite-dimensional complex representations of $G$,
$\operatorname{Ch}_{\mathbb C}^{b}$ for bounded cochain complexes of
finite-dimensional complex vector spaces, and $\operatorname{OS}^{\mathrm{fd}}$
for finite-dimensional operator spaces with completely bounded maps. For
$C,D\in\operatorname{Ch}_{\mathbb C}^{b}$, define
\[
(C\otimes D)^n=\bigoplus_{p+q=n}C^p\otimes D^q,
\qquad
d(c\otimes e)=dc\otimes e+(-1)^pc\otimes de
\]
for $c\in C^p$, and define the graded interchange by
$c\otimes e\mapsto(-1)^{pq}e\otimes c$ for $e\in D^q$.

For finite factors $M,N$, an \textbf{$M$--$N$ correspondence} is a Hilbert
space $H$ with commuting normal unital representations of $M$ and
$N^{\mathrm{op}}$, the algebra with the multiplication of $N$ reversed. Write
$\operatorname{Corr}(M,N)$ for their category with bounded intertwiners. For
an $M$--$N$ correspondence $H$ and an $N$--$P$ correspondence $K$, write
$H\mathbin{\overline\otimes_N}K$ for their \textbf{Connes fusion}, an
$M$--$P$ correspondence. A \textbf{bicategory} has objects, a category of
morphisms between each ordered pair of objects, and horizontal composition
that is associative and unital up to coherent natural isomorphisms.

\begin{exercise}[Tensor products have different symmetries]
\label{ex:tensor-structures-synthesis}\leavevmode\par
\textit{Prove that
$\operatorname{Vect}_{\mathbb C}^{\mathrm{fd}}$,
$\operatorname{Rep}_{\mathbb C}^{\mathrm{fd}}(G)$,
$\operatorname{Ch}_{\mathbb C}^{b}$, and $\operatorname{Vect}(X)$ for
compact Hausdorff $X$ are
symmetric rigid monoidal categories. Identify their tensor units and duals,
and prove that their categorical traces are respectively the ordinary trace,
the ordinary trace of an equivariant endomorphism, the supertrace of a
cochain endomorphism $f=(f^n)_n$,}
\[
\operatorname{str}(f)=\sum_n(-1)^n\trace(f^n),
\]
\textit{and the fiberwise trace of a bundle endomorphism. Prove the Hopf
trace identity}
\[
\sum_n(-1)^n\trace(f^n)
=\sum_n(-1)^n\trace\bigl(H^n(f)\bigr)
\]
\textit{for an endomorphism of a bounded complex.}

\textit{Prove that $\operatorname{OS}^{\mathrm{fd}}$ is monoidal under
$\otimes_h$ with unit $\mathbb C$, but that the ordinary flip cannot define
a symmetry: use Exercise~\ref{ex:row-column} and}
\[
C_n\otimes_hR_n\cong M_n,
\qquad
R_n\otimes_hC_n\cong S_1^n.
\]

\textit{Construct the bicategory whose objects are finite factors, whose
hom-categories are $\operatorname{Corr}(M,N)$, and whose horizontal
composition is Connes fusion. Prove that $L^2(M)$ is the unit and that
finite-index correspondences are dualizable} \cite{jones-sunder}.
\end{exercise}

\begin{exercise}[Tensor structure controls physical observables]
\leavevmode\par
\textit{Express the selection rule of
Exercise~\ref{ex:electric-dipole-selection} as the nonvanishing of}
\[
\operatorname{Hom}_G
\bigl(1,V_f^*\otimes V_\mu\otimes V_i\bigr),
\]
\textit{and explain why this formulation is independent of bases and
Clebsch--Gordan coordinates.}

\textit{For the pointed $\mathbb Z/3$ category of
Exercise~\ref{ex:anyonic-monodromy}, prove that}
\[
S_{rs}=\trace(M_{a^r,a^s})
=\exp(2\pi i rs/3)
\]
\textit{is an invertible Fourier matrix. Relate its entries to the measured
$2\pi/3$ monodromy phase and to the modularity hypothesis in
Exercise~\ref{ex:rt-tqft}} \cite{nakamura-anyons}.

\textit{For a finite-index inclusion $N\subseteq M$ of type $\mathrm{II}_1$
factors, prove that the $N$--$M$ correspondence
${}_NL^2(M)_M$ is dualizable. Define the statistical dimension of a
dualizable correspondence $H$ by $d(H)=\trace(1_H)$ and prove}
\[
d({}_NL^2(M)_M)=\sqrt{[M:N]}.
\]
\textit{Identify this dimension with the Temperley--Lieb loop parameter
$\delta$, and relate categorical duality to the discrete values in
Exercise~\ref{ex:jones-index}} \cite{jones-index}.
\end{exercise}

\subsection{Generalized cohomology and index}

A pointed space has a distinguished basepoint. Its \textbf{wedge} is the
disjoint union of pointed spaces with their basepoints identified. For a
pointed space $(X,x_0)$, its \textbf{reduced suspension} is
\[
\Sigma X
=X\times[0,1]\big/
\bigl(X\times\{0,1\}\cup\{x_0\}\times[0,1]\bigr).
\]
For a pointed space $Y$, its \textbf{loop space} $\Omega Y$ is the space of
continuous based maps $S^1\to Y$.

A \textbf{reduced generalized cohomology theory} on pointed CW complexes is
a sequence of contravariant functors $\widetilde E^n$ to abelian groups that
is invariant under based homotopy, carries wedges to products, has natural
suspension isomorphisms
\[
\widetilde E^n(X)\cong\widetilde E^{n+1}(\Sigma X),
\]
and assigns to every pointed CW subcomplex $A\subseteq X$ a natural long
exact sequence
\[
\cdots\longrightarrow\widetilde E^n(X/A)
\longrightarrow\widetilde E^n(X)
\longrightarrow\widetilde E^n(A)
\longrightarrow\widetilde E^{n+1}(X/A)
\longrightarrow\cdots.
\]
It satisfies the \textbf{dimension axiom} if
$\widetilde E^n(S^0)=0$ for $n\ne0$. For an unpointed space $X$, set
$E^n(X)=\widetilde E^n(X_+)$, where $X_+$ is $X$ with a disjoint basepoint.

An \textbf{$\Omega$-spectrum} is a sequence of pointed spaces $E_n$ with
pointed homotopy equivalences $E_n\simeq\Omega E_{n+1}$. It
\textbf{represents} $\widetilde E$ when
\[
\widetilde E^n(X)\cong[X,E_n]_*.
\]
Let $U=\varinjlim U(n)$ be the stable unitary group.

\begin{exercise}[Spectra represent cohomology and K-theory]
\leavevmode\par
\textit{Prove Brown representability: every reduced generalized cohomology
theory satisfying the wedge axiom is represented by an $\Omega$-spectrum}
\cite{brown-cohomology}. \textit{For an abelian group $A$, identify ordinary
cohomology with the Eilenberg--Mac Lane spectrum $HA$, whose $n$th space
$K(A,n)$ has $\pi_n\cong A$ and all other homotopy groups zero. Identify the
spaces of the complex K-theory spectrum $KU$ as}
\[
KU_{2j}\simeq\mathbb Z\times BU,
\qquad
KU_{2j+1}\simeq U.
\]
\textit{Prove that ordinary cohomology satisfies the dimension axiom, that
complex K-theory does not, and that the two-periodicity of $KU$ is Bott
periodicity. For a finite CW complex $X$, prove that the Chern character
induces the rational isomorphisms} \cite{atiyah-k-theory}
\[
K^0(X)\otimes\mathbb Q
\cong\bigoplus_jH^{2j}(X;\mathbb Q),
\qquad
K^1(X)\otimes\mathbb Q
\cong\bigoplus_jH^{2j+1}(X;\mathbb Q).
\]
\end{exercise}

\begin{exercise}[K-theory has three models]
\label{ex:k-models-synthesis}\leavevmode\par
\textit{For a finite group $G$, define $K_G^0(X)$ as the Grothendieck group
of finite-rank complex $G$-equivariant vector bundles over a compact
$G$-space $X$. Define the representation ring $R(G)$ as the Grothendieck
group of finite-dimensional complex representations, with multiplication
induced by tensor product, and prove}
\[
K_G^0(\mathrm{pt})\cong R(G).
\]
\textit{For a compact finite CW complex $X$ and an infinite-dimensional
separable Hilbert space $H$, prove the natural
identifications} \cite{atiyah-k-theory,swan-vector-bundles}
\[
K^0(X)
\cong[X,\mathbb Z\times BU]
\cong K_0(C(X))
\cong[X,\operatorname{Fred}(H)],
\]
\[
K^1(X)
\cong[X,U]
\cong K_1(C(X)).
\]
\textit{Given a gapped class A Hamiltonian $H:X\to\End(V)$, follow its
class through the negative spectral projection, occupied vector bundle,
classifying map to $\mathbb Z\times BU$, projection over $C(X)$, and
Fredholm-family models. Prove that the Chern character is independent of the
chosen model.}
\end{exercise}

\begin{exercise}[Generalized cohomology quantizes observables]
\leavevmode\par
\textit{Place the following constructions in the generalized-cohomology
framework above. First, interpret the Aharonov--Bohm phase in
Exercise~\ref{ex:aharonov-bohm} as the exponential of the pairing between a
degree-one ordinary cohomology class and a one-cycle. Second, use
Exercise~\ref{ex:k-models-synthesis} to prove that the first Chern number and
TKNN Hall response in Exercise~\ref{ex:class-a-insulators} depend only on the
corresponding $KU$-class. Third, prove that the connecting morphism for the
Calkin extension in Exercise~\ref{ex:calkin-index} extends the $KU$ cofiber
boundary from commutative $C^*$-algebras and sends a Fredholm class to its
integer index.}

\textit{For an elliptic operator $D$ on a closed manifold, follow its symbol
class from $K_c^0(T^*M)$ through K-theoretic pushforward and prove that the
result agrees with the analytic boundary-map index by
Exercise~\ref{ex:atiyah-singer}. Deduce that Aharonov--Bohm holonomy is
invariant under gauge change and homology of loops in the field-free region,
that the Hall integer is invariant under gap-preserving homotopy, and that
the Fredholm index is invariant under norm-continuous Fredholm homotopy.
Identify the distinct generalized cohomology theories responsible for these
quantized quantities.}
\end{exercise}

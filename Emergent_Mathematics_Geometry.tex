\part{Geometry}

\section{Symplectic Geometry}

\begin{exercise}[Differential forms and Cartan calculus]
Let $M$ be a smooth manifold.  Its cotangent space at $p$ is
$T_p^*M=(T_pM)^*$, and $T^*M$ is the union of these spaces over $p\in M$.

A \emph{differential $k$-form} on $M$ is a smooth section of
$\Lambda^kT^*M$; write $\Omega^k(M)$ for the space of differential
$k$-forms.  The exterior
products on the cotangent spaces define
\[
\wedge:\Omega^r(M)\times\Omega^s(M)\longrightarrow\Omega^{r+s}(M).
\]
For a smooth map $F:M\to N$, define
\[
(F^*\alpha)_p(v_1,\ldots,v_k)
=\alpha_{F(p)}(dF_pv_1,\ldots,dF_pv_k).
\]
For a vector field $X$, define contraction by
\[
(\iota_X\alpha)_p(v_2,\ldots,v_k)
=\alpha_p(X_p,v_2,\ldots,v_k).
\]
The \emph{exterior derivative} is the family
$d:\Omega^k(M)\to\Omega^{k+1}(M)$ satisfying $df(X)=Xf$, the graded
Leibniz rule, $d^2=0$, and $F^*d=dF^*$.
For a vector field $X$ with local flow $\Phi_t$, define
\[
\mathcal L_X\alpha
=\left.\frac d{dt}\right|_{t=0}\Phi_t^*\alpha.
\]

Construct $d$ locally and prove its uniqueness.  For contraction
$\iota_X$, prove Cartan's identity
\[
\mathcal L_X=d\iota_X+\iota_Xd.
\]
For $A\in\Omega^1(M)$, put $F=dA$.  Prove that replacing $A$ by
$A+d\chi$, with $\chi\in C^\infty(M)$, leaves $F$ unchanged.  Use
Cartan's identity to derive
\[
\mathcal L_XA=\iota_XF+d(\iota_XA).
\]
\end{exercise}

\begin{exercise}[Local Poincar\'e lemma]
Let $U$ be a star-shaped open neighborhood of $0$ in a finite-dimensional
real vector space.  For $\alpha\in\Omega^k(U)$ with $k>0$, define
\[
(K\alpha)_x(v_1,\ldots,v_{k-1})
=\int_0^1t^{k-1}\alpha_{tx}(x,v_1,\ldots,v_{k-1})\,dt.
\]
Use Cartan's identity for the radial flow to prove
\[
dK+Kd=\id
\]
on positive-degree forms.  Deduce that every closed form of positive degree
on $U$ is exact, with a primitive vanishing at the origin.
\end{exercise}

\begin{exercise}[Cotangent phase space]
For the projection $\pi:T^*Q\to Q$, define the tautological one-form by
\[
\theta_\alpha(v)=\alpha(d\pi(v)),
\qquad \alpha\in T^*Q,\quad v\in T_\alpha(T^*Q).
\]
The canonical two-form on $T^*Q$ is $\omega=d\theta$.

For a finite-dimensional real vector space $V$, construct the canonical
identifications
\[
TV\cong V\times V,
\qquad
T^*V\cong V\times V^*.
\]
Prove that a phase $S\in C^\infty(Q)$ determines the momentum
$p=dS_q\in T_q^*Q$.  For a diffeomorphism
$F:Q\to Q$, construct its cotangent lift and prove that it preserves
$\theta$ and $d\theta$.  Under $T^*V\cong V\times V^*$, prove
\[
\omega\bigl((q,p),(q',p')\bigr)=p(q')-p'(q).
\]
\end{exercise}

\begin{exercise}[Symplectic manifolds and Hamiltonian fields]
A \emph{symplectic form} on a smooth manifold $M$ is a closed
nondegenerate two-form $\omega$.  A symplectic vector space as defined in
Symmetry carries the corresponding constant symplectic form.  For
$f\in C^\infty(M)$, define
\[
\iota_{X_f}\omega=-df,
\qquad
\{f,g\}=-\omega(X_f,X_g).
\]
The local flow of $X_f$ is the \emph{Hamiltonian flow} of $f$.
A diffeomorphism $F:(M,\omega_M)\to(N,\omega_N)$ is a
\emph{canonical transformation} when $F^*\omega_N=\omega_M$.

Prove that the canonical two-form $d\theta$ on $T^*Q$ is nondegenerate and
hence symplectic.
Prove that the Poisson bracket is antisymmetric,
satisfies the Leibniz rule and Jacobi identity, and is preserved by
canonical transformations.  For the Hamiltonian flow $\Phi_t$ of $H$,
prove
\[
\frac{d}{dt}f(\Phi_t(x))=\{f,H\}(\Phi_t(x)).
\]
\end{exercise}

\begin{exercise}[Liouville's theorem]
\label{ex:liouville-theorem}
On a $2n$-dimensional symplectic manifold, prove that Hamiltonian flow
preserves $\omega$, the volume form $\omega^n/n!$, and every density
depending only on the Hamiltonian.  On a symplectic vector space, prove
that the infinitesimal generator of a linear Hamiltonian flow has trace
zero and that the flow has determinant one.
\end{exercise}

\begin{exercise}[Contact manifolds and symplectization]
Let $Y$ be a smooth manifold of dimension $2n+1$.  A one-form $\alpha$ is
a \emph{contact form} when
\[
\alpha\wedge(d\alpha)^n
\]
is nowhere zero.  Its kernel $\xi=\ker\alpha$ is the associated cooriented
\emph{contact structure}; contact forms defining the same cooriented
contact structure differ by multiplication by a positive smooth function.  The
\emph{Reeb vector field} $R_\alpha$ is defined by
\[
\alpha(R_\alpha)=1,
\qquad
\iota_{R_\alpha}d\alpha=0.
\]
An immersed $n$-manifold $\iota:L\to Y$ is \emph{Legendrian} when
$\iota^*\alpha=0$.

Let $(M,\omega=d\lambda)$ be an exact symplectic manifold.  On
$M\times\R$, with coordinate function $u$ on the second factor, prove that
\[
\alpha=du-\lambda
\]
is a contact form and that $R_\alpha$ is the infinitesimal generator of
translation in the $\R$ factor.  For a manifold $X$, identify
$J^1X=T^*X\times\R$ with this contactification and prove that
\[
j^1f:X\longrightarrow J^1X,
\qquad
j^1f(x)=(df_x,f(x)),
\]
is Legendrian for every $f\in C^\infty(X)$.  Prove that every section of
$J^1X\to X$ with Legendrian image is of this form.

For a contact manifold $(Y,\alpha)$ and the positive coordinate $r$ on
$\R_{>0}$, prove that
\[
\Omega=d(r\alpha)
\]
is symplectic on $\R_{>0}\times Y$.  Determine the vector field $Z$
satisfying $\iota_Z\Omega=r\alpha$ and prove that multiplication of
$\alpha$ by a positive function gives a symplectomorphic symplectization.
\end{exercise}

\begin{exercise}[Integration of forms and Stokes' theorem]
An \emph{orientation} of an $n$-manifold is a continuous choice of component
of the nonzero elements in $\Lambda^nT_p^*M$. The integral $\int_M\omega$ of a
compactly supported top form is defined using orientation-preserving charts
and a partition of unity.

Prove that the integral is independent of charts and partition of unity.
For an oriented manifold with boundary, prove
\[
\int_Md\omega=\int_{\partial M}\omega.
\]
Deduce the fundamental theorem of calculus, Green's theorem, the divergence
theorem, and the classical curl theorem. For time-dependent magnetic
two-form $B$ and electric one-form $E$, prove the equivalence of
\[
\int_{\partial S}E=-\frac d{dt}\int_SB
\qquad\text{and}\qquad
dE=-\partial_tB,
\]
recovering the observed flux--emf law.
\end{exercise}

\begin{exercise}[de Rham cohomology]
A differential form $\omega$ is \emph{closed} if $d\omega=0$ and
\emph{exact} if $\omega=d\eta$ for some form $\eta$. Since $d^2=0$, the
\emph{de Rham cohomology} of $M$ is
\[
H_{\mathrm{dR}}^k(M)
=\frac{\ker(d:\Omega^k(M)\to\Omega^{k+1}(M))}
{\im(d:\Omega^{k-1}(M)\to\Omega^k(M))}.
\]
Pullback induces maps on de Rham cohomology, and the wedge product makes
$H_{\mathrm{dR}}^*(M)$ a graded-commutative algebra.

Use the local Poincar\'e lemma and Stokes' theorem
to prove that an exact $k$-form has zero integral over
every closed oriented $k$-manifold mapped smoothly into $M$. Prove
$H_{\mathrm{dR}}^1(S^1)\cong\R$, with the isomorphism given by integration.

On $\R^2\setminus\{0\}$, show that
\[
\alpha=\frac{-y\,dx+x\,dy}{x^2+y^2}
\]
is closed and has integral $2\pi$ around the unit circle, hence is not exact.
Compute $H_{\mathrm{dR}}^1(\R^2\setminus\{0\})=\R\cdot[\alpha]$. Deduce that a
closed electromagnetic field strength has local potentials, while it has a
global potential $F=dA$ exactly when $[F]=0$ in de Rham cohomology.
\end{exercise}


\begin{exercise}[Connections, curvature, and holonomy]
A \emph{connection} on a vector bundle $E\to M$ is an $\R$-bilinear map
$\nabla:\Gamma(TM)\times\Gamma(E)\to\Gamma(E)$, linear over smooth
functions in the first variable and satisfying
$\nabla_X(fs)=X(f)s+f\nabla_Xs$. Along a curve $\gamma$ it induces a
covariant derivative $D/dt$; its solutions $Ds/dt=0$ define
\emph{parallel transport} between endpoint fibers.

The \emph{curvature} of $\nabla$ is the $\operatorname{End}(E)$-valued
two-form
\[
R^\nabla(X,Y)s
=\nabla_X\nabla_Ys-\nabla_Y\nabla_Xs-\nabla_{[X,Y]}s.
\]
The \emph{holonomy group} at $p$ consists of the automorphisms of $E_p$
obtained by parallel transport around loops based at $p$.

Prove existence, uniqueness, composition, and inverse properties of parallel
transport. Prove that curvature is tensorial and gives the leading failure
of transport around an infinitesimal parallelogram to return a vector to
itself. Deduce that a flat connection has path-independent transport on every
simply connected coordinate domain.
\end{exercise}

\begin{exercise}[Principal connections and gauge covariance]
For a Lie group $G$, a \emph{principal $G$-bundle} is a smooth map
$P\to M$ with a free right $G$-action, transitive on each fiber, and local
$G$-equivariant trivializations.  For a representation
$G\to\operatorname{GL}(W)$, its
associated vector bundle is
$P\times_GW=(P\times W)/((pg,w)\sim(p,gw))$.
A \emph{principal connection} is an
equivariant $\mathfrak g$-valued one-form $A$ on $P$ reproducing fundamental
vertical fields. It induces connections on every associated bundle, and its
\emph{curvature} is
$F_A=dA+\frac12[A\wedge A]$. A \emph{gauge transformation} is a
$G$-equivariant bundle automorphism covering $\id_M$.

Prove $d_AF_A=0$.  For a gauge transformation $\Phi$, prove
$F_{\Phi^*A}=\Phi^*F_A$ and that holonomy changes by conjugacy.  Prove that
a flat connection has trivial holonomy on contractible loops and is
gauge-equivalent to a product connection on a connected base exactly when
its holonomy representation is trivial.  For $G=U(1)$, compute the holonomy
around a
generator of the fundamental group and exhibit the obstruction to removing
a flat connection globally.
\end{exercise}



\begin{exercise}[Integral curvature and circle cocycles]
A \emph{good cover} of a manifold is an open cover
$\{U_i\}_{i\in I}$ for which every nonempty finite intersection is
contractible.  A \emph{smooth complex vector bundle} is a smooth vector
bundle locally modeled on a finite-dimensional complex vector space by
fiberwise complex-linear product maps; it is a \emph{complex line bundle}
when its complex rank is one.  A \emph{Hermitian vector bundle} carries a
smoothly varying positive-definite Hermitian form, and a connection on it
is \emph{unitary} when it preserves that form.  Let
$F\in\Omega^2(M;i\R)$ be closed and choose, on a good cover,
\[
dA_i=F|_{U_i}.
\]
On $U_i\cap U_j$, choose $f_{ij}:U_i\cap U_j\to i\R$ with
$A_j-A_i=df_{ij}$ and $f_{ji}=-f_{ij}$.  Prove that
\[
c_{ijk}=f_{ij}+f_{jk}+f_{ki}
\]
is constant on every nonempty triple intersection.

Use the following good-cover form of the integral de Rham theorem as
theorem input:
\[
\frac{1}{2\pi i}\int_\Sigma h^*F\in\Z
\quad\text{for every smooth }h:\Sigma\to M
\]
from a closed oriented surface if and only if the $f_{ij}$ may be chosen
so that $c_{ijk}\in2\pi i\Z$.  Under this condition put
\[
g_{ij}=e^{f_{ij}}.
\]
Prove the cocycle identities for the $g_{ij}$, construct the associated
Hermitian line bundle, and show that the local operators $d+A_i$ define a
unitary connection of curvature $F$.  Conversely, compare local unitary
frames of a Hermitian line bundle with connection and recover the integral
period condition from its transition functions.
\end{exercise}

\begin{exercise}[Connections, curvature, and Chern--Weil forms]
Let $E\to M$ be a smooth complex vector bundle of rank $r$.  Define its
direct sum, tensor product, dual, and pullback by the corresponding
fiberwise operations.  Call a connection defined above \emph{complex} when
$\nabla_X(is)=i\nabla_Xs$.  For a complex connection, extend $\nabla$ to
$E$-valued forms by the graded Leibniz rule and write
\[
F_\nabla=\nabla^2\in\Omega^2(M;\operatorname{End}E)
\]
for its curvature.

Prove the Bianchi identity $\nabla F_\nabla=0$.  Using the intrinsic
trace and determinant developed earlier, define
\[
\operatorname{ch}(E,\nabla)
=\operatorname{Tr}\exp\!\left(\frac{iF_\nabla}{2\pi}\right),
\qquad
c(E,\nabla)
=\det\!\left(I+\frac{iF_\nabla}{2\pi}\right)
=1+c_1(E,\nabla)+\cdots+c_r(E,\nabla).
\]
Use Bianchi and cyclicity of trace to prove that the homogeneous components
of $\operatorname{ch}(E,\nabla)$ are closed.  For a path $\nabla_t$ of
connections, derive
\[
\frac d{dt}\operatorname{Tr}
\exp\!\left(\frac{iF_{\nabla_t}}{2\pi}\right)
=d\,\operatorname{Tr}\!\left(
\frac{i\dot\nabla_t}{2\pi}
\exp\!\left(\frac{iF_{\nabla_t}}{2\pi}\right)\right).
\]
Use as theorem input the corresponding Chern--Weil transgression formula
for the determinant invariant polynomials.  Deduce that the de Rham classes
$\operatorname{ch}(E)$ and $c_j(E)$ are independent of the connection,
natural under pullback, and satisfy
\[
c(E\oplus F)=c(E)c(F),
\qquad
\operatorname{ch}(E\oplus F)=\operatorname{ch}(E)+\operatorname{ch}(F),
\qquad
\operatorname{ch}(E\otimes F)=\operatorname{ch}(E)\operatorname{ch}(F).
\]
Prove also that
\[
\operatorname{ch}_q(E^\vee)=(-1)^q\operatorname{ch}_q(E).
\]
Recover the first Chern form of a line bundle used above.
\end{exercise}

\begin{exercise}[Spectral projections and Berry bundles]
Let $W$ be a finite-dimensional Hermitian vector space, let $B$ be a smooth
manifold, and let $H:B\to\operatorname{End}(W)$ be a smooth family of
self-adjoint operators.  Suppose a fixed contour $\Gamma$ encloses an
isolated spectral cluster of constant multiplicity.  Define
\[
P_b=\frac{1}{2\pi i}\oint_\Gamma(z-H(b))^{-1}\,dz,
\qquad
\mathcal E=\coprod_{b\in B}\operatorname{im}P_b.
\]
Prove that $P_b$ is a smooth family of orthogonal projections and that
$\mathcal E\to B$ is a smooth Hermitian vector bundle.  Construct its local
product maps from the restrictions of nearby projections to a fixed fiber,
without choosing a frame.  Define the Berry connection by
\[
\nabla^{\mathrm B}s=P\,ds
\]
for sections satisfying $Ps=s$, and prove
\[
F_{\mathrm B}=P(dP\wedge dP)P,
\qquad
c_1(\mathcal E)=
\left[\frac{i}{2\pi}\operatorname{Tr}P(dP\wedge dP)\right].
\]
\end{exercise}

\begin{exercise}[Two-level Berry degree and gap protection]
For the distinguished two-band Hamiltonian on the Brillouin torus $B=T^2$,
\[
H(k)=d_1(k)\sigma_1+d_2(k)\sigma_2+d_3(k)\sigma_3,
\qquad d(k)\ne0,
\]
let $\widehat d=d/|d|$ and let $\mathcal E_-$ be the negative-energy line
bundle.  From $P_-=(I-\widehat d\cdot\sigma)/2$, derive
\[
\int_{T^2}c_1(\mathcal E_-)
=\frac1{4\pi}\int_{T^2}
\widehat d\cdot
(\partial_{k_1}\widehat d\times\partial_{k_2}\widehat d)
\,dk_1\,dk_2
=\deg(\widehat d).
\]
Show that this integer cannot change under a smooth deformation that keeps
the spectral gap open.
\end{exercise}

\begin{exercise}[Elliptic pendulum period]
On $T^*S^1$ consider
\[
H(\theta,p)=\frac{p^2}{2m\ell^2}+mg\ell(1-\cos\theta).
\]
For an oscillation of amplitude $0<\theta_0<\pi$, set
$k=\sin(\theta_0/2)$ and define
\[
K(k)=\int_0^{\pi/2}\frac{d\phi}
{\sqrt{1-k^2\sin^2\phi}}.
\]
Use energy conservation to prove
\[
T(\theta_0)=4\sqrt{\frac{\ell}{g}}\,K(k).
\]
Apply the generalized binomial theorem with $a=\frac12$.  For
\[
I_m=\int_0^{\pi/2}\sin^{2m}\phi\,d\phi,
\]
use integration by parts to prove
\[
I_0=\frac{\pi}{2},
\qquad
I_m=\frac{2m-1}{2m}I_{m-1}\quad(m\geq1),
\]
and obtain
\[
K(k)=\frac{\pi}{2}\sum_{m=0}^{\infty}
\left(\frac{(1/2)_m}{m!}\right)^2k^{2m},
\qquad |k|<1.
\]
Deduce
\[
T(\theta_0)=2\pi\sqrt{\frac{\ell}{g}}
\left(1+\frac{\theta_0^2}{16}+O(\theta_0^4)\right).
\]
Deduce the experimentally testable increase of period with amplitude and
the logarithmic divergence as $\theta_0\uparrow\pi$.
\end{exercise}

\begin{exercise}[Prequantum line bundles]
A \emph{prequantum line bundle} over $(M,\omega)$ is a Hermitian line bundle
$L\to M$ with a unitary connection $\nabla$ satisfying
\[
F_\nabla=-\frac{i}{\hbar}\omega.
\]
Prove that a prequantum line bundle exists if and only if the period
condition
\[
\frac1{2\pi\hbar}\int_\Sigma f^*\omega\in\Z
\]
holds for every smooth map from a closed oriented surface
$f:\Sigma\to M$.  Apply the circle-cocycle construction above to
$F=-i\omega/\hbar$, identify its local connection forms with symplectic
potentials, and prove that the changes
\[
A_i\longmapsto A_i+d\phi_i,
\qquad
f_{ij}\longmapsto f_{ij}+\phi_j-\phi_i
\]
produce an isomorphic prequantum line bundle with connection.  Show that
two arbitrary prequantum line bundles with this curvature differ by a flat
Hermitian line bundle, so curvature alone need not determine the
prequantization up to isomorphism.
\end{exercise}

\begin{exercise}[Kostant--Souriau commutators]
For $f\in C^\infty(M)$, define the Kostant--Souriau operator by
\[
\mathcal Q_{\rm pre}(f)=-i\hbar\nabla_{X_f}+f.
\]

Prove
\[
[\mathcal Q_{\rm pre}(f),\mathcal Q_{\rm pre}(g)]
=i\hbar\mathcal Q_{\rm pre}(\{f,g\}).
\]
\end{exercise}

\begin{exercise}[Contactification]
Let $\pi:S(L)\to M$ be the unit-circle bundle of $L$, let
$A\in\Omega^1(S(L);i\R)$ be its connection form normalized by
$A(\xi)=i$ on the circle generator $\xi$, and put $\alpha=-iA$.
Prove
\[
d\alpha=-\frac1\hbar\pi^*\omega,
\]
that $\alpha$ is a contact form with Reeb vector field $\xi$, and that
sections of $L$ correspond to functions $F:S(L)\to\C$ satisfying
$F(ue^{i\theta})=e^{-i\theta}F(u)$.
\end{exercise}

\begin{exercise}[Polarized quantum spaces]
The vertical polarization on $T^*V$ is the tangent distribution to the
fibers of $T^*V\to V$.  A complex polarization is an involutive Lagrangian
subbundle $P\subset T_\C M$; its annihilator and canonical line are
\[
P^\circ=\{\alpha\in T_\C^*M:\alpha|_P=0\},
\qquad K_P=\Lambda^{\rm top}P^\circ.
\]
A half-form bundle is a line bundle $\delta_P$ with
$\delta_P^{\otimes2}\cong K_P$.  The corrected polarized sections of
$L\otimes\delta_P$ are covariantly constant along $P$ and square-integrable
for the half-form pairing.

For $T^*V$ with the vertical polarization, recover $L^2(V)$ and the
operators
\[
\psi(x)\longmapsto \ell(x)\psi(x),
\qquad
\psi(x)\longmapsto-i\hbar\,d\psi_x(v).
\]
For a compact
K\"ahler manifold with $P=T^{0,1}M$, prove $P^\circ=T^{*1,0}M$ and
$K_P=K_M$.  For a positive prequantum line bundle, identify the
corrected quantum space with
$H^0(M,L\otimes K_M^{1/2})$.
\end{exercise}


\begin{exercise}[Theta functions and Landau levels]
Let
$Q=\R^2/(L_x\Z\oplus L_y\Z)$ have area $A=L_xL_y$, let a particle of
charge $e$ experience the constant magnetic field $B\,dx\wedge dy$, and
assume $eB>0$.  Put
\[
N=\frac{eBA}{2\pi\hbar},
\qquad
\tau=i\frac{L_y}{L_x},
\qquad
z=\frac{x}{L_x}+\tau\frac{y}{L_y}.
\]
Use the prequantization condition to prove that a magnetic line bundle
$L\to Q$ exists exactly when $N\in\Z$.  For $N>0$ and
$r\in\mathbb Z/N\mathbb Z$, define the level-$N$ theta function
\[
\theta_{N,r}(z\mid\tau)
=\sum_{n\in\mathbb Z}
\exp\!\left(
\pi iN\tau(n+r/N)^2+2\pi iN(n+r/N)z
\right).
\]
Use the holomorphic-gauge automorphy laws
\[
f(z+1)=f(z),
\qquad
f(z+\tau)=e^{-\pi iN\tau-2\pi iNz}f(z)
\]
to prove the magnetic-translation decomposition
\[
H^0(Q,L)
=\bigoplus_{r\in\Z/N\Z}
\C\,\theta_{N,r}(z\mid\tau).
\]
Derive the automorphy and magnetic-translation eigenvalue laws directly
from the defining series.  Relate the holomorphic-gauge sections to
physical wavefunctions by their common unitary-gauge factor.

For magnetic translations by $L_x/N$ and $L_y/N$, use holonomy around a
fundamental magnetic cell to prove
\[
UV=e^{2\pi i/N}VU.
\]
For the covariant Landau Hamiltonian, construct the raising and lowering
operators and prove
\[
E_n=\hbar\frac{eB}{m}\left(n+\frac12\right),
\qquad
\dim E_{\widehat H}(\{E_n\})H=N.
\]
Deduce the cyclotron resonance spacing, the degeneracy density
$eB/(2\pi\hbar)$, and the magnetic-translation interference phase.
State how measurements of these three quantities separately test the
energy, flux-integrality, and theta-function predictions.
\end{exercise}


\begin{exercise}[Poisson manifolds and star products]
A \emph{Poisson manifold} is a smooth manifold $M$ with a bracket on
$C^\infty(M)$ that is a Lie bracket and a derivation in each argument.
Equivalently, there is a bivector $\Pi\in\Gamma(\Lambda^2TM)$ satisfying
\[
\{f,g\}=\Pi(df,dg),
\qquad [\Pi,\Pi]_{\rm Sch}=0.
\]

A \emph{differential star product} on a Poisson manifold is an associative
$\C[[\hbar]]$-bilinear product on $C^\infty(M)[[\hbar]]$ of the form
\[
f\star g=fg+\sum_{r\geq1}\hbar^rC_r(f,g),
\]
where the $C_r$ are bidifferential operators, $1$ is a unit, and
\[
C_1(f,g)-C_1(g,f)=i\{f,g\}.
\]
Let $(E,\omega)$ be a symplectic vector space and let
$\Pi:E^*\to E$ be the inverse of $v\mapsto\omega(v,-)$, and write
$\Pi(k,\ell)=\ell(\Pi(k))$.  Let $D_\Pi$ be the constant bidifferential
operator characterized on decomposable functions by
\[
D_\Pi(f\otimes g)(x,y)
=d_yg\bigl(\Pi(d_xf)\bigr).
\]
On $C^\infty(E)[[\hbar]]$, define
\[
f\star g
=m\circ\exp\!\left(\frac{i\hbar}{2}D_\Pi\right)(f\otimes g),
\]
where $m$ restricts a function on $E\times E$ to the diagonal.  Prove
associativity from the commutation of the induced constant bidifferential
operators on $E^3$, and prove
\[
f\star g=fg+\frac{i\hbar}{2}\{f,g\}+O(\hbar^2).
\]
Verify that this product is a differential star product in the preceding
sense.
For $k,\ell\in E^*$ put $e_k(x)=e^{ik(x)}$ and derive
\[
e_k\star e_\ell
=e^{-i\hbar\Pi(k,\ell)/2}e_{k+\ell}.
\]
Prove associativity directly by checking the multiplier identity for the
phase.  Differentiate in $k$ and $\ell$ to recover the canonical
commutation relations.  Prove that every affine symplectic transformation
of $E$ preserves $\star$.  For $E=V\oplus V^*$, prove that the restriction
to Schwartz functions is the formal $\hbar$-expansion of the Weyl product
defined by operator composition above.
\end{exercise}

\begin{exercise}[Magnetic lattice Hamiltonians]
Let $U,V$ be unitary translations satisfying
\[
UV=e^{2\pi i\alpha}VU,
\]
where $\alpha$ is the magnetic flux through one lattice cell in units of
the flux quantum.  The magnetic nearest-neighbor Hamiltonian is
\[
H_\alpha=U+U^*+V+V^*.
\]
In the position realization on $\ell^2(\Z)$, with transverse phase
$\theta$, it is the Harper operator
\[
(H_{\alpha,\theta}\psi)_m
=\psi_{m+1}+\psi_{m-1}
+2\cos(2\pi\alpha m+\theta)\psi_m.
\]

Derive the magnetic-translation relation from the Moyal plane-wave
identity.  Prove that the commutator phase $e^{2\pi i\alpha}$ is the phase
accumulated around one lattice cell.  Show that nonintegral total flux on a
periodic sample obstructs consistent torus boundary phases, while integral
total flux makes their corner cocycle trivial.  Prove that nonzero integral
flux still obstructs a single globally periodic vector potential.

For rational flux $\alpha=p/q$, prove that the Harper equation
has $q$-periodic coefficients and reduce it by the Floquet condition
$\psi_{m+q}=e^{iqk}\psi_m$ to a $q$-dimensional spectral problem.
At zero flux, use the Chebyshev identity
\[
T_q(\cos t)=\cos(qt)
\]
to derive the $q$-step discriminant
\[
\Delta_{0,q}(E,\theta)
=2T_q\!\left(\frac{E-2\cos\theta}{2}\right).
\]
At half a flux quantum, compute the two bands

E_\pm(k,\theta)

\end{exercise}

\section{Schemes}

\begin{exercise}[Affine schemes, localization, and structure sheaves]
Let $A$ be a commutative ring.  An ideal
$\mathfrak p\subsetneq A$ is \emph{prime} when $ab\in\mathfrak p$ implies
$a\in\mathfrak p$ or $b\in\mathfrak p$.  Define
\[
\operatorname{Spec}A=\{\mathfrak p:\mathfrak p\text{ is prime}\},
\qquad
V(I)=\{\mathfrak p:I\subseteq\mathfrak p\}.
\]
Prove that the sets $V(I)$ are the closed sets of a topology and that the
principal open sets
\[
D(f)=\{\mathfrak p:f\notin\mathfrak p\}
\]
form a family closed under finite intersections and covering every open
set.

For $f\in A$, define the localization $A_f$ by the universal property that
$f$ becomes invertible and every homomorphism from $A$ in which $f$ is
invertible factors uniquely through $A_f$.  A \emph{presheaf} of rings on
a space $X$ assigns a ring $\mathcal F(U)$ to each open set and compatible
restriction maps.  It is a \emph{sheaf} when compatible local sections on
an open cover glue uniquely.  Set
\[
\mathcal O_{\operatorname{Spec}A}(D(f))=A_f
\]
with the localization restriction maps.  Prove the sheaf gluing condition
on principal-open covers and extend this prescription uniquely to a sheaf
on $\operatorname{Spec}A$.  Prove
\[
A\cong\Gamma(\operatorname{Spec}A,\mathcal O_{\operatorname{Spec}A}).
\]

For a sheaf of rings $\mathcal F$ and $x\in X$, define its stalk by
\[
\mathcal F_x=\varinjlim_{x\in U}\mathcal F(U).
\]
A commutative ring is \emph{local} when it has a unique maximal ideal.  A
\emph{locally ringed space} is a space $X$ with a sheaf of commutative
rings $\mathcal O_X$ whose stalks are local rings.  Prove
\[
\mathcal O_{\operatorname{Spec}A,\mathfrak p}\cong A_{\mathfrak p}
\]
and that its unique maximal ideal is
$\mathfrak pA_{\mathfrak p}$.  For a continuous map $f:X\to Y$, define
$(f_*\mathcal O_X)(U)=\mathcal O_X(f^{-1}(U))$.  A \emph{morphism of
sheaves of rings} is a family of ring homomorphisms on open sets commuting
with restriction maps.  A \emph{morphism of
locally ringed spaces}
\[
(f,f^\#):(X,\mathcal O_X)\longrightarrow(Y,\mathcal O_Y)
\]
consists of $f$ and a morphism of sheaves of rings
$f^\#:\mathcal O_Y\to f_*\mathcal O_X$ whose induced homomorphism on each
stalk is local, meaning that the inverse image of the maximal ideal is the
maximal ideal.  An isomorphism of locally ringed spaces is such a morphism
admitting an inverse.

An \emph{affine scheme} is a locally ringed space isomorphic to
$(\operatorname{Spec}A,\mathcal O_{\operatorname{Spec}A})$ for a
commutative ring $A$.  A \emph{scheme} is a locally ringed space admitting
an open cover by affine schemes.  A \emph{morphism of schemes} is a
morphism of their underlying locally ringed spaces.  A \emph{complex
scheme} is a scheme equipped with a morphism to
$\operatorname{Spec}\mathbb C$; its morphisms are required to commute with
these structure morphisms.  It is \emph{of finite type} when it admits a
finite affine open cover by spectra of finitely generated commutative
$\mathbb C$-algebras.  Except in the general definitions above, schemes
in this section are complex and of finite type.

Use as theorem input the weak Nullstellensatz: every maximal ideal of
$\mathbb C[x_1,\ldots,x_n]$ is the kernel of evaluation at a unique point
of $\mathbb C^n$.  For
$A=\mathbb C[x_1,\ldots,x_n]/I$, identify its closed points with the common
zero set of $I$.  Show that replacing $I$ by its radical preserves the
closed points but can change the structure sheaf.  In particular, compare
$\operatorname{Spec}\mathbb C[\varepsilon]/(\varepsilon^2)$ with a reduced
point and identify the nilpotent section that records its infinitesimal
thickening.
\end{exercise}

\begin{exercise}[Morphisms, local rings, tangent spaces, and singularities]
Let $A$ and $B$ be finitely generated commutative $\mathbb C$-algebras.  A
$\mathbb C$-algebra homomorphism $\varphi:A\to B$ induces

\varphi^*:\operatorname{Spec}B\longrightarrow\operatorname{Spec}A,
\qquad
\mathfrak q\longmapsto\varphi^{-1}(\mathfrak q).
\]
Prove continuity, construct the compatible maps on localized structure
sheaves, verify the locality condition on stalks, and prove that composition
of algebra homomorphisms becomes composition in the reverse direction.
Prove the affine anti-equivalence

\operatorname{Hom}_{\rm Sch}(\operatorname{Spec}B,\operatorname{Spec}A)
\cong\operatorname{Hom}_{\mathbb C\text{-alg}}(A,B).
\]

For $p\in X=\operatorname{Spec}A$ corresponding to $\mathfrak p$, define
the residue field, cotangent space, and tangent space by
\[
\kappa(p)=A_{\mathfrak p}/\mathfrak pA_{\mathfrak p},
\]
\[
T_p^*X=\mathfrak pA_{\mathfrak p}/(\mathfrak pA_{\mathfrak p})^2,
\qquad
T_pX=\operatorname{Hom}_{\kappa(p)}(T_p^*X,\kappa(p)).
\]
Define the Krull dimension of a ring as the supremum of the lengths of
strict chains of prime ideals and define the local dimension of $X$ at
$p$ as $\dim A_{\mathfrak p}$.

For an $A$-module $M$, define the module of K\"ahler differentials
$\Omega^1_{A/\mathbb C}$ by a derivation
$d:A\to\Omega^1_{A/\mathbb C}$ universal among $\mathbb C$-linear
derivations from $A$ to $A$-modules.  If
\[
A=\mathbb C[x_1,\ldots,x_n]/(f_1,\ldots,f_r),
\]
derive the conormal sequence and prove at a closed point $p$ that

T_pX
=\{v\in\mathbb C^n:df_j(p)(v)=0\text{ for }1\leq j\leq r\}.
\]
Use as theorem input the Jacobian criterion over $\mathbb C$: $p$ is
smooth exactly when $\dim_{\mathbb C}T_pX=\dim\mathcal O_{X,p}$.
Determine the smooth and singular points of

y^2-x^3,
\qquad
y^2-x^2(x+1),
\qquad
xy,

\end{exercise}

\begin{exercise}[Projective spectra, twisting sheaves, and Hilbert polynomials]
Let $A=\bigoplus_{n\geq0}A_n$ be a finitely generated graded
$\mathbb C$-algebra generated by $A_1$, and let
$A_+=\bigoplus_{n>0}A_n$.  Define
\[
\operatorname{Proj}A
=\{\mathfrak p\subset A:\mathfrak p\text{ is homogeneous prime and }
A_+\not\subseteq\mathfrak p\}.
\]
For homogeneous $f$ of positive degree, define
\[
D_+(f)=\{\mathfrak p:f\notin\mathfrak p\},
\qquad
\mathcal O_{\operatorname{Proj}A}(D_+(f))=(A_f)_0.
\]
Prove that these affine charts glue to a scheme.  For a finite-dimensional
complex vector space $V$, prove
\[
\mathbb P(V)=\operatorname{Proj}(\operatorname{Sym}V^\vee)
\]
and recover its interpretation as the space of complex lines in $V$.

An \emph{algebraic line bundle} on a scheme $X$ is an
$\mathcal O_X$-module locally isomorphic to $\mathcal O_X$.  For
$n\in\mathbb Z$, define the twisting sheaf $\mathcal O(n)$ by the
degree-zero parts $(A(n)_f)_0$ on the displayed charts.  For a homogeneous
ideal $I$, define the projective subscheme $\operatorname{Proj}(A/I)$.
Prove intrinsically for $n\geq0$ that
\[
\Gamma(\mathbb P(V),\mathcal O(n))
\cong\operatorname{Sym}^nV^\vee,
\]
and that $\mathcal O(-1)$ is the tautological line bundle whose fiber over
a line $\ell\subset V$ is $\ell$.
Show that homogenization of an affine ideal gives its projective closure
and identify the points added at infinity for a parabola and a hyperbola.

Define the Hilbert function of $A$ by $H_A(n)=\dim_{\mathbb C}A_n$.
Use as theorem input the Hilbert polynomial theorem: for $n$ sufficiently
large, $H_A(n)$ is a polynomial whose degree is $\dim\operatorname{Proj}A$
and whose leading coefficient is $\deg(\operatorname{Proj}A)/d!$ when
$d=\dim\operatorname{Proj}A$.  For a degree-$e$ hypersurface in
$\mathbb P(V)$, derive

\sum_{n\geq0}H_A(n)t^n
=\frac{1-t^e}{(1-t)^{\dim V}}
\]
and recover its dimension and degree.
\end{exercise}

\begin{exercise}[Intersection multiplicity, resultants, and B\'ezout's theorem]
Let $C=V(F)$ and $D=V(G)$ be projective plane curves with no common

I_p(C,D)

Prove that this length is finite, is invariant under multiplication of

For a two-dimensional vector space $W$ and binary forms

\operatorname{Sym}^{n-1}W^\vee\oplus
\operatorname{Sym}^{m-1}W^\vee
\longrightarrow
\operatorname{Sym}^{m+n-1}W^\vee,

\end{exercise}

\begin{exercise}[Branched covers and the Riemann--Hurwitz formula]
For a compact connected oriented surface $S$, define its genus $g(S)$ by
\[
\chi(S)=2-2g(S).
\]
Let $f:C\to D$ be a nonconstant holomorphic map between compact connected
Riemann surfaces.  At $p\in C$, choose local parameters $z$ at $p$ and $w$
at $f(p)$.  Define the \emph{ramification index} $e_p\geq1$ by
\[
w\circ f=z^{e_p}u(z),
\qquad u(0)\ne0,
\]
and define the ramification divisor
\[
R_f=\sum_{p\in C}(e_p-1)[p].
\]
Show that $e_p$ is independent of the local parameters and that the number
of points in a nearby regular fiber, counted with multiplicity, is a
constant $d$, the degree of $f$.

Triangulate $D$ so that every branch value is a vertex, lift the
triangulation to $C$, and compare vertices, edges, and faces to prove the
Riemann--Hurwitz formula
\[
2g(C)-2=d\bigl(2g(D)-2\bigr)+\deg R_f.
\]
For a square-free polynomial $q\in\mathbb C[x]$ of degree $2g+1$ or
$2g+2$, let $C_q$ be the nonsingular projective curve obtained from
$y^2=q(x)$.  Show that the projection $x:C_q\to\mathbb P^1$ has degree
two, determine its branch points including the point or points over
infinity, and deduce that $C_q$ has genus $g$.
\end{exercise}

\section{Riemann--Roch Theory}

\begin{exercise}[Divisors, line bundles, and first Chern classes]
On a compact Riemann surface $C$, a \emph{divisor} is a finite formal sum
\[
D=\sum_{p\in C}n_p[p],
\qquad n_p\in\mathbb Z,
\qquad
\deg D=\sum_p n_p.
\]
For a nonzero meromorphic function $f$, define
$\operatorname{div}(f)=\sum_p\operatorname{ord}_p(f)[p]$.  Use the argument
principle to prove
\[
\deg\operatorname{div}(f)=0.
\]

A \emph{holomorphic line bundle} on $C$ is a complex line bundle with local
product maps whose transition functions are nowhere-vanishing holomorphic
functions $g_{ij}$ satisfying $g_{ij}g_{jk}=g_{ik}$.  For a divisor
$D=\sum_pn_p[p]$, define $\mathcal O(D)$ to be the sheaf of meromorphic
functions $h$ satisfying $\operatorname{div}(h)+D\geq0$ on each open set.
Show that this sheaf is the sheaf of holomorphic sections of a holomorphic
line bundle.
Two divisors are \emph{linearly equivalent} when their difference is
principal, namely the divisor of a nonzero meromorphic function.  Prove
\[
\mathcal O(D)\cong\mathcal O(D')
\quad\Longleftrightarrow\quad
D\text{ and }D'\text{ are linearly equivalent}.
\]

For a nonzero meromorphic section $s$, define $\operatorname{div}(s)$ from
its local representatives.  Use as theorem input the meromorphic-section
theorem: every holomorphic line bundle on a compact Riemann surface has a
nonzero meromorphic section.  Prove that
\[
\deg L:=\deg\operatorname{div}(s)
\]
is independent of $s$, that tensor products add degrees, and that
$L\cong\mathcal O(\operatorname{div}(s))$.

For a Hermitian connection $\nabla$ on $L$, define
\[
c_1(L)=\left[\frac{i}{2\pi}F_\nabla\right]\in H_{\mathrm{dR}}^2(C;\mathbb R).
\]
Show that this class is independent of the Hermitian connection and is
natural under pullback.  Compare connection forms on overlaps and use
Stokes' theorem around the zeros and poles of a meromorphic section to
prove
\[
\int_C c_1(L)=\deg L.
\]
\end{exercise}

\begin{exercise}[\v{C}ech cohomology and exact sequences]
Let $\mathcal F$ be a sheaf of complex vector spaces on a compact Riemann
surface $C$, and let $\mathfrak U=(U_i)_{i\in I}$ be a finite ordered open
cover.  Define the alternating \v{C}ech cochains and coboundary by
\[
\check C^q(\mathfrak U,\mathcal F)
=\prod_{i_0<\cdots<i_q}
\mathcal F(U_{i_0}\cap\cdots\cap U_{i_q}),
\]
\[
(\delta s)_{i_0\ldots i_{q+1}}
=\sum_{r=0}^{q+1}(-1)^r
s_{i_0\ldots\widehat{i_r}\ldots i_{q+1}}
\big|_{U_{i_0}\cap\cdots\cap U_{i_{q+1}}}.
\]
Prove $\delta^2=0$ and define
$\check H^q(\mathfrak U,\mathcal F)=\ker\delta/\operatorname{im}\delta$.
Show directly from the sheaf gluing axiom that
\[
\check H^0(\mathfrak U,\mathcal F)=\mathcal F(C).
\]

Use as theorem input that a finite cover whose nonempty intersections are
Stein computes the cohomology $H^q(C,\mathcal F)$ of a coherent analytic
sheaf, that coherent sheaves on $C$ have finite-dimensional cohomology and
vanish in degrees $q>1$, and that a short exact sequence of coherent sheaves
induces a long exact cohomology sequence.  For a holomorphic line bundle
$L$, a point $p\in C$, and its inclusion $i_p:\{p\}\to C$, prove that
evaluation at $p$ gives an exact sequence
\[
0\longrightarrow L(-p)\longrightarrow L
\longrightarrow(i_p)_*L_p\longrightarrow0.
\]
Writing $h^q(L)=\dim H^q(C,L)$ and
$\chi(L)=h^0(L)-h^1(L)$, use its long exact sequence to prove

\chi(L)=\chi(L(-p))+1.
\]
Iterate this identity through the positive and negative parts of a divisor
and use the meromorphic-section theorem to deduce
\[
\chi(L)=\deg L+\chi(\mathcal O_C)
\]
for every holomorphic line bundle on $C$.
\end{exercise}

\begin{exercise}[Serre duality and the Riemann--Roch theorem]
Define the canonical line bundle $K_C$ to be the holomorphic cotangent
bundle of a compact connected Riemann surface $C$.  Let $g$ be its
topological genus and use as theorem input the analytic genus identity
$\dim H^0(C,K_C)=g$.
Use as theorem input Serre duality in the form of a natural perfect pairing
\[
H^1(C,L)\otimes H^0(C,K_C\otimes L^{-1})
\longrightarrow\mathbb C,
\qquad

Show that every holomorphic function on $C$ is constant and deduce

\chi(\mathcal O_C)=1-g.

h^0(L)=\deg L+1-g

and that a negative-degree line bundle has no nonzero holomorphic sections.

\begin{exercise}[Abel's theorem and theta divisors]

\begin{exercise}[Theta sections and genus-one Riemann--Roch]

\begin{exercise}[Surface-group representations and character varieties]
Let $\Sigma_g$ be a closed oriented surface of genus $g$, let $G$ be a
complex reductive affine algebraic group, and use the presentation
\[
\pi_1(\Sigma_g)
=\left\langle a_1,b_1,\ldots,a_g,b_g\ \middle|\
\prod_{j=1}^g[a_j,b_j]=1\right\rangle.

Define the representation variety by
\[
\mathcal R(\Sigma_g,G)
=\operatorname{Hom}(\pi_1(\Sigma_g),G)
=\left\{(A_1,B_1,\ldots,A_g,B_g)\in G^{2g}:
\prod_{j=1}^g[A_j,B_j]=1\right\}.
\]
Show that changing the value of a representation by conjugation changes
only the framing at the base point.  If an algebraic group $G$ acts on an
affine algebraic set $X$, define its affine quotient by
\[
\mathbb C[X/\!/G]=\mathbb C[X]^G,
\]
and define the $G$-character variety by
\[
\mathcal X(\Sigma_g,G)=\mathcal R(\Sigma_g,G)/\!/G.
\]

quotient correspond to closed conjugacy orbits.  For $G=SL_n(\mathbb C)$,
prove that every loop $\gamma$ defines a regular function
\[

\]
depending only on the free homotopy class of $\gamma$.  Show that these
are invariant under conjugation.  More generally, for a finite-dimensional
algebraic representation $V$ of $G$, define
$W_\gamma^V([\rho])=\operatorname{Tr}_V\rho(\gamma)$ and prove
\[
W_\gamma^V W_\gamma^W=W_\gamma^{V\otimes W},
\qquad
W_\gamma^{V\oplus W}=W_\gamma^V+W_\gamma^W.
\]
Show that orientation reversal sends $W_\gamma$ to
$\operatorname{tr}(\rho(\gamma)^{-1})$.
\end{exercise}

\begin{exercise}[Hyperelliptic curves and holomorphic differentials]
Let
\[
C:\quad y^2=\prod_{r=1}^{2g+1}(x-e_r),
\qquad e_r\ne e_s\quad(r\ne s),

denote its nonsingular projective completion.  Use Riemann--Hurwitz to prove
that $C$ has genus $g$.  Analyze the finite branch points and the point over
infinity to show that the distinguished differentials
\[
\eta_j=\frac{x^{j-1}dx}{y},
\qquad 1\leq j\leq g,
\]
are holomorphic.  Compare their vanishing orders at infinity and use
$\dim H^0(C,K_C)=g$ to show that they exhaust the holomorphic
differentials.

Choose a symplectic marking
$(a_1,\ldots,a_g,b_1,\ldots,b_g)$ of $H_1(C;\mathbb Z)$, define the
period maps
\[
\mathsf A(\eta)=\left(\int_{a_1}\eta,\ldots,
\int_{a_g}\eta\right),
\qquad
\mathsf B(\eta)=\left(\int_{b_1}\eta,\ldots,
\int_{b_g}\eta\right),
\]
and prove from the Riemann bilinear identity that $\mathsf A$ is an

satisfying $\int_{a_i}\omega_j=\delta_{ij}$ and define

\[
\Omega^{\mathsf T}=\Omega,
\qquad
\operatorname{Im}\Omega>0.
\]
\end{exercise}

\begin{exercise}[Jacobians, Abel--Jacobi maps, and Riemann theta functions]
For a compact Riemann surface $C$, define its Jacobian intrinsically by
\[
\operatorname{Jac}(C)
=H^0(C,K_C)^\vee
\big/
\left\{\eta\mapsto\int_\gamma\eta:
\gamma\in H_1(C;\mathbb Z)\right\}.

\mathbb C^g/(\mathbb Z^g+\Omega\mathbb Z^g).
\]
Define the Riemann theta function by the normally convergent series
\[
\vartheta(z\mid\Omega)
=\sum_{n\in\mathbb Z^g}
\exp\!\left(\pi i\,n^{\mathsf T}\Omega n
+2\pi i\,n^{\mathsf T}z\right).
\]
Prove its holomorphy and automorphy laws
\[
\vartheta(z+m\mid\Omega)=\vartheta(z\mid\Omega),
\]
\[
\vartheta(z+\Omega n\mid\Omega)
=e^{-\pi i n^{\mathsf T}\Omega n-2\pi i n^{\mathsf T}z}
\vartheta(z\mid\Omega).
\]

\mathcal A_{p_0}(p)
=\left[\eta\longmapsto\int_{p_0}^p\eta\right]
\in\operatorname{Jac}(C).

theorem input the Abel--Jacobi theorem: a degree-zero divisor is principal
exactly when its Abel--Jacobi image vanishes, and every point of the
Jacobian is represented by a degree-zero divisor.  Deduce

\]
Use as theorem input Riemann's vanishing theorem to identify a translate of
the zero locus of $\vartheta$ with the Abel--Jacobi images of effective

statements above when $g=1$.
\end{exercise}

\begin{exercise}[Jacobi elliptic functions and elliptic uniformization]
For $0<k<1$, define $\operatorname{sn}(u;k)$ near $u=0$ by

\frac{dt}{\sqrt{(1-t^2)(1-k^2t^2)}}

$\operatorname{cn}(u;k)$ and $\operatorname{dn}(u;k)$ by

\operatorname{cn}^2=1-\operatorname{sn}^2,
\qquad
\operatorname{dn}^2=1-k^2\operatorname{sn}^2,
\qquad
\operatorname{cn}(0;k)=\operatorname{dn}(0;k)=1.
\]
Derive
\[
\operatorname{sn}'=\operatorname{cn}\operatorname{dn},
\qquad
\operatorname{cn}'=-\operatorname{sn}\operatorname{dn},
\qquad
\operatorname{dn}'=-k^2\operatorname{sn}\operatorname{cn}.

x(u)=m\operatorname{sn}^2(u;k),
\qquad
y(u)=m\operatorname{sn}(u;k)\operatorname{cn}(u;k)
\operatorname{dn}(u;k).

Prove

\qquad
\frac{dx}{y}=2\,du.

uniformization of the Legendre curve constructed earlier.
\end{exercise}

\begin{exercise}[Floquet monodromy and Bloch curves]
Let

\Delta(E)=\operatorname{tr}M(E)

\begin{exercise}[Lam\'e theta functions and the spectral curve]

\begin{exercise}[Baker--Akhiezer functions and finite-gap reconstruction]

\]
Use as theorem input the finite-gap KdV evolution theorem, which replaces
$Ux+Z$ in the reconstructed potential by $Ux+Vt+Z$ for the normalized
second-kind KdV period vector $V$.  Show that the genus-one case gives the
cnoidal wave above and that $k\uparrow1$ gives the solitary wave
\[
u(x,t)=a+2\kappa^2
\operatorname{sech}^2\!\bigl(\kappa(x-(6a+4\kappa^2)t)\bigr).
\]
Predict the wavelength, phase speed, and harmonic-amplitude ratios of a
weakly dispersive one-dimensional wave train from $K$, $E$, and the nome.
State the KdV scaling, weak-nonlinearity, long-wavelength, negligible-loss,
one-dimensional, and finite-window assumptions under which a wave-tank or
transmission-line experiment tests these predictions.
\end{exercise}

\section{Relativistic Geometry}

\begin{exercise}[Levi--Civita connections, geodesics, and the exponential map]
Let $(M,g)$ be a Riemannian manifold.  For a connection on $TM$, define its
torsion by
\[
T_\nabla(X,Y)=\nabla_XY-\nabla_YX-[X,Y].
\]
Call the connection metric-compatible when
\[
Xg(Y,Z)=g(\nabla_XY,Z)+g(Y,\nabla_XZ).
\]
Show that torsion-freeness and metric compatibility force the Koszul
formula
\[
\begin{split}
2g(\nabla_XY,Z)={}&Xg(Y,Z)+Yg(Z,X)-Zg(X,Y)\\
&-g(X,[Y,Z])+g(Y,[Z,X])+g(Z,[X,Y]),
\end{split}
\]
and use it to prove existence and uniqueness of the Levi--Civita
connection.

Define a geodesic to be a curve $\gamma$ satisfying
$\nabla_{\dot\gamma}\dot\gamma=0$.  For a fixed-endpoint variation, derive
the first-variation formula for
\[
E(\gamma)=\frac12\int g(\dot\gamma,\dot\gamma)\,dt
\]
and prove that its critical curves are precisely the constant-speed
geodesics.  Prove local existence and uniqueness from initial data
$(p,v)\in TM$, define
\[
\exp_p(v)=\gamma_v(1),
\]
and show that $d(\exp_p)_0$ is the identity on $T_pM$.  Deduce that short
geodesics uniquely minimize length by proving the Gauss lemma for radial
geodesics in a normal neighborhood.
\end{exercise}

\begin{exercise}[Curvature, Jacobi fields, and geodesic deviation]
Define the Riemann curvature endomorphism by
\[
R(X,Y)Z
=\nabla_X\nabla_YZ-\nabla_Y\nabla_XZ-\nabla_{[X,Y]}Z.
\]
Derive its skew symmetries and the first Bianchi identity.  Define
\[
\operatorname{Ric}(X,Y)
=\operatorname{Tr}\bigl(Z\mapsto R(Z,X)Y\bigr),
\qquad
\mathcal R=\operatorname{Tr}_g\operatorname{Ric},
\]
and, for a two-plane spanned by $v,w$, define its sectional curvature by
\[
K(v,w)=
\frac{g(R(v,w)w,v)}{g(v,v)g(w,w)-g(v,w)^2}.
\]

For a variation through geodesics, let $J$ be its variation field and prove
the Jacobi equation
\[
\frac{D^2J}{dt^2}+R(J,\dot\gamma)\dot\gamma=0.
\]
On a manifold of constant sectional curvature $K$, solve this equation for
a perpendicular field with $J(0)=0$ and $DJ/dt(0)=w$ to obtain the factors
\[
\frac{\sin(\sqrt Kt)}{\sqrt K},
\qquad t,
\qquad
\frac{\sinh(\sqrt{-K}t)}{\sqrt{-K}}
\]
for $K>0$, $K=0$, and $K<0$.  Relate their zeros or growth to conjugate
points and the focusing or spreading of neighboring rays.
\end{exercise}

\begin{exercise}[Hodge theory, Euler forms, and Gauss--Bonnet]
On an oriented Riemannian $n$-manifold, define the Hodge operator by
\[
\alpha\wedge\star\beta
=\langle\alpha,\beta\rangle_g\,d\operatorname{vol}_g.
\]
Derive $\star\star\alpha=(-1)^{q(n-q)}\alpha$ on real $q$-forms, define
the formal adjoint $d^*$ from $\star$ and the Hodge Laplacian
\[
\Delta_{\mathrm H}=dd^*+d^*d,
\]
and prove the energy identity
\[
\langle\Delta_{\mathrm H}\alpha,\alpha\rangle_{L^2}
=\|d\alpha\|_{L^2}^2+\|d^*\alpha\|_{L^2}^2
\]
on a closed manifold.  Call a form $\alpha$ \emph{harmonic} when
$\Delta_{\mathrm H}\alpha=0$, and use as theorem input the Hodge theorem
identifying harmonic forms with de Rham cohomology.

For a closed oriented Riemannian surface $S$, use rotation by $\pi/2$ to
regard $TS$ as a complex line bundle.  Show from the Levi--Civita curvature
that its first Chern form is
\[
c_1(TS)=\frac{K_g}{2\pi}\,dA_g.
\]
Use as theorem input the Poincar\'e--Hopf identification
$\int_Sc_1(TS)=\chi(S)$ to prove Gauss--Bonnet:
\[
\int_S K_g\,dA_g=2\pi\chi(S).
\]
Deduce the area $4\pi R^2$ of the round sphere of curvature $R^{-2}$ and
the area $4\pi(g-1)R^2$ of a closed genus-$g>1$ surface of curvature
$-R^{-2}$.
\end{exercise}

\begin{exercise}[Lorentzian metrics and the Einstein equation]
A Lorentzian metric on a four-manifold $M$ is a smooth nondegenerate
symmetric form of signature $(-,+,+,+)$.  Define timelike, null, and
spacelike tangent vectors by the sign of $g(v,v)$, and for a timelike curve
define proper time by
\[
\tau(\gamma)=\frac1c\int\sqrt{-g(\dot\gamma,\dot\gamma)}\,dt.
\]
Show that the timelike cone has two components, that the orthogonal
complement of a timelike vector is spacelike, and that a nonzero null
vector lies on the boundary of either cone component.
Show that the Koszul argument gives a unique Levi--Civita connection and
that freely falling worldlines are the critical curves of proper time.

Define the Einstein tensor by
\[
G=\operatorname{Ric}-\frac12\mathcal Rg.
\]
Contract the differential Bianchi identity to prove
$\nabla^aG_{ab}=0$.  For a symmetric stress--energy tensor $T$, state the
Einstein equation with cosmological constant,
\[
G+\Lambda g=\frac{8\pi G_{\mathrm N}}{c^4}T,
\]
and deduce that it requires $\nabla^aT_{ab}=0$.
In the static weak-field, slow-motion regime, put
$g_{tt}=-(1+2\Phi/c^2)c^2$ and use as theorem input the linearized
curvature formula.  Show that the time-time equation reduces to
\[
\nabla^2\Phi=4\pi G_{\mathrm N}\rho
\]
when $T_{tt}\simeq\rho c^4$ and $|\Lambda|$ is negligible.
\end{exercise}

\begin{exercise}[Schwarzschild geometry and gravitational redshift]
Let
\[
r_s=\frac{2G_{\mathrm N}M}{c^2},
\qquad
f(r)=1-\frac{r_s}{r},
\]
and on $(r_s,\infty)\times\mathbb R\times S^2$ consider
\[
g=-f(r)c^2dt^2+f(r)^{-1}dr^2+r^2g_{S^2}.
\]
Use as theorem input Birkhoff's theorem: every spherically symmetric vacuum
solution with $\Lambda=0$ is locally isometric to a member of this family.
Compute its Ricci tensor and verify the vacuum Einstein equation.  Compute
the invariant
\[
R_{abcd}R^{abcd}
=\frac{48G_{\mathrm N}^2M^2}{c^4r^6}
\]
and use it to distinguish the curvature singularity at $r=0$ from the
coordinate singularity at $r=r_s$.

For the timelike Killing field $\partial_t$, prove that
$-g(k,\partial_t)$ is conserved along a null geodesic with tangent $k$.
Compare it with the frequency measured by static observers to derive
\[
\frac{\nu_{\rm rec}}{\nu_{\rm emit}}
=\sqrt{\frac{f(r_{\rm emit})}{f(r_{\rm rec})}}.
\]
In particular, for reception at infinity predict
$\nu_\infty/\nu_r=\sqrt{1-r_s/r}$.  State the static-emitter,
static-receiver, geometric-optics, negligible-backreaction, and
spectral-resolution assumptions under which a clock or spectral-line
measurement tests the formula.
\end{exercise}

\begin{exercise}[Weierstrass elliptic functions and Schwarzschild light rays]
Restrict a Schwarzschild null geodesic to an equatorial great circle.  Let
$\mathcal E=f(r)c^2\dot t$ and $L=r^2\dot\phi$ be its conserved energy and
angular momentum per affine parameter, put $b=cL/\mathcal E$, and set
$u=1/r$.  Derive
\[
\left(\frac{du}{d\phi}\right)^2
=\frac1{b^2}-u^2+r_su^3,
\qquad
\frac{d^2u}{d\phi^2}+u=\frac32r_su^2.
\]
Define the Weierstrass elliptic function by
\[
(\wp')^2=4\wp^3-g_2\wp-g_3,
\qquad
\wp(z)=z^{-2}+O(z^2),
\]
and prove that every nonradial orbit is obtained by analytic continuation
of
\[
u(\phi)=\frac{1/3+4\wp(\phi-\phi_0;g_2,g_3)}{r_s},
\qquad
g_2=\frac1{12},
\qquad
g_3=\frac1{216}-\frac{r_s^2}{16b^2}.
\]
Show that its nontrivial finite-impact-parameter discriminant degeneration
occurs at
\[
r_{\rm ph}=\frac32r_s=\frac{3G_{\mathrm N}M}{c^2},
\qquad
b_c=\frac{3\sqrt3G_{\mathrm N}M}{c^2},
\]
the unstable circular photon orbit and critical capture impact parameter.
Expand the orbit equation for $r_s/b\ll1$ to derive the deflection angle
\[
\delta\phi=\frac{4G_{\mathrm N}M}{bc^2}
+O\!\left(\frac{G_{\mathrm N}^2M^2}{b^2c^4}\right).
\]
Predict the far-field shadow angular radius $b_c/D$ for an observer at
distance $D$.  State the vacuum, nonrotating, geometric-optics,
thin-lens, far-observer, and source-model assumptions separately from the
bending and shadow predictions.
\end{exercise}

\begin{exercise}[Regge--Wheeler equations and confluent-Heun scattering]
Define the tortoise coordinate by
\[
r_*=r+r_s\log\!\left(\frac r{r_s}-1\right).
\]
For integer spin $s\in\{0,1,2\}$ and $\ell\geq s$, use as theorem input
the scalar, electromagnetic, and axial gravitational separation theorem:
after expansion in modes with time dependence $e^{-i\omega t}$ and the
corresponding spherical harmonics,
the radial master field satisfies
\[
\frac{d^2\psi}{dr_*^2}
+\left(\frac{\omega^2}{c^2}-V_{\ell,s}(r)\right)\psi=0,
\]
\[
V_{\ell,s}(r)
=f(r)\left(\frac{\ell(\ell+1)}{r^2}
+(1-s^2)\frac{r_s}{r^3}\right).
\]
Show that $r=r_s$ and $r=\infty$ are respectively regular and irregular
singular endpoints.  After putting $z=r/r_s$ and factoring the ingoing or
outgoing exponential and power behavior, transform the radial equation to
\[
y''+\left(\alpha+\frac{\beta+1}{z}
+\frac{\gamma+1}{z-1}\right)y'
+\left(\frac\mu z+\frac\nu{z-1}\right)y=0.
\]
Define its locally normalized solutions to be the confluent Heun functions
and determine $\alpha,\beta,\gamma,\mu,\nu$ from $\omega r_s/c$,
$\ell$, and $s$ for the chosen asymptotic factors.

Define the horizon-ingoing and infinity-outgoing solutions by
\[
\psi_{\rm in}\sim e^{-i\omega r_*/c}\quad(r_*\to-\infty),
\qquad
\psi_{\rm out}\sim e^{+i\omega r_*/c}\quad(r_*\to+\infty),
\]
and define the connection determinant
\[
D_{\ell,s}(\omega)
=W(\psi_{\rm in},\psi_{\rm out}).
\]
Use constancy of the Wronskian to express reflection and transmission
amplitudes as confluent-Heun connection coefficients and prove flux
conservation for real $\omega$.
\end{exercise}

\begin{exercise}[Black-hole resonances, photon spheres, and ringdown]
Define a Schwarzschild quasinormal frequency to be a zero of the
confluent-Heun connection determinant $D_{\ell,s}(\omega)$ in the lower
half-plane.  Show that it is equivalently a mode ingoing at the horizon and
outgoing at infinity, and that its time dependence is a damped oscillation.

For $\ell\to\infty$, locate the maximum of the Regge--Wheeler potential at
the photon sphere.  Apply the leading parabolic-barrier WKB connection
formula, with the Langer replacement
$\ell(\ell+1)\sim(\ell+\tfrac12)^2$, to derive
\[
\omega_{\ell n}
=\Omega_{\rm ph}\left(\ell+\frac12\right)
-i\lambda_{\rm ph}\left(n+\frac12\right)
+O(\Omega_{\rm ph}\ell^{-1}),
\]
\[
\Omega_{\rm ph}=\lambda_{\rm ph}
=\frac{c^3}{3\sqrt3G_{\mathrm N}M},
\qquad n=0,1,2,\ldots.
\]
Relate $\Omega_{\rm ph}$ to the critical impact parameter by
$\Omega_{\rm ph}=c/b_c$.  If the source has cosmological redshift
$z_{\rm cos}$ and angular-diameter distance $D_A$, predict the observed
ringdown frequency, e-folding time, and shadow angular radius
\[
f_{\ell n}^{\rm obs}
=\frac{\operatorname{Re}\omega_{\ell n}}{2\pi(1+z_{\rm cos})},
\qquad
\tau_{\ell n}^{\rm obs}
=\frac{1+z_{\rm cos}}{|\operatorname{Im}\omega_{\ell n}|},
\qquad
\theta_{\rm sh}=\frac{b_c}{D_A},
\]
all from the same photon-sphere geometry.  State the isolated, nonrotating,
asymptotically flat,
linear-perturbation, eikonal, detector-frame, cosmological-redshift, and
mode-identification assumptions under which shadow and ringdown
measurements test these linked predictions.
\end{exercise}

\section{Algebraic Curves}

\begin{exercise}[Pendulum periods on the Legendre curve]
On $T^*S^1$ consider
\[
H(\theta,p)=\frac{p^2}{2M\ell^2}+Mg\ell(1-\cos\theta).
\]
For an oscillation of amplitude $0<\theta_0<\pi$, set
$k=\sin(\theta_0/2)$ and $m=k^2$.  Consider the complex plane curve
\[
C_m:\quad y^2=x(x-1)(x-m).
\]
Use energy conservation and the real period of $dx/y$ to prove
\[
T(\theta_0)=4\sqrt{\frac{\ell}{g}}\,K(k).
\]
Apply the generalized binomial theorem with $a=\frac12$.  For
\[
I_n=\int_0^{\pi/2}\sin^{2n}\phi\,d\phi,
\]
use integration by parts to prove
\[
I_0=\frac{\pi}{2},
\qquad
I_n=\frac{2n-1}{2n}I_{n-1}\quad(n\geq1),
\]
and obtain
\[
K(k)=\frac{\pi}{2}\sum_{n=0}^{\infty}
\left(\frac{(1/2)_n}{n!}\right)^2k^{2n},
\qquad |k|<1.
\]
Deduce
\[
T(\theta_0)=2\pi\sqrt{\frac{\ell}{g}}
\left(1+\frac{\theta_0^2}{16}+O(\theta_0^4)\right).
\]
Deduce the experimentally testable increase of period with amplitude and
the logarithmic divergence as $\theta_0\uparrow\pi$.
\end{exercise}

\begin{exercise}[Affine schemes and the Legendre family]
Replace $m$ by an indeterminate and form the finitely generated
$\mathbb C[m]$-algebra
\[
A=\mathbb C[m,x,y]\big/(y^2-x(x-1)(x-m)).
\]

For a commutative ring $R$, an ideal
$\mathfrak p\subsetneq R$ is \emph{prime} when $ab\in\mathfrak p$ implies
$a\in\mathfrak p$ or $b\in\mathfrak p$.  Define
\[
\operatorname{Spec}R=\{\mathfrak p:\mathfrak p\text{ is prime}\},
\qquad
V(I)=\{\mathfrak p:I\subseteq\mathfrak p\},
\qquad
D(f)=\{\mathfrak p:f\notin\mathfrak p\}.
\]
Prove that the sets $V(I)$ are the closed sets of a topology and that the
principal open sets $D(f)$ are closed under finite intersections and form
a basis for the topology.

For $f\in R$, define the localization $R_f$ by the universal property that
$f$ becomes invertible and every homomorphism from $R$ in which $f$ is
invertible factors uniquely through $R_f$.  A \emph{presheaf} of rings on
a space $X$ assigns a ring $\mathcal F(U)$ to each open set and compatible
restriction maps.  It is a \emph{sheaf} when compatible local sections on
an open cover glue uniquely.  Set
\[
\mathcal O_{\operatorname{Spec}R}(D(f))=R_f
\]
with the localization restriction maps.  Prove the sheaf gluing condition
on principal-open covers, extend this prescription uniquely to a sheaf on
$\operatorname{Spec}R$, and prove
\[
R\cong\Gamma(\operatorname{Spec}R,\mathcal O_{\operatorname{Spec}R}).
\]

For a sheaf of rings $\mathcal F$ and $p\in X$, define its stalk by
\[
\mathcal F_p=\varinjlim_{p\in U}\mathcal F(U).
\]
A commutative ring is \emph{local} when it has a unique maximal ideal.  A
\emph{locally ringed space} is a space $X$ with a sheaf of commutative
rings $\mathcal O_X$ whose stalks are local rings.  Prove
\[
\mathcal O_{\operatorname{Spec}R,\mathfrak p}\cong R_{\mathfrak p},
\qquad
\mathfrak m_{\mathfrak p}=\mathfrak pR_{\mathfrak p}.
\]
For a continuous map $f:X\to Y$, set
$(f_*\mathcal O_X)(U)=\mathcal O_X(f^{-1}(U))$.  A morphism of locally
ringed spaces
\[
(f,f^\#):(X,\mathcal O_X)\longrightarrow(Y,\mathcal O_Y)
\]
consists of $f$ and a morphism of sheaves
$f^\#:\mathcal O_Y\to f_*\mathcal O_X$ whose induced homomorphisms on
stalks are local.  Here a morphism of sheaves of rings is a family of ring
homomorphisms on open sets commuting with restriction maps, and a
homomorphism of local rings is \emph{local} when the inverse image of the
maximal ideal is the maximal ideal.

An \emph{affine scheme} is a locally ringed space isomorphic to
$(\operatorname{Spec}R,\mathcal O_{\operatorname{Spec}R})$.  A
\emph{scheme} is a locally ringed space admitting an open cover by affine
schemes, and a \emph{scheme morphism} is a morphism of locally ringed
spaces.  A \emph{complex scheme} is a scheme equipped with a morphism to
$\operatorname{Spec}\mathbb C$, and its morphisms are required to commute
with these structure morphisms.  It is \emph{of finite type} when it
admits a finite affine open cover by spectra of finitely generated
commutative $\mathbb C$-algebras.

For a homomorphism $\varphi:R\to S$, prove that inverse image of prime
ideals and localization induce a scheme morphism
\[
\varphi^*:\operatorname{Spec}S\longrightarrow\operatorname{Spec}R
\]
and establish the affine anti-equivalence
\[
\operatorname{Hom}_{\rm Sch}(\operatorname{Spec}S,\operatorname{Spec}R)
\cong\operatorname{Hom}_{\rm Ring}(R,S).
\]
The inclusion $\mathbb C[m]\to A$ therefore defines the Legendre-family
morphism
\[
\pi:\mathcal E_{\rm aff}=\operatorname{Spec}A
\longrightarrow\mathbb A^1_m=\operatorname{Spec}\mathbb C[m].
\]
Prove that both schemes are complex and of finite type and that $\pi$
commutes with their structure morphisms to $\operatorname{Spec}\mathbb C$.
For $m_0\in\mathbb C$, define its algebraic fiber by
\[
\mathcal E_{m_0}^{\rm aff}
=\operatorname{Spec}\!\left(
A\otimes_{\mathbb C[m]}\mathbb C[m]/(m-m_0)\right)
\]
Use as theorem input the weak Nullstellensatz~\cite{hartshorne}: every maximal ideal of
$\mathbb C[z_1,\ldots,z_n]$ is the kernel of evaluation at a unique point
of $\mathbb C^n$.  Identify the closed points of the fiber with
\[
y^2=x(x-1)(x-m_0).
\]
Show that replacing an ideal by its radical preserves the closed points but
can change the structure sheaf.  Compare
$\operatorname{Spec}\mathbb C[t]/(t)$ with
$\operatorname{Spec}\mathbb C[t]/(t^2)$, identify the nonzero nilpotent
class in the latter ring, and compute its square.
\end{exercise}

\begin{exercise}[Local rings, tangent spaces, and singular fibers]
For $p\in X=\operatorname{Spec}R$ corresponding to $\mathfrak p$, define
the residue field, cotangent space, and tangent space by
\[
\kappa(p)=R_{\mathfrak p}/\mathfrak pR_{\mathfrak p},
\qquad
T_p^*X=\mathfrak pR_{\mathfrak p}/(\mathfrak pR_{\mathfrak p})^2,
\qquad
T_pX=\operatorname{Hom}_{\kappa(p)}(T_p^*X,\kappa(p)).
\]
Define the Krull dimension of a ring as the supremum of the lengths of
strict chains of prime ideals and define the local dimension of $X$ at
$p$ as $\dim R_{\mathfrak p}$.

For a $\mathbb C$-algebra $R$, define the module of K\"ahler
differentials $\Omega^1_{R/\mathbb C}$ by a derivation
$d:R\to\Omega^1_{R/\mathbb C}$ universal among $\mathbb C$-linear
derivations from $R$ to $R$-modules.  If
\[
R=\mathbb C[x_1,\ldots,x_n]/(f_1,\ldots,f_r),
\]
derive the conormal sequence and prove at a closed point $p$ that
\[
T_pX
=\{v\in\mathbb C^n:df_j(p)(v)=0\text{ for }1\leq j\leq r\}.
\]
Use as theorem input the Jacobian criterion over
$\mathbb C$~\cite{hartshorne}: $p$ is
smooth exactly when
$\dim_{\mathbb C}T_pX=\dim\mathcal O_{X,p}$.

Apply the criterion to the fibers $\mathcal E_{m}^{\rm aff}$ constructed
above.  Prove that they are smooth for $m\notin\{0,1\}$ and that the
fibers at $m=0$ and $m=1$ have singular points $(0,0)$ and $(1,0)$,
respectively.  Compute the local dimension at every closed point of these
fibers and compare it with the tangent-space dimension at the two singular
points.  Define the tangent cone of a plane curve at a point by the
lowest nonzero homogeneous part of a translated local equation and prove
that both singular fibers have two distinct tangent directions over
$\mathbb C$.

Determine the smooth and singular points of
\[
y^2-x^3,\qquad y^2-x^2(x+1),\qquad xy,
\]
and distinguish the cusp, node, and crossing by their local rings and
tangent cones.  For the morphism
\[
\operatorname{Spec}\mathbb C[u,t]/(t^2-u)
\longrightarrow\operatorname{Spec}\mathbb C[u],
\]
show that every fiber over $u\ne0$ consists of two reduced points, whereas
the fiber over $u=0$ is $\operatorname{Spec}\mathbb C[t]/(t^2)$, and
compute its tangent space and length-two local algebra.
\end{exercise}

\begin{exercise}[Projective curves and the Legendre compactification]
Let $R=\bigoplus_{n\geq0}R_n$ be a finitely generated graded
$\mathbb C$-algebra generated by $R_1$, and let
$R_+=\bigoplus_{n>0}R_n$.  Define
\[
\operatorname{Proj}R
=\{\mathfrak p\subset R:\mathfrak p\text{ is homogeneous prime and }
R_+\not\subseteq\mathfrak p\}.
\]
For homogeneous $f$ of positive degree, define
\[
D_+(f)=\{\mathfrak p:f\notin\mathfrak p\},
\qquad
\mathcal O_{\operatorname{Proj}R}(D_+(f))=(R_f)_0.
\]
Prove that these affine charts glue to a scheme.  For a finite-dimensional
complex vector space $V$, prove
\[
\mathbb P(V)=\operatorname{Proj}(\operatorname{Sym}V^\vee)
\]
and recover its interpretation as the space of complex lines in $V$.

For a homogeneous ideal $I\subset R$, define the projective subscheme
$\operatorname{Proj}(R/I)$.  Show that homogenization of an affine ideal
gives its projective closure and identify the points added at infinity for
a parabola and a hyperbola.  Call a complex scheme \emph{projective} when
it is isomorphic to a closed subscheme of some $\mathbb P(V)$.  Define a
\emph{projective algebraic curve} to be a projective complex scheme of
finite type whose irreducible components all have Krull dimension one.
For a projective plane curve $C=V(F)$
defined by a homogeneous polynomial, define $\deg C=\deg F$.

For $m\in\mathbb C\setminus\{0,1\}$, construct the projective closure
\[
E_m:\quad Y^2Z=X(X-Z)(X-mZ)
\subset\mathbb P^2.
\]
Prove that $E_m$ is a projective algebraic curve of degree three.  Use the
Jacobian criterion on its affine charts to prove that $E_m$ is nonsingular
and that it has the unique point
$O=[0:1:0]$ over the line at infinity.  Prove that the rational function
$x=X/Z$ extends to a morphism
\[
\pi_m:E_m\longrightarrow\mathbb P^1,
\qquad
\pi_m([X:Y:Z])=[X:Z]\quad([X:Y:Z]\ne O),
\qquad
\pi_m(O)=[1:0].
\]
\end{exercise}

\begin{exercise}[Intersection multiplicity, resultants, and discriminants]
Let $C=V(F)$ and $D=V(G)$ be projective plane curves with no common
component.  At $p\in C\cap D$, let $F_p,G_p\in\mathcal O_{\mathbb P^2,p}$
be local equations and define their local intersection
multiplicity by
\[
I_p(C,D)
=\dim_{\mathbb C}
\mathcal O_{\mathbb P^2,p}/(F_p,G_p).
\]
Prove that this length is finite, is invariant under multiplication of
$F$ and $G$ by local units, and equals one exactly for a transverse
intersection of smooth branches.

For a two-dimensional vector space $W$ and binary forms
$F\in\operatorname{Sym}^mW^\vee$ and
$G\in\operatorname{Sym}^nW^\vee$, define their resultant through the
determinant on the highest exterior powers of
\[
\operatorname{Sym}^{n-1}W^\vee\oplus
\operatorname{Sym}^{m-1}W^\vee
\longrightarrow
\operatorname{Sym}^{m+n-1}W^\vee,
\qquad
(A,B)\longmapsto AF+BG.
\]
Prove that it vanishes exactly when $F$ and $G$ have a common projective
zero.  Define the discriminant of a one-variable polynomial by the
resultant with its derivative and prove that
\[
\operatorname{Disc}(t^3+at+b)=-4a^3-27b^2.
\]
Compute also
\[
\operatorname{Disc}\bigl(x(x-1)(x-m)\bigr)
=m^2(1-m)^2
\]
and recover algebraically the two singular Legendre fibers found by the
Jacobian criterion.

Use as theorem input B\'ezout's theorem for plane
curves~\cite{hartshorne}:
\[
\sum_{p\in C\cap D}I_p(C,D)=\deg C\deg D.
\]
Derive the possible intersection patterns of a line with a smooth cubic
and prove that tangency is detected by a repeated root and hence by the
corresponding discriminant locus.
\end{exercise}

\begin{exercise}[Caustics and Airy--Pearcey diffraction]
For a monochromatic scalar wave with wave number $k=2\pi/\lambda$, define
the fold and cusp canonical fields as oscillatory integrals
\[
\mathcal A_{\rm fold}(a;k)
=\int_{\mathbb R}^{\rm osc}
e^{ik(t^3/3-at)}\,dt,
\]
\[
\mathcal A_{\rm cusp}(a,b;k)
=\int_{\mathbb R}^{\rm osc}
e^{ik(t^4/4+at^2/2+bt)}\,dt.
\]
For a phase function, call the image of its degenerate stationary points
in parameter space its \emph{caustic}.  Prove that the fold caustic is
$a=0$ and that the cusp caustic is the discriminant curve
\[
4a^3+27b^2=0
\]
of the stationary cubic $t^3+at+b$.

Using the Airy function, prove
\[
\mathcal A_{\rm fold}(a;k)
=2\pi k^{-1/3}\operatorname{Ai}(-k^{2/3}a).
\]
Define the Pearcey function by
\[
\operatorname{Pe}(A,B)
=\int_{\mathbb R}^{\rm osc}
e^{i(t^4/4+At^2/2+Bt)}\,dt
\]
and derive
\[
\mathcal A_{\rm cusp}(a,b;k)
=k^{-1/4}\operatorname{Pe}(k^{1/2}a,k^{3/4}b).
\]
Deduce that fold dark fringes occur at
$a=a_nk^{-2/3}$, where $\operatorname{Ai}(-a_n)=0$, and that the two
cusp control scales are $k^{-1/2}$ and $k^{-3/4}$.  Predict the normalized
intensities $|\operatorname{Ai}|^2$ and $|\operatorname{Pe}|^2$ near fold
and cusp caustics.  State the scalar-wave, short-wavelength, canonical-form,
finite-aperture, and detector-resolution assumptions under which measured
fringe locations test the discriminant geometry and its wavelength
scaling.
\end{exercise}

\begin{exercise}[Branched covers and the Riemann--Hurwitz formula]
For a compact connected oriented surface $S$, define its genus $g(S)$ by
\[
\chi(S)=2-2g(S).
\]
Let $f:C\to D$ be a nonconstant holomorphic map between compact connected
Riemann surfaces.  At $p\in C$, choose local parameters $z$ at $p$ and $w$
at $f(p)$.  Define the \emph{ramification index} $e_p\geq1$ by
\[
w\circ f=z^{e_p}u(z),
\qquad u(0)\ne0,
\]
and define the ramification divisor
\[
R_f=\sum_{p\in C}(e_p-1)[p].
\]
Show that $e_p$ is independent of the local parameters and that the number
of points in a nearby regular fiber, counted with multiplicity, is a
constant $d$, the degree of $f$.

Triangulate $D$ so that every branch value is a vertex, lift the
triangulation to $C$, and compare vertices, edges, and faces to prove the
Riemann--Hurwitz formula
\[
2g(C)-2=d\bigl(2g(D)-2\bigr)+\deg R_f.
\]
For a square-free polynomial $q\in\mathbb C[x]$ of degree $2g+1$ or
$2g+2$, let $C_q$ be the nonsingular projective curve obtained from
$y^2=q(x)$.  Show that the projection $x:C_q\to\mathbb P^1$ has degree
two, determine its branch points including the point or points over
infinity, and deduce that $C_q$ has genus $g$.
\end{exercise}

\begin{exercise}[Legendre ramification and holomorphic differentials]
For $m\in\mathbb C\setminus\{0,1\}$, let
$\pi_m:E_m\to\mathbb P^1$ be the projective Legendre curve and morphism
constructed above.  Prove that it has degree two and is ramified exactly
over $0,1,m,\infty$.  Use Riemann--Hurwitz to obtain
\[
\chi(E_m)=2\chi(\mathbb P^1)-4=0
\]
and hence that $E_m$ has genus one.

On the affine chart $Z=1$, put
\[
\eta_m=\frac{dx}{y},
\qquad
y^2=x(x-1)(x-m).
\]
Check at each ramification point and at infinity that $\eta_m$ extends to a
nowhere-vanishing holomorphic one-form on $E_m$.  For $0<m<1$, let
$\gamma_m$ be the closed real oval over $[0,m]$.  With $m=k^2$, prove
that its orientation may be chosen so that
\[
\oint_{\gamma_m}\eta_m
=4K(k),
\qquad
T(\theta_0)=\sqrt{\frac{\ell}{g}}
\oint_{\gamma_m}\eta_m.
\]
\end{exercise}

\begin{exercise}[Riemann bilinear identity and period lattices]
For a compact genus-one Riemann surface $C$ and a nowhere-vanishing
holomorphic one-form $\eta$, define
the period homomorphism and period subgroup by
\[
\operatorname{per}_\eta:H_1(C;\mathbb Z)\longrightarrow\mathbb C,
\qquad
[\gamma]\longmapsto\int_\gamma\eta,
\qquad
\Lambda_\eta=\operatorname{im}(\operatorname{per}_\eta).
\]

Let $\alpha,\beta$ be oriented simple closed curves meeting once
positively and generating the first homology of $C$, and put
\[
A=\int_\alpha\eta,
\qquad
B=\int_\beta\eta.
\]
Cut $C$ along $\alpha\cup\beta$, choose a primitive of $\eta$ on the
resulting polygon, and apply Stokes' theorem to its boundary to prove the
genus-one Riemann bilinear identity
\[
\frac{i}{2}\int_C\eta\wedge\overline\eta
=\operatorname{Im}(\overline A B)>0.
\]
Deduce that $A$ and $B$ are linearly independent over $\mathbb R$ and
that
\[
\Lambda_\eta=A\mathbb Z+B\mathbb Z
\]
is a discrete subgroup of rank two.  Apply this to $(E_m,\eta_m)$ and
identify the real generator $A=4K(\sqrt m)$ for $0<m<1$.
\end{exercise}

\begin{exercise}[Abel maps and elliptic uniformization]
Let $\Lambda_{\eta_m}$ be the period lattice obtained above.  For a fixed
$p_0\in E_m$, define
\[
\mathcal A_{p_0}(p)
=\left[\int_{p_0}^{p}\eta_m\right]\in\mathbb C/\Lambda_{\eta_m}.
\]
Show that this is independent of the path modulo periods, is holomorphic,
and has nowhere-vanishing differential.  Show that it is therefore a
finite covering.  Identify its induced map on first homology with
$\operatorname{per}_{\eta_m}:H_1(E_m;\mathbb Z)\to\Lambda_{\eta_m}$ and
use the period-pairing calculation to prove that the covering
has one sheet.  Conclude that it is a biholomorphism
\[
E_m\cong\mathbb C/\Lambda_{\eta_m}.
\]
Under this identification, show that translation on the complex torus
defines an intrinsic group law on $E_m$ with identity $p_0$.
\end{exercise}

\begin{exercise}[Picard--Fuchs equations, monodromy, and pendulum periods]
Let $U=\mathbb C\setminus\{0,1\}$.  Along a path in $U$, transport a closed
cycle continuously through the family $E_m$ and call the resulting
identification of the groups $H_1(E_m;\mathbb Z)$ \emph{Gauss--Manin
transport}.  Transport around a loop defines its \emph{monodromy}.

For a transported cycle $\gamma_m$, put
\[
\Pi(m)=\oint_{\gamma_m}\eta_m.
\]
Differentiate under the integral sign and reduce differentials modulo exact
differentials on $E_m$ to prove the Picard--Fuchs equation
\[
m(1-m)\Pi''(m)+(1-2m)\Pi'(m)-\frac14\Pi(m)=0.
\]
Identify this with the Gauss hypergeometric equation.  Prove that the
projective cubic has discriminant
\[
\Delta(m)=16m^2(1-m)^2
\]
in the displayed normalization and that the family has singular fibers at
$m=0,1,\infty$.

Show that local monodromy at $m=1$ fixes one analytic solution and sends a
logarithmic companion to itself plus a multiple of the analytic solution.
For the real oval, use $\Pi(m)=4K(\sqrt m)$ and the hypergeometric connection
formula to obtain
\[
K(\sqrt m)
=\log\frac4{\sqrt{1-m}}
+O\bigl((1-m)|\log(1-m)|\bigr)
\qquad(m\uparrow1).
\]
For the pendulum of length $\ell$, where
$m=\sin^2(\theta_0/2)$, deduce
\[
T(\theta_0)
=4\sqrt{\frac{\ell}{g}}\log\frac4{\cos(\theta_0/2)}
+O\!\left(\sqrt{\frac{\ell}{g}}\,
\cos^2\frac{\theta_0}{2}
\left|\log\cos^2\frac{\theta_0}{2}\right|
\right).
\]
State how a period-versus-amplitude measurement tests the location and
logarithmic type of the singular fiber.
\end{exercise}

\begin{exercise}[Divisors, line bundles, and first Chern classes]
For a scheme $X$, an \emph{algebraic line bundle} is an
$\mathcal O_X$-module locally isomorphic to $\mathcal O_X$.  For
$X=\operatorname{Proj}R$ and $n\in\mathbb Z$, define the graded shift by
$R(n)_d=R_{n+d}$ and define the twisting sheaf
$\mathcal O_X(n)$ on the standard affine charts by
\[
\mathcal O_X(n)(D_+(f))=(R(n)_f)_0.
\]
For projective space, prove intrinsically
for $n\geq0$ that
\[
\Gamma(\mathbb P(V),\mathcal O(n))
\cong\operatorname{Sym}^nV^\vee,
\]
and that $\mathcal O(-1)$ is the tautological line bundle whose fiber over
a line $\ell\subset V$ is $\ell$.  For a smooth projective plane curve
$C$, define $\mathcal O_C(n)$ by restriction.
Use as theorem input~\cite{hartshorne,forster} that an algebraic line bundle on a smooth complex
projective curve determines a holomorphic line bundle and that this
correspondence preserves tensor products and global sections.

On a compact Riemann surface $C$, a \emph{divisor} is a finite formal sum
\[
D=\sum_{p\in C}n_p[p],
\qquad n_p\in\mathbb Z,
\qquad
\deg D=\sum_p n_p.
\]
For a nonzero meromorphic function $f$, define
$\operatorname{div}(f)=\sum_p\operatorname{ord}_p(f)[p]$.  Use the argument
principle to prove
\[
\deg\operatorname{div}(f)=0.
\]

A \emph{holomorphic line bundle} on $C$ is a complex line bundle with local
product maps whose transition functions are nowhere-vanishing holomorphic
functions $g_{ij}$ satisfying $g_{ij}g_{jk}=g_{ik}$.  For a divisor
$D=\sum_pn_p[p]$, define $\mathcal O(D)$ to be the sheaf of meromorphic
functions $h$ satisfying $\operatorname{div}(h)+D\geq0$ on each open set.
Show that this sheaf is the sheaf of holomorphic sections of a holomorphic
line bundle.
Two divisors are \emph{linearly equivalent} when their difference is
principal, namely the divisor of a nonzero meromorphic function.  Prove
\[
\mathcal O(D)\cong\mathcal O(D')
\quad\Longleftrightarrow\quad
D\text{ and }D'\text{ are linearly equivalent}.
\]

For a nonzero meromorphic section $s$, define $\operatorname{div}(s)$ from
its local representatives.  Use as theorem input the meromorphic-section
theorem~\cite{forster}: every holomorphic line bundle on a compact Riemann surface has a
nonzero meromorphic section.  Prove that
\[
\deg L:=\deg\operatorname{div}(s)
\]
is independent of $s$, that tensor products add degrees, and that
$L\cong\mathcal O(\operatorname{div}(s))$.
Use B\'ezout's theorem to prove
\[
\deg\mathcal O_C(1)=\deg C
\]
for every smooth projective plane curve $C$.

For a Hermitian connection $\nabla$ on $L$, define
\[
c_1(L)=\left[\frac{i}{2\pi}F_\nabla\right]\in H_{\mathrm{dR}}^2(C;\mathbb R).
\]
Show that this class is independent of the Hermitian connection and is
natural under pullback.  Compare connection forms on overlaps and use
Stokes' theorem around the zeros and poles of a meromorphic section to
prove
\[
\int_C c_1(L)=\deg L.
\]
\end{exercise}

\begin{exercise}[\v{C}ech cohomology and exact sequences]
Let $\mathcal F$ be a sheaf of complex vector spaces on a compact Riemann
surface $C$, and let $\mathfrak U=(U_i)_{i\in I}$ be a finite ordered open
cover.  Define the alternating \v{C}ech cochains and coboundary by
\[
\check C^q(\mathfrak U,\mathcal F)
=\prod_{i_0<\cdots<i_q}
\mathcal F(U_{i_0}\cap\cdots\cap U_{i_q}),
\]
\[
(\delta s)_{i_0\ldots i_{q+1}}
=\sum_{r=0}^{q+1}(-1)^r
s_{i_0\ldots\widehat{i_r}\ldots i_{q+1}}
\big|_{U_{i_0}\cap\cdots\cap U_{i_{q+1}}}.
\]
Prove $\delta^2=0$ and define
$\check H^q(\mathfrak U,\mathcal F)=\ker\delta/\operatorname{im}\delta$.
Show directly from the sheaf gluing axiom that
\[
\check H^0(\mathfrak U,\mathcal F)=\mathcal F(C).
\]

Use as theorem input~\cite{forster} that a finite cover whose nonempty intersections are
Stein computes the cohomology $H^q(C,\mathcal F)$ of a coherent analytic
sheaf, that coherent sheaves on $C$ have finite-dimensional cohomology and
vanish in degrees $q>1$, and that a short exact sequence of coherent sheaves
induces a long exact cohomology sequence.  For a holomorphic line bundle
$L$, a point $p\in C$, and its inclusion $i_p:\{p\}\to C$, prove that
evaluation at $p$ gives an exact sequence
\[
0\longrightarrow L(-p)\longrightarrow L
\longrightarrow(i_p)_*L_p\longrightarrow0.
\]
Writing $h^q(L)=\dim H^q(C,L)$ and
$\chi(L)=h^0(L)-h^1(L)$, use its long exact sequence to prove
\[
\chi(L)=\chi(L(-p))+1.
\]
Iterate this identity through the positive and negative parts of a divisor
and use the meromorphic-section theorem to deduce
\[
\chi(L)=\deg L+\chi(\mathcal O_C)
\]
for every holomorphic line bundle on $C$.
\end{exercise}

\begin{exercise}[Serre duality and the Riemann--Roch theorem]
Define the canonical line bundle $K_C$ to be the holomorphic cotangent
bundle of a compact connected Riemann surface $C$.  Let $g$ be its
topological genus and use as theorem input~\cite{forster} the analytic genus identity
$\dim H^0(C,K_C)=g$.
Use as theorem input Serre duality~\cite{forster} in the form of a natural perfect pairing
\[
H^1(C,L)\otimes H^0(C,K_C\otimes L^{-1})
\longrightarrow\mathbb C,
\qquad
H^1(C,L)\cong H^0(C,K_C\otimes L^{-1})^\vee.
\]
Show that every holomorphic function on $C$ is constant and deduce
\[
\chi(\mathcal O_C)=1-g.
\]
Combine this with the \v{C}ech calculation above to prove the
Riemann--Roch theorem
\[
h^0(L)-h^0(K_C\otimes L^{-1})=\deg L+1-g.

Apply it to $L=K_C$ to prove $\deg K_C=2g-2$.  Deduce that

h^0(L)=\deg L+1-g
\qquad\text{when }\deg L>2g-2,
\]
and that a negative-degree line bundle has no nonzero holomorphic sections.

$h^0(L)=N$.
\end{exercise}

\begin{exercise}[Polarized quantum spaces]
Let $(M,\omega)$ be prequantizable.  A \emph{complex polarization} is an
involutive Lagrangian subbundle $P\subset T_{\mathbb C}M$.  Define its
annihilator and canonical line by
\[
P^\circ=\{\alpha\in T_{\mathbb C}^*M:\alpha|_P=0\},
\qquad
K_P=\Lambda^{\rm top}P^\circ.
\]
A \emph{half-form bundle} is a line bundle $\delta_P$ with
$\delta_P^{\otimes2}\cong K_P$.  Use as theorem input~\cite{woodhouse} the induced partial
connection on $\delta_P$ along $P$ and the half-form pairing.  Define the
corrected polarized quantum space to be the square-integrable sections of
$L\otimes\delta_P$ annihilated by the tensor-product partial connection
along $P$.

For the vertical polarization on $T^*V$, defined as the tangent
distribution to the fibers of $T^*V\to V$, recover $L^2(V)$ and the
operators
\[
\psi(x)\longmapsto \ell(x)\psi(x),
\qquad
\psi(x)\longmapsto-i\hbar\,d\psi_x(v)
\]
for $\ell\in V^*$ and $v\in V$.

A \emph{K\"ahler manifold} is a complex manifold $(M,J)$ with a symplectic
form $\omega$ satisfying
\[
\omega(Ju,Jv)=\omega(u,v),
\qquad
g(u,v)=\omega(u,Jv)
\]
for a Riemannian metric $g$.  For a compact K\"ahler manifold with
$P=T^{0,1}M$, prove
\[
P^\circ=T^{*1,0}M,
\qquad
K_P=K_M:=\Lambda^{\rm top}T^{*1,0}M.
\]
Call a Hermitian holomorphic line bundle \emph{positive} when its first
Chern form satisfies $c_1(L,\nabla)(u,Ju)>0$ for every nonzero tangent
vector $u$.  When the prequantum line bundle is positive and a square root
$K_M^{1/2}$ exists, identify the corrected quantum space with
\[
H^0(M,L\otimes K_M^{1/2}),
\]
where $H^0$ denotes holomorphic sections.  On an elliptic curve, use the
trivialization by $dz$ to show that the half-form correction does not
change the theta-section count.
\end{exercise}

\begin{exercise}[Abel's theorem and theta divisors]
Let $E_\tau=\mathbb C/(\mathbb Z+\tau\mathbb Z)$ and put
\[
\vartheta_1(z\mid\tau)
=\vartheta\!\begin{bmatrix}\frac12\\[1pt]\frac12\end{bmatrix}
(z\mid\tau).
\]
Use the theta product and automorphy laws to prove that
$\vartheta_1$ has one simple zero modulo the period lattice, at the
origin.

For a degree-zero divisor $D=\sum_pn_p[p]$, define its Abel sum by

\mathfrak a(D)=\sum_p n_pp\in E_\tau.
\]
Show that this is independent of the chosen lifts of the points to
$\mathbb C$.  If
\[
D=[p_1]+\cdots+[p_N]-[q_1]-\cdots-[q_N]
\]
and $\mathfrak a(D)=0$, choose lifts with
$\sum p_j-\sum q_j=m+n\tau$ and prove that
\[
f_D(z)=e^{-2\pi inz}
\frac{\prod_{j=1}^N\vartheta_1(z-p_j\mid\tau)}
{\prod_{j=1}^N\vartheta_1(z-q_j\mid\tau)}
\]
is elliptic and satisfies $\operatorname{div}(f_D)=D$.  Conversely, apply
the argument principle to $f'/f$ and $zf'/f$ around a period
parallelogram to prove that a principal divisor has degree zero and Abel

\[
D\text{ is principal}
\quad\Longleftrightarrow\quad
\deg D=0\text{ and }\mathfrak a(D)=0.
\]

Let $\operatorname{Pic}^d(E_\tau)$ denote the isomorphism classes of

E_\tau\longrightarrow\operatorname{Pic}^0(E_\tau),
\qquad
p\longmapsto\mathcal O([p]-[0]),
\]
is an isomorphism of complex groups.  Show that tensoring with
$\mathcal O(N[0])$ identifies $\operatorname{Pic}^0(E_\tau)$ with
$\operatorname{Pic}^N(E_\tau)$.  Identify effective divisors of degree
$N$ with unordered $N$-tuples of theta zeros and show that their Abel sum

\end{exercise}

\begin{exercise}[Theta sections and genus-one Riemann--Roch]
For $a,b\in\R$, $z\in\C$, and $\tau$ in the upper half-plane, define

\vartheta\!\begin{bmatrix}a\\b\end{bmatrix}(z\mid\tau)

1,&L\cong\mathcal O,\\
0,&L\not\cong\mathcal O
\end{cases}

\begin{exercise}[Theta-function Landau spectra]

\begin{exercise}[Floquet operators and Bloch spectral curves]
Let
\[
L=-\frac{d^2}{du^2}+V(u)
\]
with real smooth potential of period $P$.  For $E\in\mathbb C$, let
$\mathcal S_E$ be the two-dimensional solution space of
$L\psi=E\psi$, and define the monodromy automorphism
\[
M(E):\mathcal S_E\longrightarrow\mathcal S_E,
\qquad
\psi(u)\longmapsto\psi(u+P).
\]
Use constancy of the Wronskian to prove $\det M(E)=1$.  Define
\[
\Delta(E)=\operatorname{tr}M(E)
\]
and call $\rho$ a Bloch multiplier when $M(E)\psi=\rho\psi$ for a nonzero
solution.  Prove
\[
\rho+\rho^{-1}=\Delta(E)
\]
Let $H^2([0,P])$ be the space of $L^2$ functions whose first two
distributional derivatives belong to $L^2$, and use the one-dimensional
Sobolev trace theorem~\cite{hormander3} to evaluate a function and its
first derivative at the endpoints.  For $\rho\in\mathbb C^\times$,
define the twisted Sobolev domain
\[
H^2_\rho
=\{\psi\in H^2([0,P]):
\psi(P)=\rho\psi(0),
\psi'(P)=\rho\psi'(0)\}.
\]
Regard
\[
\mathcal L_{\rho,E}:H^2_\rho\longrightarrow L^2([0,P]),
\qquad
\mathcal L_{\rho,E}\psi=(L-E)\psi,
\]
as a bounded operator.  For a bounded operator $T:H_0\to H_1$, define
its cokernel by
\[
\operatorname{coker}T=H_1/\operatorname{im}T.
\]
Use variation of parameters to prove that the image of
$\mathcal L_{\rho,E}$ is closed and that its kernel and cokernel are
finite-dimensional.  Prove
\[
\ker\mathcal L_{\rho,E}
\cong\ker(M(E)-\rho I)
\]
and that $\mathcal L_{\rho,E}$ is invertible exactly when
$M(E)-\rho I$ is invertible.  Deduce that the noninvertibility locus of
the family $\mathcal L_{\rho,E}$ is the Bloch spectral curve
\[
\mathcal B_L:\quad \rho^2-\Delta(E)\rho+1=0.
\]
For $E\in\mathbb R$ and $|\rho|=1$, use Green's identity to prove that
$\mathcal L_{\rho,E}$ is self-adjoint as an operator on $L^2([0,P])$
with domain $H^2_\rho$, and deduce
\[
\dim\ker\mathcal L_{\rho,E}
-\dim\operatorname{coker}\mathcal L_{\rho,E}=0.
\]
Show that its branch points occur at the periodic and antiperiodic energies
$\Delta(E)=\pm2$, and that a real energy supports a propagating Bloch wave
exactly when $|\Delta(E)|\leq2$.
\end{exercise}

\begin{exercise}[Hyperelliptic curves and holomorphic differentials]
Let
\[
C:\quad y^2=\prod_{r=1}^{2g+1}(x-e_r),
\qquad e_r\ne e_s\quad(r\ne s),
\]
denote its nonsingular projective completion.  Use Riemann--Hurwitz to prove
that $C$ has genus $g$.  Use the projective construction and Jacobian
criterion to verify the asserted
nonsingularity.  Analyze the finite branch points and the point over
infinity to show that the distinguished differentials
\[
\eta_j=\frac{x^{j-1}dx}{y},
\qquad 1\leq j\leq g,
\]
are holomorphic.  Compare their vanishing orders at infinity and use
$\dim H^0(C,K_C)=g$ to show that they exhaust the holomorphic
differentials.

Choose a symplectic marking
$(a_1,\ldots,a_g,b_1,\ldots,b_g)$ of $H_1(C;\mathbb Z)$, define the
period maps
\[
\mathsf A(\eta)=\left(\int_{a_1}\eta,\ldots,
\int_{a_g}\eta\right),
\qquad
\mathsf B(\eta)=\left(\int_{b_1}\eta,\ldots,
\int_{b_g}\eta\right),
\]
and prove from the Riemann bilinear identity that $\mathsf A$ is an
isomorphism.  Let $\omega_j$ be the uniquely normalized differentials
satisfying $\int_{a_i}\omega_j=\delta_{ij}$ and define
\[
\Omega_{ij}=\int_{b_i}\omega_j.
\]
Use the bilinear identity to prove
\[
\Omega^{\mathsf T}=\Omega,
\qquad
\operatorname{Im}\Omega>0.
\]
\end{exercise}

\begin{exercise}[Jacobians, Abel--Jacobi maps, and Riemann theta functions]
For a compact Riemann surface $C$, define its Jacobian intrinsically by
\[
\operatorname{Jac}(C)
=H^0(C,K_C)^\vee
\big/
\left\{\eta\mapsto\int_\gamma\eta:
\gamma\in H_1(C;\mathbb Z)\right\}.
\]
Using the symplectic marking and normalized periods above, identify it with
\[
\mathbb C^g/(\mathbb Z^g+\Omega\mathbb Z^g).
\]
Define the Riemann theta function by the normally convergent series
\[
\vartheta(z\mid\Omega)
=\sum_{n\in\mathbb Z^g}
\exp\!\left(\pi i\,n^{\mathsf T}\Omega n
+2\pi i\,n^{\mathsf T}z\right).
\]
Prove its holomorphy and automorphy laws
\[
\vartheta(z+m\mid\Omega)=\vartheta(z\mid\Omega),
\]
\[
\vartheta(z+\Omega n\mid\Omega)
=e^{-\pi i n^{\mathsf T}\Omega n-2\pi i n^{\mathsf T}z}
\vartheta(z\mid\Omega).
\]

Fix $p_0\in C$ and define the Abel--Jacobi map by
\[
\mathcal A_{p_0}(p)
=\left[\eta\longmapsto\int_{p_0}^p\eta\right]
\in\operatorname{Jac}(C).
\]
Prove that it is well defined and extend it additively to divisors.  Use as
theorem input the Abel--Jacobi theorem~\cite{forster}: a degree-zero divisor is principal
exactly when its Abel--Jacobi image vanishes, and every point of the
Jacobian is represented by a degree-zero divisor.  Deduce
\[
\operatorname{Pic}^0(C)\cong\operatorname{Jac}(C).
\]
Use as theorem input Riemann's vanishing theorem~\cite{fay-theta} to identify a translate of
the zero locus of $\vartheta$ with the Abel--Jacobi images of effective
divisors of degree $g-1$.  Recover the genus-one Abel and theta-divisor
statements above when $g=1$.
\end{exercise}

\begin{exercise}[Jacobi elliptic functions and elliptic uniformization]
For $0<k<1$, define $\operatorname{sn}(u;k)$ near $u=0$ by
\[
u=\int_0^{\operatorname{sn}(u;k)}
\frac{dt}{\sqrt{(1-t^2)(1-k^2t^2)}}
\]
and extend it by analytic continuation.  Define
$\operatorname{cn}(u;k)$ and $\operatorname{dn}(u;k)$ by
\[
\operatorname{cn}^2=1-\operatorname{sn}^2,
\qquad
\operatorname{dn}^2=1-k^2\operatorname{sn}^2,
\qquad
\operatorname{cn}(0;k)=\operatorname{dn}(0;k)=1.
\]
Derive
\[
\operatorname{sn}'=\operatorname{cn}\operatorname{dn},
\qquad
\operatorname{cn}'=-\operatorname{sn}\operatorname{dn},
\qquad
\operatorname{dn}'=-k^2\operatorname{sn}\operatorname{cn}.
\]

Put $m=k^2$ and define
\[
x(u)=m\operatorname{sn}^2(u;k),
\qquad
y(u)=m\operatorname{sn}(u;k)\operatorname{cn}(u;k)
\operatorname{dn}(u;k).
\]
Prove
\[
y(u)^2=x(u)(x(u)-1)(x(u)-m),
\qquad
\frac{dx}{y}=2\,du.
\]
Identify the period lattice of $(x(u),y(u))$ and recover the
uniformization of the Legendre curve constructed earlier.
\end{exercise}

\begin{exercise}[Lam\'e theta functions and the spectral curve]
Let $K=K(k)$, $K'=K(\sqrt{1-k^2})$, and
$\tau=iK'/K$.  Using the theta functions with characteristics, put

H(u)=
\vartheta\!\begin{bmatrix}\frac12\\[1pt]\frac12\end{bmatrix}
\left(\frac{u}{2K}\,\middle|\,\tau\right),
\qquad
\Theta(u)=
\vartheta\!\begin{bmatrix}0\\[1pt]\frac12\end{bmatrix}
\left(\frac{u}{2K}\,\middle|\,\tau\right),

For the Lam\'e operator

L_k=-\frac{d^2}{du^2}+2k^2\operatorname{sn}^2(u;k),

use the theta addition identities to verify the Hermite solutions

\psi_\pm(u,a)
=\frac{H(u\pm a)}{\Theta(u)}
\exp\bigl(\mp uZ(a)\bigr),

\[
L_k\psi_\pm(\,\cdot\,,a)=E(a)\psi_\pm(\,\cdot\,,a),
\qquad

From the theta automorphy laws, derive their Bloch multipliers over the

\end{exercise}

\begin{exercise}[Baker--Akhiezer functions and finite-gap reconstruction]
Let
\[
\Sigma:\quad
\mu^2=\prod_{j=0}^{2g}(E-E_j),
\qquad
E_0<E_1<\cdots<E_{2g},
\]
be a nonsingular real hyperelliptic curve with marked point $\infty$, and
let $D$ be an effective divisor of degree $g$.  Let $d\Omega_1$ be the
unique meromorphic differential with zero $a$-periods, no pole away from
$\infty$, and principal part
$d\sqrt E$ at $\infty$.  Put
\[
\Omega_1(p)=\int^p d\Omega_1,
\qquad
U_j=\frac1{2\pi i}\int_{b_j}d\Omega_1,
\]
take the Abel map $\mathcal A$ to be based at $\infty$, and let
$Z\in\operatorname{Jac}(\Sigma)$ be the theta translate determined by
$D$ through Riemann's vanishing theorem.

For $x\in\mathbb R$, let $F_x$ be the flat holomorphic line bundle whose
holonomy around a closed curve $\gamma$ is
\[
\operatorname{hol}_{F_x}(\gamma)
=\exp\left(-x\int_\gamma d\Omega_1\right).
\]
Call $(D,Z)$ \emph{admissible} when, for every $x\in\mathbb R$,
\[
H^0\bigl(\Sigma,K_\Sigma(-D)\otimes F_x^{-1}\bigr)=0,
\qquad
\vartheta(Ux+Z\mid\Omega)\ne0,
\]
and
$\frac{d^2}{dx^2}\log\vartheta(Ux+Z\mid\Omega)$ is real.  Since
$\deg F_x=0$, use Riemann--Roch to prove
\[
\dim H^0\bigl(\Sigma,\mathcal O(D)\otimes F_x\bigr)=1.
\]

For admissible $(D,Z)$, define
\[
\psi(x,p)=
\frac{\vartheta(\mathcal A(p)+Ux+Z\mid\Omega)\vartheta(Z\mid\Omega)}
{\vartheta(\mathcal A(p)+Z\mid\Omega)
 \vartheta(Ux+Z\mid\Omega)}
e^{x\Omega_1(p)}.
\]
Use the theta automorphy laws to prove that the apparent monodromy of the
quotient cancels that of the exponential, so $\psi$ is a function on
$\Sigma\setminus\{\infty\}$ with poles prescribed by $D$.  Prove that
$\psi(x,p)e^{-x\Omega_1(p)}$ is the unique section of
$\mathcal O(D)\otimes F_x$ normalized to have value $1$ at $\infty$ in
the local trivialization used in the theta formula.
Deduce from the Riemann--Roch dimension count that $\psi$ is the unique
normalized Baker--Akhiezer function with these divisor, monodromy, and
asymptotic data.

Use as theorem input the finite-gap reconstruction
theorem~\cite{belokolos}: there is a
constant $c$ determined by the expansion at infinity such that
\[
u(x)=c-2\frac{d^2}{dx^2}
\log\vartheta(Ux+Z\mid\Omega)
\]
is real and smooth for admissible $(D,Z)$, and
\[
\left(-\frac{d^2}{dx^2}+u(x)\right)\psi(x,p)
=E(p)\psi(x,p).
\]
Show from the two points over a regular $E$ and their Bloch multipliers that
the spectrum is
\[
[E_0,E_1]\cup[E_2,E_3]\cup\cdots\cup
[E_{2g-2},E_{2g-1}]\cup[E_{2g},\infty).
\]
Specialize to $g=1$ and recover, up to translation and an additive energy
constant, the Lam\'e potential and Hermite theta solutions above.
\end{exercise}

\begin{exercise}[Cnoidal KdV waves and finite-gap transport]
For the normalized Korteweg--de Vries equation
\[
u_t+6uu_x+u_{xxx}=0,
\]
put $\xi=x-vt$ and
\[
u(x,t)=a+2k^2\kappa^2
\operatorname{cn}^2(\kappa\xi;k),
\qquad 0<k<1.
\]
Derive
\[
\frac{d^2}{dz^2}\operatorname{cn}^2(z;k)
=2(1-k^2)+4(2k^2-1)\operatorname{cn}^2(z;k)
-6k^2\operatorname{cn}^4(z;k)
\]
and prove that the displayed wave solves KdV exactly when
\[
v=6a+4\kappa^2(2k^2-1).
\]
Deduce its amplitude, wavelength, and speed:
\[
A=2k^2\kappa^2,
\qquad
\lambda=\frac{2K(k)}{\kappa},
\qquad
v=6a+4\kappa^2(2k^2-1).
\]
Define the complete elliptic integral of the second kind and the elliptic
nome by
\[
E(k)=\int_0^{\pi/2}\sqrt{1-k^2\sin^2\theta}\,d\theta,
\qquad
q=e^{-\pi K'(k)/K(k)}.
\]
Prove the Fourier expansion
\[
\operatorname{cn}^2(z;k)
=\frac{E(k)-(1-k^2)K(k)}{k^2K(k)}
+\frac{2\pi^2}{k^2K(k)^2}
\sum_{n\geq1}\frac{nq^n}{1-q^{2n}}
\cos\frac{n\pi z}{K(k)}.
\]
Use as theorem input the finite-gap KdV evolution
theorem~\cite{belokolos}, which replaces
$Ux+Z$ in the reconstructed potential by $Ux+Vt+Z$ for the normalized
second-kind KdV period vector $V$.  Show that the genus-one case gives the
cnoidal wave above and that $k\uparrow1$ gives the solitary wave
\[
u(x,t)=a+2\kappa^2
\operatorname{sech}^2\!\bigl(\kappa(x-(6a+4\kappa^2)t)\bigr).
\]
Predict the wavelength, phase speed, and harmonic-amplitude ratios of a
weakly dispersive one-dimensional wave train from $K$, $E$, and the nome.

\end{exercise}

\begin{exercise}[Fredholm operators and the Calkin algebra]
Let $H_0$ and $H_1$ be complex Hilbert spaces.  Write
$\mathcal B(H_0,H_1)$ for the bounded operators and define
$\mathcal K(H_0,H_1)$ to be the operator-norm closure of the finite-rank
operators.  Call its elements \emph{compact operators}.  Prove that
composition on either side by a bounded operator preserves compactness and
that $\mathcal K(H)=\mathcal K(H,H)$ is a closed two-sided ideal in
$\mathcal B(H)=\mathcal B(H,H)$.  Define the \emph{Calkin algebra} by
\[
\mathcal Q(H)=\mathcal B(H)/\mathcal K(H).
\]

A bounded operator $T:H_0\to H_1$ is \emph{Fredholm} when its kernel and
cokernel are finite-dimensional and its image is closed.  Define
\[
\operatorname{ind}_{\rm an}(T)
=\dim\ker T-\dim\operatorname{coker}T.
\]
Use as theorem input Atkinson's theorem~\cite{conway-functional}: $T$ is Fredholm exactly when
there is a bounded $S:H_1\to H_0$ for which
\[
ST-I_{H_0}\in\mathcal K(H_0),
\qquad
TS-I_{H_1}\in\mathcal K(H_1).
\]
For $H_0=H_1$, identify this condition with invertibility of the image of
$T$ in the Calkin algebra.

Prove that compact perturbations preserve the Fredholm property and the
index, that
\[
\operatorname{ind}_{\rm an}(T^*)=-\operatorname{ind}_{\rm an}(T),
\qquad
\operatorname{ind}_{\rm an}(T_2T_1)
=\operatorname{ind}_{\rm an}(T_2)+\operatorname{ind}_{\rm an}(T_1),
\]
and that the index is constant along a norm-continuous path of Fredholm
operators.  On $\ell^2(\mathbb N)$, let $S$ be the unilateral shift
$Se_n=e_{n+1}$.  Construct a parametrix modulo finite-rank operators and
compute
\[
\operatorname{ind}_{\rm an}(S)=-1,
\qquad
\operatorname{ind}_{\rm an}(S^*)=1.
\]

For the twisted periodic Schr\"odinger operators
\[
\mathcal L_{\rho,E}:H^2_\rho\longrightarrow L^2([0,P])
\]
with $|\rho|=1$ and $E\in\mathbb R$, use the closed-range and
kernel--cokernel calculations to prove
\[
\operatorname{ind}_{\rm an}(\mathcal L_{\rho,E})=0,
\qquad
\ker\mathcal L_{\rho,E}\ne0
\Longleftrightarrow
\rho^2-\Delta(E)\rho+1=0.
\]
\end{exercise}

\begin{exercise}[Elliptic operators and analytic index]
Let $E,F\to M$ be smooth complex vector bundles over a closed smooth
manifold.  For $f\in C^\infty(M)$ let $M_f$ denote multiplication by $f$.
A linear map
\[
D:\Gamma^\infty(E)\longrightarrow\Gamma^\infty(F)
\]
is a \emph{differential operator of order at most $m$} when every
$(m+1)$-fold iterated commutator with multiplication operators vanishes.
For an operator of order $m$, $x\in M$, $\xi\in T_x^*M$, and
$v\in E_x$, choose $f$ with $df_x=\xi$ and a section $s$ with $s(x)=v$,
and define
\[
\sigma_m(D)(x,\xi)v
=\frac1{m!}\operatorname{ad}_{M_f}^{\,m}(D)s(x),
\qquad
\operatorname{ad}_{M_f}(D)=[D,M_f].
\]
Prove that this is independent of $f$ and $s$, depends homogeneously of
degree $m$ on $\xi$, and defines the \emph{principal symbol}
\[
\sigma_m(D):\pi^*E\longrightarrow\pi^*F
\quad\text{over }T^*M.
\]
Call $D$ \emph{elliptic} when $\sigma_m(D)(x,\xi)$ is invertible for every
$\xi\ne0$.

Choose Hermitian and Riemannian metrics and a compatible connection on
$E$.  Using the spectral calculus for the nonnegative
connection Laplacian $\Delta_E=\nabla^*\nabla$, define $H^s(M;E)$ as the
completion of smooth sections for the norm
\[
\lVert u\rVert_{H^s}
=\lVert(I+\Delta_E)^{s/2}u\rVert_{L^2}.
\]
Use as theorem input~\cite{hormander3} that different choices give equivalent norms.  Use as
theorem input~\cite{hormander3} the coordinate-invariant elliptic regularity, symbol
quantization, and parametrix theorem: for every real $s$, an elliptic $D$
extends to
\[
D_s:H^s(M;E)\longrightarrow H^{s-m}(M;F)
\]
and admits a bounded parametrix whose two remainders are compact.  Deduce
from Atkinson's theorem that $D_s$ is Fredholm, that its index is
independent of $s$, and use symbol quantization to prove that it depends
only on the homotopy class of the elliptic principal symbol.  Define this
integer to be
$\operatorname{ind}_{\rm an}(D)$ and prove
\[
\operatorname{ind}_{\rm an}(D^*)=-\operatorname{ind}_{\rm an}(D).
\]

For a holomorphic line bundle $L$ on a compact Riemann surface $C$, show
that
\[
\bar\partial_L:\Omega^{0,0}(C;L)\longrightarrow\Omega^{0,1}(C;L)
\]
is elliptic.  Use as theorem input~\cite{forster,hormander3} the Dolbeault identification of its
kernel and cokernel with $H^0(C,L)$ and $H^1(C,L)$.  Recover from
Riemann--Roch the analytic-index formula
\[
\operatorname{ind}_{\rm an}(\bar\partial_L)
=\deg L+1-g(C).
\]
\end{exercise}

\begin{exercise}[Homotopy, suspension, and clutching]
A \emph{based space} is a space $X$ with a distinguished point $x_0$.
Define based maps, based homotopies, and the set $[X,Y]_*$ of based
homotopy classes.  Define the reduced suspension and based loop space by
\[
\Sigma X
=X\times[0,1]\big/
\bigl(X\times\{0,1\}\cup\{x_0\}\times[0,1]\bigr),
\]
\[
\Omega Y=\{\gamma:[0,1]\to Y:
\gamma(0)=\gamma(1)=y_0\}.
\]
Construct the suspension--loop adjunction
\[
[\Sigma X,Y]_*\cong[X,\Omega Y]_*.
\]

Let $V$ be a finite-dimensional complex vector space and let
$g:X\to\operatorname{Aut}(V)$ be a based map.  Glue the trivial bundles on
the two reduced cones on $X$ by $g$ along their common copy of $X$ and
denote the resulting bundle on $\Sigma X$ by $E_g$.  Prove that based
homotopic clutching maps give isomorphic bundles and that direct sum of
clutching maps corresponds to direct sum of bundles.  Call bundles
$E$ and $F$ \emph{stably isomorphic} when
\[
E\oplus\underline{\mathbb C}^{\,r}
\cong F\oplus\underline{\mathbb C}^{\,s}
\]
for some $r,s$.  Use as theorem input the clutching
theorem~\cite{atiyah-k-theory}: for compact
Hausdorff $X$, stable bundles on $\Sigma X$ are classified by stable
homotopy classes of clutching maps.

Take $X=S^1$, $V=\mathbb C$, and let $g(z)$ be multiplication by $z$.
Identify $\Sigma S^1$ with $S^2$, identify $E_g$ with the Hopf line bundle
after choosing the orientation compatible with the clutching convention,
and use the circle-cocycle calculation to prove
\[
\int_{S^2}c_1(E_g)=1.
\]
\end{exercise}

\begin{exercise}[Connections, curvature, and Chern--Weil forms]
Let $E\to M$ be a smooth complex vector bundle of rank $r$.  Define its
direct sum, tensor product, dual, and pullback by the corresponding
fiberwise operations.  Call a connection \emph{complex} when
$\nabla_X(is)=i\nabla_Xs$.  For a complex connection, extend $\nabla$ to
$E$-valued forms by the graded Leibniz rule and write
\[
F_\nabla=\nabla^2\in\Omega^2(M;\operatorname{End}E)
\]
for its curvature.

Prove the Bianchi identity $\nabla F_\nabla=0$.  Using trace and determinant,
define
\[
\operatorname{ch}(E,\nabla)
=\operatorname{Tr}\exp\!\left(\frac{iF_\nabla}{2\pi}\right),
\qquad
c(E,\nabla)
=\det\!\left(I+\frac{iF_\nabla}{2\pi}\right)
=1+c_1(E,\nabla)+\cdots+c_r(E,\nabla).
\]
Use Bianchi and cyclicity of trace to prove that the homogeneous components
of $\operatorname{ch}(E,\nabla)$ are closed.  For a path $\nabla_t$ of
connections, derive
\[
\frac d{dt}\operatorname{Tr}
\exp\!\left(\frac{iF_{\nabla_t}}{2\pi}\right)
=d\,\operatorname{Tr}\!\left(
\frac{i\dot\nabla_t}{2\pi}
\exp\!\left(\frac{iF_{\nabla_t}}{2\pi}\right)\right).
\]
Use as theorem input~\cite{bott-tu} the corresponding Chern--Weil transgression formula
for the determinant invariant polynomials.  Deduce that the de Rham classes
$\operatorname{ch}(E)$ and $c_j(E)$ are independent of the connection,
natural under pullback, and satisfy
\[
c(E\oplus F)=c(E)c(F),
\qquad
\operatorname{ch}(E\oplus F)=\operatorname{ch}(E)+\operatorname{ch}(F),
\qquad
\operatorname{ch}(E\otimes F)=\operatorname{ch}(E)\operatorname{ch}(F).
\]
Prove also that
\[
\operatorname{ch}_q(E^\vee)=(-1)^q\operatorname{ch}_q(E).
\]
Recover the first Chern form of a line bundle used above.
\end{exercise}

\begin{exercise}[Spectral projections and Berry bundles]
Let $W$ be a finite-dimensional Hermitian vector space, let $B$ be a smooth

P_b=\frac{1}{2\pi i}\oint_\Gamma(z-H(b))^{-1}\,dz,

Prove that $P_b$ is a smooth family of orthogonal projections and that

\nabla^{\mathrm B}s=P\,ds

for sections satisfying $Ps=s$, and prove

F_{\mathrm B}=P(dP\wedge dP)P,

\end{exercise}

\begin{exercise}[Two-level Berry degree and gap protection]
For the Pauli endomorphisms and the

\begin{exercise}[Topological \(K\)-theory and Bott periodicity]
For a compact Hausdorff space $X$, let $\operatorname{Vect}_{\mathbb C}(X)$
be the commutative monoid of isomorphism classes of finite-rank complex
vector bundles under direct sum.  Define $K^0(X)$ as its Grothendieck
group: elements are formal differences $[E]-[F]$, with
\[
[E]-[F]=[E']-[F']
\]
when there is a bundle $G$ for which
\[
E\oplus F'\oplus G\cong E'\oplus F\oplus G.
\]
Use tensor product to make $K^0(X)$ a ring.  For based spaces define the
smash product
\[
X\wedge Y=X\times Y/(X\vee Y),
\qquad
X\vee Y=X\times\{y_0\}\cup\{x_0\}\times Y,
\]
and define the exterior product in $K^0(X\times Y)$ by
\[
a\boxtimes b
=\operatorname{pr}_X^*a\cdot\operatorname{pr}_Y^*b.
\]
Define
\[
\widetilde K^0(X)
=\ker\!\left(K^0(X)\longrightarrow K^0(\{x_0\})\right).
\]
For a closed based subspace $A\subset Z$, define
\[
K^0(Z,A)=\widetilde K^0(Z/A).
\]
For $a=[E]-[E']\in\widetilde K^0(X)$, stabilize and choose an
isomorphism $E_{x_0}\cong E'_{x_0}$, and do the same for a representative
of $b\in\widetilde K^0(Y)$.  Show that $a\boxtimes b$, together with the
induced stable trivialization on $X\vee Y$, determines a relative class
in
\[
K^0(X\times Y,X\vee Y)
\cong\widetilde K^0(X\wedge Y),
\]
independent of the stabilizations and isomorphisms.  Deduce the bilinear
reduced exterior product
\[
\widetilde K^0(X)\otimes\widetilde K^0(Y)
\longrightarrow\widetilde K^0(X\wedge Y).
\]
Define
\[
\widetilde K^{-n}(X)=\widetilde K^0(\Sigma^nX)
\quad(n\geq0),
\qquad
K^{-n}(X)=\widetilde K^{-n}(X_+),
\qquad
K^1(X)=K^{-1}(X).
\]
Here $X_+=X\sqcup\{*\}$ for an unbased compact space $X$.

Let $BU(r)$ be the classifying space characterized by the natural
correspondence between rank-$r$ complex bundles on a compact space and
homotopy classes of maps to $BU(r)$.  Let $BU$ and $U$ denote the stable
classifying and unitary spaces, and write $[X,Y]$ for unbased homotopy
classes.  Use as theorem input~\cite{atiyah-k-theory} the stable
classification
\[
K^0(X)\cong[X,\mathbb Z\times BU],
\qquad
K^1(X)\cong[X,U],
\]
and complex Bott periodicity
\[
\Omega(\mathbb Z\times BU)\simeq U,
\qquad
\Omega U\simeq\mathbb Z\times BU.
\]
Let
\[
\beta=[E_g]-[\underline{\mathbb C}]
\in\widetilde K^0(S^2)
\]
be the Bott class from the Hopf bundle above.  Prove that exterior product
with $\beta$ gives
\[
\widetilde K^0(\Sigma^nX)
\cong
\widetilde K^0(\Sigma^nX\wedge S^2)
\cong\widetilde K^0(\Sigma^{n+2}X)
\qquad(n\geq0).
\]
Use this isomorphism to extend reduced and unreduced (K)-theory
two-periodically to all degrees.
Deduce
\[
\widetilde K^0(S^{2r})\cong\mathbb Z,
\quad
\widetilde K^0(S^{2r+1})=0,
\quad
K^1(S^{2r})=0,
\quad
K^1(S^{2r+1})\cong\mathbb Z.
\]
Define a \emph{CW complex} inductively by attaching disks along their
boundary spheres, and call an inclusion $A\hookrightarrow X$ a
\emph{cofibration} when it has the homotopy-extension property.  Use as
theorem input~\cite{atiyah-k-theory} the six-term exact sequence in reduced \(K\)-theory for a
cofibration and the standard cell attachment
$S^1\vee S^1\to T^2\to S^2$ to compute
\[
K^0(T^2)\cong\mathbb Z^2,
\qquad
K^1(T^2)\cong\mathbb Z^2.
\]
When $X$ is a compact smooth manifold, use as theorem
input~\cite{bott-tu} that the
Chern--Weil classes $c_j(E)$ are the real images of integral cohomology
classes.  Use their rational polynomial relation with the Chern character
and the Chern--Weil identities to prove that
\[
\operatorname{ch}([E]-[F])
=\operatorname{ch}(E)-\operatorname{ch}(F)
\]
defines a natural ring homomorphism
\[
\operatorname{ch}:K^0(X)\longrightarrow H^{\rm even}(X;\mathbb Q).
\]
Use as theorem input~\cite{atiyah-k-theory} that these classes define the same natural
transformation for finite CW complexes and that it becomes an isomorphism
after tensoring with $\mathbb Q$.

For a continuous finite-rank spectral projection $P(x)$ of a gapped
Hamiltonian over a compact parameter space $X$, form its image bundle and
class in $K^0(X)$.  Prove that this class is unchanged by a gap-preserving
homotopy and identify, for $X=T^2$, its rank and first-Chern components
with the two integers in the displayed computation.
\end{exercise}

\begin{exercise}[Spectral flow and Toeplitz index]
Let $A_t$, $0\leq t\leq1$, be a norm-continuous path of bounded
self-adjoint Fredholm operators with invertible endpoints.  Choose a
partition $0=t_0<\cdots<t_N=1$ and numbers $\varepsilon_j>0$ so that the
finite-rank spectral projections
\[
\mathbf1_{[-\varepsilon_j,\varepsilon_j]}(A_t)
\]
vary continuously for $t\in[t_{j-1},t_j]$.  Define the
\emph{spectral flow} by
\[
\operatorname{sf}(A_t)
=\sum_{j=1}^N
\left(
\operatorname{rank}\mathbf1_{[0,\varepsilon_j)}(A_{t_j})
-\operatorname{rank}\mathbf1_{[0,\varepsilon_j)}(A_{t_{j-1}})
\right).
\]
Prove independence of the choices, additivity under concatenation, and
invariance under endpoint-fixed homotopies through such paths.

Let $W$ be a finite-dimensional Hermitian vector space, let
$u:S^1\to U(W)$ be smooth, and define
\[
\operatorname{wind}(\det u)
=\frac1{2\pi i}\int_{S^1}(\det u)^{-1}d(\det u).
\]
Let $H^2(S^1;W)\subset L^2(S^1;W)$ be the closed span of the nonnegative
Fourier modes, let $P_+$ be its orthogonal projection, and define the
Toeplitz operator
\[
T_u=P_+M_u\big|_{H^2(S^1;W)}.
\]
Use as theorem input~\cite{atiyah-k-theory} that $T_u$ is Fredholm and that homotopies of $u$
produce norm-continuous Fredholm paths.  For
$W=\mathbb C\oplus W'$ compute the index when
$u(z)=z^n\oplus I_{W'}$ from the unilateral shift.  Use as theorem
input~\cite{atiyah-k-theory}
that $SU(r)$ is simply connected for $r\geq2$ and use the determinant
fibration
\[
SU(r)\longrightarrow U(r)\xrightarrow{\det}U(1)
\]
to prove that determinant induces $\pi_1(U(r))\cong\mathbb Z$.  Combine
this with index homotopy invariance and stabilization to prove
\[
\operatorname{ind}_{\rm an}(T_u)
=-\operatorname{wind}(\det u).
\]
Define the odd \(K\)-theory pairing on the circle by
\[
\langle[u],[S^1]\rangle
=\operatorname{wind}(\det u)
=-\operatorname{ind}_{\rm an}(T_u).
\]
For a loop $A_t$ of self-adjoint Fredholm operators based at an invertible
operator, define its suspended operator on periodic paths by
\[
\mathcal D_A=\frac d{dt}+A_t.
\]
Use as theorem input the suspension theorem~\cite{atiyah-k-theory}
\[
\operatorname{ind}_{\rm an}(\mathcal D_A)
=\operatorname{sf}(A_t)
\]
and identify both integers with the corresponding $K^1$ pairing.
\end{exercise}

\begin{exercise}[Thom classes and the topological index]
For a locally compact space $Y$, define compactly supported \(K\)-theory by

K_c^j(Y)=\widetilde K^j(Y^+),
\qquad j\in\mathbb Z/2,

K^j(X)\longrightarrow K_c^j(E),

\operatorname{ind}_{\rm top}:

\begin{exercise}[Atiyah--Singer index theorem]

\operatorname{Td}(E)

\widehat A(V)

A \emph{spin structure} on an oriented Riemannian $n$-manifold is a lift

\operatorname{Spin}^c(n)

\mathcal D_{\bar\partial,E}^+

\begin{exercise}[Monopole harmonics and spherical Landau levels]
Let $S_R^2$ be the round sphere of radius $R$, let $N\in\mathbb Z$, and
let $L_N\to S_R^2$ be a Hermitian line bundle with unitary connection of
constant curvature
\[
F_\nabla=-\frac{iN}{2R^2}\,dA,
\qquad
\frac{i}{2\pi}\int_{S_R^2}F_\nabla=N.
\]
For a particle of charge $e$, impose the convention
$F_\nabla=-ieB\,dA/\hbar$ and deduce the magnetic-flux quantization
\[
\Phi_B=\int_{S_R^2}B\,dA=\frac{Nh}{e}.
\]
On the northern and southern polar charts use the connection forms
\[
A_{\rm N}=-\frac{iN}{2}(1-\cos\theta)\,d\phi,
\qquad
A_{\rm S}=\frac{iN}{2}(1+\cos\theta)\,d\phi,
\]
and determine their transition function on the overlap.

For $\alpha,\beta>-1$, define the Jacobi polynomial
$P_n^{(\alpha,\beta)}(x)$ as the degree-$n$ polynomial solution of
\[
(1-x^2)y''
+[\beta-\alpha-(\alpha+\beta+2)x]y'
+n(n+\alpha+\beta+1)y=0
\]
normalized by $P_n^{(\alpha,\beta)}(1)=\binom{n+\alpha}{n}$.
Prove its orthogonality for distinct degrees with respect to
$(1-x)^\alpha(1+x)^\beta\,dx$ on $[-1,1]$.

Put $q=|N|/2$.  Separate the covariant Laplacian in the two monopole
gauges, extract the endpoint powers imposed by regularity and the
transition function, and reduce the polar equation to the Jacobi equation.
Call the normalized global eigensections obtained in this way the
\emph{monopole harmonics}.  Identify sections of $L_N$ with functions
$F:SU(2)\to\mathbb C$ satisfying
\[
F(gh)=\chi_N(h)^{-1}F(g)
\]
under the identification $S^2\cong SU(2)/U(1)$, where
\[
h_\theta=
\begin{pmatrix}e^{i\theta}&0\\0&e^{-i\theta}\end{pmatrix},
\qquad
\chi_N(h_\theta)=e^{iN\theta}.
\]
Use Peter--Weyl theory to prove their orthogonality and completeness and
to show that
\[
\ell=q+n,\qquad n=0,1,2,\ldots,
\qquad
\dim\mathcal H_\ell=2\ell+1=|N|+2n+1.
\]
For the scalar magnetic Hamiltonian
\[
H_N=\frac{\hbar^2}{2m}\nabla^*\nabla,
\]
derive
\[
E_n
=\frac{\hbar^2}{2mR^2}
\left((q+n)(q+n+1)-q^2\right),
\qquad
E_{n+1}-E_n
=\frac{\hbar^2}{mR^2}(n+q+1).
\]

For $N\geq0$, equip $S^2\cong\mathbb P^1$ with its complex structure,
identify $L_N$ with $\mathcal O(N)$, and define the spin$^c$
Dolbeault--Pauli Hamiltonian
\[
H_+=\frac{\hbar^2}{m}\bar\partial_\nabla^*
\bar\partial_\nabla.
\]
Use Riemann--Roch and the Dolbeault index theorem to prove
\[
\dim\ker H_+=N+1,
\]
identify these zero modes with the lowest monopole-harmonic multiplet, and
predict the complete level degeneracies and spacings above.  State the
uniform-field, round-sphere, fixed-charge, spin$^c$ coupling, noninteracting,
and spectral-resolution assumptions under which spherical Landau-level
spectroscopy tests the Jacobi-polynomial and index predictions.
\end{exercise}

\begin{exercise}[Family index and quantized Hall transport]
Let $B$ be compact and let $T_b:H_0\to H_1$ be a norm-continuous family
of Fredholm operators.  Use as theorem input~\cite{atiyah-singer-families} that there is a
finite-dimensional subspace $V\subset H_1$ for which
\[
H_0\oplus V\longrightarrow H_1,
\qquad
(u,v)\longmapsto T_bu+v
\]
is surjective for every $b\in B$.  Prove that its kernels form a vector
bundle $\mathcal K_T\to B$ and define the \emph{family index} by
\[
\operatorname{Ind}(T)=[\mathcal K_T]-[B\times V]\in K^0(B).
\]
Prove independence of $V$, homotopy invariance, and additivity under
direct sum.  Show that the construction is local on $B$ and therefore
extends to continuous Fredholm morphisms between Hilbert bundles.

Let $E_\tau$ carry a degree-$N>0$ magnetic line bundle $L_N$, and let
\[
\widehat E_\tau=\operatorname{Hom}(H_1(E_\tau;\mathbb Z),U(1))
\]
be the torus of boundary-condition twists.  Let
$\mathcal P\to E_\tau\times\widehat E_\tau$ be the normalized Poincar\'e
line bundle, whose restriction to $E_\tau\times\{\chi\}$ is the flat line
bundle $F_\chi$.  Orient $\widehat E_\tau$ so that
\[
\frac12\int_{E_\tau}c_1(\mathcal P)^2
\]
is its positive integral generator.  Define the lowest-Landau-level bundle
and filled-level line by
\[
\mathcal H_N\big|_\chi
=H^0(E_\tau,L_N\otimes F_\chi),
\qquad
\mathcal D_N=\bigwedge^N\mathcal H_N.
\]
Use the theta automorphy laws to realize the fibers by level-$N$ theta
functions and Riemann--Roch to prove that $\mathcal H_N$ has rank $N$.
Show that the Sobolev completions of
$\bar\partial_{L_N\otimes F_\chi}$ form a Fredholm family and that
Riemann--Roch identifies its family index with $[\mathcal H_N]$.

Use as theorem input the families Atiyah--Singer
formula~\cite{atiyah-singer-families}: for a
holomorphic submersion $p:X\to B$ with compact fibers and a holomorphic
bundle $\mathcal L\to X$,
\[
\operatorname{ch}\operatorname{Ind}(\bar\partial_{\mathcal L})
=\int_{X/B}
\operatorname{ch}(\mathcal L)\operatorname{Td}(T^{1,0}(X/B)),
\]
where the right side is integration along the compact complex fibers.
Apply it to
$X=E_\tau\times\widehat E_\tau$, $B=\widehat E_\tau$, and
$\mathcal L=\operatorname{pr}_1^*L_N\otimes\mathcal P$.  Use the
triviality of the elliptic-curve tangent bundle to prove
\[
\operatorname{ch}(\mathcal H_N)
=\int_{E_\tau}
\exp\bigl(c_1(L_N)+c_1(\mathcal P)\bigr).
\]
Extract its degree-zero and degree-two parts to prove
\[
\operatorname{rank}\mathcal H_N=N,
\qquad
\int_{\widehat E_\tau}c_1(\mathcal D_N)=1.
\]
Use as theorem input~\cite{tknn} the gapped noninteracting Kubo--Chern formula and the
adiabatic theorem.  Deduce for one completely filled Landau level
\[
\sigma_{xy}=\frac{e^2}{h},
\qquad
Q_{\rm pump}=e
\]
for a unit winding of a boundary twist.  Express the Berry connection of
$\mathcal D_N$ through the normalized theta sections and identify the same
integer from their automorphy.  Identify the pumped charge with the
spectral-flow and \(K^1\)-pairing integer.  State
separately the spectral-gap,
adiabatic, noninteracting, filled-level, mobility, contact, temperature,
and finite-size assumptions under which Hall-voltage and charge-pump
measurements test these predictions.
\end{exercise}

\section{Cobordism}

\begin{exercise}[Oriented, spin, and complex bordism rings]
For an oriented manifold $M$, write $-M$ for the same manifold with the
opposite orientation.  An \emph{oriented bordism} from a closed oriented
$n$-manifold $M_0$ to a closed oriented $n$-manifold $M_1$ is a compact
oriented $(n+1)$-manifold $W$ with an orientation-preserving identification
\[
\partial W\cong(-M_0)\sqcup M_1,
\]
product neighborhoods of its boundary components, and the boundary
orientation determined by the outward-normal-first convention.  Call
$M_0$ and $M_1$ \emph{oriented bordant} when such a $W$ exists.

Prove that oriented bordism is an equivalence relation.  Let
$\Omega_n^{SO}$ be the set of its equivalence classes.  Prove that
disjoint union makes $\Omega_n^{SO}$ an abelian group with zero the empty
manifold and inverse $[M]\mapsto[-M]$.  Use the product orientation to
make
\[
\Omega_*^{SO}=\bigoplus_{n\geq0}\Omega_n^{SO}
\]
a graded-commutative ring with unit the positively oriented point.  Prove
\[
\Omega_0^{SO}\cong\mathbb Z,
\qquad
\Omega_1^{SO}=0,
\qquad
\Omega_2^{SO}=0
\]
by signed point count and disks, and by using as theorem input the
classification of closed oriented surfaces and constructing oriented
handlebodies.

For a spin manifold with boundary, use the outward unit normal to induce
a spin structure on its boundary.  Define \emph{spin bordism} by requiring
the boundary identification to preserve these structures, and construct
the graded ring $\Omega_*^{\operatorname{Spin}}$.  A \emph{stable complex
structure} on $M$ is a complex-vector-bundle structure on
$TM\oplus\underline{\mathbb R}^{\,r}$ for some $r$, considered modulo
addition of trivial complex line bundles and homotopy.  Define stable
complex bordism and its ring $\Omega_*^U$.  Prove that products inherit
the corresponding structures and construct the forgetful homomorphisms
\[
\Omega_*^{\operatorname{Spin}}\longrightarrow\Omega_*^{SO},
\qquad
\Omega_*^U\longrightarrow\Omega_*^{SO}.
\]
\end{exercise}

\begin{exercise}[Pontryagin--Thom collapse and Thom spectra]
For $G=SO,\operatorname{Spin}$, let $BG(k)$ be the classifying space for
rank-$k$ real vector bundles with $G$-structure and let
$\gamma_k^G\to BG(k)$ be the universal bundle.  Put
\[
MG(k)=\operatorname{Th}(\gamma_k^G),
\qquad
\pi_n(MG)=\varinjlim_k\pi_{n+k}(MG(k)),
\]
where stabilization of the universal bundles induces the structure maps.
For complex rank-$k$ bundles, put
\[
MU(k)=\operatorname{Th}(\gamma_k^U),
\qquad
\pi_n(MU)=\varinjlim_k\pi_{n+2k}(MU(k)).
\]
Call the resulting sequences of Thom spaces the spectra $MSO$,
$M\operatorname{Spin}$, and $MU$.

Choose an embedding $M^n\hookrightarrow\mathbb R^{n+k}$ of a closed
$G$-manifold, give its normal bundle the stable $G$-structure complementary
to that on $TM$, and let its classifying map pull back $\gamma_k^G$ to
the normal bundle.  Collapse the complement of a tubular neighborhood to
construct
\[
S^{n+k}\longrightarrow\operatorname{Th}(\nu_M)
\longrightarrow MG(k).
\]
Prove that isotopy, stabilization, and a $G$-bordism do not change the
resulting stable homotopy class.  Use as theorem input the transversality
and regular-value construction in the Pontryagin--Thom
theorem~\cite{thom-cobordism} to prove that collapse induces ring
isomorphisms
\[
\Omega_n^{SO}\cong\pi_n(MSO),
\qquad
\Omega_n^{\operatorname{Spin}}\cong\pi_n(M\operatorname{Spin}),
\qquad
\Omega_n^U\cong\pi_n(MU).
\]

A \emph{stable framing} of $M$ is a trivialization of its stable normal
bundle.  Let $\Omega_n^{\mathrm{fr}}$ be framed bordism and define the
$n$th stable stem by
\[
\pi_n^S=\varinjlim_k[S^{n+k},S^k]_*.
\]
Show that a framing turns the collapse map into a map
$S^{n+k}\to S^k$ and recover
\[
\Omega_n^{\mathrm{fr}}\cong\pi_n^S.
\]
\end{exercise}

\begin{exercise}[Pontryagin numbers and the signature genus]
For a real vector bundle $V$, define its Pontryagin classes by
\[
p_j(V)=(-1)^j c_{2j}(V\otimes_{\mathbb R}\mathbb C)
\in H_{\mathrm{dR}}^{4j}(M).
\]
For a closed oriented $4k$-manifold and a partition
$I=(i_1,\ldots,i_r)$ of $k$, define the Pontryagin number
\[
p_I[M]
=\int_Mp_{i_1}(TM)\wedge\cdots\wedge p_{i_r}(TM).
\]
If $M=\partial W$ as an oriented manifold, use
\[
TW|_M\cong TM\oplus\underline{\mathbb R}
\]
and choose a connection that is a product near the boundary.  Use Stokes'
theorem for its Chern--Weil representatives to prove
$p_I[M]=0$.  Deduce that every Pontryagin number defines a homomorphism
$\Omega_{4k}^{SO}\to\mathbb Q$.  Use as theorem input Thom's rational
bordism theorem~\cite{thom-cobordism}: two oriented manifolds determine
the same class in $\Omega_*^{SO}\otimes\mathbb Q$ exactly when all their
Pontryagin numbers agree.

For a closed oriented $4k$-manifold, use as theorem input Poincar\'e
duality~\cite{bott-tu} to prove that
\[
([\alpha],[\beta])\longmapsto\int_M\alpha\wedge\beta
\]
is a nondegenerate symmetric form on
$H_{\mathrm{dR}}^{2k}(M)$, and define $\operatorname{sign}(M)$ to be its
signature.  If the formal Pontryagin roots satisfy
$p(TM)=\prod_j(1+x_j^2)$, define
\[
L(TM)=\prod_j\frac{x_j}{\tanh x_j}.
\]
Use as theorem input the Hirzebruch signature
theorem~\cite{hirzebruch-topological}
\[
\operatorname{sign}(M)=\int_ML(TM)
\]
and deduce that the signature is a ring homomorphism
$\Omega_*^{SO}\to\mathbb Z$.  In dimension four derive
\[
\operatorname{sign}(M)=\frac13\int_Mp_1(TM).
\]
For $h=c_1(\mathcal O(1))$ on $\mathbb{CP}^2$, use as theorem input the
Euler sequence
\[
0\longrightarrow\mathcal O
\longrightarrow\mathcal O(1)^{\oplus3}
\longrightarrow T\mathbb{CP}^2\longrightarrow0
\]
and prove
\[
c(T\mathbb{CP}^2)=(1+h)^3,
\qquad
\int_{\mathbb{CP}^2}p_1=3,
\qquad
\operatorname{sign}(\mathbb{CP}^2)=1,
\]
and conclude that $[\mathbb{CP}^2]$ has infinite order in
$\Omega_4^{SO}$.
\end{exercise}

\begin{exercise}[Dirac and Dolbeault bordism genera]
For a commutative ring $R$, a \emph{genus} is a ring homomorphism
from a bordism ring to $R$.  Given a formal power series
$Q(x)\in R[[x]]$ with $Q(0)=1$, use the splitting principle to construct
the multiplicative characteristic class $\prod_jQ(x_j)$ from the
appropriate formal roots.  Prove that its top-degree characteristic number
is additive under disjoint union and multiplicative under products, and
determine the condition under which stabilization leaves it unchanged.

For a closed spin $4k$-manifold, define
\[
\widehat A[M]=\int_M\widehat A(TM).
\]
Use the Atiyah--Singer theorem and the product formula for spinor bundles
to prove
\[
\widehat A[M]=\operatorname{ind}_{\rm an}(\mathcal D_M^+)
\in\mathbb Z
\]
and that $\widehat A$ is a genus
$\Omega_*^{\operatorname{Spin}}\to\mathbb Z$.  Prove its bordism
invariance directly from the stable tangent bundle of a boundary and
Stokes' theorem.

For a closed stable complex manifold $X$, define
\[
\operatorname{Td}[X]=\int_X\operatorname{Td}(TX).
\]
Use as theorem input the canonical spin$^c$ Dirac operator associated to
a stable complex structure~\cite{lawson-michelsohn}, and apply the index
theorem to prove that $\operatorname{Td}[X]$ is integral and defines a
genus $\Omega_*^U\to\mathbb Z$.  When $X$ is a compact complex manifold,
identify it with
\[
\operatorname{Td}[X]
=\sum_q(-1)^q\dim H^q(X,\mathcal O_X).
\]
From the degree-zero and degree-two terms of the Todd class, prove
\[
\operatorname{Td}[\mathrm{pt}]=1,
\qquad
\operatorname{Td}[C]=1-g
\]
for a compact Riemann surface $C$ of genus $g$.
\end{exercise}

\begin{exercise}[Atiyah--Patodi--Singer index and spectral eta functions]
Let $W$ be a compact even-dimensional spin Riemannian manifold whose
metric is a product near its boundary $M$.  Let $E\to W$ be a Hermitian
bundle with unitary connection of product type there.  Write
$\mathcal D_{W,E}^+$ for the chiral Dirac operator and $A_{M,E}$ for its
self-adjoint tangential boundary operator.  For
$\operatorname{Re}s$ sufficiently large, define the eta function
\[
\eta_{A_{M,E}}(s)
=\sum_{\lambda\ne0}
\operatorname{sign}(\lambda)|\lambda|^{-s},
\]
where the spectral values are repeated with multiplicity.  Use as theorem
input~\cite{aps1} that this function has a meromorphic continuation regular
at $s=0$.

Let $P_{\geq0}(A_{M,E})$ be the spectral projection onto the nonnegative
spectrum and impose the Atiyah--Patodi--Singer boundary condition
\[
P_{\geq0}(A_{M,E})(\psi|_M)=0.
\]
Use as theorem input the Atiyah--Patodi--Singer index
theorem~\cite{aps1}
\[
\operatorname{ind}_{\rm APS}(\mathcal D_{W,E}^+)
=\int_W\widehat A(TW)\operatorname{ch}(E)
-\frac{\eta_{A_{M,E}}(0)+h_{M,E}}2,
\qquad
h_{M,E}=\dim\ker A_{M,E}.
\]
Define the reduced eta invariant by
\[
\overline\eta(A)=\frac{\eta_A(0)+\dim\ker A}{2}
\quad\text{in }\mathbb R/\mathbb Z.
\]
Deduce
\[
\overline\eta(A_{M,E})
=\int_W\widehat A(TW)\operatorname{ch}(E)
\pmod{\mathbb Z}.
\]
Prove
\[
\overline\eta(-A)=-\overline\eta(A)\pmod{\mathbb Z}
\]
from spectral symmetry under orientation reversal.  When two product-type
bordisms are glued along oppositely oriented copies of the same boundary,
use additivity of the characteristic integrals to show that their reduced
eta corrections cancel modulo $\mathbb Z$.  Recover the closed
Atiyah--Singer formula when the boundary is empty.
\end{exercise}

\begin{exercise}[Hurwitz zeta functions and flux spectral asymmetry]
For $0<a<1$, let
\[
A_a=-i\frac d{d\theta}+a
\]
act on periodic functions on $\mathbb R/2\pi\mathbb Z$.  Prove that its
spectral values are $n+a$, $n\in\mathbb Z$.  Define the Hurwitz zeta
function for $\operatorname{Re}s>1$ by
\[
\zeta(s,a)=\sum_{n\geq0}(n+a)^{-s}.
\]
Separate the positive and negative spectral values to prove
\[
\eta_{A_a}(s)=\zeta(s,a)-\zeta(s,1-a).
\]
Use as theorem input the analytic continuation identity
\[
\zeta(0,a)=\frac12-a
\]
to derive
\[
\eta_{A_a}(0)=1-2a,
\qquad
\overline\eta(A_a)=\frac12-a\pmod{\mathbb Z}.
\]

For a single chiral massless charged Dirac channel on a coherent ring of
circumference $\ell$ threaded by flux $\Phi$, put
\[
a=\frac{\Phi}{\Phi_0}\pmod{\mathbb Z},
\qquad
\Phi_0=\frac he,
\qquad
E_n=\frac{2\pi\hbar v}{\ell}(n+a).
\]
Define the eta-regularized vacuum charge by
\[
Q_{\rm vac}=-\frac e2\eta_{A_a}(0).
\]
Predict
\[
\frac{Q_{\rm vac}}e=a-\frac12\pmod{\mathbb Z}
\]
and prove that insertion of one flux quantum produces unit spectral flow.
State the single-channel, linear-dispersion, coherent-ring, charge
regularization, flux-calibration, temperature, and level-resolution
assumptions under which charge sensing and flux-resolved spectroscopy test
the prediction.
\end{exercise}

\begin{exercise}[Jacobi theta functions and the K3 elliptic genus]
For a closed complex manifold $X$ of complex dimension $d$ with formal
Chern roots $x_1,\ldots,x_d$, define its two-variable elliptic genus by
\[
\operatorname{Ell}_X(\tau,z)
=\int_X\prod_{j=1}^d
x_j\,
\frac{\vartheta_1\!\left(\frac{x_j}{2\pi i}-z\,\middle|\,\tau\right)}
{\vartheta_1\!\left(\frac{x_j}{2\pi i}\,\middle|\,\tau\right)},
\qquad
\operatorname{Im}\tau>0.
\]
Use as theorem input~\cite{witten-elliptic} that this characteristic
series defines a stable complex bordism genus with values in holomorphic
functions of $(\tau,z)$.

Define
\[
\begin{aligned}
\vartheta_2(z\mid\tau)
&=\vartheta\!\begin{bmatrix}\frac12\\0\end{bmatrix}(z\mid\tau),&
\vartheta_3(z\mid\tau)
&=\vartheta\!\begin{bmatrix}0\\0\end{bmatrix}(z\mid\tau),&
\vartheta_4(z\mid\tau)
&=\vartheta\!\begin{bmatrix}0\\\frac12\end{bmatrix}(z\mid\tau).
\end{aligned}
\]
A \emph{K3 surface} is a compact connected complex surface $X$ with
$K_X\cong\mathcal O_X$ and $H^1(X,\mathcal O_X)=0$.  Use Serre duality
and the Dolbeault index formula to prove
\[
\operatorname{Td}[X]=2,
\qquad
\int_Xc_2(TX)=24.
\]
Use the Chern-root formula, the theta addition identities, and the K3
conditions $c_1=0$, $c_2[X]=24$ to derive the K3 identity
from~\cite{eguchi-k3}
\[
\operatorname{Ell}_{K3}(\tau,z)
=8\left[
\left(\frac{\vartheta_2(z\mid\tau)}{\vartheta_2(0\mid\tau)}\right)^2
+\left(\frac{\vartheta_3(z\mid\tau)}{\vartheta_3(0\mid\tau)}\right)^2
+\left(\frac{\vartheta_4(z\mid\tau)}{\vartheta_4(0\mid\tau)}\right)^2
\right].
\]
Put $q=e^{2\pi i\tau}$ and $y=e^{2\pi iz}$.  Derive
\[
\operatorname{Ell}_{K3}(\tau,z)
=2y+20+2y^{-1}+O(q),
\qquad
\operatorname{Ell}_{K3}(\tau,0)=24.
\]

For an $\mathcal N=(4,4)$ supersymmetric sigma model with K3 target,
use as theorem input~\cite{witten-elliptic,eguchi-k3} the Ramond-sector
identity
\[
\operatorname{Ell}_{K3}(\tau,z)
=\operatorname{Tr}_{\mathcal H_{RR}}
\left((-1)^{F_L+F_R}y^{J_0}
q^{L_0-c/24}\bar q^{\bar L_0-\bar c/24}\right)
\]
and prove that right-moving supersymmetry cancels every non-ground-state
pair, making the trace independent of $\bar q$.  Predict $24$ protected
Ramond ground states with charge-resolved multiplicities $2,20,2$ at
charges $1,0,-1$.  State the exact-supersymmetry, Ramond-boundary,
charge-normalization, unbroken-supersymmetry, continuum-scaling,
finite-volume, temperature, and symmetry-resolved spectroscopy assumptions
under which a programmable quantum system tests these protected counts.
\end{exercise}

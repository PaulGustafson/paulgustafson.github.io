\part{Algebraic Geometry}

\section{Elliptic Functions}

\begin{exercise}[Affine and projective curves]
For a commutative unital ring $A$, define prime ideals, localization, the
Zariski space $\operatorname{Spec}A$, and its structure sheaf by the rings
$A_f$ on distinguished opens.  Prove that every stalk is local and define a
locally ringed space, morphisms of locally ringed spaces, affine schemes, and
schemes by affine covers.  Define Noetherian rings, Krull dimension, and a
regular Noetherian local ring by
$\dim_{A/\mathfrak m}(\mathfrak m/\mathfrak m^2)=\dim A$.  Construct
$\operatorname{Proj}$ from homogeneous prime ideals and homogeneous
localization.  Define a projective curve over
$\C$ to be a smooth connected one-dimensional closed subscheme of a
projective space, with smoothness expressed by regular local rings.  Construct
the analytification of a smooth projective curve from its local rings and
prove it is a compact Riemann surface.  Define divisors by valuations of the
function field and prove that principal divisors have degree zero by the
argument principle.
\end{exercise}

\begin{exercise}[Homology, periods, and de Rham's theorem]
Define singular chains, boundary, homology, cochains, cup product,
orientations, and the fundamental class of an oriented closed manifold.  For
the differential forms developed with Hamiltonian actions, define integration
over smooth singular simplices.  Prove Stokes's theorem and show that
integration gives a cochain map.  Construct a good cover, compare the singular
and de Rham \v Cech double complexes with partitions of unity, and prove
\[
 H^*_{\rm dR}(M)\simeq H^*(M;\R)
\]
as graded rings.  Derive Mayer--Vietoris and Poincar\'e duality from the
integration pairing.  For a compact Riemann surface, prove that the
intersection form on $H_1$ is unimodular and alternating and construct a
symplectic marking.  Define its genus $g$ by
$\operatorname{rank}H_1=2g$ and identify this integer with its topological
handle number.
\end{exercise}

\begin{exercise}[Elliptic curves and translation symmetry]
For a nonsingular cubic
$E:y^2=4x^3-g_2x-g_3$ with discriminant $g_2^3-27g_3^2\ne0$, construct the
group law by divisors and prove the chord--tangent formula.  Show that the
holomorphic form $\dd x/y$ has no zeros and that its period map identifies
\[
 E(\C)\simeq\C/(\Z\omega_1+\Z\omega_2).
\]
Construct the Weierstrass function from its normally convergent lattice sum,
prove $\wp'^2=4\wp^3-g_2\wp-g_3$, and show that $z\mapsto(\wp(z),\wp'(z))$
is the inverse uniformization.  Derive the addition formula from translation
on $E$.
\end{exercise}

\begin{exercise}[Branched covers and Riemann--Hurwitz]
Define a branched cover of compact curves and its ramification divisor.
Define the canonical divisor as the divisor of any nonzero meromorphic
differential and prove that its divisor class is independent of that
differential.  Prove Riemann--Hurwitz by comparing canonical divisors.  For the normalization
of $y^2=\prod_{j=1}^{2g+1}(x-e_j)$, include the chart at infinity and prove
smoothness and genus $g$.
\end{exercise}

\begin{exercise}[Abel--Jacobi decomposition]
Construct a symplectic marking of first homology,
normalize the holomorphic differentials, and define the Abel map and Jacobian.
Prove Abel's theorem and Jacobi inversion by the argument principle and the
Riemann bilinear relations.
\end{exercise}

\begin{exercise}[Elliptic integrals and Jacobi functions]
For $0<k<1$ define
\[
 K(k)=\int_0^{\pi/2}\frac{\dd\theta}
 {\sqrt{1-k^2\sin^2\theta}},\qquad
 E(k)=\int_0^{\pi/2}\sqrt{1-k^2\sin^2\theta}\dd\theta.
\]
Invert the Abel integral
$u=\int_0^x\dd t/\sqrt{(1-t^2)(1-k^2t^2)}$ on its elliptic curve to define
$\operatorname{sn}(u;k)$ globally; define $\operatorname{cn}$ and
$\operatorname{dn}$ by the algebraic relations with branches fixed at zero.
Derive their differential equations, addition laws, periods, Fourier series,
and the hypergeometric representation
$K(k)=\frac\pi2{}_2F_1(1/2,1/2;1;k^2)$.
\end{exercise}

\begin{exercise}[Nonlinear pendulum periods]
For a pendulum of length $L$ in uniform gravity $g$, use energy conservation
and the elliptic Abel map to prove that oscillations of amplitude
$0<\theta_0<\pi$ have period
\[
 T(\theta_0)=4\sqrt{\frac Lg}
 K\!\left(\sin\frac{\theta_0}{2}\right).
\]
Expand the hypergeometric series to obtain
\[
 T=2\pi\sqrt{\frac Lg}
 \left(1+\frac{\theta_0^2}{16}
 +\frac{11\theta_0^4}{3072}+O(\theta_0^6)\right)
\]
and derive the logarithmic divergence at the separatrix.  Identify $L,g$,
$\theta_0$, and the measured return time, and distinguish ideal rigid,
frictionless motion from experimental corrections.
\end{exercise}

\section{Bergman Kernels}

\begin{exercise}[Line-bundle automorphisms and sheaf cohomology]
Define a line bundle $p:L\to X$ and immediately define a section over
$U\subset X$ as a map $s:U\to L$ with $p\circ s=\id_U$; write
$\Gamma(U,L)$ for all such sections.  Define the sheaf of sections, locally
free and coherent sheaves, tensor product, dual, and pullback.  Define
$\operatorname{Aut}(X,L)$ to consist of pairs formed by an automorphism
$\phi$ of $X$ and a fiberwise-linear bundle isomorphism
$\widetilde\phi:L\to L$ covering it, and prove
that
$(\phi,\widetilde\phi)s=\widetilde\phi\circ s\circ\phi^{-1}$ acts on sections
and functorially on every sheaf-cohomology group.  Construct \v Cech
cohomology from an open cover.  Prove the affine vanishing theorem for
quasi-coherent sheaves directly from localization, deduce that affine covers
compute their sheaf cohomology, prove independence of the chosen affine cover
by the refinement double complex, and derive the long exact sequence from a
short exact sequence of sheaves.  Identify divisors modulo principal divisors
with the Picard group.
\end{exercise}

\begin{exercise}[Serre duality and Riemann--Roch]
For a smooth projective curve $X$, construct the residue pairing
\[
 H^1(X,L)\otimes H^0(X,K_X\otimes L^{-1})\longrightarrow\C
\]
from \v Cech representatives and prove it is perfect.  Prove Riemann--Roch by
induction on a divisor using the exact sequence at a point:
\[
 h^0(L)-h^1(L)=\deg L+1-g.
\]
Define the holomorphic Euler characteristic
$\chi(X,L)=\sum_q(-1)^q\dim H^q(X,L)$.
Deduce $\deg K_X=2g-2$, the vanishing theorem for $\deg L>2g-2$, and the
dimension of meromorphic functions with prescribed poles.
\end{exercise}

\begin{exercise}[Dolbeault cohomology and Serre duality]
Define a complex manifold by a holomorphic atlas and a holomorphic vector
bundle by its locally free sheaf of sections.  Decompose complex differential
forms into bidegrees and define $\partial$ and $\bar\partial$.  Prove the
$\bar\partial$-Poincar\'e lemma, construct the fine Dolbeault resolution, and
identify coherent sheaf cohomology of a holomorphic bundle $E$ with
$H^{0,q}_{\bar\partial}(X,E)$.  On a compact complex manifold with Hermitian
metric, construct the formal adjoint and Dolbeault Laplacian from the metric
volume form, establish elliptic regularity from local Fourier estimates and a
partition of unity, and prove the harmonic-representative theorem.  Use the
Hodge star and integration pairing to prove the perfect duality
\[
 H^q(X,E)\otimes H^{d-q}(X,K_X\otimes E^\vee)\longrightarrow\C
\]
for $d=\dim_\C X$, recovering the curve theorem as the case $d=1$.
\end{exercise}

\begin{exercise}[Bott--Borel--Weil for $SL_2$]
Identify $SL_2(\C)/B\simeq\mathbb P^1$ and compute
\[
 H^q(\mathbb P^1,\mathcal O(n))=
 \begin{cases}
  \operatorname{Sym}^n(\C^2)^\vee,&q=0, n\ge0,\\
  \operatorname{Sym}^{-n-2}(\C^2),&q=1, n\le-2,\\
  0,&\text{otherwise},
 \end{cases}
\]
using Riemann--Roch and Serre duality.  Recover the shifted reflection
$s\mathbin\cdot n=-n-2$.
\end{exercise}

\begin{exercise}[Bott--Borel--Weil for reductive groups]
For general reductive $G$, cover $G/B$ by Bruhat
cells, filter its \v Cech complex by Bruhat order, identify the associated
graded terms with Verma modules, and prove that singular $\lambda+\rho$ gives
zero while regular $\lambda+\rho$ gives
\[
 H^{\ell(w)}(G/B,L_\lambda)
 \simeq V_{w(\lambda+\rho)-\rho}^\vee
\]
for the unique $w$ carrying it to the dominant chamber.
\end{exercise}

\begin{exercise}[Twisted differential operators on $\mathbb P^1$]
Derive from the M\"obius action on $\mathbb P^1$ the operators on
$\Gamma(\mathbb A^1,\mathcal O(n))$
\[
 e=\partial_z,\qquad h=-2z\partial_z+n,\qquad
 f=-z^2\partial_z+nz,
\]
and verify the $\mathfrak{sl}_2$ relations and invariance of polynomials of
degree at most $n$.  Define
$\Omega=\tfrac12h^2+ef+fe$, compute its scalar, and define the resulting
infinitesimal character.  Construct the twisted differential-operator sheaf
$\mathcal D_n$ by gluing the displayed chart operators.
\end{exercise}

\begin{exercise}[Beilinson--Bernstein localization]
For a weight $\lambda$, define $\mathcal D_\lambda$ by twisting differential
operators with $L_\lambda$.  Construct the order filtration, identify its
associated graded with functions
on $T^*(G/B)$.  For a finite-dimensional space of commuting endomorphisms,
construct its Koszul complex intrinsically by exterior powers, prove its
exactness for a regular sequence, and apply it to the moment-map fibers to
prove
\[
 \Gamma(G/B,\mathcal D_\lambda)
 \simeq U(\mathfrak g)/U(\mathfrak g)\ker\chi_\lambda.
\]
For regular dominant $\lambda$, prove exactness and faithfulness of global
sections by the Bruhat filtration and deduce the localization equivalence.
Recover Borel--Weil from the localized highest-weight module.
\end{exercise}

\begin{exercise}[Bergman kernels]
Give $X$ a Hermitian area form and $L$ a positive Hermitian holomorphic line
bundle.  Define the orthogonal projection
$L^2(X,L)\to H^0(X,L)$ by its intrinsic integral kernel $B_L(x,y)$.
Prove its reproducing, Hermitian, and projection identities.  Define
$\Lambda$ as contraction with the area form.  Derive the
extremal characterization
$B_L(x,x)=\sup\{\abs{s(x)}^2:\norm s_{L^2}=1\}$ and, from rescaling the heat
kernel of the Dolbeault Laplacian, prove for $L^k$
\[
 B_{L^k}(x,x)=\frac{k}{2\pi}\ii\Lambda F_L+O(1)
\]
uniformly.  On $\mathbb P^1$ and on a complex torus compute the kernel
explicitly, reusing rather than redefining the earlier theta functions.
\end{exercise}

\begin{exercise}[Landau-level degeneracy and density]
Define a magnetic line bundle for a charge $q$ on a compact Riemann surface
to be a positive Hermitian holomorphic line bundle $L$.  Define its Chern
connection as the unique unitary connection whose $(0,1)$ part is
$\bar\partial_L$, prove existence and uniqueness, and require its curvature
to be $F_L=-\ii qB/\hbar$, so that
$\frac{q}{2\pi\hbar}\int_XB=N\in\Z$.  Define its unitary gauge group as the
unitary bundle automorphisms over $\id_X$, prove that it acts on the quantum
state space, and show that automorphisms in
$\operatorname{Aut}(X,L)$ preserving the metric and connection commute with
the magnetic Hamiltonian.  Factor its magnetic
Laplacian through the Dolbeault operator and identify the lowest Landau level
with $H^0(X,L)$.  When $N>2g-2$, prove
\[
 \dim H^0(X,L)=N+1-g,\qquad
 \rho(x)=B_L(x,x).
\]
For $X=\mathbb P^1$ and $L=\mathcal O(N)$ with its invariant metric, use the
$SU(2)$ action, Bott--Borel--Weil, and the Bergman projection identity to
identify the state representation with
$\operatorname{Sym}^N(\C^2)^\vee$ and prove the constant density
$\rho=(N+1)/\operatorname{area}(X)$.  Thus predict both the exact number of
lowest-level states and their local particle density; state flux
quantization, curvature, boundary, interaction, and field-uniformity
assumptions.
\end{exercise}

\section{Baker--Akhiezer Functions}

\begin{exercise}[KdV Lax flows]
Let the translation circle act on the phase space
$\mathcal U=C^\infty(\R/a\Z,\R)$ of periodic potentials by pullback.  On
periodic smooth functions define
$L=-\partial_x^2+u$ and
$P=-4\partial_x^3+3(u\partial_x+\partial_xu)$.  Compute the commutator and
prove that $L_t=[P,L]$ is
\[
 u_t-6uu_x+u_{xxx}=0.
\]
Prove that the Lax vector field commutes with translations.  Construct the
monodromy of $L\psi=E\psi$, prove that its discriminant is
constant under the flow, and show that the periodic spectrum is preserved.
Derive the commuting odd operator for a stationary higher KdV flow and the
Burchnall--Chaundy polynomial relation.
\end{exercise}

\begin{exercise}[Hyperelliptic spectral curves]
Normalize the projective closure of
$y^2=\prod_{j=0}^{2g}(E-E_j)$ and include its smooth chart at infinity.
Choose real ordered branch points and construct the real ovals.  Normalize
the holomorphic differentials by a symplectic homology marking and prove from
the bilinear relations that the period matrix is symmetric with positive
imaginary part.  Define
\[
 \Theta(z\mid\Omega)=\sum_{n\in\Z^g}
 e^{\pi\ii n^T\Omega n+2\pi\ii n^Tz}
\]
and derive its quasi-periodicity.  State and prove the symplectic change of
marking formula and the invariance of the reconstructed potential below.
\end{exercise}

\begin{exercise}[Baker--Akhiezer functions]
At the point over $E=\infty$, put $k=\sqrt E$.  Construct normalized
second-kind differentials with principal parts $\dd(\ii k)$ and
$\dd(4\ii k^3)$ and period vectors $U,V$.  By Riemann--Roch prove uniqueness
of the function with a prescribed nonspecial pole divisor and essential
factor $e^{\ii kx+4\ii k^3t}$.  Derive its theta formula and show directly
that it solves $L\psi=E\psi$ and $\psi_t=P\psi$.  Compare the coefficient at
infinity to prove
\[
 u(x,t)=c-2\partial_x^2\log\Theta(Ux+Vt+Z\mid\Omega)
\]
and hence the stated sign convention for KdV.
\end{exercise}

\begin{exercise}[Bloch--Floquet decomposition]
For a real periodic potential of period $a$, define the translation by $a$
and decompose $L^2(\R)$ over its $U(1)$ characters.  Prove that
$\Delta(E)=\Tr M(E)$ satisfies
$\Delta(E)=2\cos(ka)$ on a Bloch wave and that the spectrum is exactly
$\{E:\abs{\Delta(E)}\le2\}$.  Show that simple branch points of the spectral
curve are band edges and that closing all but $g$ gaps is equivalent to the
finite-genus polynomial relation above.
\end{exercise}

\begin{exercise}[Lam\'e band-gap spectroscopy]
For
\[
 L_k=-\frac{\dd^2}{\dd x^2}+2k^2\operatorname{sn}^2(x;k)
\]
of period $2K(k)$, use the Jacobi identities to prove that
$\operatorname{dn}$, $\operatorname{cn}$, and $\operatorname{sn}$ have band
edge energies $k^2,1,1+k^2$.  Prove
\[
 \spec(L_k)=[k^2,1]\cup[1+k^2,\infty).
\]
After restoring a spatial scale $\kappa$ and factor $\hbar^2/(2m)$, predict
the two band widths and the gap width
$\hbar^2\kappa^2k^2/(2m)$.  State how Bloch spectroscopy measures the three
edge energies and distinguish the one-gap periodic-potential model from
lattice, interaction, and finite-size corrections.
\end{exercise}

\part{Algebraic Topology}

\section{Monopole Harmonics}

\begin{exercise}[Homotopy, CW complexes, and vector bundles]
Define pointed homotopy, homotopy equivalence, loop and suspension spaces,
homotopy groups, and CW complexes.  Prove cellular approximation and the long
exact homotopy sequence of a fibration.  Define a topological complex vector
bundle intrinsically by local product structure and fiberwise linear
operations.  Construct the infinite Grassmannian from its finite-dimensional
subspace functor and prove that homotopy classes $[X,BU(n)]$ classify rank-$n$
bundles over a finite CW complex.  Define $K^0(X)$ by group completion under
direct sum, prove homotopy invariance and the splitting principle, and derive
its reduced suspension and clutching descriptions.  When $X$ is smooth,
define a smooth vector bundle and prove by approximation and partitions of
unity that every topological bundle is isomorphic to one with compatible
smooth structure.
\end{exercise}

\begin{exercise}[$U(1)$ gauge symmetry and characteristic classes]
For the smooth vector bundles just constructed, define connections, curvature,
and gauge transformations intrinsically.  Prove that the gauge group acts on
the affine space of connections and by conjugation on every naturally coupled
elliptic operator, preserving its spectrum and index.  From invariant
polynomials and the Bianchi
identity prove that $\Tr\exp(\ii F/2\pi)$ is closed and connection-independent
in cohomology.  Construct integral Chern classes from transition functions
and prove that their real images agree with Chern--Weil forms.  Use the
splitting principle to derive the Chern character and Todd and $\widehat A$
classes and prove their multiplicative identities.
\end{exercise}

\begin{exercise}[Fredholm operators and elliptic symbols]
Define Fredholm operators and their analytic index, prove stability under
compact perturbations, and construct the symbol class of an elliptic operator
in $K^0_c(T^*M)$.  Construct the bounded transform and prove that its index
depends only on the compactly supported symbol class.
\end{exercise}

\begin{exercise}[Bott periodicity and the topological index]
Construct the harmonic-oscillator Clifford symbol on a complex vector space,
compute its one-dimensional kernel, and prove that external product with its
class gives the Bott isomorphism by radial contraction and Mayer--Vietoris.
Construct Thom isomorphisms by the same Clifford class and define the
topological index by embedding $M$ in Euclidean space.  Prove independence of
the embedding by tubular neighborhoods and isotopy.
\end{exercise}

\begin{exercise}[Atiyah--Singer index equality]
Prove equality of analytic and topological indices first for the Bott
operator, then for external products and open excision, and finally for every
compact manifold by embedding and retraction.  Apply the splitting principle
and Chern--Weil theory to deduce
\[
 \operatorname{ind}\bar\partial_E
 =\int_M\operatorname{ch}(E)\operatorname{Td}(T^{1,0}M)
\]
on a compact complex manifold, and derive the $\widehat A$ formula for a
spin Dirac operator.
\end{exercise}

\begin{exercise}[K3 elliptic genus]
Define a K3 surface as a simply connected compact complex surface with
trivial canonical bundle.  Apply the Dolbeault index formula to the trivial
bundle, use Serre duality, and prove $\int_{K3}c_2=24$ and
$\widehat A(K3)=2$.  With
$y=e^{2\pi\ii z}$ and $q=e^{2\pi\ii\tau}$, define its elliptic genus by
\[
 y^{-1}\chi\!\left(K3,\bigotimes_{n\ge1}
 \Lambda_{-yq^{n-1}}T^*\otimes\Lambda_{-y^{-1}q^n}T
 \otimes S_{q^n}T^*\otimes S_{q^n}T\right).
\]
Here $\Lambda_t$ and $S_t$ are the total exterior- and symmetric-power
operations in completed $K$-theory.
Extend the theta constants to Jacobi theta functions by inserting
$e^{2\pi\ii(n+a)z}$ in their characteristic lattice sums, and thereby define
$\vartheta_1(z;\tau)$.  Put
\[
 \phi_{0,1}(z,\tau)=4\left[
 \frac{\vartheta_2(z;\tau)^2}{\vartheta_2(0;\tau)^2}
 +\frac{\vartheta_3(z;\tau)^2}{\vartheta_3(0;\tau)^2}
 +\frac{\vartheta_4(z;\tau)^2}{\vartheta_4(0;\tau)^2}\right].
\]
Use Riemann--Roch to derive the Chern-root theta product.  Expand it,
integrate, and prove
$\operatorname{Ell}_{K3}(z,\tau)=2\phi_{0,1}(z,\tau)$.  Derive its first
Fourier--Jacobi coefficients from the theta identities already established.
\end{exercise}

\begin{exercise}[Monopole harmonics]
Let $L_N\to S^2$ be the homogeneous line bundle associated with the weight
$N$ character of $U(1)\subset SU(2)$.  Use Peter--Weyl, rather than polar
gauges, to decompose $L^2(S^2,L_N)$ into the $SU(2)$ representations
$V_j$, $j=\abs N/2,\abs N/2+1,\ldots$.  Define the monopole harmonics as the
corresponding distinguished matrix coefficients and derive their
orthogonality, completeness, addition kernel, and magnetic-Laplacian values
\[
 \lambda_j=\frac1{R^2}\left(j(j+1)-\frac{N^2}{4}\right).
\]
Identify the lowest level and its degeneracy $\abs N+1$ with Riemann--Roch.
\end{exercise}

\begin{exercise}[Integer quantum Hall conductance]
For a gapped family of Fermi projections $P(k)$ over the Brillouin torus,
derive the Kubo response from adiabatic perturbation theory and prove
\[
 \sigma_{xy}=\frac{e^2}{h}\frac{\ii}{2\pi}
 \int_{\T^2}\Tr(P\dd P\wedge\dd P)
 =C\frac{e^2}{h},\qquad C\in\Z.
\]
Relate $C$ to spectral flow by applying the index theorem to a flux insertion.
For filled monopole-harmonic Landau levels, compute $C$ from the associated
line bundle and predict the Hall plateau and level degeneracy.  Separate the
gap, noninteracting-band, zero-temperature, and disorder-localization
assumptions from the measured transverse conductance.
\end{exercise}

\section{Eta Functions}

\begin{exercise}[Spectra and stable homotopy]
Using pointed CW complexes and the loop--suspension adjunction already
developed, construct spectra and stable homotopy groups by iterated
suspension and prove the Freudenthal stable range.  For an abelian group $A$,
construct $K(A,n)$ inductively by classifying spaces, prove that its only
nonzero homotopy group is $\pi_n\simeq A$, and use the loop equivalences
$K(A,n)\simeq\Omega K(A,n+1)$ to define the Eilenberg--Mac Lane spectrum
$HA$.  Identify ordinary homology and cohomology with the theories represented
by $HA$.  Construct generalized
homology and cohomology from a spectrum and prove their exactness and
suspension axioms.
\end{exercise}

\begin{exercise}[$SO$, $\operatorname{Spin}$, and $U$ structures]
Construct the orthogonal frame bundle of a Riemannian manifold with its free
right $O(d)$ action.  Define a tangential $G$-structure as a lift of the
stable tangent classifier through $BG\to BO$, equivalently as the appropriate
reduction of this frame symmetry.  Construct orientation, spin, and stable
complex
structures as the cases $SO$, $\operatorname{Spin}$, and $U$, and identify
the obstruction cocycles over the skeleta of $BO$.  Define their universal
classes $w_1,w_2$, prove naturality and the Whitney product formula, and show
that they are precisely the orientation and spin obstructions; identify the
stable complex lifting condition.
Define $G$-bordism by compact manifolds with compatible boundary structure
and prove it is a graded ring under disjoint union and product.
\end{exercise}

\begin{exercise}[Higher categories and bordisms]
Define an $(\infty,n)$-category as a complete $n$-fold Segal space: state the
Segal composition maps, completeness condition, and the requirement that
$k$-morphisms above level $n$ be invertible.  Define symmetric monoidal
structures by commutative algebra objects in this model and construct the
$(\infty,n)$-category whose objects are compact $0$-manifolds and whose
$k$-morphisms are collared $k$-bordisms, diffeomorphisms, isotopies, and
higher isotopies.  Prove associativity of gluing from contractibility of
collar choices and identify disjoint union as its symmetric tensor product.
\end{exercise}

\begin{exercise}[Pontryagin--Thom decomposition]
For an embedding of a closed $G$-manifold in Euclidean space, collapse the
complement of a tubular neighborhood and obtain a map to the Thom space of
its normal bundle.  Prove independence under stabilization and bordism.
Conversely, make a stable map transverse to the zero section and recover a
$G$-manifold.  Prove the isomorphism between $G$-bordism and the stable
homotopy groups of the Thom spectrum $MG$.  Derive complex orientations and
the formal group law of $MU$ from tensor product of line bundles.
\end{exercise}

\begin{exercise}[Fully extended field theories]
Define adjoints for $k$-morphisms in a symmetric monoidal
$(\infty,n)$-category and
call an object fully dualizable when it has evaluation and coevaluation at
every level with all required adjoints.  Construct framed bordisms by handles
and prove that evaluation on a point gives an equivalence between fully
extended framed topological field theories and fully dualizable objects.
Add homotopy fixed-point data for a tangential $G$-structure.  Verify in
dimension one that the snake identities defining finite-dimensional vector
spaces at the beginning of the manuscript are exactly the required point
data.
\end{exercise}

\begin{exercise}[Eta functions and the APS boundary correction]
For a self-adjoint elliptic operator $D$ with discrete nonzero eigenvalues,
define
\[
 \eta_D(s)=\sum_{\lambda\ne0}\operatorname{sign}(\lambda)\abs\lambda^{-s}
\]
for $\Re s$ large.  Use the heat kernel Mellin transform to continue it to
$s=0$ and prove regularity there.  Attach a cylinder to a compact manifold
with boundary, analyze separated boundary modes, and prove
\[
 \operatorname{ind}D^+_{\rm APS}
 =\int_M\widehat A(TM)\operatorname{ch}(E)
 -\frac{\eta_{D_\partial}(0)+\dim\ker D_\partial}{2}.
\]
For the circle operator with holonomy $e^{2\pi\ii a}$ derive
$\eta(0)=1-2a$ for $0<a<1$ as follows: define
$\zeta(s,a)=\sum_{n\geq0}(n+a)^{-s}$ for $\Re s>1$, use the Mellin transform
of the shifted Gaussian and Poisson summation to continue it and derive its
functional equation, and evaluate the resulting difference at $s=0$.
\end{exercise}

\begin{exercise}[Invertible phases and eta-response predictions]
Define an invertible reflection-positive field theory as a symmetric
monoidal map from the appropriate bordism category to invertible complex
lines, and use stacking to form its classification group.  Apply
Pontryagin--Thom to identify deformation classes with maps from the relevant
Thom spectrum to the Picard spectrum obtained by iteratively delooping the
grouplike symmetric monoidal $\infty$-groupoid of invertible phases, including the
symmetry structure in the tangential data.  For a
massive fermion whose boundary operator is $D_\partial$, derive the relative
partition phase
\[
 Z_{\rm top}=\exp\!\left[-\frac{\pi\ii}{2}
 (\eta_{D_\partial}(0)+\dim\ker D_\partial)\right].
\]
Use the circle calculation to predict its change under one unit of flux and
the resulting protected boundary spectral flow.  State locality, gap,
symmetry, and interaction assumptions separately from the measurable phase
and crossing parity.
\end{exercise}

\part{Tensor Categories}

\section{Hall--Littlewood Polynomials}

\begin{exercise}[Loop groups and Schubert cells]
For a complex reductive group $G$, define
$G((z))$, $G[[z]]$, the faithfully flat quasi-compact topology on affine
schemes, and sheafification of a functor of points.  Define the fpqc quotient
$\operatorname{Gr}_G=G((z))/G[[z]]$ and prove that $G((z))$ acts transitively
with stabilizer $G[[z]]$.  For the induced $G[[z]]$ action, prove the Cartan
decomposition
\[
 \operatorname{Gr}_G=\bigsqcup_{\lambda\in X_*(T)_+}
 G[[z]]z^\lambda
\]
and compute the orbit dimension $\langle2\rho,\lambda\rangle$.  Construct
the Schubert varieties by orbit closure and derive their dominance order and
convolution correspondence from modification of $G$-bundles on a formal disk.
\end{exercise}

\begin{exercise}[Constructible derived categories and Verdier duality]
For a finite stratification of a locally compact space, define bounded
constructible complexes of sheaves and chain homotopies.  Localize the
homotopy category at quasi-isomorphisms to construct the constructible derived
category.  Resolve by injective and soft sheaves to construct
$Rf_*,Rf_!,Lf^*,f^!,\otimes^L$, and $R\mathcal Hom$; derive their adjunctions,
the projection formula, and proper base change from the corresponding
statements for the resolutions.  Construct the dualizing complex from compactly
supported cochains and define Verdier duality
$\mathbb D\mathcal F=R\mathcal Hom(\mathcal F,\omega_X)$.  Prove biduality and
the interchange of $Rf_*$ with $Rf_!$ under $\mathbb D$.
\end{exercise}

\begin{exercise}[Perverse sheaves and convolution]
For the Schubert stratification, define the perverse truncation by the support
and cosupport inequalities.  Prove that intersection cohomology
extensions are simple perverse sheaves.  Define convolution through the
proper multiplication correspondence, prove perversity by semismallness, and
derive associativity, commutativity from fusion on two moving points, and
rigid duals from inversion and Verdier duality.
\end{exercise}

\begin{exercise}[Satake fiber functor]
Let $\operatorname{Sat}_G$ be the category of $G[[z]]$-equivariant perverse
sheaves on $\operatorname{Gr}_G$.  Prove exactness and faithfulness of total
cohomology by the weight-functor decomposition over the semi-infinite orbits
$S_\mu=N((z))z^\mu$, where $N$ is the unipotent radical selected by the
positive roots; prove the required dimension and transversality statements
for their intersections with Schubert varieties.
Use convolution to make it a symmetric monoidal fiber functor.  Construct the
group of its tensor automorphisms, recover its coordinate Hopf algebra from
matrix coefficients and rigid duals, and prove the Tannaka equivalence with
representations of a reductive group.
\end{exercise}

\begin{exercise}[Geometric Satake equivalence]
Compute the roots of this Tannaka group from the
rank-one Schubert varieties and prove that its root datum is dual to that of
$G$.  Deduce the monoidal equivalence
\[
 \operatorname{Sat}_G\simeq\operatorname{Rep}(G^\vee)
\]
and identify intersection cohomology of $\overline{\operatorname{Gr}}_\lambda$
with the irreducible highest-weight representation $V_\lambda$.
\end{exercise}

\begin{exercise}[Hall--Littlewood polynomials]
For a partition $\lambda$ define
\[
 P_\lambda(x;t)=\frac1{v_\lambda(t)}
 \sum_{w\in S_r}w\!\left(x^\lambda
 \prod_{i<j}\frac{x_i-tx_j}{x_i-x_j}\right)
\]
with $v_\lambda(t)$ fixed by monicity.  Prove polynomiality, triangularity,
orthogonality, the Cauchy identity, and the Pieri rule.  Over a finite field,
stratify the convolution fibers in $\operatorname{Gr}_{GL_r}$ and show that
their point counts give the Hall--Littlewood structure constants at
$t=\abs{\mathbb F}^{-1}$.  At $t=0$ recover the Weyl characters.
\end{exercise}

\begin{exercise}[Gauge fields and 't Hooft defects]
Choose a compact form $K$ of the complex reductive group $G$ above and a
compatible maximal torus $T_K\subset K$ whose complexification is $T$.  Define a
principal $K$-bundle $P\to M$ over an oriented Riemannian four-manifold as a
smooth fiber bundle with a free transitive right $K$-action on each fiber.
Define a gauge field as a $K$-equivariant horizontal distribution on $P$,
equivalently a connection form $A$ satisfying
$R_k^*A=\Ad(k^{-1})A$ and $A(\xi_P)=\xi$, where $\xi_P$ is the
fundamental vertical vector field generated by $\xi\in\mathfrak k$.  Define
its curvature by
$F_A=\dd A+\frac12[A\wedge A]$ and a gauge transformation as a
$K$-equivariant bundle automorphism over $\id_M$.  Derive the induced action
on $A$ and $F_A$ and prove gauge invariance of the Yang--Mills action
\[
 S(A)=\frac{1}{2e^2}\int_M\langle F_A\wedge *F_A\rangle
\]
for an $\Ad K$-invariant inner product on $\mathfrak k$.

For an oriented embedded curve $C\subset M$, define an 't Hooft defect of
charge $\lambda\in X_*(T)$ by requiring a Dirac-type singularity on every
small two-sphere linking $C$: its reductions to $T_K$ over the northern and
southern hemispheres are glued on the equator by
$e^{\ii\theta}\mapsto\lambda(e^{\ii\theta})$.  Treat this reduction and its
singular asymptotics, rather than merely the isomorphism class of the
underlying $K$-bundle, as the boundary condition.  Prove that conjugate
cocharacters give gauge-equivalent defects and that every charge has a unique
dominant representative modulo the Weyl group.

Define the formal holomorphic local model of this boundary condition on a
transverse disk with parameter $z$ to be a $G$-bundle on
$\operatorname{Spec}\C[[z]]$ together with a trivialization over
$\operatorname{Spec}\C((z))$.  Show that its transition function lies in
$G((z))$, that changes of regular trivialization act by $G[[z]]$, and that the
modification of charge $\lambda$ lies in the Schubert orbit of
$z^\lambda$ in $\operatorname{Gr}_G$.  Show that two ordered defects on the
disk determine two successive modifications, so bringing their support
together gives precisely the convolution correspondence constructed above.
State separately the compact-gauge-group, topological--holomorphic reduction,
and point-defect idealization assumptions in this local model.
\end{exercise}

\begin{exercise}['t Hooft-line fusion]
Model the protected magnetic line operators constructed above by
$\operatorname{Sat}_G$, and let $L_\lambda$ be the Grothendieck class of the
intersection-cohomology sheaf of
$\overline{\operatorname{Gr}}_\lambda$.  Use the convolution description of
collision and geometric Satake to prove in the protected fusion ring that
\[
 L_\lambda L_\mu=\sum_\nu
 \dim\Hom_{G^\vee}(V_\nu,V_\lambda\otimes V_\mu)L_\nu.
\]
Interpret the coefficient as the dimension of the corresponding space of
local junction operators.  Compute it from Hall--Littlewood multiplication at
$t=0$ and specialize to $G=PGL_2$, with $V_n$ the irreducible
$SU(2)$-representation of highest weight $n$, to obtain
\[
 L_mL_n=\sum_{j=0}^{\min(m,n)}L_{m+n-2j}.
\]
Thus predict the allowed magnetic charges, the unit multiplicity of every
allowed junction channel, and the forbidden parity and triangle-inequality
channels.  State how defect-correlation measurements resolve these channels,
and separate the protected multiplicities from coupling-dependent operator
normalizations and from corrections to the local-defect idealization.
\end{exercise}

\section{Modular S-Matrices}

\begin{exercise}[Rigid and unitary fusion categories]
Define a unitary fusion category as a finite semisimple rigid
$\C$-linear $C^*$-tensor category with simple tensor unit.  Define evaluation,
coevaluation, pivotal and spherical structures, quantum dimensions, and the
global dimension.  Derive the snake, rigidity, Frobenius reciprocity, and
$\sum_cd_{ab}^cd_c=d_ad_b$ identities intrinsically.  Define braidings,
twists, and nondegeneracy through the categorical trace of double braiding.
Show that convolution categories of topological line defects furnish
examples, while noninvertible simple objects require no underlying group.
For simple objects $a_1,\ldots,a_n,b$, define the fusion state space
\[
 \mathcal H(a_1,\ldots,a_n;b)
 =\Hom(b,a_1\otimes\cdots\otimes a_n).
\]
Use $f^*g=\langle f,g\rangle1_b$ to make it a Hilbert space.  Prove that
associators and braidings induce unitary maps among these fusion spaces and
derive the resulting braid-group action and its invariance under isotopy.
\end{exercise}

\begin{exercise}[Temperley--Lieb categories and fusion semisimplification]
Define $\operatorname{TL}_d$ by planar pairings and loop value $d$, and prove
the tensor and $*$ structures by gluing diagrams.  Construct the Jones--Wenzl
idempotents recursively and derive
$f_1\otimes f_n\simeq f_{n-1}\oplus f_{n+1}$.  At
$d=2\cos(\pi/(k+2))$, identify the negligible ideal generated by $f_{k+1}$
and prove that the semisimplification has simples $0,\ldots,k$ and fusion
\[
 a\otimes b=\bigoplus_{\substack{c=\abs{a-b}\ (2)\\
 c\le\min(a+b,\,2k-a-b)}}c.
\]
Construct the ribbon functor to the root-of-unity tilting quotient developed
in Quantum Groups by sending one strand to $V_1$, prove that its kernel is the
negligible ideal generated by $f_{k+1}$, and identify the two
semisimplifications.
\end{exercise}

\begin{exercise}[Factorization homology and the Levin--Wen model]
Define a disk algebra in a symmetric monoidal higher category and construct
factorization homology by the colimit over embedded disks.  Prove excision by
cofinality of collar gluings.  Define the Drinfeld center $Z(\mathcal C)$ to
have objects $(x,c_{x,-})$ consisting of an object and a natural half-braiding
satisfying the hexagon relation.  From the simple labels, quantum dimensions,
and associators of $\mathcal C$, construct the Turaev--Viro state sum and
prove its invariance under Pachner moves from the pentagon and spherical
identities.  Construct its bordism maps by gluing, regard the result as the
three-dimensional Turaev--Viro theory, and prove that the category it assigns
to a circle is $Z(\mathcal C)$.  Triangulate a closed surface,
define the Levin--Wen vertex and plaquette projectors from the $F$-symbols,
and prove their commutativity by the pentagon identity.  Show that the
ground-state space is the corresponding factorization-homology state space.
Verify explicitly that $\mathcal C=SU(2)_1$ is the semion category and its
Levin--Wen theory is doubled semion, not the toric code.
\end{exercise}

\begin{exercise}[Modular $S$-matrix and the Verlinde decomposition]
For $\mathcal C=SU(2)_k$, compute the categorical trace of double braiding
from the root-of-unity ribbon structure constructed in Quantum Groups and
recover the finite sine transform
obtained from affine characters:
\[
 S_{ab}=\sqrt{\frac{2}{k+2}}
 \sin\!\left(\frac{(a+1)(b+1)\pi}{k+2}\right),
 \qquad
 \theta_a=e^{\pi\ii a(a+2)/(2(k+2))}.
\]
Prove directly that the two constructions agree, then use its orthogonality
to deduce nondegeneracy of braiding and the Verlinde formula
\[
 N_{ab}^{\phantom{ab}c}=\sum_x
 \frac{S_{ax}S_{bx}\overline{S_{cx}}}{S_{0x}}.
\]
Recover the truncated fusion rule and compute the torus ground-state
degeneracy.  Relate these trigonometric special functions to the root-of-unity
limit of the quantum-group characters without redefining them.
\end{exercise}

\begin{exercise}[Anyon interferometry]
For simple anyons $a,b$, prove that a probe $a$ encircling total charge $b$
has normalized monodromy
\[
 M_{ab}=\frac{S_{ab}S_{00}}{S_{0a}S_{0b}}.
\]
In a balanced two-path interferometer with controllable phase $\phi$, derive
\[
 P_\pm(b)=\frac12\left(1\pm\Re(e^{\ii\phi}M_{ab})\right).
\]
When $\mathcal C$ is modular, identify the simple charges of the Drinfeld
center realized by its Levin--Wen model with pairs and prove
\[
 S^{Z(\mathcal C)}_{(a,b),(c,d)}
 =S_{ac}\overline{S_{bd}}.
\]
Insert the $SU(2)_k$ sine matrix into the chiral and doubled monodromy formulas
to predict every fringe displacement and visibility and show which charges
are distinguishable by a fixed probe.
State coherence, braiding adiabaticity, charge-superselection, path balance,
and detector-resolution assumptions separately from these probabilities.
\end{exercise}

\section{Heisenberg Group}

\subsection{Central Extensions}


\begin{exercise}[Weyl multipliers]
A \emph{normalized multiplier} on a locally compact group $G$ is a
continuous map
$\sigma:G\times G\to\T$ satisfying
\[
\sigma(x,e)=\sigma(e,x)=1,
\qquad
\sigma(x,y)\sigma(xy,z)=\sigma(y,z)\sigma(x,yz).
\]
Let $(E,\omega)$ be a symplectic vector space.  For
$E=V\oplus V^*$, the canonical symplectic form used below is
\[
\omega\bigl((q,p),(q',p')\bigr)=p(q')-p'(q).
\]
For $\lambda\in\R$, define
\[
\sigma_{\lambda\omega}(z,z')=e^{i\lambda\omega(z,z')/2}.
\]

Prove that $\sigma_{\lambda\omega}$ is a normalized multiplier for every
$\lambda\in\R$.
\end{exercise}


\begin{exercise}[Commutator bicharacter]
For a multiplier $\sigma$ on an abelian group $A$, define
\[
c_\sigma(x,y)=\sigma(x,y)\overline{\sigma(y,x)}.
\]
Multipliers $\sigma$ and $\sigma'$ are cohomologous when there is a
continuous map $b:A\to\T$, $b(e)=1$, with
\[
\sigma'(x,y)=b(x)b(y)\overline{b(x+y)}\,\sigma(x,y).
\]
For $E=V\oplus V^*$, call $\sigma_{\omega/\hbar}$ the Weyl multiplier.

Prove that $c_\sigma$ is an alternating bicharacter and depends only on the
cohomology class of $\sigma$.  For the Weyl multiplier on $V\oplus V^*$,
prove
\[
c_{\sigma_{\omega/\hbar}}\bigl((q,p),(q',p')\bigr)
=e^{i(p(q')-p'(q))/\hbar}.
\]
Prove
\[
c_{\sigma_{\lambda\omega}}(z,z')=e^{i\lambda\omega(z,z')}.
\]
\end{exercise}


\begin{exercise}[Heisenberg groups and central characters]
A $\sigma$-projective unitary representation of a group $E$ is a strongly
continuous map $U:E\to U(H)$ satisfying
$U_zU_{z'}=\sigma(z,z')U_{z+z'}$ and $U_0=I$.

For a symplectic vector space $(E,\omega)$, its \emph{Heisenberg group} is
\[
\mathbb H(E,\omega)=E\times\R
\]
with multiplication
\[
(z,s)(z',s')
=(z+z',s+s'+\tfrac12\omega(z,z')).
\]
Write
$\mathbb H_n=\mathbb H(\R^n\oplus(\R^n)^*,\omega)$ for the canonical
symplectic form on $\R^n\oplus(\R^n)^*$.  Its Lie algebra is
$\mathfrak h(E,\omega)=E\oplus\R Z$ with
\[
[(z,sZ),(z',s'Z)]=\omega(z,z')Z.
\]
Prove that $\mathbb H(E,\omega)$ is a locally compact group with center
$\{0\}\times\R$.  Prove that projective representations of $E$ with
multiplier $e^{i\lambda\omega/2}$ correspond to representations of
$\mathbb H(E,\omega)$ with central character $e^{i\lambda s}$.
\end{exercise}

\begin{exercise}[Heisenberg coadjoint orbits and Schr\"odinger polarization]
For a connected simply connected nilpotent group $G$ and
$\ell\in\mathfrak g^*$, a \emph{polarization} is a Lie subalgebra
$\mathfrak p\subseteq\mathfrak g$ satisfying
\[
\ell([\mathfrak p,\mathfrak p])=0,
\qquad
\dim\mathfrak p
=\frac{\dim\mathfrak g+\dim\mathfrak g_\ell}{2}.
\]
For $P=\exp(\mathfrak p)$, set
$\chi_\ell(\exp X)=e^{i\ell(X)}$.  The induced representation
$\operatorname{Ind}_P^G\chi_\ell$ acts by left translation on the
$L^2$-sections over $G/P$ satisfying
$F(gp)=\chi_\ell(p)^{-1}F(g)$.

Identify
\[
\mathfrak h(E,\omega)^*=E^*\oplus\R Z^*.
\]
For $(\xi,\lambda)\in E^*\oplus\R Z^*$, prove
\[
\Ad_{(z,s)}^*(\xi,\lambda)
=\bigl(\xi-\lambda\omega(z,{\cdot}),\lambda\bigr).
\]
Deduce that the orbits with $\lambda=0$ are points and that, for
$\lambda\ne0$,
\[
\mathcal O_\lambda
=\{(\xi,\lambda):\xi\in E^*\}.
\]
Prove that
\[
E\longrightarrow\mathcal O_\lambda,
\qquad
z\longmapsto\bigl(-\lambda\omega(z,{\cdot}),\lambda\bigr),
\]
pulls the Kirillov form back to $\lambda\omega$.

For $E=V\oplus V^*$ and
$\mathfrak p=V^*\oplus\R Z$, prove that $\mathfrak p$ is a polarization
at $(0,\lambda)$.  Identify
$\mathbb H(E,\omega)/\exp(\mathfrak p)$ with $V$ and compute the
induced action to obtain
\[
\bigl(\pi_\lambda(q,p,s)\psi\bigr)(x)
=e^{i\lambda(s+p(x-q/2))}\psi(x-q).
\]
Prove that this representation has central character $e^{i\lambda s}$.
For $\lambda=0$, recover every character of $E$ from a point orbit.
\end{exercise}

\begin{exercise}[Moyal algebra of a Heisenberg orbit]
Let $(E,\omega)$ be a symplectic vector space, let
$\Pi:E^*\to E$ be the inverse of $v\mapsto\omega(v,-)$, and define
\[
\{f,g\}=dg\bigl(\Pi(df)\bigr),
\qquad
\Pi(k,\ell)=\ell(\Pi(k)).
\]
For $\lambda\ne0$, use the orbit map
$E\to\mathcal O_\lambda$ to identify $\omega$ with the pullback of
$\lambda^{-1}$ times the Kirillov form.

A \emph{differential formal deformation} of $C^\infty(E)$ is an
associative $\mathbb C[[\hbar]]$-bilinear product on
$C^\infty(E)[[\hbar]]$ of the form
\[
f\star g=fg+\sum_{r\geq1}\hbar^rC_r(f,g),
\]
where the $C_r$ are bidifferential operators, $1$ is a unit, and
\[
C_1(f,g)-C_1(g,f)=i\{f,g\}.
\]
Let $D_\Pi$ be the constant bidifferential operator characterized on
decomposable functions by
\[
D_\Pi(f\otimes g)(x,y)=d_yg\bigl(\Pi(d_xf)\bigr).
\]
Define
\[
f\star g
=m\circ\exp\!\left(\frac{i\hbar}{2}D_\Pi\right)(f\otimes g),
\]
where $m$ restricts a function on $E\times E$ to the diagonal.  Prove
associativity from the commutation of the induced constant
bidifferential operators on $E^3$, and prove
\[
f\star g=fg+\frac{i\hbar}{2}\{f,g\}+O(\hbar^2).
\]

For $k,\ell\in E^*$ put $e_k(x)=e^{ik(x)}$ and derive
\[
e_k\star e_\ell
=e^{-i\hbar\Pi(k,\ell)/2}e_{k+\ell}.
\]
Check the multiplier identity directly and prove that every affine
symplectic transformation of $E$ preserves $\star$.  Call this intrinsic
formal algebra the \emph{Moyal algebra} of the Heisenberg orbit.
\end{exercise}

\begin{exercise}[Canonical commutation and Weyl relations]
For $\hbar>0$, $\ell\in V^*$, and $v\in V$, define on $\mathcal S(V)$
\[
Q_\ell\psi(x)=\ell(x)\psi(x),
\qquad
P_v\psi(x)=-i\hbar\,d\psi_x(v).
\]
For
\[
(T_q\psi)(x)=\psi(x-q),
\qquad
(M_p^\hbar\psi)(x)=e^{ip(x)/\hbar}\psi(x),
\]
prove that the corresponding one-parameter groups have infinitesimal
generators $-iP_v/\hbar$ and $iQ_\ell/\hbar$, respectively, and prove
\[
M_p^\hbar T_q=e^{ip(q)/\hbar}T_qM_p^\hbar.
\]
For
\[
W_\hbar(q,p)=e^{-ip(q)/(2\hbar)}M_p^\hbar T_q,
\]
prove
\[
W_\hbar(z)W_\hbar(z')
=e^{i\omega(z,z')/(2\hbar)}W_\hbar(z+z')
\]
and identify $W_\hbar(z)$ with
$\pi_{1/\hbar}(z,0)$ under the canonical set-theoretic section
$z\mapsto(z,0)$.  Prove that this section is not a homomorphism and that
its failure to preserve multiplication is exactly the displayed multiplier.
Differentiate the Weyl relations on $\mathcal S(V)$ to prove
\[
[Q_\ell,P_v]=i\hbar\ell(v)I.
\]
\end{exercise}


\subsection{Central-Character Decomposition}

\begin{exercise}[Irreducibility of the Schr\"odinger representation]
Prove that an operator on $L^2(V)$ commuting with every translation and
every modulation is scalar: first use the Fourier transform to identify the
commutant of translations, and then impose commutation with modulations.
Deduce that $\pi_\lambda$ is irreducible for $\lambda\ne0$.
\end{exercise}

\begin{exercise}[Stone--von Neumann theorem]
Let $U$ be a strongly continuous irreducible unitary representation of
$\mathbb H(V\oplus V^*,\omega)$ with central character
$U(0,s)=e^{i\lambda s}I$, where $\lambda\ne0$.  Integrate Schwartz
functions on $V\oplus V^*$ against the canonical section of $U$.  Verify in
the Schr\"odinger model that Fourier inversion sends such an integrated
operator with Schwartz symbol to an integral operator with Schwartz kernel.
Prove the Stone--von Neumann theorem~\cite{folland-phase}:
every such $U$ is
unitarily equivalent to $\pi_\lambda$.  Deduce that the nonzero central
character $\lambda$ determines the irreducible representation up to unitary
equivalence.
\end{exercise}

\begin{exercise}[Trace and Hilbert--Schmidt classes]
For $x,y\in H$, let
\[
\theta_{x,y}(\xi)=\langle y,\xi\rangle x.
\]
The \emph{trace class} $S_1(H)$ consists of operators admitting a representation
\[
A=\sum_{j=1}^{\infty}\theta_{x_j,y_j},
\qquad
\sum_{j=1}^{\infty}\|x_j\|\,\|y_j\|<\infty,
\]
where the series converges in operator norm, with norm the infimum of the
displayed sums.  Define
\[
\Tr(A)=\sum_{j=1}^{\infty}\langle y_j,x_j\rangle.
\]
The \emph{Hilbert--Schmidt class} is
\[
S_2(H)=\{A\in B(H):A^*A\in S_1(H)\},
\qquad
\|A\|_2^2=\Tr(A^*A).
\]

Prove that $S_1(H)$, $\Tr$, and $S_2(H)$ are well-defined and that every
trace-class operator is compact.  For a positive compact operator $A$,
use its spectral projections to prove
\[
A\in S_1(H)
\quad\Longleftrightarrow\quad
\sum_j\lambda_j(A)<\infty,
\qquad
\Tr(A)=\sum_j\lambda_j(A),
\]
where the nonzero spectral values are repeated with multiplicity.  Prove
that $\langle A,B\rangle_{S_2}=\Tr(A^*B)$ makes $S_2(H)$ a Hilbert space
and that the pairing $(A,T)\mapsto\Tr(AT)$ identifies $S_1(H)^*$
isometrically with $B(H)$.
\end{exercise}


\begin{exercise}[Coefficient and Fourier--Wigner transforms]
Let $V$ be an $n$-dimensional real vector space with Lebesgue measure $dq$,
let $\hbar>0$, and equip $V^*$ with the dual measure $dp$ for which
$(2\pi\hbar)^{-n/2}\int_Ve^{-ip(q)/\hbar}f(q)\,dq$ is unitary on $L^2$.
Use the translation, modulation, and Weyl operators
$T_q,M_p^\hbar,W_\hbar(q,p)$ already normalized by the Weyl relations.
The \emph{Fourier--Wigner transform} is
\[
\mathcal W_\hbar:S_1(L^2(V))\longrightarrow C_b(V\oplus V^*),
\qquad
\mathcal W_\hbar(A)(z)=(2\pi\hbar)^{-n/2}\Tr(AW_\hbar(z)).
\]

Prove
\[
W_\hbar(z)^*=W_\hbar(-z).
\]
For $A,B\in S_1(L^2(V))\cap S_2(L^2(V))$, prove the Fourier--Wigner
orthogonality identity
\[
\int_{V\oplus V^*}
\overline{\mathcal W_\hbar(A)(z)}\mathcal W_\hbar(B)(z)\,dz
=\Tr(A^*B).
\]
Deduce that $\mathcal W_\hbar$ extends uniquely to a unitary map
\[
S_2(L^2(V))\longrightarrow L^2(V\oplus V^*)
\]
and, whenever $\mathcal W_\hbar(A)\in L^1(V\oplus V^*)$, prove
\[
A=(2\pi\hbar)^{-n/2}\int_{V\oplus V^*}
\mathcal W_\hbar(A)(z)W_\hbar(z)^*\,dz
\]
in the weak operator topology.
\end{exercise}


\begin{exercise}[Schr\"odinger character]
For $V=\R^n$, let $f\in C^\infty(\mathbb H_n)$ satisfy
\[
(q,p,s)\longmapsto f\bigl(\exp(q,p,s)\bigr)
\in\mathcal S(\R^{2n+1})
\]
under the canonical identification
$\mathfrak h_n=\R^n\oplus(\R^n)^*\oplus\R$.  With Haar measure
$dq\,dp\,ds$, put
\[
\pi_\lambda(f)
=\int_{\mathbb H_n}f(g)\pi_\lambda(g)\,dg.
\]
Prove
\[
\Tr\pi_\lambda(f)
=(2\pi)^n|\lambda|^{-n}
\int_\R f(0,0,s)e^{i\lambda s}\,ds.
\]
Derive this identity directly from the Schr\"odinger kernel and identify
the factor $|\lambda|^{-n}$ with the symplectic-volume scaling of
$\mathcal O_\lambda$.
\end{exercise}

\begin{exercise}[Heisenberg Plancherel and inversion]
For $f\in L^1(\mathbb H_n)\cap L^2(\mathbb H_n)$, use Haar measure
$dq\,dp\,ds$ and the scalar Fourier transform
$\widehat h(\lambda)=\int_\R h(s)e^{-i\lambda s}\,ds
=(2\pi)^{1/2}(\mathcal Fh)(\lambda)$, so its inverse carries
$(2\pi)^{-1}$, and define
\[
\widehat f(\lambda)
=\int_{\mathbb H_n}f(g)\pi_\lambda(g)^*\,dg.
\]
Compute the integral kernel of $\widehat f(\lambda)$ and apply Euclidean
Plancherel in the central and phase-space variables.  Deduce the Plancherel measure
$(2\pi)^{-(n+1)}|\lambda|^n\,d\lambda$ and prove
\[
\|f\|_2^2
=(2\pi)^{-(n+1)}\int_{\R\setminus\{0\}}
\|\widehat f(\lambda)\|_{\rm HS}^2|\lambda|^n\,d\lambda.
\]
Prove the inversion formula and the distributional orthogonality of Weyl
operators.  Prove that representations with trivial central character have
zero Plancherel measure.
\end{exercise}

\begin{exercise}[Weyl quantization and Wigner distributions]
For $a\in\mathcal S(V\oplus V^*)$, define
$\operatorname{Op}_\hbar^W(a):\mathcal S(V)\to\mathcal S(V)$ by
\[
\bigl(\operatorname{Op}_\hbar^W(a)\psi\bigr)(x)
=\frac{1}{(2\pi\hbar)^n}
\int_{V\times V^*}
e^{ip(x-y)/\hbar}
a\!\left(\frac{x+y}{2},p\right)\psi(y)\,dy\,dp.
\]
For polynomial symbols, use the same formula as an oscillatory integral on
$\mathcal S(V)$.

For $\psi,\varphi\in\mathcal S(V)$, define
\[
W_\hbar(\psi,\varphi)(q,p)
=\frac{1}{(2\pi\hbar)^n}
\int_Ve^{-ip(y)/\hbar}
\overline{\psi(q-y/2)}\,\varphi(q+y/2)\,dy.
\]
Write $W_\hbar[\psi]=W_\hbar(\psi,\psi)$.

For $\ell\in V^*$ and $v\in V$, prove
\[
\operatorname{Op}_\hbar^W\bigl((q,p)\mapsto\ell(q)\bigr)=Q_\ell,
\qquad
\operatorname{Op}_\hbar^W\bigl((q,p)\mapsto p(v)\bigr)=P_v.
\]
Prove
\[
\operatorname{Op}_\hbar^W(a)^*
=\operatorname{Op}_\hbar^W(\overline a),
\qquad
\|\operatorname{Op}_\hbar^W(a)\|_{\rm HS}
=(2\pi\hbar)^{-n/2}\|a\|_2,
\]
and
\[
\langle\psi,\operatorname{Op}_\hbar^W(a)\varphi\rangle
=\int_{V\oplus V^*}a(q,p)W_\hbar(\psi,\varphi)(q,p)\,dq\,dp.
\]
Prove that for $a,b\in\mathcal S(V\oplus V^*)$ there is a unique
$a\star_\hbar b\in\mathcal S(V\oplus V^*)$ satisfying
\[
\operatorname{Op}_\hbar^W(a\star_\hbar b)
=\operatorname{Op}_\hbar^W(a)\operatorname{Op}_\hbar^W(b).
\]
Derive the oscillatory-integral formula for $\star_\hbar$ and prove
\[
a\star_\hbar b
=ab+\frac{i\hbar}{2}\{a,b\}+O(\hbar^2),
\qquad
\frac{a\star_\hbar b-b\star_\hbar a}{i\hbar}
=\{a,b\}+O(\hbar^2)
\]
in the Schwartz-symbol topology.
Prove that the complete asymptotic expansion agrees coefficient by
coefficient with the Moyal product.  First prove that every coefficient
of that product preserves Schwartz functions, so
for $E=V\oplus V^*$ it restricts to a formal product on
$\mathcal S(V\oplus V^*)[[\hbar]]$.  For
$k\in(V\oplus V^*)^*$ put $e_k(z)=e^{ik(z)}$.  Extend Weyl quantization to
these oscillatory symbols and prove that $\operatorname{Op}_\hbar^W(e_k)$
is unitary and that
\[
\operatorname{Op}_\hbar^W(e_k)\operatorname{Op}_\hbar^W(e_\ell)
=e^{-i\hbar\Pi(k,\ell)/2}
\operatorname{Op}_\hbar^W(e_{k+\ell}).
\]
Deduce the operator realization of the Moyal plane-wave identity.
\end{exercise}

\begin{exercise}[Groenewold--van Hove obstruction]
Specialize to $V=\R$ and write $Q=Q_{\id_\R}$ and $P=P_1$.
Prove that the polynomials of degree at most two form a Lie subalgebra for
the Poisson bracket and that Weyl quantization is an exact Lie-algebra
quantization on this subalgebra.  Quantize $q^2p^2$ using
\[
q^2p^2=\frac19\{q^3,p^3\}
=\frac13\{q^2p,qp^2\}
\]
and prove that there is no linear quantization of the polynomial Poisson
algebra satisfying
\[
\mathcal Q(1)=I,\qquad
\mathcal Q(q)=Q,\qquad
\mathcal Q(p)=P,\qquad
\mathcal Q(q^m)=Q^m,\qquad
\mathcal Q(p^m)=P^m,
\]
together with
\[
\mathcal Q(\{f,g\})=(i\hbar)^{-1}
[\mathcal Q(f),\mathcal Q(g)]
\]
on a common invariant domain.
\end{exercise}


\begin{exercise}[Classical and quantum evolution]
Let $H\in C^\infty(V\oplus V^*,\R)$ be subquadratic, meaning that every
derivative of order at least two is bounded.  Define its Hamiltonian field by
\[
da(X_H)=\{a,H\}
\]
for every smooth $a$, let $\Phi_t$ be the flow of $X_H$, and let
$\operatorname{Op}_\hbar^W(H)$ denote its self-adjoint realization.  Define
\[
U_\hbar(t)=e^{-it\operatorname{Op}_\hbar^W(H)/\hbar}.
\]
For $a\in\mathcal S(V\oplus V^*)$, define
\[
a_t=a\circ\Phi_t,
\qquad
A_\hbar(t)=U_\hbar(t)^*\operatorname{Op}_\hbar^W(a)U_\hbar(t).
\]

Prove
\[
\frac{d}{dt}a_t=\{a_t,H\},
\qquad
\frac{d}{dt}A_\hbar(t)
=\frac{i}{\hbar}
[\operatorname{Op}_\hbar^W(H),A_\hbar(t)].
\]
Prove~\cite{hormander3} the semiclassical operator estimate that the Moyal
commutator remainder obtained from symbols with the displayed derivative
bounds has operator norm $O(\hbar)$, uniformly for $|t|\leq T$, after
division by $i\hbar$.  Apply Duhamel's formula to prove, for every $T>0$,
Egorov's theorem
\[
\sup_{|t|\leq T}
\left\|
A_\hbar(t)-\operatorname{Op}_\hbar^W(a\circ\Phi_t)
\right\|=O_T(\hbar).
\]
Prove exact equality for quadratic $H$.  Derive Ehrenfest's equations for
the expectations of $Q_\ell$ and $P_v$ for every $\ell\in V^*$ and
$v\in V$.
\end{exercise}

\begin{exercise}[Linear symplectic covariance]
Let $S$ be a linear symplectic automorphism of $V\oplus V^*$.  Construct
unitary implementers for cotangent lifts of linear automorphisms of $V$,
for symplectic shears defined by quadratic forms, and, after fixing a
positive inner product $b:V\to V^*$, for the symplectic exchange
$(q,p)\mapsto(-b^{-1}p,bq)$ implemented by the $b$-Fourier transform.  Prove for
these generators that
\[
\mu(S)\operatorname{Op}_\hbar^W(a)\mu(S)^*
=\operatorname{Op}_\hbar^W(a\circ S^{-1}),
\qquad
W_\hbar[\mu(S)\psi]=W_\hbar[\psi]\circ S^{-1}.
\]
Deduce the covariance for their products and prove that each implementer is
determined up to phase.  Prove the metaplectic lifting
theorem~\cite{folland-phase}: the multiplier of these implementers has a
canonical double cover on which the projective representation becomes
unitary, and its two lifts of a fixed symplectic map differ by sign.  Use
the Groenewold--van Hove obstruction to show why exact covariance cannot
extend to every nonlinear canonical transformation.
\end{exercise}

\subsection{Laguerre Functions}

\begin{exercise}[Wigner--Laguerre identity]
Use the generalized Laguerre polynomials and their orthogonality.  For
\[
H_{\rm cl}(q,p)=\frac{p^2}{2m}+\frac{m\omega^2q^2}{2},
\qquad
\psi_n(q)=\left(\frac{m\omega}{\hbar}\right)^{1/4}
h_n\!\left(\sqrt{\frac{m\omega}{\hbar}}q\right),
\]
prove
\[
W_\hbar[\psi_n](q,p)
=\frac{(-1)^n}{\pi\hbar}
e^{-2H_{\rm cl}(q,p)/(\hbar\omega)}
L_n\!\left(\frac{4H_{\rm cl}(q,p)}{\hbar\omega}\right).
\]
Verify the position and momentum marginals and identify the zeros and sign
changes of the Wigner distribution from those of $L_n$.
\end{exercise}


\subsection{Homodyne Tomography}

\begin{exercise}[Laguerre tomography]
For $F\in\mathcal S(\R^2)$, $n_\theta=(\cos\theta,\sin\theta)$, and
$n_\theta^\perp=(-\sin\theta,\cos\theta)$, define the Radon transform by
\[
(\mathcal RF)(\theta,s)
=\int_\R F(sn_\theta+tn_\theta^\perp)\,dt.
\]
With
\[
\widehat h(k)=\int_\R h(s)e^{-iks}\,ds,
\qquad
\widehat F(\xi)=\int_{\R^2}F(z)e^{-i\xi\cdot z}\,dz,
\]
which are the same unnormalized transforms and hence differ from the
unitary transform $\mathcal F$ by $(2\pi)^{1/2}$ and $2\pi$ respectively,
prove the Fourier-slice identity
\[
\widehat{\mathcal RF}(\theta,k)=\widehat F(kn_\theta)
\]
and the inversion formula
\[
F(z)=\frac1{(2\pi)^2}\int_0^\pi\int_\R
|k|\widehat{\mathcal RF}(\theta,k)e^{ikn_\theta\cdot z}\,dk\,d\theta.
\]
For the dimensionless quadrature
\[
X_\theta=\sqrt{\frac{m\omega}{\hbar}}Q\cos\theta
+\frac{P}{\sqrt{m\hbar\omega}}\sin\theta,
\]
prove that its probability density in a state $\psi$ is the Radon
transform of $W_\hbar[\psi]$ along the corresponding phase-space lines.
Apply the Fourier-slice inversion formula to reconstruct the Wigner
distribution from the family of quadrature densities.  For the oscillator
state $\psi_n$, use the Wigner--Laguerre identity to prove that the
reconstructed radial profile is
\[
\frac{(-1)^n}{\pi\hbar}e^{-2H_{\rm cl}/(\hbar\omega)}
L_n\!\left(\frac{4H_{\rm cl}}{\hbar\omega}\right),
\]
while every quadrature density is a scaled
$e^{-x^2}H_n(x)^2$.  Deduce the predicted number of quadrature nodes, the
radial zero circles determined by $L_n$, and the sign at the phase-space
origin.  State how homodyne quadrature data distinguish adjacent number
states and detect Wigner negativity after inverse Radon reconstruction.
State the harmonic-oscillator, number-state-preparation, quadrature-phase,
detection-efficiency, inverse-reconstruction, sampling, and spatial-resolution
assumptions under which these nodes, zero circles, and signs are observable.
\end{exercise}


\section{Poincar\'e Group}

\subsection{Minkowski Isometries}

\begin{exercise}[Minkowski momentum orbits and relativistic dispersion]
A \emph{time-oriented Minkowski space} is a real four-dimensional vector
space $M$ with a symmetric nondegenerate form $g$ of signature $(1,3)$
and a chosen component of the cone $\{v:g(v,v)>0\}$.  Let
$SO^+(M,g)$ be the group preserving $g$, the orientation, and the time
orientation.  Define the proper orthochronous Poincar\'e group by
\[
\mathcal P^+(M,g)=M\rtimes SO^+(M,g),
\qquad
(a,A)(b,B)=(a+Ab,AB).
\]
Prove that $(a,A)$ acts on the affine space underlying $M$ by
$x\mapsto a+Ax$, that this identifies $\mathcal P^+(M,g)$ with its
orientation- and time-orientation-preserving affine isometries, and that
the induced action on momentum covectors factors through $SO^+(M,g)$.
Use $g$ to identify $M^\vee$ with $M$.  For $m\geq0$, define the
positive mass shell
\[
\mathcal O_m^+
=\{p\in M^\vee:g^{-1}(p,p)=m^2c^2,\ p^\sharp\text{ is future-pointing}\},
\]
with the zero covector removed when $m=0$.

Let $u$ be a future-pointing observer with $g(u,u)=c^2$.  Define
\[
E_u(p)=g(p^\sharp,u),
\qquad
\mathbf p_u=p^\sharp-\frac{E_u(p)}{c^2}u,
\qquad
|\mathbf p_u|^2=-g(\mathbf p_u,\mathbf p_u).
\]
Prove
\[
E_u(p)^2=c^2|\mathbf p_u|^2+m^2c^4
\]
and show that $SO^+(M,g)$ acts transitively on each $\mathcal O_m^+$.
For $m>0$, prove that the stabilizer of a point is the rotation group of
its negative-definite orthogonal complement and hence is isomorphic to
$SO(3)$.  For $m=0$, show that the stabilizer preserves the Euclidean
plane $p^\perp/\mathbb Rp$ and is isomorphic to
$SO(2)\ltimes\mathbb R^2$.

Put $Q(p)=g^{-1}(p,p)$.  For a Lorentz-invariant volume form $dV$ on
$M^\vee$, choose along the shell a vector field $N$ satisfying
$dQ(N)=1$.  Define
\[
(\iota_NdV)(v_1,v_2,v_3)=dV(N,v_1,v_2,v_3)
\]
and restrict this three-form to the shell.  Prove that the
restriction is independent of $N$ and constructs an invariant measure
$d\mu_m$, unique up to scale.  Show that in the observer decomposition it
is proportional to
\[
\frac{d^3\mathbf p_u}{E_u(\mathbf p)}.
\]
Deduce that the mass-shell relation and this measure are independent of
the observer used to express them.
\end{exercise}

\begin{exercise}[The Lorentz spin cover and finite-dimensional fields]

\subsection{Wigner Classification}

\begin{exercise}[Wigner induction and the Poincar\'e particle classification]
Let

\widetilde{\mathcal P}=M\rtimes SL(S)

be the spin Poincar\'e group.  Fix a Lorentz orbit
$\mathcal O\subset M^\vee$, a point $p_0\in\mathcal O$, its little group

\mathcal E_\sigma=SL(S)\times_KV_\sigma
\longrightarrow\mathcal O

intrinsically

[U(a,A)\psi](p)
=e^{-ip(a)/\hbar}\,A\cdot\psi(A^{-1}p),

translation spectral measure is supported on $\mathcal O$.

translation spectral measure is supported in the closed future cone.

theorem~\cite{mackey-induced,wigner-poincare}: every
irreducible positive-energy unitary representation whose translation
spectrum is supported on one orbit is obtained by this construction, and

little-group representation data are equivalent.  Deduce that a massive

$V_j$, so it has $2j+1$ rest-frame polarizations.  For a massless orbit,

subgroup are labelled by helicity
$h\in\tfrac12\mathbb Z$, while nontrivial translation action produces the
continuous-spin family and an additional positive scale.

On the common invariant space of smooth vectors, let $P$ be the
translation generator and let $J$ be the Lorentz bivector generator.
Use the isomorphism

\Lambda^2M\longrightarrow\mathfrak{so}(M,g),
\qquad
u\wedge v\longmapsto\bigl[w\mapsto g(v,w)u-g(u,w)v\bigr]

in the massive representation and

\qquad

\]

classification constrains masses, spins, helicities, and polarization
counts but does not determine the numerical masses or interactions.
\end{exercise}

\subsection{Euclidean Covariance}

\begin{exercise}[Outgoing Green kernels and the Lippmann--Schwinger equation]
Let $\mu>0$ be a reduced mass, let $k>0$, put
\[
E=\frac{\hbar^2k^2}{2\mu},
\qquad
H=-\frac{\hbar^2}{2\mu}\Delta+V,
\]
and initially take $V$ to be a real Schwartz function on $\R^3$.  For an
incident direction $\omega\in S^2$, call a solution of $(H-E)\psi=0$
\emph{outgoing} when
\[
\psi(x)=e^{ik\omega\cdot x}
+f(\widehat x,\omega;k)\frac{e^{ik|x|}}{|x|}
+o(|x|^{-1}),
\qquad \widehat x=\frac{x}{|x|},
\]
and call $f$ its \emph{scattering amplitude}.  For outgoing direction
$\omega'$, define the scattering angle by
$\cos\theta=\omega'\cdot\omega$.

Define the Euclidean group and its unitary action by
\[
E(3)=\R^3\rtimes SO(3),
\qquad
(U_{a,R}\phi)(x)=\phi(R^{-1}(x-a)).
\]
Prove that the free Hamiltonian commutes with this action, that $H$ commutes
with every rotation when $V$ is radial, and that translations are broken
unless $V$ is constant.  For radial $V$, prove
\[
f(R\omega',R\omega;k)=f(\omega',\omega;k).
\]

Prove in $\mathcal D'(\R^3)$ that
\[
G_k^+(x)=\frac{e^{ik|x|}}{4\pi|x|},
\qquad
(-\Delta-k^2)G_k^+=\delta_0,
\]
and verify the Sommerfeld condition
$(\partial_r-ik)G_k^+=o(r^{-1})$.  Derive the Lippmann--Schwinger equation
\[
\psi(x)=e^{ik\omega\cdot x}
-\frac{2\mu}{\hbar^2}
\int_{\R^3}G_k^+(x-y)V(y)\psi(y)\,dy.
\]
First take $V$ compactly supported and use
\[
G_k^+(x-y)=\frac{e^{ik|x|}}{4\pi|x|}
e^{-ik\widehat x\cdot y}+O(|x|^{-2})
\]
for bounded $y$ to prove
\[
f(\omega',\omega;k)
=-\frac{\mu}{2\pi\hbar^2}
\int_{\R^3}e^{-ik\omega'\cdot y}V(y)\psi(y)\,dy.
\]
Extend the formula to Schwartz $V$ by controlling the tails with its decay.
\end{exercise}

\begin{exercise}[Probability current, Born series, and screened-Coulomb counts]

\subsection{Scattering Decomposition}

\begin{exercise}[Scattering operators and elastic unitarity]
For a real radial potential supported in $\{x:\|x\|\leq R\}$, define the
on-shell scattering operator $S(k)$ on angular wave packets by sending the
incoming far-field coefficient to the outgoing one, with the free propagation
phases normalized so that $V=0$ gives $S(k)=I$.  Use conservation of
probability current to prove that $S(k)$ is unitary, and use rotation covariance
together with the multiplicity-free decomposition
\[
L^2(S^2)=\mathop{\widehat{\bigoplus}}_{\ell\geq0}\mathcal H_\ell
\]
where $\mathcal H_\ell$ is the degree-$\ell$ spherical-harmonic summand, to
prove
\[
S(k)|_{\mathcal H_\ell}=S_\ell(k)I,
\qquad
S_\ell(k)=e^{2i\delta_\ell(k)}
\]
for real phase shifts $\delta_\ell$, determined modulo $\pi$.
\end{exercise}

\begin{exercise}[Partial waves, Legendre amplitudes, and the optical theorem]
Put
\[
j_\ell(z)=\sqrt{\frac{\pi}{2z}}J_{\ell+1/2}(z).
\]
Project the Euclidean Fourier kernel onto the spherical-harmonic summands and
derive the distinguished expansion
\[
e^{ikr\cos\theta}
=\sum_{\ell\geq0}i^\ell(2\ell+1)j_\ell(kr)P_\ell(\cos\theta),
\]
where $P_\ell=P_\ell^{(0,0)}$ is the Legendre polynomial.  Separate the radial
equation and show that the regular radial function $u_\ell$ satisfies
\[
u_\ell''(r)+\left(k^2-\frac{\ell(\ell+1)}{r^2}
-\frac{2\mu}{\hbar^2}V(r)\right)u_\ell(r)=0,
\]
with
\[
u_\ell(r)\sim
\sin\left(kr-\frac{\ell\pi}{2}+\delta_\ell(k)\right)
\qquad(r\to\infty).
\]
Match incoming and outgoing spherical waves to derive
\[
f(\theta;k)
=\frac1{2ik}\sum_{\ell\geq0}(2\ell+1)
(S_\ell(k)-1)P_\ell(\cos\theta)
=\frac1k\sum_{\ell\geq0}(2\ell+1)e^{i\delta_\ell}
\sin\delta_\ell P_\ell(\cos\theta).
\]
Use Legendre orthogonality and $P_\ell(1)=1$ to prove
\[
\sigma_{\rm el}(k)
=\frac{4\pi}{k^2}\sum_{\ell\geq0}(2\ell+1)\sin^2\delta_\ell(k),
\qquad
\sigma_{\rm el}(k)=\frac{4\pi}{k}\operatorname{Im}f(0;k),
\]
and obtain the partial-wave unitarity bound
\[
\sigma_\ell(k)\leq\frac{4\pi}{k^2}(2\ell+1).
\]
Identify exactly where reality of the potential and absence of inelastic
channels enter the proof.
\end{exercise}

\begin{exercise}[Square-well thresholds and Breit--Wigner resonances]

\begin{exercise}[Jost functions, Levinson's theorem, and bound-state counting]
Let $V$ be a real radial potential supported in $[0,R]$ and suppose

\begin{exercise}[Marchenko reconstruction and the one-soliton potential]
For the Schr\"odinger operator $L=-d^2/dx^2+V$ with a real potential on
the line satisfying
$\int_\R(1+|x|)|V(x)|\,dx<\infty$, let $f_+(k,x)$ and $f_-(k,x)$ be the
Jost solutions normalized by
\[
f_+(k,x)e^{-ikx}\longrightarrow1\quad(x\to+\infty),
\qquad
f_-(k,x)e^{ikx}\longrightarrow1\quad(x\to-\infty).
\]
For real $k\ne0$, define the right-incident reflection and transmission
coefficients by the asymptotics
\[
\psi_R(k,x)\sim e^{-ikx}+R(k)e^{ikx}\quad(x\to+\infty),
\qquad
\psi_R(k,x)\sim T(k)e^{-ikx}\quad(x\to-\infty).
\]
The bound states are the poles $i\kappa_j$, $\kappa_j>0$, of the
meromorphic transmission coefficient.  Normalize their real eigenfunctions
so that
\[
\|\psi_j\|_2=1,
\qquad
\psi_j(x)\sim c_j^{1/2}e^{-\kappa_jx}\quad(x\to+\infty),
\]
and call $c_j>0$ the right norming constants.  Define
\[
\mathcal F(s)=\frac1{2\pi}\int_\R R(k)e^{iks}\,dk
+\sum_jc_je^{-\kappa_js}.
\]
The Marchenko kernel is the solution of
\[
K(x,y)+\mathcal F(x+y)
+\int_x^\infty K(x,z)\mathcal F(z+y)\,dz=0,
\qquad y\geq x,
\]
and the reconstructed potential is
\[
V(x)=-2\frac d{dx}K(x,x).
\]
Assume $R(-k)=\overline{R(k)}$, $\kappa_j,c_j>0$, and that
$\mathcal F$ is real and absolutely continuous with
$\mathcal F,\mathcal F'\in L^1([a,\infty))$ for every $a$.  On
$L^2([x,\infty))$ define
\[
(H_xh)(y)=\int_x^\infty\mathcal F(y+z)h(z)\,dz,
\]
and assume $I+H_x$ is strictly positive for every $x$.  Prove the
Marchenko existence and uniqueness theorem~\cite{deift-trubowitz}: the
integral equation has a unique solution, its diagonal defines a real
potential with the stated weighted integrability, and its right
scattering data are exactly $(R,\{\kappa_j,c_j\})$.  Prove that two
potentials with the same data therefore coincide.

In the reflectionless one-bound-state case put
\[
R(k)=0,
\qquad
\mathcal F(s)=2\kappa e^{2\kappa x_0}e^{-\kappa s}.
\]
Solve the integral equation and prove
\[
K(x,y)=
-\frac{2\kappa e^{2\kappa x_0}e^{-\kappa(x+y)}}
{1+e^{-2\kappa(x-x_0)}},
\qquad
V(x)=-2\kappa^2\operatorname{sech}^2(\kappa(x-x_0)).
\]
Compute its transmission coefficient
\[
T(k)=\frac{k+i\kappa}{k-i\kappa}
\]
and recover the bound-state pole from transmission data.  Let the norming
constant evolve by $c(t)=c(0)e^{8\kappa^3t}$ and show that
$u(x,t)=-V(x,t)$ solves the Korteweg--de Vries equation
\[
u_t+6uu_x+u_{xxx}=0
\]
and is the soliton
\[
u(x,t)=2\kappa^2
\operatorname{sech}^2\!\bigl(\kappa(x-x_0-4\kappa^2t)\bigr).
\]
State the one-dimensional, reflectionless, full-line, coherent-scattering,
and weakly dispersive assumptions under which the measured pole and pulse
speed test the inverse-scattering reconstruction.
\end{exercise}

\subsection{Coulomb Functions}

\begin{exercise}[Coulomb radial functions and phase shifts]
For the repulsive Coulomb potential $V(r)=\kappa/r$, with $\kappa>0$, put
\[
\rho=kr,
\qquad
\eta=\frac{\mu\kappa}{\hbar^2k},
\qquad
C_\ell(\eta)
=\frac{2^\ell e^{-\pi\eta/2}|\Gamma(\ell+1+i\eta)|}{(2\ell+1)!}.
\]
Verify that the regular radial solution is
\[
F_\ell(\eta,\rho)
=C_\ell(\eta)\rho^{\ell+1}e^{-i\rho}
M(\ell+1-i\eta,2\ell+2,2i\rho).
\]
Derive from the Kummer series and the beta integral that
\[
M(a,b,z)=\frac{\Gamma(b)}{\Gamma(a)\Gamma(b-a)}
\int_0^1e^{zt}t^{a-1}(1-t)^{b-a-1}\,dt
\]
when $\operatorname{Re}b>\operatorname{Re}a>0$.  Analyze its two endpoints
for the displayed Coulomb parameters and prove
\[
F_\ell(\eta,\rho)
=\sin\left(\rho-\eta\log(2\rho)-\frac{\ell\pi}{2}
+\sigma_\ell(\eta)\right)+o(1),
\qquad
\sigma_\ell(\eta)=\arg\Gamma(\ell+1+i\eta),
\]
where the argument is chosen continuously from $\sigma_\ell(0)=0$.
Deduce the Coulomb partial-wave factor
\[
S_\ell^{\rm C}
=e^{2i\sigma_\ell}
=\frac{\Gamma(\ell+1+i\eta)}{\Gamma(\ell+1-i\eta)}.
\]
\end{exercise}

\subsection{Rutherford Scattering}

\begin{exercise}[Coulomb amplitude and Rutherford detector counts]
Because the Coulomb potential is not short-range, use the logarithmically
modified radial phase above and define its nonforward amplitude by Abel
regularization of the partial-wave expression.

For $\operatorname{Re}s>0$, use the hypergeometric form of $P_\ell$ to prove
\[
\int_{-1}^1(1-x)^{s-1}P_\ell(x)\,dx
=2^s(-1)^\ell
\frac{\Gamma(s)^2}{\Gamma(s-\ell)\Gamma(s+\ell+1)}.
\]
Continue in $s$ and use the Legendre generating function to obtain, for
$-1<x<1$, the Abel-summed identity
\[
\lim_{r\uparrow1}\sum_{\ell\geq0}(2\ell+1)
(S_\ell^{\rm C}-1)r^\ell P_\ell(x)
=-2i\eta\,2^{i\eta}
\frac{\Gamma(1+i\eta)}{\Gamma(1-i\eta)}
(1-x)^{-1-i\eta}.
\]
With $x=\cos\theta$, derive the exact Coulomb amplitude away from the forward
direction,
\[
f_{\rm C}(\theta)
=-\frac{\eta}{2k\sin^2(\theta/2)}
\frac{\Gamma(1+i\eta)}{\Gamma(1-i\eta)}
\exp\!\left(-i\eta\log\sin^2\frac\theta2\right),
\]
and hence the Rutherford law
\[
\frac{d\sigma_{\rm C}}{d\Omega}
=\left(\frac{\kappa}{4E\sin^2(\theta/2)}\right)^2.
\]
For incident particle rate $\dot N_{\rm in}$, target areal density $n_T$,
acquisition time $T$, detector efficiency $\epsilon(\theta)$, and a narrow
solid-angle bin $\Delta\Omega$ away from $\theta=0$, derive
\[
N(\theta;\Delta\Omega)
\simeq \dot N_{\rm in}n_TT\epsilon(\theta)
\frac{d\sigma_{\rm C}}{d\Omega}\Delta\Omega.
\]
State separately the assumptions of nonrelativistic spinless two-body motion,
a thin target, pointlike unscreened charges, single scattering, negligible
inelasticity, far-field detection, and sufficiently fine angular bins.  State
how screening, finite charge radius, recoil calibration, multiple scattering,
and detector acceptance alter the comparison with measured counts.
\end{exercise}

\section{Schwarzschild Spacetime}

\subsection{Schwarzschild Isometries}

\begin{exercise}[Schwarzschild geometry and gravitational redshift]
A \emph{Lorentzian metric} on a four-manifold is a smooth
nondegenerate symmetric form of signature $(+,-,-,-)$.  Call a tangent
vector timelike, null, or spacelike according as its squared length is
positive, zero, or negative.  Prove the fundamental theorem
of pseudo-Riemannian geometry: there is a unique torsion-free
metric-compatible connection $\nabla$, characterized by the Koszul
formula.  Define geodesics by $\nabla_{\dot\gamma}\dot\gamma=0$, curvature
by
\[
R(X,Y)Z=\nabla_X\nabla_YZ-\nabla_Y\nabla_XZ-\nabla_{[X,Y]}Z,
\]
and $\operatorname{Ric}(X,Y)=\Tr(Z\mapsto R(Z,X)Y)$.  Let
\[
r_s=\frac{2G_{\mathrm N}M}{c^2},
\qquad
f(r)=1-\frac{r_s}{r},
\]
and on $(r_s,\infty)\times\mathbb R\times S^2$ consider
\[
g=f(r)c^2dt^2-f(r)^{-1}dr^2-r^2g_{S^2}.
\]
Prove Birkhoff's theorem~\cite{wald-gr}: every
spherically symmetric Lorentzian vacuum solution with
$\operatorname{Ric}=0$ is locally isometric to a member of this family.
Compute its Ricci tensor and verify $\operatorname{Ric}=0$.  Compute
the invariant
\[
R_{abcd}R^{abcd}
=\frac{48G_{\mathrm N}^2M^2}{c^4r^6}
\]
and use it to distinguish the curvature singularity at $r=0$ from the
coordinate singularity at $r=r_s$.

Prove that time translation preserves $g$.  For its timelike Killing
field $\partial_t$, prove that
$g(k,\partial_t)$ is conserved along a null geodesic with tangent $k$.
Compare it with the frequency measured by static observers to derive
\[
\frac{\nu_{\rm rec}}{\nu_{\rm emit}}
=\sqrt{\frac{f(r_{\rm emit})}{f(r_{\rm rec})}}.
\]
In particular, for reception at infinity predict
$\nu_\infty/\nu_r=\sqrt{1-r_s/r}$.  State the static-emitter,
static-receiver, geometric-optics, negligible-backreaction, and
spectral-resolution assumptions under which a clock or spectral-line
measurement tests the formula.
\end{exercise}

\subsection{Geodesic Reduction}

\begin{exercise}[Weierstrass elliptic functions and Schwarzschild light rays]
Restrict a Schwarzschild null geodesic to an equatorial great circle.  Let
$\mathcal E=f(r)c^2\dot t$ and $L=r^2\dot\phi$ be its conserved energy and
angular momentum per affine parameter, put $b=cL/\mathcal E$, and set
$u=1/r$.  Derive
\[
\left(\frac{du}{d\phi}\right)^2
=\frac1{b^2}-u^2+r_su^3,
\qquad
\frac{d^2u}{d\phi^2}+u=\frac32r_su^2.
\]
For $g_2^3-27g_3^2\ne0$, prove the Weierstrass
uniformization theorem: there is a lattice $\Lambda\subset\C$, unique up
to homothety with the prescribed invariants, and a unique even
$\Lambda$-periodic meromorphic function satisfying
\[
(\wp')^2=4\wp^3-g_2\wp-g_3,
\qquad
\wp(z)=z^{-2}+O(z^2),
\]
and this function gives the universal covering of the nonsingular cubic.
For a nonradial orbit whose cubic is nonsingular, prove from uniqueness of
solutions on that cubic that it is given by
\[
u(\phi)=\frac{1/3+4\wp(\phi-\phi_0;g_2,g_3)}{r_s},
\qquad
g_2=\frac1{12},
\qquad
g_3=\frac1{216}-\frac{r_s^2}{16b^2}.
\]
Show that its nontrivial finite-impact-parameter discriminant degeneration
occurs at
\[
r_{\rm ph}=\frac32r_s=\frac{3G_{\mathrm N}M}{c^2},
\qquad
b_c=\frac{3\sqrt3G_{\mathrm N}M}{c^2},
\]
the unstable circular photon orbit and critical capture impact parameter.
Expand the orbit equation for $r_s/b\ll1$ to derive the deflection angle
\[
\delta\phi=\frac{4G_{\mathrm N}M}{bc^2}
+O\!\left(\frac{G_{\mathrm N}^2M^2}{b^2c^4}\right).
\]
Predict the far-field shadow angular radius $b_c/D$ for an observer at
distance $D$.  State the vacuum, nonrotating, geometric-optics,
thin-lens, far-observer, and source-model assumptions separately from the
bending and shadow predictions.
\end{exercise}

\subsection{Confluent Heun Functions}

\begin{exercise}[Regge--Wheeler equations and confluent-Heun scattering]
Define the tortoise coordinate by
\[
r_*=r+r_s\log\!\left(\frac r{r_s}-1\right).
\]
For integer spin $s\in\{0,1,2\}$ and $\ell\geq s$, insert the scalar,
Maxwell, and linearized Einstein equations into the spin-weighted
spherical-harmonic decomposition and prove the separation theorem
\cite{regge-wheeler,chandrasekhar}:
after expansion in modes with time dependence $e^{-i\omega t}$ and the
corresponding spherical harmonics,
the radial master field satisfies
\[
\frac{d^2\psi}{dr_*^2}
+\left(\frac{\omega^2}{c^2}-V_{\ell,s}(r)\right)\psi=0,
\]
\[
V_{\ell,s}(r)
=f(r)\left(\frac{\ell(\ell+1)}{r^2}
+(1-s^2)\frac{r_s}{r^3}\right).
\]
Show that $r=r_s$ and $r=\infty$ are respectively regular and irregular
singular endpoints.  After putting $z=r/r_s$ and factoring the ingoing or
outgoing exponential and power behavior, transform the radial equation to
\[
y''+\left(\alpha+\frac{\beta+1}{z}
+\frac{\gamma+1}{z-1}\right)y'
+\left(\frac\mu z+\frac\nu{z-1}\right)y=0.
\]
When the exponent difference is nonintegral, define $H_0$ and $H_1$ to be
the Frobenius solutions with exponent zero normalized by
$H_0(0)=H_1(1)=1$; in resonant cases use their analytic limits.  Call these
the local confluent Heun functions.  Substitute their power series into
the equation to derive the three-term coefficient recurrences and their
radii of convergence.  Determine $\alpha,\beta,\gamma,\mu,\nu$ from
$\omega r_s/c$, $\ell$, and $s$ for the chosen asymptotic factors.

Define the horizon-ingoing and infinity-outgoing solutions by
\[
\psi_{\rm in}\sim e^{-i\omega r_*/c}\quad(r_*\to-\infty),
\qquad
\psi_{\rm out}\sim e^{+i\omega r_*/c}\quad(r_*\to+\infty),
\]
and define the connection determinant
\[
D_{\ell,s}(\omega)
=W(\psi_{\rm in},\psi_{\rm out}).
\]
Use constancy of the Wronskian to express reflection and transmission
amplitudes as confluent-Heun connection coefficients and prove flux
conservation for real $\omega$.
\end{exercise}

\subsection{Black-Hole Ringdown}

\begin{exercise}[Black-hole resonances, photon spheres, and ringdown]
Define a Schwarzschild quasinormal frequency to be a zero of the
confluent-Heun connection determinant $D_{\ell,s}(\omega)$ in the lower
half-plane.  Show that it is equivalently a mode ingoing at the horizon and
outgoing at infinity, and that its time dependence is a damped oscillation.

For $\ell\to\infty$, locate the maximum of the Regge--Wheeler potential at
the photon sphere.  Prove the leading parabolic-barrier WKB
connection formula: if $V$ has a nondegenerate maximum $V_0$ and
$V_0''=d^2V/dr_*^2<0$ there, an outgoing resonance satisfies
\[
\frac{\omega^2/c^2-V_0}{\sqrt{-2V_0''}}
=-i\left(n+\frac12\right)+O(\ell^{-1})
\]
in the eikonal scaling.  Apply it with the Langer replacement
$\ell(\ell+1)\sim(\ell+\tfrac12)^2$, to derive
\[
\omega_{\ell n}
=\Omega_{\rm ph}\left(\ell+\frac12\right)
-i\lambda_{\rm ph}\left(n+\frac12\right)
+O(\Omega_{\rm ph}\ell^{-1}),
\]
\[
\Omega_{\rm ph}=\lambda_{\rm ph}
=\frac{c^3}{3\sqrt3G_{\mathrm N}M},
\qquad n=0,1,2,\ldots.
\]
Relate $\Omega_{\rm ph}$ to the critical impact parameter by
$\Omega_{\rm ph}=c/b_c$.  If the source has cosmological redshift
$z_{\rm cos}$ and angular-diameter distance $D_A$, predict the observed
ringdown frequency, e-folding time, and shadow angular radius
\[
f_{\ell n}^{\rm obs}
=\frac{\operatorname{Re}\omega_{\ell n}}{2\pi(1+z_{\rm cos})},
\qquad
\tau_{\ell n}^{\rm obs}
=\frac{1+z_{\rm cos}}{|\operatorname{Im}\omega_{\ell n}|},
\qquad
\theta_{\rm sh}=\frac{b_c}{D_A},
\]
all from the same photon-sphere geometry.  State the isolated, nonrotating,
asymptotically flat,
linear-perturbation, eikonal, detector-frame, cosmological-redshift, and
mode-identification assumptions under which shadow and ringdown
measurements test these linked predictions.
For the axial gravitational fundamental mode $(\ell,n)=(2,0)$, compute
\[
\frac{GM\omega}{c^3}\approx0.4811-0.0962i
\]
from the eikonal formula.  Numerically continue the two local Heun
solutions, locate the corresponding zero of $D_{2,2}$, and compare with
\[
\frac{GM\omega}{c^3}\approx0.3737-0.0890i.
\]
Record that the leading formula overestimates this low-multipole real part
by about $29\%$ and is a controlled prediction only as $\ell\to\infty$.
\end{exercise}

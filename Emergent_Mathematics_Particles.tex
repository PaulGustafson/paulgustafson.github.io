\section{Heisenberg Group}

\begin{exercise}[Central extensions of phase space]
Let $(W,\omega)$ be a finite-dimensional real symplectic vector space.
Define the Heisenberg group
\[
 \operatorname{Heis}(W)=W\times\R,
 \qquad (w,s)(w',s')=(w+w',s+s'+\tfrac12\omega(w,w')).
\]
Prove associativity, identify its center and commutator, and show that its Lie
algebra satisfies $[(w,s),(w',s')]=(0,\omega(w,w'))$.  Prove that every
continuous central extension of the additive group $W$ with this commutator
is equivalent to this one after changing its section by a continuous
coboundary.
\end{exercise}

\begin{exercise}[Weyl relations and the Schr\"odinger representation]
For a real finite-dimensional vector space $Q$, put
$W=Q\oplus Q^*$ and
$\omega((q,p),(q',p'))=p(q')-p'(q)$.  On $L^2(Q)$ with Lebesgue measure fixed
by a density, define intrinsically
\[
 (T_qf)(x)=f(x-q),\qquad (M_pf)(x)=e^{\ii p(x)/\hbar}f(x),
\]
and $W(q,p)=e^{-\ii p(q)/(2\hbar)}M_pT_q$.  Prove
\[
 W(w)W(w')=e^{\ii\omega(w,w')/(2\hbar)}W(w+w').
\]
Differentiate on the Schwartz core to obtain
$[\widehat p,\widehat q]=-\ii\hbar,p(q)$ and prove the Robertson uncertainty
inequality directly from positivity of
$\norm{(A-\langle A\rangle)x+\ii t(B-\langle B\rangle)x}^2$.
\end{exercise}

\begin{exercise}[Stone--von Neumann decomposition]
Call a unitary representation of $\operatorname{Heis}(W)$ regular if its
one-parameter subgroups are strongly continuous and assume its center acts by
$e^{\ii s/\hbar}$.  For a Lagrangian subspace $Q^*\subset W$, construct the
projection-valued measure obtained from its commuting translations.  Prove
the imprimitivity relation for the complementary translations, identify its
spectral multiplicity as constant, and construct a unitary intertwiner with a
Hilbert direct sum of Schr\"odinger representations.  Deduce that every
irreducible regular representation with this central character is unitarily
equivalent to the Schr\"odinger representation.
\end{exercise}

\begin{exercise}[Self-adjoint generators and Hermite functions]
Prove Stone's theorem by constructing the spectral measure of a strongly
continuous one-parameter unitary group, and define its self-adjoint generator.
For $Q=\R$, take the closures of
$Xf=xf$ and $Pf=-\ii\hbar f'$ from the Schwartz core.  For
\[
 H=\frac{P^2}{2m}+\frac{m\omega^2X^2}{2},\qquad
 a=\frac{m\omega X+\ii P}{\sqrt{2m\hbar\omega}},
\]
prove $[a,a^*]=1$ and $H=\hbar\omega(a^*a+1/2)$.  Define the Hermite
polynomials by
$H_n(u)=(-1)^ne^{u^2}(\dd/\dd u)^ne^{-u^2}$ and the normalized Hermite
functions $h_n$.  Derive their differential equation, recurrence,
orthogonality, Mehler kernel, and completeness without calling them a basis.
\end{exercise}

\begin{exercise}[Oscillator spectroscopy]
Use the Hermite decomposition to prove
$E_n=\hbar\omega(n+1/2)$.  Couple the oscillator to a weak electric field by
$V(t)=-qE_0X\cos(\Omega t)$ and derive from the ladder relations that only
$n\leftrightarrow n\pm1$ transitions occur, with
\[
 \abs{\langle n+1|X|n\rangle}^2
 =\frac{\hbar}{2m\omega}(n+1).
\]
For a thermal state compute the Green-function absorption profile and show
that its lines are centered at $\Omega=\omega$, have the resolution width
from Hilbert Space, and have intensities in the ratios prescribed above.
These positions and ratios are the spectroscopic prediction.
\end{exercise}

\section{Poincar\'e Group}

\begin{exercise}[Minkowski isometries and their Casimirs]
Let Minkowski spacetime be an oriented, time-oriented affine space modeled on
$(M,g)$ with signature $(-,+,+,+)$.  Define
$\mathcal P^+=M\rtimes SO^+(M,g)$, derive its Lie algebra, and construct its
metric volume form and Hodge star, then construct its two Casimirs from
momentum $P$ and the Pauli--Lubanski covector
$W=* (P^\flat\wedge J)$, fixing this normalization.  Prove
$P^2=-m^2c^2$ and $W^2=-m^2c^2\hbar^2j(j+1)$ on a massive spin-$j$
irreducible representation.
\end{exercise}

\begin{exercise}[Wigner classification by little groups]
For a transitive action of a locally compact group $G$ on $G/H$, define the
representation induced from a unitary $H$-representation as square-integrable
equivariant sections of the associated bundle.  Prove the imprimitivity
theorem by disintegrating the translation spectral measure.  Apply it to the
mass shells in $M^*$: identify the massive little group with $SU(2)$ after
passing to the double cover, the massless little group with
$\operatorname{Spin}(2)\ltimes\R^2$, where
$\operatorname{Spin}(2)=U(1)\to SO(2)$ is the angle-doubling cover, and
derive the massive spin and massless
helicity families.  State explicitly the exclusion of continuous-spin
representations when the translational part of the massless little group acts
trivially.
\end{exercise}

\begin{exercise}[Klein--Gordon Bessel functions]
Put $\mu=mc/\hbar$ and
$\Box=-c^{-2}\partial_t^2+\Delta$.  Derive
$(\Box-\mu^2)\phi=0$ from the massive Casimir.  Define
$\Gamma(s)=\int_0^\infty t^{s-1}e^{-t}\dd t$ for $\Re s>0$, derive its
functional equation and meromorphic continuation, and define
\[
 J_\nu(z)=\sum_{k\ge0}\frac{(-1)^k(z/2)^{2k+\nu}}
 {k!\Gamma(k+\nu+1)},
 \qquad
 K_\nu(z)=\int_0^\infty e^{-z\cosh u}\cosh(\nu u)\dd u
\]
in their stated domains and continue by their differential equation.  Use
the Fourier decomposition of the positive mass shell to derive, at equal
time and spatial separation $r>0$,
\[
 \Delta_+(0,r)=\frac{mc}{4\pi^2\hbar r}K_1(\mu r).
\]
Derive the small- and large-argument asymptotics needed below.
\end{exercise}

\begin{exercise}[Relativistic correlation length]
For a free massive scalar field, define the vacuum two-point observable by
the positive-energy mass-shell integral and derive the preceding $K_1$
formula without choosing an inertial frame until evaluating the invariant
spacelike distance.  Prove
\[
 \Delta_+(x,y)\sim
 \frac{\sqrt m}{4\sqrt2\,\pi^{3/2}
 (\hbar/c)^{1/2}\,r^{3/2}}e^{-r/\lambda_C},
 \qquad \lambda_C=\frac\hbar{mc},
\]
up to the normalization fixed in the previous exercise.  Show that the slope
of $\log(r^{3/2}\Delta_+)$ against invariant separation is $-1/\lambda_C$;
this is a quantitative mass determination from spacelike correlation data.
\end{exercise}

\section{Potential Scattering}

\begin{exercise}[$SO(3)$ symmetry and radial operators]
On $L^2(\R^3)$ let $SO(3)$ act by pullback.  For a central potential $V(r)$,
prove that the Schr\"odinger operator commutes with this action.  Use the
spherical-harmonic decomposition from Compact Harmonic Analysis to obtain
the intrinsic orthogonal sum over angular momentum and the radial operators
\[
 H_\ell=-\frac{\hbar^2}{2m}\frac{\dd^2}{\dd r^2}
 +\frac{\hbar^2\ell(\ell+1)}{2mr^2}+V(r)
\]
on $L^2(\R_+,\dd r)$ with the regular boundary condition.
\end{exercise}

\begin{exercise}[Partial-wave decomposition and phase shifts]
For a short-range real potential, construct the regular radial solution by a
Volterra equation and prove that for $k>0$ it has the asymptotic form
\[
 u_\ell(k,r)=A_\ell(k)
 \sin(kr-\ell\pi/2+\delta_\ell(k))+o(1).
\]
Define $S_\ell(k)=e^{2\ii\delta_\ell(k)}$, fix
$\delta_\ell=0$ for the free operator, and use the conserved Wronskian to
prove $\abs{S_\ell}=1$.  Derive
\[
 f(\theta)=\frac1{2\ii k}\sum_{\ell\ge0}(2\ell+1)
 (S_\ell-1)P_\ell(\cos\theta),
 \quad \sigma_{\rm tot}=\frac{4\pi}{k^2}
 \sum_{\ell\ge0}(2\ell+1)\sin^2\delta_\ell.
\]
\end{exercise}

\begin{exercise}[Coulomb hypergeometric functions]
For $V(r)=\kappa/r$ set $\rho=kr$ and
$\eta=m\kappa/(\hbar^2k)$.  Put
$(a)_n=\Gamma(a+n)/\Gamma(a)$ and define
$\mathop{{}_1F_1}(a;c;z)=\sum_{n\ge0}(a)_nz^n/((c)_nn!)$.  Define the regular
Coulomb function by
\[
 F_\ell(\eta,\rho)=C_\ell(\eta)\rho^{\ell+1}e^{-\ii\rho}
 {}_1F_1(\ell+1-\ii\eta;2\ell+2;2\ii\rho),
\]
choosing $C_\ell(\eta)>0$ so that its large-$\rho$ sine amplitude is one.
Derive the radial equation, the Euler integral and connection formula for
${}_1F_1$, and
$\sigma_\ell=\arg\Gamma(\ell+1+\ii\eta)$ with
$\sigma_\ell-\sigma_{\ell-1}=\arctan(\eta/\ell)$.
Sum the resulting Legendre series by its generating function.
\end{exercise}

\begin{exercise}[Rutherford scattering]
Show that Coulomb scattering has amplitude, up to an unobservable constant
phase,
\[
 f_C(\theta)=-\frac{\eta}{2k\sin^2(\theta/2)}
 \exp\!\left[-2\ii\eta\log\sin\frac\theta2+2\ii\sigma_0\right].
\]
Hence derive
\[
 \frac{\dd\sigma}{\dd\Omega}=
 \left(\frac{Z_1Z_2e^2}{16\pi\epsilon_0E}\right)^2
 \csc^4\frac\theta2
\]
for repulsive charges with center-of-mass kinetic energy $E$.  Identify beam
energy, scattering angle, and detector solid angle as the measured quantities,
and state the point-charge and nonrelativistic limitations of the prediction.
\end{exercise}

\section{Schwarzschild Spacetime}

\begin{exercise}[Lorentzian geometry and spherical vacuum metrics]
Define a Lorentzian metric, its Levi--Civita connection by the Koszul identity,
curvature, Ricci tensor, and geodesics.  Derive the Jacobi equation from a
variation of geodesics.  For a static spherically symmetric metric, use the
area radius and proper-time normalization at infinity to solve
$\operatorname{Ric}=0$, obtaining
\[
 g=-\left(1-\frac{r_s}{r}\right)c^2\dd t^2
 +\left(1-\frac{r_s}{r}\right)^{-1}\dd r^2+r^2\dd\Omega^2,
 \qquad r_s=\frac{2GM}{c^2}.
\]
Prove that its connected exterior isometry group is
$\R\times SO(3)$.
\end{exercise}

\begin{exercise}[Regge--Wheeler decomposition]
Set $G=c=1$ temporarily and define the tortoise coordinate by
$\dd r_*/\dd r=(1-2M/r)^{-1}$.  Separate the scalar wave equation using the
$SO(3)$ decomposition and prove that $u_{\ell m}=r\phi_{\ell m}$ satisfies
\[
 \left(-\partial_t^2+\partial_{r_*}^2-V_\ell(r)\right)u_{\ell m}=0,
 \qquad
 V_\ell(r)=\left(1-\frac{2M}{r}\right)
 \left(\frac{\ell(\ell+1)}{r^2}+\frac{2M}{r^3}\right).
\]
Construct the conserved Wronskian and the ingoing/outgoing scattering
coefficients, and prove their flux identity.
\end{exercise}

\begin{exercise}[Confluent Heun functions and quasinormal modes]
Define a confluent Heun function as the Frobenius solution at a regular
singular point of
\[
 y''+\left(\alpha+\frac{\beta+1}{z}+\frac{\gamma+1}{z-1}\right)y'
 +\left(\frac\mu z+\frac\nu{z-1}\right)y=0,
\]
with its parameters and initial exponent specified.  Transform the separated
Schwarzschild radial equation to this form and derive the three-term
recurrence of its Frobenius coefficients.  Define a quasinormal frequency as
a zero of the connection coefficient multiplying the incoming solution at
infinity for the solution ingoing at the horizon.  Express that coefficient
as the limit of the recurrence's continued fraction and prove its convergence
under the minimal-solution condition.
\end{exercise}

\begin{exercise}[Black-hole ringdown in the eikonal limit]
For $\ell\to\infty$, locate the maximum of $V_\ell$ at the unstable photon
orbit $r=3M$.  Match parabolic-cylinder solutions across the maximum and
derive, for fixed overtone $n$,
\[
 \frac{GM\omega_{\ell n}}{c^3}=
 \frac{\ell+1/2-\ii(n+1/2)}{3\sqrt3}+O(\ell^{-1}).
\]
Show that the predicted waveform has oscillation frequency
$\Re\omega/(2\pi)$ and damping time $1/\abs{\Im\omega}$.  Explain how the
confluent-Heun continued fraction supplies the finite-$\ell$ correction and
why the displayed asymptotic value must not be substituted as an exact
$\ell=2$ prediction.
\end{exercise}

\section{\texorpdfstring{$SL_2(\mathbb R)$}{SL2(R)}}

\begin{exercise}[Hyperbolic isometries and the Casimir]
Let $SL_2(\R)$ act on $\mathbb H=\{x+\ii y:y>0\}$ by M\"obius maps.  Prove
that it preserves $\dd s^2=(\dd x^2+\dd y^2)/y^2$, identify the stabilizer of
$\ii$ with $SO(2)$, and derive the Iwasawa and Cartan decompositions.  With
the nonnegative convention define
\[
 \Delta_{\mathbb H}=-y^2(\partial_x^2+\partial_y^2)
\]
and show that it is the action of the quadratic Casimir on right-$SO(2)$
invariant functions.
\end{exercise}

\begin{exercise}[Principal, complementary, and discrete series]
Construct the spherical principal series by unitary induction from the upper
triangular subgroup, the complementary series by its invariant kernel, and
the holomorphic and antiholomorphic discrete series on weighted Bergman
spaces.  Determine their Casimir and $SO(2)$ spectra, including the signs of
the discrete-series weights.  Prove irreducibility by cyclicity of a
$K$-finite vector and classify the irreducible unitary representations by
the possible lowest/highest weight or continuous Casimir parameter.
\end{exercise}

\begin{exercise}[$SL_2(\R)$ Plancherel decomposition]
Construct the standard intertwining operator between opposite principal
series, compute its scalar on every $SO(2)$ type from the raising and lowering
relations, and determine its poles and residues.  Shift its contour to recover
the discrete-series summands and derive the group Plancherel measure, with
both even and odd principal-series densities.  Prove inversion first for
compactly supported smooth $K$-finite functions and extend it to $L^2$.
\end{exercise}

\begin{exercise}[Legendre functions and the Mehler--Fock transform]
Define
$\mathop{{}_2F_1}(a,b;c;z)=\sum_{n\ge0}(a)_n(b)_nz^n/((c)_nn!)$ in the unit
disk and continue it by its differential equation.  With fixed branches,
define
\[
 P_\nu^\mu(z)=\frac1{\Gamma(1-\mu)}
 \left(\frac{z+1}{z-1}\right)^{\mu/2}
 {}_2F_1\!\left(-\nu,\nu+1;1-\mu;\frac{1-z}{2}\right).
\]
Derive the associated Legendre equation and show that the spherical function
at spectral parameter $\lambda$ is
$P_{-1/2+\ii\lambda}(\cosh r)$.  From the radial Sturm--Liouville problem
prove the Mehler--Fock inversion and Plancherel formulas, including the
density $\lambda\tanh(\pi\lambda)$.  Derive the analogous fixed-$SO(2)$-type
transform with $P_{-1/2+\ii\lambda}^{-m}$.
\end{exercise}

\begin{exercise}[Hyperbolic-drum spectra]
Let a membrane occupy a geodesic disk of curvature radius $a$ and hyperbolic
radius $R$, with wave speed $v$ and Dirichlet boundary.  Separate its wave
equation into $SO(2)$ types and prove that its resonances are
\[
 \omega_{m,k}=\frac va\sqrt{\frac14+\lambda_{m,k}^2},
 \qquad
 P_{-1/2+\ii\lambda_{m,k}}^{-\abs m}(\cosh R)=0.
\]
Prove the $m\ne0$ double degeneracy and the Weyl asymptotic from the
Mehler--Fock spectral measure.  The zero locations, degeneracies, and area
coefficient are the three measurable predictions.
\end{exercise}

\section{\texorpdfstring{$SL_2(\mathbb Z)$}{SL2(Z)}}

\begin{exercise}[Modular group and its fundamental domain]
Define $SL_2(\Z)$, its action on $\mathbb H$, and
$S\tau=-1/\tau$, $T\tau=\tau+1$.  Prove the presentation
$PSL_2(\Z)=\langle S,T\mid S^2=(ST)^3=1\rangle$ by reducing points to the
standard fundamental domain.  Compute its hyperbolic area and the cusp,
elliptic-point, and orbifold stabilizer data.
\end{exercise}

\begin{exercise}[Eisenstein series and the scattering coefficient]
Construct $L^2(SL_2(\Z)\backslash\mathbb H)$ with hyperbolic measure and the
nonnegative Laplacian.  Define the Eisenstein series
$E(\tau,s)=\sum_{\Gamma_\infty\backslash\Gamma}(\Im\gamma\tau)^s$ for
$\Re s>1$.  Unfold its inner products, continue it using the theta Mellin
transform, compute its constant term, and prove its functional equation and
unitarity on $\Re s=1/2$.
\end{exercise}

\begin{exercise}[Automorphic spectral and trace decomposition]
Develop the Friedrichs realization of the nonnegative Laplacian at the cusp.
Truncate the Eisenstein series, prove the Maass--Selberg relation, and obtain
the orthogonal decomposition into constants, cusp forms, and the continuous
Eisenstein integral.  Build the automorphic kernel from a compactly supported
point-pair invariant, unfold it by conjugacy classes, and derive the Selberg
trace identity with its identity, elliptic, parabolic, and hyperbolic terms.
Recover the Weyl area term.
\end{exercise}

\begin{exercise}[Modular forms, theta constants, and the eta function]
Define modular forms by their weight, multiplier, and cusp growth.  Starting
from the theta transformation proved in Fourier Analysis, define
$\vartheta_2,\vartheta_3,\vartheta_4$ by their shifted or signed lattice sums
and derive all their $S,T$ transformations without re-proving Poisson
summation.  Define
$\eta(\tau)=q^{1/24}\prod_{n\ge1}(1-q^n)$ with
$q=e^{2\pi\ii\tau}$, and derive its transformations from the Jacobi triple
product.  Construct $E_4,E_6$, prove the valence formula, and show
\[
 \C[E_4,E_6]=\bigoplus_{k\ge0}M_k,\qquad
 \Delta=\eta^{24}=\frac{E_4^3-E_6^2}{1728}.
\]
\end{exercise}

\begin{exercise}[The modular invariant and monstrous moonshine]
Define $j=E_4^3/\Delta$ and prove from the valence formula that every modular
function holomorphic away from the cusp is a polynomial in $j$.  Compute
\[
 J(\tau)=j(\tau)-744=q^{-1}+196884q+21493760q^2+\cdots.
\]
Construct the graded traces of the moonshine module from the vertex-operator
axioms, derive the denominator recursion from its Borcherds product, and use
that recursion and genus-zero invariance to prove that the identity trace is
$J$.  Verify the first decompositions
$196884=1+196883$ and
$21493760=1+196883+21296876$ into irreducible Monster dimensions.
\end{exercise}

\begin{exercise}[Conformal spectra and the Cardy prediction]
Define a unitary two-dimensional conformal theory on a circle of circumference
$L$ by commuting Virasoro actions with central charges $c,\bar c$ and weights
$h,\bar h$.  Derive the cylinder Hamiltonian and momentum
\[
 H=\frac{2\pi v}{L}\left(L_0+\bar L_0-\frac{c+\bar c}{24}\right),
 \qquad P=\frac{2\pi}{L}(L_0-\bar L_0).
\]
Show that the torus trace is modular invariant when gravitational anomaly
constraints are met.  Apply the $S$ transformation and a saddle-point
estimate to prove, for effective chiral central charge $c_{\rm eff}$,
\[
 \log\rho(\Delta)=2\pi\sqrt{\frac{c_{\rm eff}\Delta}{6}}+o(\sqrt\Delta).
\]
Thus predict both the finite-size ground-energy shift
$-\pi v(c+\bar c)/(12L)$ and the high-energy level-counting rate, specifying
$v,L$, and the measured energies.
\end{exercise}

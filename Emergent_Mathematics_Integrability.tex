\begin{exercise}[$\mathbb Z_2$ spin symmetry and the partition function]
For a finite graph $\Gamma=(V,E)$ and spins $\sigma:V\to\{\pm1\}$ define
\[
 Z_\Gamma(K,h)=\sum_\sigma
 \exp\!\left(K\sum_{ij\in E}\sigma_i\sigma_j
 +h\sum_{i\in V}\sigma_i\right).
\]
Identify the global $\mathbb Z_2$ action at $h=0$.  Expand each edge factor
into even and odd parts and prove the high-temperature loop expansion.
Expand about the two ordered configurations to obtain the dual domain-wall
expansion.  On the square lattice derive Kramers--Wannier duality
$\sinh(2K)\sinh(2K^*)=1$, including all finite-torus boundary sectors.
\end{exercise}

\begin{exercise}[Transfer-matrix decomposition]
For an $M\times N$ periodic square lattice, construct the row Hilbert space
as the tensor product of the two-dimensional spin object and define the
positive transfer operator $T_M(K)$.  Prove
$Z_{M,N}=\Tr(T_M^N)$ and that $T_M$ commutes with global spin reversal and
translation.  Define the four spin structures as the periodic or
antiperiodic boundary choices around the two torus cycles.  Introduce the
Clifford algebra with the fixed convention
$\{\gamma_a,\gamma_b\}=2\delta_{ab}$, construct the Jordan--Wigner operators,
and reduce $T_M$ in each spin structure to commuting two-dimensional momentum
blocks.  Derive their rapidities
\[
 \cosh\gamma(k)=\cosh(2K^*)\cosh(2K)
 -\sinh(2K^*)\sinh(2K)\cos k.
\]
\end{exercise}

\begin{exercise}[Onsager free energy]
Put $\beta=(k_BT)^{-1}$.  Take the thermodynamic limit of the largest transfer
eigenvalue and derive
the isotropic free energy
\[
 -\beta f(K)=\frac12\log(2\sinh2K)
 +\frac1{2\pi}\int_0^\pi
 \operatorname{arcosh}(\cosh2K\coth2K-\cos\theta)\dd\theta.
\]
Locate its only nonanalytic point at
$K_c=\tfrac12\log(1+\sqrt2)$ and derive the logarithmic heat-capacity
singularity.
\end{exercise}

\begin{exercise}[Toeplitz determinants and spontaneous magnetization]
Express a long-distance spin correlation as a Toeplitz determinant.  Derive
the scalar symbol by the Clifford decomposition, factor it into inside and
outside analytic parts, and prove the strong Szeg\H{o} limit needed for its
ordered phase.  Obtain
\[
 M(K)=\lim_{r\to\infty}\langle\sigma_0\sigma_r\rangle^{1/2}
 =\left(1-\sinh^{-4}(2K)\right)^{1/8},\qquad K>K_c.
\]
\end{exercise}

\begin{exercise}[Virasoro characters of the Ising transfer matrix]
Use the convention $q=e^{2\pi\ii\tau}$ from $SL_2(\Z)$.  Take the scaling
limit of the four transfer-matrix spin structures and factor their fermionic
mode products by the Jacobi triple product.  Identify the resulting three
functions with the $c=1/2$ Virasoro characters
$\chi_0,\chi_{1/2},\chi_{1/16}$.  Match the lattice
vacuum, energy, and spin sectors to the weights $0,1/2,1/16$, derive their
coefficient multiplicities, and use their $S,T$ matrices to construct the
modular-invariant torus partition function
$\abs{\chi_0}^2+\abs{\chi_{1/2}}^2+\abs{\chi_{1/16}}^2$.
\end{exercise}

\begin{exercise}[Ising criticality]
Restore $K=J/(k_BT)$.  Predict

 k_BT_c=\frac{2J}{\log(1+\sqrt2)},\qquad

On a cylinder of circumference $L$, use the three Ising characters and the

\end{exercise}

\section{Affine Characters}

\begin{exercise}[Central extensions of loop algebras]
Let $K_c$ be a simply connected compact Lie group whose complexified Lie
algebra is the complex simple Lie algebra $\mathfrak g$.  Define its smooth
loop group $LK_c=C^\infty(S^1,K_c)$ and let it act on the loop-field
configuration space $LK_c$ by left multiplication, with $S^1$ acting by
rotation of the source circle; prove that these actions combine to an
$LK_c\rtimes S^1$ action.  Normalize the invariant
form $(\ ,\ )$ so that long roots have squared length two.  Define the
polynomial loop algebra $L\mathfrak g=\mathfrak g\otimes\C[z,z^{-1}]$ and
\[
 \widehat{\mathfrak g}=L\mathfrak g\oplus\C K\oplus\C d
\]
by
\[
 [x\otimes z^m,y\otimes z^n]
 =[x,y]\otimes z^{m+n}+m\delta_{m+n,0}(x,y)K,
 \qquad [d,x\otimes z^n]=n x\otimes z^n.
\]
Obtain the central term from the residue cocycle
$\operatorname{Res}_{z=0}(\partial_zx(z),y(z))\dd z$ and prove the Jacobi
identity.  For $K_c$ construct the corresponding $U(1)$-central extension of
its smooth loop group at an integer
level $k$ by exponentiating the cocycle.  If $\vartheta$ is the left
Maurer--Cartan form, define the Cartan three-form as the scalar multiple of
$(\vartheta,[\vartheta,\vartheta])$ that integrates to one on every long-root
$SU(2)$ subgroup, and prove independence of the filling disk from its integral
cohomology class.
\end{exercise}

\begin{exercise}[Integrable highest-weight decomposition]
Extend a Cartan algebra of $\mathfrak g$ by $K$ and $d$.  Define the null
root by $\delta(d)=1$ and by zero on the other Cartan summands.  With highest
root $\theta$, set
$\alpha_0=\delta-\theta$ and $\alpha_0^\vee=K-\theta^\vee$, and define the
affine Weyl group, level, and dominant affine weights.  Apply the
rank-one $\mathfrak{sl}_2$ strings for every affine simple root to prove that
an irreducible highest-weight module is integrable at level $k\in\N$ exactly
when its finite highest weight $\lambda$ satisfies
\[
 \langle\lambda,\theta^\vee\rangle\leq k.
\]
Call a unitary loop-group representation positive-energy when the rotation
energy $-d$ has spectrum bounded below with finite-dimensional eigenspaces.
Prove that every such representation decomposes orthogonally into these
modules.  For
$\widehat{\mathfrak{sl}}_2$ identify the level-$k$ summands
$L_0,L_1,\ldots,L_k$.
\end{exercise}

\begin{exercise}[Sugawara Virasoro decomposition]
Let $\mathcal C_{\mathfrak g}\in\mathfrak g\otimes\mathfrak g$ be the inverse
tensor to the invariant form and define the dual Coxeter number $h^\vee$ by
the scalar $2h^\vee$ through which the associated Casimir acts on
$\mathfrak g$.  Write $J_{x,n}$ for the action of $x\otimes z^n$ and set
the linear map
$\mathcal N_{p,q}:\mathfrak g\otimes\mathfrak g\to\End(L)$ to send
$x\otimes y$ to $J_{x,p}J_{y,q}$ for $p<0$ and to
$J_{y,q}J_{x,p}$ for $p\geq0$.  Define the Sugawara operators intrinsically
by contraction with $\mathcal C_{\mathfrak g}$:
\[
 L_n=\frac1{2(k+h^\vee)}
 \sum_{m\in\Z}\mathcal N_{n-m,m}(\mathcal C_{\mathfrak g}).
\]
Prove from the affine commutators that these operators satisfy the Virasoro
relations and
\[
 c=\frac{k\dim\mathfrak g}{k+h^\vee},\qquad
 h_\lambda=\frac{(\lambda,\lambda+2\rho)}{2(k+h^\vee)}.
\]
For $\widehat{\mathfrak{sl}}_2$ derive
$c=3k/(k+2)$ and $h_\ell=\ell(\ell+2)/(4(k+2))$.
\end{exercise}

\begin{exercise}[Weyl--Kac characters]
Put $y=e^{2\pi\ii z}$ and, without changing the modular nome convention,
extend the level theta series by
\[
 \Theta_{m,K}(z,\tau)=\sum_{n\in\Z}
 q^{K(n+m/(2K))^2}y^{K(n+m/(2K))}.
\]
Define
$\chi_\ell^{(k)}(z,\tau)=
\operatorname{Tr}_{L_\ell}(q^{L_0-c/24}y^{H/2})$, where $H$ is the
$\mathfrak{sl}_2$ coroot operator.  Construct the affine BGG resolution by
reflections in the walls of the
fundamental alcove and take its Euler character to prove the Weyl--Kac
formula.  For $\widehat{\mathfrak{sl}}_2$ obtain
\[
 \chi_\ell^{(k)}(z,\tau)=
 \frac{\Theta_{\ell+1,k+2}(z,\tau)-
       \Theta_{-\ell-1,k+2}(z,\tau)}
      {\Theta_{1,2}(z,\tau)-\Theta_{-1,2}(z,\tau)},
 \qquad 0\leq\ell\leq k.
\]
Use Poisson summation to prove that their continuous values at $z=0$
transform by
\[
 S_{\ell m}=\sqrt{\frac2{k+2}}
 \sin\!\frac{(\ell+1)(m+1)\pi}{k+2},\qquad
 T_{\ell\ell}=e^{2\pi\ii(h_\ell-c/24)}.
\]
Derive orthogonality and completeness of this finite sine transform.
\end{exercise}

\begin{exercise}[Knizhnik--Zamolodchikov conformal blocks]
For finite-dimensional $\mathfrak g$-modules $V_1,\ldots,V_n$, let
$\Omega_{ij}$ be the invariant Casimir contraction on the $i$th and $j$th
factors.  On the configuration space of distinct points define
\[
 \nabla_{KZ}=\dd-\frac1{k+h^\vee}
 \sum_{i<j}\Omega_{ij}\dd\log(z_i-z_j).
\]
Prove the infinitesimal braid relations for the $\Omega_{ij}$ and hence
flatness.  For $k\geq2$ and four fundamental $\mathfrak{sl}_2$ modules,
decompose the
invariant tensors through the two intermediate-spin channels, place three
points at $0,1,\infty$, and reduce the horizontal-section equation to the
Gauss hypergeometric equation.  Derive its Euler integral
from the Gamma integral, use it to obtain the connection matrix between the
$0$ and $1$ channels, and compute
the two braid eigenvalues.  Record the resulting braid representation by its
eigenvalue ratio and channel-connection matrix.
\end{exercise}

\begin{exercise}[Wess--Zumino--Witten finite-size spectroscopy]
Define the diagonal $SU(2)_k$ Wess--Zumino--Witten state space as
$\bigoplus_{\ell=0}^kL_\ell\otimes\overline{L_\ell}$.  For a periodic
critical system whose scaling limit has this state space, use the Sugawara
Hamiltonian and the affine characters to derive
\[
 E_0(L)=e_\infty L-\frac{\pi v}{6L}\frac{3k}{k+2}+o(L^{-1}),
\]
\[
 E_{\ell,N,\bar N}(L)-E_0(L)
 =\frac{2\pi v}{L}
 \left(2h_\ell+N+\bar N\right)+o(L^{-1}),
\]
where $e_\infty$ is the bulk ground-energy density and
$N,\bar N\in\Z_{\geq0}$ are
descendant levels, with every degeneracy given by the corresponding character
coefficients.  For an antiferromagnetic spin-$1/2$ Heisenberg chain of
$N_s$ sites, set the lattice spacing to one, take $k=1$, $L=N_s$, and
$v=\pi J/2$ to predict the leading primary gap $\pi^2J/(2N_s)$ and
ground-energy correction $-\pi^2J/(12N_s)$.  Separate the WZW scaling
hypothesis, periodic boundary condition, velocity calibration, and
logarithmic finite-size corrections from the measured gaps and degeneracies.
\end{exercise}

\begin{exercise}[The Hopf algebra $U_v(\mathfrak{sl}_2)$]
For $v\in\C^\times$ not a root of unity, define
$U_v(\mathfrak{sl}_2)$ by
\[
 KEK^{-1}=v^2E,\quad KFK^{-1}=v^{-2}F,\quad
 [E,F]=\frac{K-K^{-1}}{v-v^{-1}},
\]
with
\[
 \Delta K=K\otimes K,\quad
 \Delta E=E\otimes K+1\otimes E,\quad
 \Delta F=F\otimes1+K^{-1}\otimes F,
\]
and construct its counit and antipode from the Hopf identities.  Define
$[n]_v=(v^n-v^{-n})/(v-v^{-1})$ and construct every finite-dimensional
type-one irreducible $V_n$ from a highest-weight vector.  Prove the deformed
Clebsch--Gordan rule $V_m\otimes V_n\simeq
\bigoplus_{j=0}^{\min(m,n)}V_{m+n-2j}$.  For any ordered collection of
type-one modules, use the iterated coproduct to define the diagonal
$U_v(\mathfrak{sl}_2)$ action on their tensor-product state space and identify
quantum-group-invariant observables with its commuting endomorphisms.
\end{exercise}

\begin{exercise}[Universal $R$-matrix]
Write $K=v^H$ formally and, in the completion acting on type-one modules,
define
\[
 \mathcal R=v^{H\otimes H/2}
 \sum_{r\ge0}v^{r(r-1)/2}
 \frac{(v-v^{-1})^r}{[r]_v!}E^r\otimes F^r.
\]
Verify directly on highest-weight modules that
$\mathcal R\Delta(a)=\Delta^{\rm op}(a)\mathcal R$ and
\[
 (\Delta\otimes\id)\mathcal R=\mathcal R_{13}\mathcal R_{23},
 \qquad
 (\id\otimes\Delta)\mathcal R=\mathcal R_{13}\mathcal R_{12}.
\]
Define the braid group $B_n$ by generators
$\sigma_i$ with the far-commutativity and braid relations; deduce its actions
on tensor powers and the quantum Yang--Baxter equation.
On the two fundamental channels, where the sum terminates and has a regular
root-of-unity value, specialize at $v=e^{\pi\ii/(k+2)}$.  Prove that the
ratio of its two braid eigenvalues and its recoupling matrix agree with the
framing-independent Knizhnik--Zamolodchikov braid data.
\end{exercise}

\begin{exercise}[Root-of-unity tilting modules and ribbon quotients]
Construct the divided-power integral form of
$U_v(\mathfrak{sl}_2)$ over $\Z[v,v^{-1}]$ and specialize it at
$v=e^{\pi\ii/(k+2)}$.  Define Weyl modules, dual Weyl modules, and tilting
modules through filtrations by both.  Use the antipode, evaluation, and
coevaluation to define their rigid duals and prove closure of the tilting
subcategory under tensor products and duals.  Construct the pivotal element
from the square of the antipode and use it to define the categorical trace.
Define a morphism to be
negligible when its categorical trace vanishes after every composable
morphism, prove that negligible morphisms form a tensor ideal, and form the
Karoubi-completed quotient by formally adjoining the images of idempotents.
Show that it is semisimple with simple objects
$V_0,\ldots,V_k$, quantum dimensions
\[
 d_a=\frac{\sin((a+1)\pi/(k+2))}{\sin(\pi/(k+2))},
\]
and truncated Clebsch--Gordan products.  Construct the ribbon element from
the specialized $R$-matrix, define its compatibility with the coproduct,
antipode, and pivotal element, prove the ribbon identities in the quotient,
and derive
\[
 \theta_a=\exp\!\left(\frac{\pi\ii a(a+2)}{2(k+2)}\right).
\]
\end{exercise}

\begin{exercise}[Quantum Peter--Weyl decomposition]
Define the restricted Hopf dual
$\mathcal O_v(SL_2)$ as the span of matrix coefficients.  Prove its algebraic
Peter--Weyl decomposition
\[
 \mathcal O_v(SL_2)=\bigoplus_{n\ge0}V_n^\vee\otimes V_n
\]
and derive Schur orthogonality for the invariant positive functional when
$0<v<1$, after equipping it with the compact real-form involution and writing
the resulting Hopf $*$-algebra as $\mathcal O_v(SU(2))$.
\end{exercise}

\begin{exercise}[Basic hypergeometric and dual $q$-Hahn functions]
Put $Q=v^2$ and keep this base distinct from the modular nome.  For
$n\in\Z_{\geq0}$ and $|Q|<1$ define
\[
 (a;Q)_n=\prod_{j=0}^{n-1}(1-aQ^j),\qquad
 (a;Q)_\infty=\prod_{j\geq0}(1-aQ^j),
 \qquad (a_1,\ldots,a_m;Q)_n=\prod_i(a_i;Q)_n,
\]
and define
\[
 {}_r\phi_s\!\left(\begin{matrix}a_1,\ldots,a_r\\
 b_1,\ldots,b_s\end{matrix};Q,z\right)
 =\sum_{k\ge0}\frac{(a_1,\ldots,a_r;Q)_k}
 {(Q,b_1,\ldots,b_s;Q)_k}
 \left((-1)^kQ^{k(k-1)/2}\right)^{1+s-r}z^k.
\]
Derive the $Q$-binomial and $Q$-Gauss sums by finite telescoping and analytic
continuation, and show that terminating identities extend rationally in $Q$
away from denominator zeros.  Define the dual $Q$-Hahn polynomials by
\[
 R_n(Q^{-x}+\gamma\delta Q^{x+1};\gamma,\delta,N\mid Q)
 ={}_3\phi_2\!\left(\begin{matrix}Q^{-n},Q^{-x},
 \gamma\delta Q^{x+1}\\ \gamma Q,Q^{-N}
 \end{matrix};Q,Q\right).
\]
Derive their difference equation and finite orthogonality.  Solve the
$U_v(\mathfrak{sl}_2)$ lowering recurrence and identify its normalized
Clebsch--Gordan coefficients with these polynomials after matching the three
integer labels and normalization.  Define the terminating $Q$-Racah family by
\[
 \mathcal R_n(Q^{-x}+\delta Q^{x-N})
 ={}_4\phi_3\!\left(\begin{matrix}
 Q^{-n},\alpha\beta Q^{n+1},Q^{-x},\delta Q^{x-N}\\
 \alpha Q,\beta\delta Q,Q^{-N}
 \end{matrix};Q,Q\right)
\]
and derive its finite orthogonality and recoupling identity from two
associations of a fourfold tensor product.
\end{exercise}

\begin{exercise}[Six-vertex transfer matrices and the XXZ chain]
Write $v=e^\eta$ and define, with additive spectral parameter $u$,

 R(u)=\begin{pmatrix}

 0&\sinh u&\sinh\eta&0\\

\end{pmatrix}.

Evaluate the finite universal $R$-matrix on $V_1\otimes V_1$, prove that its

from the braid and Hecke relations.  Form the periodic row transfer matrix, prove
commutativity for different $u$, and take its logarithmic derivative at
$u=0$ to obtain, up to scalar and scale,
\[
 H_{XXZ}=\sum_{j=1}^N
 (\sigma_j^x\sigma_{j+1}^x+\sigma_j^y\sigma_{j+1}^y

Develop the algebraic Bethe ansatz and derive the Bethe equations and energy

\end{exercise}

\section{Jack Polynomials}

\begin{exercise}[Reflection groups and commuting Dunkl operators]
Let $\mathfrak a$ be a finite-dimensional Euclidean space, let
$R\subset\mathfrak a$ be a reduced root system, and let $W$ be the reflection
group generated by
$s_\alpha x=x-2\langle\alpha,x\rangle\alpha/\langle\alpha,\alpha\rangle$.
Choose $R_+$ and a nonnegative $W$-invariant multiplicity function
$k:R\to\R_{\geq0}$.  For $\xi\in\mathfrak a$ define on polynomial functions
\[
 T_\xi f(x)=\partial_\xi f(x)+
 \sum_{\alpha\in R_+}k_\alpha\langle\alpha,\xi\rangle
 \frac{f(x)-f(s_\alpha x)}{\langle\alpha,x\rangle}.
\]
Prove that the divided differences preserve polynomials, establish
$wT_\xi w^{-1}=T_{w\xi}$, and prove $[T_\xi,T_\eta]=0$ by reducing the
commutator to the rank-two root subsystems.  For
\[
 w_k(x)=\prod_{\alpha\in R_+}|\langle\alpha,x\rangle|^{2k_\alpha},
\]
derive the integration-by-parts identity for the $T_\xi$ on the complement of
the reflection arrangement and extend it to Schwartz functions.  Contract
two copies of $\xi\mapsto T_\xi$ with the inverse Euclidean metric to define
the Dunkl Laplacian.  Restrict it to $W$-invariant functions and conjugate by
$w_k^{1/2}$ to obtain the rational Calogero Hamiltonian, including its
inverse-square coupling constants.
\end{exercise}

\begin{exercise}[Dunkl kernels]
Construct the degree-preserving intertwining operator $V_k$ recursively from
$T_\xi V_k=V_k\partial_\xi$ and $V_k1=1$.  Prove its uniqueness, its
$W$-equivariance, and its convergence on exponential power series.  Define the Dunkl kernel
$E_k(x,y)=V_k(e^{\langle\,\cdot\,,y\rangle})(x)$ and prove
\[
 T_\xi^xE_k(x,y)=\langle\xi,y\rangle E_k(x,y),
 \qquad E_k(x,y)=E_k(y,x).
\]
Use the differential--difference equation and its reflection maximum
principle to prove $|E_k(\ii x,y)|\leq1$ for real $x,y$ and $k\geq0$, and
derive the exponential growth bound needed for contour and integral
arguments.
\end{exercise}

\begin{exercise}[Dunkl transforms and generalized Bessel decomposition]
With
$c_k=\int_{\mathfrak a}e^{-|x|^2/2}w_k(x)\dd x$, define
\[
 \mathcal F_kf(y)=c_k^{-1}\int_{\mathfrak a}
 f(x)E_k(x,-\ii y)w_k(x)\dd x.
\]
Construct the Dunkl heat kernel explicitly from $E_k$, prove its semigroup
and approximate-identity properties, and use them to prove inversion and
Plancherel.  Show that
$\mathcal F_k(T_\xi f)=\ii\langle\xi,y\rangle\mathcal F_kf$.
Define the generalized Bessel kernel by
\[
 J_k(x,y)=\frac1{|W|}\sum_{w\in W}E_k(\ii x,wy),
\]
and derive the corresponding transform on $W$-invariant functions.  In rank
one identify $J_k$ with the normalized Bessel function of order $k-1/2$,
and derive its Hankel inversion formula directly from Dunkl inversion.
\end{exercise}

\begin{exercise}[Heckman--Opdam Jacobi and Jack polynomials]
Put $\rho_k=\tfrac12\sum_{\alpha\in R_+}k_\alpha\alpha$.  On the regular
complex torus of the weight lattice define the trigonometric Dunkl operators
\[
 \mathcal D_\xi=\partial_\xi-\langle\rho_k,\xi\rangle
 +\sum_{\alpha\in R_+}k_\alpha\langle\alpha,\xi\rangle
 \frac{1-s_\alpha}{1-e^{-\alpha}}.
\]
Verify their commutativity by the rank-two root-system calculation and derive
the degenerate affine-Hecke cross relations that they satisfy.  For each
dominant weight $\lambda$, define the Heckman--Opdam Jacobi polynomial as the
unique $W$-invariant orbit-sum-triangular joint eigenfunction of the
$W$-invariant polynomials in the $\mathcal D_\xi$.  Prove orthogonality and
completeness for the compact-torus weight
\[
 \delta_k(x)=\prod_{\alpha\in R_+}
 |e^{\alpha(x)/2}-e^{-\alpha(x)/2}|^{2k_\alpha},
\]
construct the multiplicity-shift intertwiners from the positive-root
directional operators, compute their adjoints, derive the norm and product
kernels.  In rank one identify these functions with Jacobi polynomials,
including the parameter correspondence.

For $R=A_{r-1}$ and constant multiplicity $g$, identify the compact Jacobi
polynomials with Jack polynomials
$P_\lambda^{(1/g)}(x_1,\ldots,x_r)$.  Derive their differential operator,
orthogonality for
$\prod_{i<j}|x_i-x_j|^{2g}$ on the unit torus, Cauchy kernel, norm, and Pieri
rule, with every normalization obtained as a specialization of the preceding
root-system construction.
\end{exercise}

\begin{exercise}[Calogero--Sutherland spectroscopy]
On the configuration space of $r$ indistinguishable particles on a circle of
radius $R$, let $S_r$ act by permuting the angular positions and define the
dimensionless operator
\[
 H_g=-\frac12\sum_{i=1}^r\partial_{\theta_i}^2
 +\frac{g(g-1)}4\sum_{i<j}
 \sin^{-2}\!\frac{\theta_i-\theta_j}{2},
 \qquad g>0.
\]
Select its self-adjoint realization in the permutation-invariant sector by
the collision behavior
$\Psi=O(|\theta_i-\theta_j|^g)$, and put
$\mathcal H_g=\hbar^2H_g/(mR^2)$.  Conjugate $H_g$ by
$\Psi_0=\prod_{i<j}|e^{\ii\theta_i}-e^{\ii\theta_j}|^g$ and identify the
result with the type-$A$ trigonometric Dunkl operator.  Prove that its
normalized eigenfunctions are $\Psi_0P_\lambda^{(1/g)}$ and that
\[
 E_\lambda=\frac12\sum_{i=1}^r
 \left(\lambda_i+g\left(\frac{r+1}{2}-i\right)\right)^2.
\]
Use the Jack Pieri rule for the density mode
$p_1=\sum_ie^{\ii\theta_i}$ to derive the allowed transitions
$\lambda\to\lambda+\square$ and their squared normalized intensities.
Thus a periodic density probe predicts the angular frequencies
$\omega_{\lambda\mu}=\hbar|E_\mu-E_\lambda|/(mR^2)$, missing lines, and
intensity ratios; state the inverse-square interaction, identical-particle
sector, radius, coupling, temperature, and spectral-resolution assumptions
separately from those observables.
\end{exercise}

\section{Macdonald Polynomials}

\begin{exercise}[Double-affine braid groups]
For the type-$A_{r-1}$ root system in
$\{x\in\R^r:\sum x_i=0\}$, construct its Weyl chambers, affine reflection
hyperplanes, and alcoves.  Identify the
affine braid group with braids in an annulus.  Show that allowing every strand
to wind around both cycles of a torus gives the double-affine braid group,
with finite braid generators $T_i$ and commuting winding families $X_j$ and
$Y_j$.  Derive its braid and winding relations directly from homotopies of
torus braids.  Impose
$(T_i-t^{1/2})(T_i+t^{-1/2})=0$ and fix the total winding commutator to $q$ to
 define $\mathsf H_{q,t}(GL_r)$.  Here $q$ is a multiplicative difference
parameter, not the modular nome; when this finite Hecke action is
compared with Quantum Groups use $t^{1/2}=v$, and otherwise keep $t$ and $v$
independent.  Filter near $q=1$, compute the associated graded
difference-reflection relations, and recover the trigonometric Dunkl algebra
acting on the torus.
\end{exercise}

\begin{exercise}[Polynomial representation and Cherednik operators]
Let $\mathcal T_r=(\C^\times)^r$ be the algebraic torus and let
\[
 \mathcal P_r=\C(q,t^{1/2})[x_1^{\pm1},\ldots,x_r^{\pm1}]
\]
be its Laurent-polynomial functions.
Let $X_i$ be multiplication, let $s_i$ exchange $x_i,x_{i+1}$, define
$T_{q,x_i}f(x_1,\ldots,x_r)=f(x_1,\ldots,qx_i,\ldots,x_r)$, and put
\[
 T_i=t^{1/2}s_i+(t^{1/2}-t^{-1/2})
 \frac{s_i-1}{x_i/x_{i+1}-1},
 \quad
 (\omega f)(x)=f(qx_r,x_1,\ldots,x_{r-1}),
\]
\[
 Y_i=T_i\cdots T_{r-1}\omega T_1^{-1}\cdots T_{i-1}^{-1}.
\]
Verify that these difference-reflection operators define the double-affine
Hecke action on the function space of $\mathcal T_r$, including preservation
of Laurent polynomials, the Hecke and affine braid
relations, and $[Y_i,Y_j]=0$.  Prove their triangular action on monomials in
Bruhat-refined dominance order.  Show on symmetric Laurent polynomials that
\[
 t^{(r-1)/2}\sum_iY_i=D_{q,t},\qquad
 D_{q,t}=\sum_i\prod_{j\ne i}\frac{tx_i-x_j}{x_i-x_j}T_{q,x_i}.
\]
\end{exercise}

\begin{exercise}[Macdonald polynomials and their kernel]
For a partition $\lambda$ of length at most $r$, define $P_\lambda$ as the
monic symmetric polynomial triangular in monomial symmetric functions and
satisfying
\[
 D_{q,t}P_\lambda=\left(\sum_{i=1}^rq^{\lambda_i}t^{r-i}\right)P_\lambda.
\]
For $0<q,t<1$, prove symmetry of $D_{q,t}$ for the torus weight
\[
 \Delta_{q,t}(x)=\prod_{i\ne j}
 \frac{(x_i/x_j;q)_\infty}{(tx_i/x_j;q)_\infty}
\]
and deduce orthogonality and completeness.  In the stable symmetric-function
ring define the power-sum pairing
\[
 \langle p_\lambda,p_\mu\rangle_{q,t}
 =\delta_{\lambda\mu}z_\lambda
 \prod_j\frac{1-q^{\lambda_j}}{1-t^{\lambda_j}}.
\]
Determine the Macdonald norm from the adjoint Pieri operators, define
$Q_\lambda$ to be dual to $P_\lambda$ for this pairing, and prove the Cauchy
kernel
\[
 \sum_\lambda P_\lambda(a;q,t)Q_\lambda(b;q,t)
 =\prod_{i,j}\frac{(ta_ib_j;q)_\infty}{(a_ib_j;q)_\infty}.
\]
Construct nonsymmetric joint $Y$-eigenfunctions by intertwiners and recover
$P_\lambda$ with the finite-Hecke symmetrizer.
\end{exercise}

\begin{exercise}[Elliptic deformation and Macdonald specialization]
For $|p|,|q|<1$ define
\[
 \theta(z;p)=(z;p)_\infty(p/z;p)_\infty,\qquad
 \Gamma(z;p,q)=\prod_{j,k\ge0}
 \frac{1-p^{j+1}q^{k+1}/z}{1-p^jq^kz}.
\]
Prove their difference and reflection identities.  Use the multiplicative
theta addition formula to construct the rank-one elliptic
difference-reflection operators and verify their dynamical braid relation.
Derive the terminating Frenkel--Turaev sum by telescoping that relation.
Take $p\to0$ to obtain the Askey--Wilson operator and its
${}_4\phi_3$ eigenpolynomials; specialize its four boundary parameters to
recover the rank-one Macdonald operator.  Record every parameter map so that
$p$ is always the elliptic nome, $q$ the difference step, and $t$ the Hecke
parameter.
\end{exercise}

\begin{exercise}[Ruijsenaars--Schneider spectrum]
Put $x_j=e^{\ii\theta_j}$ and regard the commuting symmetric polynomials in
$Y_i$ as quantum integrals of the compact trigonometric
Ruijsenaars--Schneider system.  With the positive Macdonald weight, construct
their self-adjoint realization and prove
\[
 E_\lambda=\sum_{i=1}^rq^{\lambda_i}t^{r-i}
\]
for the first normalized Hamiltonian.  Derive the first Macdonald Pieri rule
for multiplication by $x_1+\cdots+x_r$ and hence the allowed transitions
$\lambda\to\lambda+\square$ and their squared normalized intensities.
Thus a periodic density probe predicts exact line positions, missing lines,
and intensities; state the finite-particle, relativistic-difference coupling,
boundary, and spectral-resolution assumptions.
\end{exercise}

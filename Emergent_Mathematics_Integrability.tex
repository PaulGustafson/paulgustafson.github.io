\begin{exercise}[Hard-hexagon transfer matrices]
Let $\mathcal R_N$ be a finite set of allowed width-$N$ configurations and
let $T_N(z)$ be an operator on the configuration space whose matrix element
between $a,b\in\mathcal R_N$ is the weight of adjoining $b$ to $a$.  For a
height-$M$ periodic lattice define
\[
Z_{N,M}(z)=\Tr(T_N(z)^M).
\]
When $T_N(z)$ has nonnegative entries, write $\lambda_N(z)$ for its
spectral radius.

Let
\[
\mathcal R_N
=\{a\in\{0,1\}^{\Z/N\Z}:a_i a_{i+1}=0\text{ for every }i\}.
\]
Two rows $a,b\in\mathcal R_N$ are compatible when
\[
a_i b_i=0,
\qquad a_i b_{i+1}=0
\quad(i\in\Z/N\Z).
\]
For $z\geq0$, with $|a|=\sum_i a_i$, define
\[
T_N(z)_{ab}
=\mathbf1_{\{a,b\text{ compatible}\}}
z^{(|a|+|b|)/2}.
\]

Identify the horizontal, vertical, and diagonal nearest-neighbor edges
encoded by the two compatibility conditions.  Prove
\[
\Tr(T_N(z)^M)
=\sum_I z^{|I|},
\]
where the sum is over independent vertex sets of the specified triangular
$N\times M$ torus.  Prove that every coefficient is an exact state count.
Let $0$ denote the empty row.  For $z>0$, prove
\[
(T_N(z)^2)_{ab}\geq T_N(z)_{a0}T_N(z)_{0b}>0.
\]
Use as theorem input~\cite{seneta} the finite Perron--Frobenius statement that a strictly
positive matrix has a positive simple spectral radius and every other
spectral value has strictly smaller modulus.  Apply it to show that
$\lambda_N(z)$ is simple,
that every other spectral value has smaller modulus, and that
\[
\lim_{M\to\infty}\frac1{NM}\log Z_{N,M}(z)
=\frac1N\log\lambda_N(z).
\]
Put $\eta=\log z=\beta_{\rm th}\mu$ and
\[
\Psi_{N,M}(\eta)=\log Z_{N,M}(e^\eta).
\]
Give the independent sets the probability
$\mathbb P_\eta(I)=e^{\eta|I|}/Z_{N,M}(e^\eta)$ and write
$\mathbb E_\eta$ for expectation with respect to this measure.  For a
square-integrable random variable $Y$, write
$\operatorname{Var}_\eta(Y)=\mathbb E_\eta|Y-\mathbb E_\eta Y|^2$.
Set $A=\partial_\eta\Psi_{N,M}$ and prove
\[
A=\mathbb E_\eta|I|,
\qquad
\partial_\eta A=\operatorname{Var}_\eta(|I|),
\]
and derive the mean occupied fraction and isothermal finite-size
susceptibility
\[
\rho_{N,M}=\frac{A}{NM},
\qquad
\chi_{N,M}=\frac{\partial\rho_{N,M}}{\partial\mu}
=\frac{\beta_{\rm th}}{NM}\operatorname{Var}_\eta(|I|).
\]
State the lattice
geometry, boundary conditions, and finite $N,M$ limitations required when
these quantities are compared with adsorption or exclusion-process data.
\end{exercise}

\begin{exercise}[The planar Ising model and Kramers--Wannier duality]
Let $G=(V,E)$ be a finite connected graph with a fixed planar embedding.
For couplings $K_e>0$, define
\[
Z_G^{\mathrm f}(K)
=\sum_{s:V\to\{-1,1\}}
\exp\!\left(\sum_{e=uv}K_es_us_v\right).
\]
Let $G^*$ be the planar dual, fix the spin of its outer-face vertex to
$+1$, and denote the corresponding partition function by
$Z_{G^*}^{+}(K^*)$.  Let $\mathcal E(G)$ be the family of edge subsets in
which every vertex has even degree.  The dual couplings are determined by
\[
e^{-2K_{e^*}^*}=\tanh K_e.
\]

Expand each edge factor as
\[
e^{K_es_us_v}=\cosh K_e(1+s_us_v\tanh K_e)
\]
and prove
\[
Z_G^{\mathrm f}(K)
=2^{|V|}\prod_{e\in E}\cosh K_e
\sum_{\gamma\in\mathcal E(G)}
\prod_{e\in\gamma}\tanh K_e.
\]
Identify $\mathcal E(G)$ with the domain walls of dual spin configurations
whose outer-face spin is fixed, and deduce the exact planar duality
\[
Z_G^{\mathrm f}(K)
=2^{|V|}\prod_{e\in E}\bigl(\cosh K_e\,e^{-K_{e^*}^*}\bigr)
Z_{G^*}^{+}(K^*).
\]
For horizontal and vertical square-lattice couplings $K_h,K_v$, show that
the duality interchanges the directions and that its fixed locus is
\[
\sinh(2K_h)\sinh(2K_v)=1.
\]
Assuming the infinite-volume ferromagnetic model has a single transition
and that duality interchanges its high- and low-temperature phases, deduce
in the isotropic case
\[
K_c=\frac12\log(1+\sqrt2).
\]
State separately the exact finite-graph identity, the infinite-volume
assumptions, and the conclusion about the critical temperature.
\end{exercise}

\begin{exercise}[Ising star--triangle transformation]
For $K_1,K_2,K_3>0$ and boundary spins $s_1,s_2,s_3$, prove that there are
unique $A>0$ and real $L_{12},L_{23},L_{31}$ satisfying
\[
\sum_{s_0=\pm1}
e^{K_1s_0s_1+K_2s_0s_2+K_3s_0s_3}
=A e^{L_{12}s_1s_2+L_{23}s_2s_3+L_{31}s_3s_1}.
\]
Put
\[
\begin{aligned}
w_0&=2\cosh(K_1+K_2+K_3),&
w_1&=2\cosh(-K_1+K_2+K_3),\\
w_2&=2\cosh(K_1-K_2+K_3),&
w_3&=2\cosh(K_1+K_2-K_3).
\end{aligned}
\]
Derive
\[
A=(w_0w_1w_2w_3)^{1/4},
\]
\[
e^{4L_{12}}=\frac{w_0w_3}{w_1w_2},\qquad
e^{4L_{23}}=\frac{w_0w_1}{w_2w_3},\qquad
e^{4L_{31}}=\frac{w_0w_2}{w_1w_3}.
\]
Show that replacing every star of one sublattice of the honeycomb model
by a triangle changes the partition function only by the product of the
local factors $A$.  In the isotropic case derive
\[
e^{4L}=\frac{\cosh(3K)}{\cosh K}.
\]
Combine this transformation with planar duality and, under the
single-transition hypothesis, obtain
\[
K_c^{\rm tri}=\frac14\log3,
\qquad
K_c^{\rm hex}=\frac12\log(2+\sqrt3).
\]
Identify the coupling normalization and lattice geometry needed when these
critical temperatures are compared with an Ising magnet or a programmable
spin array.
\end{exercise}

\begin{exercise}[Ising transfer matrices and free fermions]
Let $\mathcal S_N=\{-1,1\}^{\Z/N\Z}$ and $K_h,K_v>0$.  For row
configurations $s,t\in\mathcal S_N$ define
\[
T_N^{\rm Is}(K_h,K_v)_{s,t}
=\exp\!\left[
\frac{K_h}{2}\sum_i(s_is_{i+1}+t_it_{i+1})
+K_v\sum_i s_it_i\right].
\]
Let $R_N$ be cyclic row translation.  On
$\mathcal H_N=\ell^2(\mathcal S_N)$ let
\[
(Z_jf)(s)=s_jf(s),
\qquad
(X_jf)(s)=f(s^{(j)}),
\qquad
Y_j=iX_jZ_j,
\]
where $s^{(j)}$ is obtained by reversing $s_j$.  Put
\[
P=\prod_{j=1}^NX_j,
\qquad
c_{2j-1}=\left(\prod_{\ell<j}X_\ell\right)Z_j,
\qquad
c_{2j}=\left(\prod_{\ell<j}X_\ell\right)Y_j.
\]
Let $K_v^*>0$ satisfy $e^{-2K_v^*}=\tanh K_v$, and set
\[
\mathcal K_+=\left\{\frac{(2m+1)\pi}{N}:0\leq m<N\right\},
\qquad
\mathcal K_-=\left\{\frac{2m\pi}{N}:0\leq m<N\right\}.
\]
Write $a_x$ and $a_t$ for the spatial and transfer-step lattice spacings
and $L=Na_x$ for the circumference.

\begin{enumerate}[label=(\alph*)]
\item Verify
\[
R_NT_N^{\rm Is}(K_h,K_v)
=T_N^{\rm Is}(K_h,K_v)R_N.
\]
\item Prove
\[
T_N^{\rm Is}(K_h,K_v)
=(2\sinh(2K_v))^{N/2}
e^{\frac{K_h}{2}\sum_jZ_jZ_{j+1}}
e^{K_v^*\sum_jX_j}
e^{\frac{K_h}{2}\sum_jZ_jZ_{j+1}}.
\]
\item Verify the Clifford relations
\[
c_rc_s+c_sc_r=2\delta_{rs}I,
\qquad
X_j=ic_{2j-1}c_{2j},
\qquad
Z_jZ_{j+1}=ic_{2j}c_{2j+1}\quad(j<N),
\]
and
\[
Z_NZ_1=-iPc_{2N}c_1.
\]
\item On the $P=+1$ and $P=-1$ sectors use, respectively, the distinguished
Fourier modes with momenta in $\mathcal K_+$ and $\mathcal K_-$ to
diagonalize the induced action on the Clifford generators.  Derive
\[
\cosh\gamma(k)
=\cosh(2K_h)\cosh(2K_v^*)
-\sinh(2K_h)\sinh(2K_v^*)\cos k,
\qquad \gamma(k)\geq0,
\]
and prove that the transfer eigenvalues are
\[
\Lambda
=(2\sinh(2K_v))^{N/2}
\exp\!\left(\frac12\sum_{k\in\mathcal K_\pm}
\epsilon_k\gamma(k)\right),
\qquad \epsilon_k\in\{-1,1\},
\]
with the fermion-parity condition determined by the chosen $P$ sector.

\item At $K_h=K_v=K_c$ prove
\[
\gamma(k)=2\operatorname{arsinh}|\sin(k/2)|
=|k|+O(|k|^3).
\]
After subtracting the sector ground energy and measuring gaps in units
$2\pi(a_x/a_t)/L$, use the parity projections to obtain the chiral
scaling products
\[
\begin{aligned}
\chi_0(q)&=\frac{q^{-1/48}}2
\left[\prod_{m\geq0}(1+q^{m+1/2})
+\prod_{m\geq0}(1-q^{m+1/2})\right],\\
\chi_{1/2}(q)&=\frac{q^{-1/48}}2
\left[\prod_{m\geq0}(1+q^{m+1/2})
-\prod_{m\geq0}(1-q^{m+1/2})\right],\\
\chi_{1/16}(q)&=q^{1/24}\prod_{m\geq1}(1+q^m).
\end{aligned}
\]
Use the Jacobi triple product to recover the theta--eta formulas in the
Ising character exercise.
\end{enumerate}
\end{exercise}

\begin{exercise}[Theta-character transfer spectra]
Let $\Lambda_0(L)>0$ be the Perron eigenvalue of
$T_N^{\rm Is}(K_c,K_c)$ and let
$\Lambda_{a,r,\bar r}(L)$ denote its low-lying symmetry-resolved eigenvalues.
Define
\[
E_{a,r,\bar r}(L)
=-\frac1{a_t}\log|\Lambda_{a,r,\bar r}(L)|.
\]
Use the critical Clifford spectrum and Euler--Maclaurin summation to prove
the finite-size asymptotic
\[
E_{a,r,\bar r}(L)
=e_\infty L+\frac{2\pi v}{L}
\left(h_a+\bar h_a+r+\bar r-\frac{c+\bar c}{24}\right)
+o(L^{-1}).
\]
If $R_N$ acts on the same state by $e^{i\mathsf p a_x}$, prove
\[
\mathsf p=\frac{2\pi}{L}(h_a-\bar h_a+r-\bar r)
\pmod{2\pi/a_x}.
\]
Prove that $v=a_x/a_t$ for the displayed isotropic transfer normalization.
Use the three Ising character products to prove that the coefficient of
$q^{h_a+r-c/24}\bar q^{\bar h_a+\bar r-\bar c/24}$ gives the predicted
multiplicity at these energy and momentum values.  Express every excitation
gap directly as
\[
E_j(L)-E_0(L)
=-\frac1{a_t}\log\left|\frac{\Lambda_j(L)}{\Lambda_0(L)}\right|.
\]
Compute the first spectral gaps and degeneracies in the three Ising
sectors.  Deduce the universal gap ratios and the $L^{-1}$ Casimir-energy
correction.  State how finite-size transfer eigenvalues determine $c$, the
primary weights, and $v$, including the symmetry-sector resolution,
anisotropy corrections, and irrelevant-operator corrections required for
the comparison.
\end{exercise}

\begin{exercise}[Harper monodromy and \texorpdfstring{$SL_2(\mathbb R)$}{SL2(R)} band geometry]
Let $\mathcal E=V\oplus V^*$ be a symplectic phase space with inverse
Poisson tensor $\Pi$.  Choose $k,\ell\in\mathcal E^*$ with
\[
-\hbar\Pi(k,\ell)=2\pi\alpha
\]
and let $U$ and $V$ be the Weyl quantizations of $e_k$ and $e_\ell$.
Use the Moyal plane-wave identity to derive
\[
UV=e^{2\pi i\alpha}VU.
\]
Interpret $\alpha$ as the magnetic flux through one lattice cell in units
of the flux quantum, and define the magnetic nearest-neighbor Hamiltonian
\[
H_\alpha=U+U^*+V+V^*.
\]
In the position realization on $\ell^2(\mathbb Z)$ with transverse phase
$\theta$, take
\[
(U\psi)_m=\psi_{m+1},
\qquad
(V\psi)_m=e^{i(2\pi\alpha m+\theta)}\psi_m
\]
and derive the Harper operator
\[
(H_{\alpha,\theta}\psi)_m
=\psi_{m+1}+\psi_{m-1}
+2\cos(2\pi\alpha m+\theta)\psi_m.
\]
Prove that the commutator phase is the holonomy around one lattice cell.
On an $N_x\times N_y$ periodic sample, show that
$N_xN_y\alpha\notin\mathbb Z$ obstructs consistent torus boundary phases,
whereas integral total flux makes their corner cocycle trivial.  Prove
that nonzero integral total flux still obstructs a single globally
periodic vector potential.

Rewrite $H_{\alpha,\theta}\psi=E\psi$ as
\[
\binom{\psi_{m+1}}{\psi_m}
=A_m(E,\theta)\binom{\psi_m}{\psi_{m-1}},
\qquad
A_m(E,\theta)=
\begin{pmatrix}
E-2\cos(2\pi\alpha m+\theta)&-1\\
1&0
\end{pmatrix}.
\]
For coprime integers $p,q$ with $q>0$, put $\alpha=p/q$ and define the
$q$-step monodromy and discriminant by
\[
M_{p/q}(E,\theta)
=A_{q-1}(E,\theta)\cdots A_0(E,\theta),
\qquad
\Delta_{p/q}(E,\theta)=\operatorname{Tr}M_{p/q}(E,\theta).
\]
Define
\[
SL_2(\mathbb R)
=\{A\in\operatorname{End}(\mathbb R^2):\det A=1\}.
\]
Prove that every $A_m(E,\theta)$ and $M_{p/q}(E,\theta)$ belongs to this
group.  For
$A=\begin{psmallmatrix}a&b\\c&d\end{psmallmatrix}$, prove that
\[
A\cdot z=\frac{az+b}{cz+d}
\]
preserves the upper half-plane.  Show that the eigenvalues of a monodromy
matrix satisfy
\[
\lambda^2-\Delta_{p/q}(E,\theta)\lambda+1=0.
\]
Call the monodromy elliptic when $|\Delta_{p/q}|<2$, parabolic when
$|\Delta_{p/q}|=2$ and $M_{p/q}\ne\pm I$, and hyperbolic when
$|\Delta_{p/q}|>2$; treat $M_{p/q}=\pm I$ as the central limiting case.
Prove that the Bloch condition
$\psi_{m+q}=e^{iq\kappa}\psi_m$ is equivalent to
\[
\Delta_{p/q}(E,\theta)=2\cos(q\kappa),
\]
that band interiors have elliptic monodromy, open band edges have parabolic
monodromy, closed gaps have central monodromy, and open gaps have
hyperbolic monodromy with growth exponent
\[
\gamma(E,\theta)
=\operatorname{arcosh}\!\left(\frac{|\Delta_{p/q}(E,\theta)|}{2}\right).
\]

For any chosen transfer period $q$ at zero flux, use
\[
T_q(\cos t)=\cos(qt)
\]
to derive
\[
\Delta_{0,q}(E,\theta)
=2T_q\!\left(\frac{E-2\cos\theta}{2}\right).
\]
At half a flux quantum, compute the two bands
\[
E_\pm(\kappa,\theta)
=\pm2\sqrt{\cos^2\kappa+\cos^2\theta}.
\]
Deduce the flux-dependent band edges, touching energy, and exponential
attenuation per period in every gap.  State the nearest-neighbor,
uniform-flux, periodicity, contact-matching, coherence, and finite-size
assumptions under which microwave, photonic, or electronic lattice
spectroscopy tests the Chebyshev discriminant, half-flux splitting, and
hyperbolic attenuation exponent.
\end{exercise}

\begin{exercise}[Poisson--Lie groups and classical Yang--Baxter equations]
A \emph{Poisson--Lie group} is a Lie group $G$ with a Poisson structure for
which multiplication $G\times G\to G$ is Poisson.  A \emph{Lie bialgebra}
is a Lie algebra $\mathfrak g$ with a linear map
\[
\delta:\mathfrak g\to\mathfrak g\wedge\mathfrak g
\]
that is a $1$-cocycle for the adjoint action and whose transpose defines a
Lie bracket on $\mathfrak g^*$.  It is \emph{coboundary} when
\[
\delta(x)=[x\otimes1+1\otimes x,r]
\]
for some $r\in\mathfrak g\otimes\mathfrak g$.

Define
\[
\operatorname{CYB}(r)
=[r_{12},r_{13}]+[r_{12},r_{23}]+[r_{13},r_{23}].
\]
Differentiate a Poisson--Lie structure at the identity and prove that it
defines a Lie bialgebra.
Prove that the co-Jacobi identity is equivalent to
$\operatorname{CYB}(r)$ being invariant under the diagonal adjoint action.
Under the additional triangular hypothesis
$\operatorname{CYB}(r)=0$, recover the classical Yang--Baxter equation;
state the modified classical Yang--Baxter equation as
$\operatorname{CYB}(r)=\Omega$ for an invariant tensor $\Omega$ and
distinguish the case $\Omega\ne0$.
\end{exercise}


\begin{exercise}[Hopf algebras and \texorpdfstring{$U_v(\mathfrak{sl}_2)$}{Uv(sl2)}]
A \emph{bialgebra} is a unital associative algebra $H$ with unital algebra
homomorphisms
\[
\Delta:H\to H\otimes H,
\qquad
\varepsilon:H\to\C
\]
satisfying coassociativity and the counit identities.  A \emph{Hopf
algebra} is a bialgebra with a linear antipode $S:H\to H$ satisfying
\[
m(S\otimes\id)\Delta
=\eta\varepsilon
=m(\id\otimes S)\Delta.
\]

Fix a quantum-group parameter $0<v<1$.

For $0<v<1$, $U_v(\mathfrak{sl}_2)$ is generated by
$E,F,K,K^{-1}$ with
\[
KEK^{-1}=v^2E,
\qquad KFK^{-1}=v^{-2}F,
\qquad [E,F]=\frac{K-K^{-1}}{v-v^{-1}},
\]
and
\[
\Delta(K)=K\otimes K,
\quad
\Delta(E)=E\otimes1+K\otimes E,
\quad
\Delta(F)=F\otimes K^{-1}+1\otimes F.
\]
Its counit, antipode, and compact real-form involution are
\[
\varepsilon(E)=\varepsilon(F)=0,
\quad \varepsilon(K)=1,
\quad
S(E)=-K^{-1}E,
\quad S(F)=-FK,
\quad S(K)=K^{-1},
\]
\[
K^*=K,
\qquad E^*=KF,
\qquad F^*=EK^{-1}.
\]

Verify that the coproduct preserves the defining relations and that the
displayed counit, antipode, and involution make
$U_v(\mathfrak{sl}_2)$ a Hopf $*$-algebra.  In the $\hbar$-adic
completion put
\[
v=e^{-\hbar},\qquad K=v^H=e^{-\hbar H}.
\]
Prove that
\[
[H,E]=2E,\qquad [H,F]=-2F,
\qquad
\frac{K-K^{-1}}{v-v^{-1}}=H+O(\hbar^2),
\]
and that the relations and coproduct tend to those of
$U(\mathfrak{sl}_2)$ as $\hbar\to0$.  Expand the coproduct through first
order and compute the resulting coboundary Lie bialgebra.
\end{exercise}

\begin{exercise}[Gaussian binomial coefficients]
For an indeterminate $u$, $n\geq0$, and $0\leq r\leq n$, define
\[
\genfrac{[}{]}{0pt}{}{n}{r}_{u}
=\prod_{j=1}^r\frac{1-u^{n-r+j}}{1-u^j},
\]
with value $1$ for $r=0$ and value $0$ outside this range.

Derive
\[
\genfrac{[}{]}{0pt}{}{n}{r}_{u}
=\genfrac{[}{]}{0pt}{}{n-1}{r}_{u}
+u^{n-r}\genfrac{[}{]}{0pt}{}{n-1}{r-1}_{u}
\]
and prove
\[
\prod_{j=0}^{n-1}(1+tu^j)
=\sum_{r=0}^n
u^{r(r-1)/2}\genfrac{[}{]}{0pt}{}{n}{r}_{u}t^r.
\]
For $YX=uXY$, prove the noncommutative identity
\[
(X+Y)^n=\sum_{r=0}^n
\genfrac{[}{]}{0pt}{}{n}{r}_{u}X^rY^{n-r}.
\]
Set $u=v^2$ and apply it to
$X=E\otimes1$ and $Y=K\otimes E$ in $U_v(\mathfrak{sl}_2)$ to obtain an
explicit formula for $\Delta(E^n)$.  Show that the limit $v\to1$ recovers
the ordinary binomial theorem and the undeformed coproduct.
\end{exercise}


\begin{exercise}[Gaussian-binomial state counts]
Apply the Gaussian binomial theorem.
Prove that the coefficient of $u^m$ in
$\genfrac{[}{]}{0pt}{}{n}{r}_u$ counts partitions
of $m$ whose diagrams fit inside an $r\times(n-r)$ rectangle.  For $n$
fermionic levels with energies $j\hbar\omega$, $0\leq j<n$, prove that the

\prod_{j=0}^{n-1}(1+tu^j)

is the exact number of $r$-particle states at excitation energy

\end{exercise}

\begin{exercise}[Compact quantum groups and $SU_v(2)$]
A \emph{compact quantum group} $\mathbb G$ is a unital $C^*$-algebra
$C(\mathbb G)$ with a unital $*$-homomorphism
\[
\Delta:C(\mathbb G)\longrightarrow
C(\mathbb G)\otimes_{\min}C(\mathbb G)
\]
where $\otimes_{\min}$ is the closure of the algebraic tensor product in
$B(H\otimes K)$ for faithful representations on $H$ and $K$.  The
coproduct satisfies
\[
(\Delta\otimes\id)\Delta=(\id\otimes\Delta)\Delta
\]
and
\[
\overline{\Delta(C(\mathbb G))(1\otimes C(\mathbb G))}
=C(\mathbb G)\otimes_{\min}C(\mathbb G)
=\overline{\Delta(C(\mathbb G))(C(\mathbb G)\otimes1)}.
\]
Here a state on a unital $C^*$-algebra is a linear functional $h$ with
$h(a^*a)\geq0$ and $h(1)=1$.  A \emph{Haar state} is a state
$h$ satisfying
\[
(h\otimes\id)\Delta(a)=h(a)1=(\id\otimes h)\Delta(a).
\]

Let $\operatorname{Pol}(SU_v(2))$ be the universal unital $*$-algebra
generated by $\alpha,\gamma$ so that
\[
u=\begin{pmatrix}
\alpha&-v\gamma^*\\
\gamma&\alpha^*
\end{pmatrix}
\]
is unitary.  Equivalently, its relations are
\[
\begin{gathered}
\alpha^*\alpha+\gamma^*\gamma=1,
\qquad \alpha\alpha^*+v^2\gamma\gamma^*=1,
\qquad \gamma^*\gamma=\gamma\gamma^*,\\
\alpha\gamma=v\gamma\alpha,
\qquad \alpha\gamma^*=v\gamma^*\alpha,
\end{gathered}
\]
Define $\Delta(u_{ij})=\sum_k u_{ik}\otimes u_{kj}$, and let
$C(SU_v(2))$ be the universal $C^*$-completion.

Prove that $\Delta$ is a coassociative $*$-homomorphism satisfying the
cancellation conditions.  On $\ell^2(\Z_{\geq0})$, define for $\xi\in\T$
\[
\pi_\xi(\gamma)e_n=\xi v^ne_n,
\qquad
\pi_\xi(\alpha)e_n=(1-v^{2n})^{1/2}e_{n-1},
\qquad e_{-1}=0.
\]
Verify the relations and prove that
\[
h(a)=(1-v^2)\sum_{n\geq0}v^{2n}
\int_\T\langle e_n,\pi_\xi(a)e_n\rangle\,dm_\T(\xi)
\]
is the Haar state.
\end{exercise}

\begin{exercise}[Quantum-group coefficient pairings]
Call a finite-dimensional $U_v(\mathfrak{sl}_2)$-module $V$ \emph{type-$1$}
when $K$ is diagonalizable and
\[
V=\bigoplus_{m\in\mathbb Z}V[m],
\qquad K|_{V[m]}=v^m\id_{V[m]}.
\]
On the distinguished Hermitian space
$V_1=\mathbb Ce_+\oplus\mathbb Ce_-$ define the fundamental type-$1$
module by
\[
Ke_\pm=v^{\pm1}e_\pm,\qquad
Ee_-=v^{-1/2}e_+,\quad Ee_+=0,\qquad
Fe_+=v^{1/2}e_-,\quad Fe_-=0.
\]
For a finite-dimensional type-$1$ module $V$,
$\xi\in V$, and $\varphi\in V^\vee$, define
\[
c_{\varphi,\xi}(x)=\varphi(x\xi)
\qquad(x\in U_v(\mathfrak{sl}_2)).
\]
Let $\mathcal O_v$ be the Hopf algebra generated by these coefficient
functionals.  Its pairing with $U_v(\mathfrak{sl}_2)$ is
\[
\langle x,c_{\varphi,\xi}\rangle=c_{\varphi,\xi}(x),
\]
with multiplication and coproduct characterized by
\[
\langle x,ab\rangle=\langle\Delta x,a\otimes b\rangle,
\qquad
\langle xy,a\rangle=\langle x\otimes y,\Delta a\rangle.
\]

Prove that the coefficient map of a tensor product is the product of the
coefficient maps, that the coefficient map of a dual module is obtained
from the antipode, and that the displayed pairing is a nondegenerate Hopf
pairing.  Apply it to the fundamental module and prove that sending its
coefficient corepresentation to the displayed matrix $u$ extends to a
Hopf $*$-isomorphism
\[
\mathcal O_v\longrightarrow\operatorname{Pol}(SU_v(2)).
\]
\end{exercise}

\begin{exercise}[$C^*$-tensor categories and corepresentations]
A finite-dimensional unitary corepresentation of a compact quantum group
$\mathbb G$ on $H_U$ is a unitary $U\in B(H_U)\otimes C(\mathbb G)$ with
$(\id\otimes\Delta)(U)=U_{12}U_{13}$.

A \emph{$\C$-linear category} $\mathcal C$ consists of objects, complex
vector spaces $\Hom_{\mathcal C}(X,Y)$, bilinear associative compositions
\[
\Hom_{\mathcal C}(Y,Z)\otimes\Hom_{\mathcal C}(X,Y)
\longrightarrow\Hom_{\mathcal C}(X,Z),
\qquad (g,f)\longmapsto gf,
\]
and identity morphisms $\id_X$.  A \emph{$\C$-linear tensor category} also
has a bilinear tensor product on objects and morphisms, a tensor unit
$\mathbf1$, and isomorphisms
\[
a_{X,Y,Z}:(X\otimes Y)\otimes Z\longrightarrow X\otimes(Y\otimes Z),
\quad
l_X:\mathbf1\otimes X\longrightarrow X,
\quad
r_X:X\otimes\mathbf1\longrightarrow X.
\]
The tensor product satisfies
\[
(f'f)\otimes(g'g)=(f'\otimes g')(f\otimes g),
\qquad
\id_X\otimes\id_Y=\id_{X\otimes Y},
\]
and the structure maps commute with morphisms; in particular,
\[
a_{X',Y',Z'}((f\otimes g)\otimes h)
=(f\otimes(g\otimes h))a_{X,Y,Z}.
\]
They satisfy the pentagon and triangle identities
\[
a_{W,X,Y\otimes Z}a_{W\otimes X,Y,Z}
=(\id_W\otimes a_{X,Y,Z})a_{W,X\otimes Y,Z}
(a_{W,X,Y}\otimes\id_Z),
\]
\[
(\id_X\otimes l_Y)a_{X,\mathbf1,Y}=r_X\otimes\id_Y.
\]
A \emph{$C^*$-category} has Banach morphism spaces and conjugate-linear
maps $f\mapsto f^*$ with
\[
f^*:Y\longrightarrow X,
\qquad
(f^*)^*=f,
\qquad
(gf)^*=f^*g^*,
\]
\[
\|gf\|\leq\|g\|\|f\|,
\qquad
\|f^*f\|=\|f\|^2.
\]
A \emph{$C^*$-tensor category} is a $\C$-linear tensor category with this
structure, with
\[
(f\otimes g)^*=f^*\otimes g^*,
\qquad
a^*=a^{-1},\quad l^*=l^{-1},\quad r^*=r^{-1}.
\]
Associators and unit maps are suppressed in subsequent tensor composites.

For a compact quantum group $\mathbb G$, let
$\operatorname{Rep}(\mathbb G)$ have finite-dimensional unitary
corepresentations as objects and
\[
\Hom_{\mathbb G}(U,V)
=\{A:H_U\to H_V:(A\otimes1)U=V(A\otimes1)\}
\]
as morphism spaces.  Composition and adjoints are those of Hilbert-space
operators.  Define
\[
U\boxtimes V=U_{13}V_{23}
\quad\text{on }H_U\otimes H_V,
\]
and take the trivial corepresentation on $\C$ as tensor unit, with the
Hilbert-space associator and unit maps.

\begin{enumerate}[label=(\alph*)]
\item Prove that the adjoint of an intertwiner is an intertwiner, that
$U\boxtimes V$ is a unitary corepresentation, and that tensor products of
intertwiners are intertwiners.
\item Verify the pentagon, triangle, and
$C^*$-identities and deduce that $\operatorname{Rep}(\mathbb G)$ is a
$C^*$-tensor category.
\item For $C(\mathbb G)=\C$, identify it with
$\operatorname{Hilb}_{\mathrm{fd}}$, the category of finite-dimensional
Hilbert spaces and all linear maps.
\end{enumerate}
\end{exercise}

\begin{exercise}[Rigidity, conjugates, and quantum dimension]
An orthogonal direct sum $X=\bigoplus_{i=1}^rX_i$ is specified by morphisms
$u_i:X_i\to X$ satisfying
\[
u_i^*u_j=\delta_{ij}\id_{X_i},
\qquad
\sum_{i=1}^ru_iu_i^*=\id_X.
\]

In a $\C$-linear tensor category, a \emph{left dual} of $V$ is an object
$V^\vee$ with morphisms
\[
\operatorname{ev}_V:V^\vee\otimes V\longrightarrow\mathbf1,
\qquad
\operatorname{coev}_V:\mathbf1\longrightarrow V\otimes V^\vee
\]
satisfying
\[
(\id_V\otimes\operatorname{ev}_V)
(\operatorname{coev}_V\otimes\id_V)=\id_V,
\]
\[
(\operatorname{ev}_V\otimes\id_{V^\vee})
(\id_{V^\vee}\otimes\operatorname{coev}_V)=\id_{V^\vee}.
\]
A \emph{right dual} ${}^\vee V$ has maps
\[
\widetilde{\operatorname{ev}}_V:V\otimes{}^\vee V\longrightarrow\mathbf1,
\qquad
\widetilde{\operatorname{coev}}_V:\mathbf1
\longrightarrow{}^\vee V\otimes V
\]
satisfying
\[
(\widetilde{\operatorname{ev}}_V\otimes\id_V)
(\id_V\otimes\widetilde{\operatorname{coev}}_V)=\id_V,
\]
\[
(\id_{{}^\vee V}\otimes\widetilde{\operatorname{ev}}_V)
(\widetilde{\operatorname{coev}}_V\otimes\id_{{}^\vee V})
=\id_{{}^\vee V}.
\]
In a $C^*$-tensor category, a \emph{conjugate} of an object $V$ is an object
$\overline V$ with morphisms
\[
R_V:\mathbf 1\longrightarrow\overline V\otimes V,
\qquad
\overline R_V:\mathbf 1\longrightarrow V\otimes\overline V
\]
satisfying
\[
(\overline R_V^*\otimes\id_V)(\id_V\otimes R_V)=\id_V,
\qquad
(R_V^*\otimes\id_{\overline V})
(\id_{\overline V}\otimes\overline R_V)=\id_{\overline V}.
\]
Call a $C^*$-tensor category \emph{rigid} when every object has a
conjugate.
Its dual maps are
\[
\operatorname{ev}_V=R_V^*,
\qquad
\operatorname{coev}_V=\overline R_V,
\qquad
\widetilde{\operatorname{ev}}_V=\overline R_V^*,
\qquad
\widetilde{\operatorname{coev}}_V=R_V.
\]
A solution of the conjugate equations is \emph{standard} when
\[
R_V^*(\id_{\overline V}\otimes T)R_V
=\overline R_V^*(T\otimes\id_{\overline V})\overline R_V
\qquad(T\in\End(V)).
\]
For a standard solution, define
\[
\operatorname{tr}_{\mathcal C}(T)
=R_V^*(\id_{\overline V}\otimes T)R_V,
\qquad
\dim_{\mathcal C}(V)=\operatorname{tr}_{\mathcal C}(\id_V).
\]
For $\mathcal C=\operatorname{Rep}(SU_v(2))$, write
$\operatorname{tr}_v$ and $\dim_v$.

\begin{enumerate}[label=(\alph*)]
\item Let two standard solutions of the conjugate equations be given for $V$.
Use the snake identities and standardness to construct the unitary
isomorphism between their conjugate objects and prove that they define the
same trace.
\item For $S:V\to W$ and $T:W\to V$, prove
\[
\operatorname{tr}_{\mathcal C}(TS)
=\operatorname{tr}_{\mathcal C}(ST),
\qquad
\operatorname{tr}_{\mathcal C}(T^*T)\geq0.
\]
\item Construct standard solutions for tensor products, conjugates, and existing
orthogonal direct sums and prove
\[
\dim_{\mathcal C}(V\otimes W)
=\dim_{\mathcal C}(V)\dim_{\mathcal C}(W),
\qquad
\dim_{\mathcal C}(\overline V)=\dim_{\mathcal C}(V),
\]
\[
\dim_{\mathcal C}(V\oplus W)
=\dim_{\mathcal C}(V)+\dim_{\mathcal C}(W).
\]
\end{enumerate}
\end{exercise}

\begin{exercise}[Basic hypergeometric series and summations]
For $m\in\N$, define
\[
(a;Q)_m=\prod_{j=0}^{m-1}(1-aQ^j),
\qquad
(a;Q)_\infty=\prod_{j=0}^{\infty}(1-aQ^j),
\]
\[
(a_1,\ldots,a_r;Q)_m=\prod_{i=1}^r(a_i;Q)_m,
\qquad
(a_1,\ldots,a_r;Q)_\infty=\prod_{i=1}^r(a_i;Q)_\infty.
\]
The \emph{basic hypergeometric series} is
\[
{}_r\phi_s\!\left(
\begin{matrix}a_1,\ldots,a_r\\b_1,\ldots,b_s\end{matrix};Q,z
\right)
=\sum_{m=0}^\infty
\frac{(a_1,\ldots,a_r;Q)_m}
     {(Q,b_1,\ldots,b_s;Q)_m}
\left((-1)^mQ^{m(m-1)/2}\right)^{1+s-r}z^m.
\]
It is terminating when one numerator parameter is $Q^{-n}$.

Prove the $Q$-Gauss identity
\[
{}_2\phi_1\!\left(
\begin{matrix}a,b\\c\end{matrix};Q,\frac{c}{ab}
\right)
=\frac{(c/a,c/b;Q)_\infty}{(c,c/(ab);Q)_\infty},
\qquad \left|\frac{c}{ab}\right|<1,
\]
and the terminating $Q$-Pfaff--Saalsch\"utz identity
\[
{}_3\phi_2\!\left(
\begin{matrix}a,b,Q^{-n}\\c,abQ^{1-n}/c\end{matrix};Q,Q
\right)
=\frac{(c/a,c/b;Q)_n}{(c,c/(ab);Q)_n}
\]
by comparing coefficients after multiplication by the denominator
$Q$-products in the first identity, and by showing that the two sides of
the second identity satisfy the same recurrence in $n$ and have the same
initial value.
\end{exercise}

\begin{exercise}[Universal $R$-matrices and braided tensor categories]
A \emph{universal $R$-matrix} for a Hopf algebra $H$ is an invertible
element $\mathcal R$ in a completion of $H\otimes H$ on which the
finite-dimensional tensor-product representations under consideration
extend, satisfying
\[
\mathcal R\Delta(x)\mathcal R^{-1}=\Delta^{\rm op}(x),
\quad
(\Delta\otimes\id)(\mathcal R)=\mathcal R_{13}\mathcal R_{23},
\quad
(\id\otimes\Delta)(\mathcal R)=\mathcal R_{13}\mathcal R_{12}.
\]
For finite-dimensional modules $V,W$, define
\[
c_{V,W}
=\tau\circ(\pi_V\otimes\pi_W)(\mathcal R):
V\otimes W\longrightarrow W\otimes V.
\]
A \emph{braiding} on a $\C$-linear tensor category $\mathcal C$ is a family
of isomorphisms
\[
c_{X,Y}:X\otimes Y\longrightarrow Y\otimes X
\]
satisfying, for $f:X\to X'$ and $g:Y\to Y'$,
\[
(g\otimes f)c_{X,Y}=c_{X',Y'}(f\otimes g),
\]
and the two hexagon identities
\[
c_{X,Y\otimes Z}
=(\id_Y\otimes c_{X,Z})(c_{X,Y}\otimes\id_Z),
\]
\[
c_{X\otimes Y,Z}
=(c_{X,Z}\otimes\id_Y)(\id_X\otimes c_{Y,Z}).
\]
The category is \emph{braided} when it has a braiding.

Prove that $c_{V,W}$ is a module intertwiner satisfying naturality and that
the two coproduct identities for $\mathcal R$ give the two hexagon
identities, so that the finite-dimensional module category is braided.
Derive
\[
\mathcal R_{12}\mathcal R_{13}\mathcal R_{23}
=\mathcal R_{23}\mathcal R_{13}\mathcal R_{12}
\]
and, on $V^{\otimes3}$,
\[
c_{12}c_{23}c_{12}=c_{23}c_{12}c_{23}.
\]
Prove that the operators $c_{j,j+1}$ define braid-group representations on
tensor powers of every finite-dimensional type-$1$ module.
\end{exercise}

\begin{exercise}[Ribbon categories and ribbon elements]
Let $\mathcal C$ be a braided rigid $\C$-linear tensor category.  A
\emph{twist} is a family of isomorphisms $\theta_X:X\to X$ satisfying
\[
\theta_Yf=f\theta_X\qquad(f:X\to Y)
\]
and
\[
\theta_{\mathbf1}=\id_{\mathbf1},
\qquad
\theta_{X\otimes Y}
=c_{Y,X}c_{X,Y}(\theta_X\otimes\theta_Y).
\]
The category is \emph{ribbon} when
\[
\theta_{X^\vee}=(\theta_X)^\vee
\]
for every object $X$.

For the universal $R$-matrix, set
\[
u=m(S\otimes\id)(\mathcal R_{21}).
\]
Let $\nu_R\in H$ be central and invertible and satisfy
\[
\nu_R^2=uS(u),
\qquad S(\nu_R)=\nu_R,
\qquad \varepsilon(\nu_R)=1,
\qquad
\Delta(\nu_R)=(\mathcal R_{21}\mathcal R)^{-1}
(\nu_R\otimes \nu_R).
\]
Prove that
\[
\theta_V=\pi_V(\nu_R^{-1})
\]
defines a ribbon twist on the category of finite-dimensional $H$-modules.
Derive invariance of the quantum trace under framed ribbon isotopy.
\end{exercise}

\begin{exercise}[Kauffman bracket skein algebra]
Let $\Sigma$ be an oriented surface and let $\mathcal L(\Sigma)$ be the
set of framed-isotopy classes of unoriented links in
$\Sigma\times[0,1]$, including the empty link.  For
$v\in\mathbb C^\times$, define $K_v(\Sigma)$ as the complex span of
$\mathcal L(\Sigma)$ modulo the following local relations.  If
$L_\times,L_0,L_\infty$ agree outside a disk and differ inside it by a
crossing and its two smoothings, impose
\[
[L_\times]=v[L_0]+v^{-1}[L_\infty].
\]
If $\bigcirc$ is a contractible component disjoint from $L$, impose
\[
[L\sqcup\bigcirc]=-(v^2+v^{-2})[L].
\]
Define multiplication by stacking links in the interval direction.

Prove that the relations are preserved by framed Reidemeister moves and
that stacking makes $K_v(\Sigma)$ an associative algebra with the empty
link as identity.  For two simple closed curves meeting once, resolve
their crossing to express both multiplication orders in terms of the two
smoothings and compute their commutator.

At $v=-1$, let $\rho:\pi_1(\Sigma)\to SL_2(\mathbb C)$ and assign to an
unoriented loop $\gamma$ the function
\[
[\gamma]\longmapsto-W_\gamma(\rho)
=-\operatorname{tr}\rho(\gamma),
\]
extending multiplicatively over disjoint components.  Use
Cayley--Hamilton to prove
\[
W_\alpha W_\beta
=W_{\alpha\beta}+W_{\alpha\beta^{-1}}
\]
and show that the assignment respects both skein relations.  Deduce that
the classical specialization is commutative and maps to the algebra of
Wilson-trace functions on flat $SL_2$ connections.

For $\Sigma=T^2$, put
\[
x=W_{1,0},\qquad y=W_{0,1},\qquad z=W_{1,1}.
\]
Use successive skein resolutions and the Euclidean algorithm to express
every $W_{p,q}$ as a polynomial in $x,y,z$, reduced modulo
\[
x^2+y^2+z^2-xyz-4=0.
\]
Use the commutator computed above to prove that $K_v(T^2)$ is
noncommutative for generic $v$ and that its specialization at $v=-1$
maps to the torus Wilson-trace algebra.
\end{exercise}

\begin{exercise}[Type-$A$ Hecke actions]

\begin{exercise}[Hecke Poincar\'e identity]

\begin{exercise}[Affine Hecke algebras and Baxterization]
The extended affine Hecke algebra
$\mathcal H_v^{\mathrm{aff}}(GL_r)$ is generated by
$T_1,\ldots,T_{r-1}$ and commuting invertible elements
$X_1,\ldots,X_r$, subject to the finite Hecke relations and

T_iX_iT_i=X_{i+1},
\qquad
T_iX_j=X_jT_i\quad(j\neq i,i+1).

Equivalently, it is the Hecke quotient of the extended affine braid group

For an invertible spectral parameter $u$, define

\check R_i(u)=uT_i-u^{-1}T_i^{-1}.

Its eigenvalues on the $v$ and $-v^{-1}$ eigenspaces of $T_i$ are

uv-u^{-1}v^{-1},
\qquad
-uv^{-1}+u^{-1}v.

Use the braid and Hecke relations for $T_i$ to prove

\check R_i(u)\check R_{i+1}(uv)\check R_i(v)
=\check R_{i+1}(v)\check R_i(uv)\check R_{i+1}(u).

Prove that $\check R_i(1)$ is a scalar multiple of the identity and compute
$\check R_i(u)\check R_i(u^{-1})$.  For $|u|=1$, prove that
$\check R_i(u)^*=\check R_i(u^{-1})$ and normalize the operators to be
unitary wherever the scalar product is nonzero.  Prove that their two
eigenvalues depend on $u$ and hence that the spectral family itself does
not satisfy one parameter-independent Hecke polynomial.
\end{exercise}

\begin{exercise}[Six-vertex transfer matrices and the XXZ spectrum]
On the fundamental module $V_1$, put

\qquad

\begin{exercise}[The Sutherland model and Jack polynomials]
For $N$ identical particles of mass $m$ on a ring of radius $R$ and
$g\geq1$, let
\[
\operatorname{Conf}_N(S^1)
=\{(\theta_1,\ldots,\theta_N):\theta_i\ne\theta_j\text{ for }i\ne j\}.
\]
On $C_c^\infty(\operatorname{Conf}_N(S^1))^{S_N}$ define
the nonnegative quadratic form
\[
q_g[\psi]=\frac{\hbar^2}{2mR^2}
\int_{(S^1)^N}\left[\sum_{j=1}^N|\partial_{\theta_j}\psi|^2
+\frac{g(g-1)}2\sum_{i<j}
\frac{|\psi|^2}{\sin^2((\theta_i-\theta_j)/2)}\right]d\theta.
\]

$g\geq1$.  Define
$\widehat H_g$ to be the unique self-adjoint operator associated with its
closure in $L^2((S^1)^N)^{S_N}$; this is the Friedrichs realization of the
displayed differential expression.  Define
\[
\Psi_0(\theta)=\prod_{i<j}|e^{i\theta_i}-e^{i\theta_j}|^g.
\]
For partitions of equal size, write $\mu<\lambda$ in dominance
order when
\[
\sum_{i=1}^k\mu_i\leq\sum_{i=1}^k\lambda_i
\]
for every $k\geq1$, with missing parts interpreted as zero, and with at
least one inequality strict.  For a partition $\lambda$ of length at most
$N$, let $m_\lambda$ be the sum of the distinct permutations of
$z_1^{\lambda_1}\cdots z_N^{\lambda_N}$.

The Jack polynomial $P_\lambda^{(1/g)}$ is the symmetric polynomial
normalized by
\[
P_\lambda^{(1/g)}=m_\lambda+
\sum_{\mu<\lambda}c_{\lambda\mu}(g)m_\mu
\]
and by satisfying
\[
\left[
\sum_i(z_i\partial_{z_i})^2
+g\sum_{i<j}\frac{z_i+z_j}{z_i-z_j}
(z_i\partial_{z_i}-z_j\partial_{z_j})
\right]P_\lambda^{(1/g)}
=\left[\sum_i
\bigl(\lambda_i^2+g(N+1-2i)\lambda_i\bigr)\right]
P_\lambda^{(1/g)};
\]
here $<$ is dominance order.

Conjugate the quantum Hamiltonian by $\Psi_0$, identify its commuting
quantum integrals, and prove that
\[
\Psi_\lambda=\Psi_0P_\lambda^{(1/g)}(e^{i\theta_1},\ldots,e^{i\theta_N})
\]
is a joint eigenfunction with
\[
E_\lambda=\frac{\hbar^2}{2mR^2}
\left[\frac{g^2N(N^2-1)}{12}
+\sum_{j=1}^N
\bigl(\lambda_j^2+g(N+1-2j)\lambda_j\bigr)\right].
\]
\end{exercise}

\begin{exercise}[Elliptic Ruijsenaars spectral tori]
